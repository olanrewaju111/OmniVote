     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { useDashboardStore } from '@/store/dashboard';
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table';
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter,
    16→  DialogHeader, DialogTitle,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    20→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  Search, Users, UserCheck, UserX, Shield, ShieldAlert, Wrench,
    27→  Loader2, FileText, Trash2, Eye, ToggleLeft, X,
    28→} from 'lucide-react';
    29→import { cn } from '@/lib/utils';
    30→import { toast } from 'sonner';
    31→
    32→interface AgentUser {
    33→  id: string;
    34→  email: string;
    35→  name: string;
    36→  role: string;
    37→  isOnline: boolean;
    38→  lastSeenAt: string | null;
    39→  createdAt: string;
    40→  _count?: { incidents: number; auditLogs: number };
    41→}
    42→
    43→interface ReportsSlideOver {
    44→  agent: AgentUser | null;
    45→  open: boolean;
    46→}
    47→
    48→export function AgentRoster() {
    49→  const queryClient = useQueryClient();
    50→  const { tenantId } = useDashboardStore();
    51→  const [search, setSearch] = useState('');
    52→  const [searchInput, setSearchInput] = useState('');
    53→  const [roleFilter, setRoleFilter] = useState('ALL');
    54→  const [addOpen, setAddOpen] = useState(false);
    55→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    56→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    57→
    58→  // Add agent form state
    59→  const [newName, setNewName] = useState('');
    60→  const [newEmail, setNewEmail] = useState('');
    61→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    62→
    63→  // Fetch agents from proper API
    64→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    65→    queryKey: ['agents', search, roleFilter, tenantId],
    66→    queryFn: () => {
    67→      const params = new URLSearchParams();
    68→      if (search) params.set('search', search);
    69→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    70→      if (tenantId) params.set('tenantId', tenantId);
    71→      return fetch(`/api/agents?${params}`).then(r => r.json());
    72→    },
    73→  });
    74→
    75→  // Add agent mutation
    76→  const addMutation = useMutation({
    77→    mutationFn: (body: { name: string; email: string; role: string }) =>
    78→      fetch(`/api/agents?tenantId=${tenantId}`, {
    79→        method: 'POST',
    80→        headers: { 'Content-Type': 'application/json' },
    81→        body: JSON.stringify(body),
    82→      }).then(r => r.json()),
    83→    onSuccess: (data) => {
    84→      if (data.error) {
    85→        toast.error(data.error);
    86→        return;
    87→      }
    88→      toast.success(`Agent "${data.user.name}" added successfully`);
    89→      queryClient.invalidateQueries({ queryKey: ['agents'] });
    90→      setAddOpen(false);
    91→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
    92→    },
    93→    onError: () => toast.error('Failed to add agent'),
    94→  });
    95→
    96→  // Action mutation (toggle online, remote wipe, change role, delete)
    97→  const actionMutation = useMutation({
    98→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
    99→      fetch(`/api/agents?tenantId=${tenantId}`, {
   100→        method: 'PATCH',
   101→        headers: { 'Content-Type': 'application/json' },
   102→        body: JSON.stringify(body),
   103→      }).then(r => r.json()),
   104→    onSuccess: (data, variables) => {
   105→      if (data.error) {
   106→        toast.error(data.error);
   107→        return;
   108→      }
   109→      if (variables.action === 'DELETE') {
   110→        toast.success(data.message || 'Agent removed');
   111→      } else if (variables.action === 'REMOTE_WIPE') {
   112→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   113→      } else if (variables.action === 'CHANGE_ROLE') {
   114→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   115→      } else {
   116→        toast.success('Status updated');
   117→      }
   118→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   119→      setConfirmAction({ type: '', agent: null });
   120→    },
   121→    onError: () => toast.error('Action failed'),
   122→  });
   123→
   124→  const users = data?.users || [];
   125→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   126→  const online = fieldAgents.filter(u => u.isOnline).length;
   127→
   128→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   129→    acc[u.role] = (acc[u.role] || 0) + 1;
   130→    return acc;
   131→  }, {});
   132→
   133→  // Search with debounce via the query key
   134→  const handleSearch = useCallback(() => {
   135→    setSearch(searchInput);
   136→  }, [searchInput]);
   137→
   138→  // Handle confirm action
   139→  const handleConfirmAction = () => {
   140→    if (!confirmAction.agent) return;
   141→    if (confirmAction.type === 'DELETE') {
   142→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   143→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   144→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   145→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   146→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   147→    }
   148→  };
   149→
   150→  // Handle add agent
   151→  const handleAddAgent = () => {
   152→    if (!newName.trim() || !newEmail.trim()) {
   153→      toast.error('Name and email are required');
   154→      return;
   155→    }
   156→    if (!newEmail.includes('@')) {
   157→      toast.error('Please enter a valid email');
   158→      return;
   159→    }
   160→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   161→  };
   162→
   163→  return (
   164→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   165→      {/* Header */}
   166→      <div className="flex items-center justify-between flex-wrap gap-3">
   167→        <div>
   168→          <h2 className="text-lg font-semibold flex items-center gap-2">
   169→            <Users className="h-5 w-5 text-emerald" />
   170→            Agent Roster
   171→          </h2>
   172→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   173→        </div>
   174→        <Button
   175→          onClick={() => setAddOpen(true)}
   176→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   177→        >
   178→          <UserCheck className="h-4 w-4" />
   179→          Add Agent
   180→        </Button>
   181→      </div>
   182→
   183→      {/* Summary cards */}
   184→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   185→        <Card className="border-border bg-card/40">
   186→          <CardContent className="p-3.5">
   187→            <div className="flex items-center gap-2 mb-1">
   188→              <Users className="h-4 w-4 text-emerald" />
   189→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   190→            </div>
   191→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   192→          </CardContent>
   193→        </Card>
   194→        <Card className="border-emerald/20 bg-emerald/5">
   195→          <CardContent className="p-3.5">
   196→            <div className="flex items-center gap-2 mb-1">
   197→              <UserCheck className="h-4 w-4 text-emerald" />
   198→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   199→            </div>
   200→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   201→          </CardContent>
   202→        </Card>
   203→        <Card className="border-amber/20 bg-amber/5">
   204→          <CardContent className="p-3.5">
   205→            <div className="flex items-center gap-2 mb-1">
   206→              <UserX className="h-4 w-4 text-amber" />
   207→              <span className="text-[11px] text-muted-foreground">Offline</span>
   208→            </div>
   209→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   210→          </CardContent>
   211→        </Card>
   212→        <Card className="border-border bg-card/40">
   213→          <CardContent className="p-3.5">
   214→            <div className="flex items-center gap-2 mb-1">
   215→              <Shield className="h-4 w-4 text-cyan" />
   216→              <span className="text-[11px] text-muted-foreground">All Users</span>
   217→            </div>
   218→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   219→          </CardContent>
   220→        </Card>
   221→      </div>
   222→
   223→      {/* Role filter + distribution badges */}
   224→      <div className="flex items-center justify-between flex-wrap gap-3">
   225→        <div className="flex items-center gap-2 flex-wrap">
   226→          {Object.entries(roleCounts).map(([role, count]) => (
   227→            <button
   228→              key={role}
   229→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   230→              className="cursor-pointer"
   231→            >
   232→              <Badge
   233→                variant="outline"
   234→                className={cn(
   235→                  'text-[11px] h-6 transition-colors',
   236→                  roleFilter === role ? 'bg-foreground/10' : '',
   237→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   238→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   239→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   240→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   241→                  'border-border text-muted-foreground'
   242→                )}
   243→              >
   244→                {role.replace(/_/g, ' ')}: {count}
   245→              </Badge>
   246→            </button>
   247→          ))}
   248→        </div>
   249→        {roleFilter !== 'ALL' && (
   250→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   251→            <X className="h-3 w-3" /> Clear filter
   252→          </button>
   253→        )}
   254→      </div>
   255→
   256→      {/* Agent table */}
   257→      <Card>
   258→        <CardHeader className="pb-2 pt-4 px-4">
   259→          <div className="flex items-center justify-between gap-3">
   260→            <CardTitle className="text-sm">
   261→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   262→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   263→            </CardTitle>
   264→            <div className="flex items-center gap-2">
   265→              <div className="relative">
   266→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   267→                <Input
   268→                  placeholder="Search agents..."
   269→                  value={searchInput}
   270→                  onChange={(e) => setSearchInput(e.target.value)}
   271→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   272→                  className="pl-8 h-8 w-48 text-xs"
   273→                />
   274→              </div>
   275→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   276→                Search
   277→              </Button>
   278→            </div>
   279→          </div>
   280→        </CardHeader>
   281→        <CardContent className="px-0 pb-0">
   282→          <div className="max-h-[480px] overflow-y-auto">
   283→            {isLoading ? (
   284→              <div className="flex items-center justify-center py-16">
   285→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   286→              </div>
   287→            ) : users.length === 0 ? (
   288→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   289→                <Users className="h-8 w-8 mb-2 opacity-40" />
   290→                <p className="text-sm">No users found</p>
   291→              </div>
   292→            ) : (
   293→              <Table>
   294→                <TableHeader>
   295→                  <TableRow className="border-border hover:bg-transparent">
   296→                    <TableHead className="text-[11px] h-9">User</TableHead>
   297→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   298→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   299→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   300→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   301→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   302→                  </TableRow>
   303→                </TableHeader>
   304→                <TableBody>
   305→                  <AnimatePresence>
   306→                    {users.map(agent => (
   307→                      <motion.tr
   308→                        key={agent.id}
   309→                        initial={{ opacity: 0, y: 4 }}
   310→                        animate={{ opacity: 1, y: 0 }}
   311→                        exit={{ opacity: 0, y: -4 }}
   312→                        className="border-border hover:bg-card/60 transition-colors"
   313→                      >
   314→                        <TableCell className="py-2.5">
   315→                          <div className="flex items-center gap-2.5">
   316→                            <div className="relative">
   317→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   318→                                {agent.name.split(' ').map(n => n[0]).join('')}
   319→                              </div>
   320→                              <span className={cn(
   321→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   322→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   323→                              )} />
   324→                            </div>
   325→                            <div className="min-w-0">
   326→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   327→                              {agent.lastSeenAt && (
   328→                                <span className="text-[10px] text-muted-foreground/60">
   329→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   330→                                </span>
   331→                              )}
   332→                            </div>
   333→                          </div>
   334→                        </TableCell>
   335→                        <TableCell className="py-2.5">
   336→                          <Badge
   337→                            variant="outline"
   338→                            className={cn(
   339→                              'text-[10px] h-5',
   340→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   341→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   342→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   343→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   344→                              'border-border text-muted-foreground'
   345→                            )}
   346→                          >
   347→                            {agent.role.replace(/_/g, ' ')}
   348→                          </Badge>
   349→                        </TableCell>
   350→                        <TableCell className="py-2.5">
   351→                          <button
   352→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   353→                            className="cursor-pointer"
   354→                            disabled={actionMutation.isPending}
   355→                          >
   356→                            <Badge
   357→                              variant="outline"
   358→                              className={cn(
   359→                                'text-[10px] h-5 transition-colors cursor-pointer',
   360→                                agent.isOnline
   361→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   362→                                  : 'border-border text-muted-foreground hover:bg-card'
   363→                              )}
   364→                            >
   365→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   366→                              {agent.isOnline ? 'Online' : 'Offline'}
   367→                            </Badge>
   368→                          </button>
   369→                        </TableCell>
   370→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   371→                        <TableCell className="py-2.5 hidden md:table-cell">
   372→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   373→                            {agent._count?.incidents ?? 0}
   374→                          </span>
   375→                        </TableCell>
   376→                        <TableCell className="py-2.5 text-right">
   377→                          <div className="flex items-center justify-end gap-0.5">
   378→                            <Button
   379→                              variant="ghost"
   380→                              size="sm"
   381→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   382→                              title="View Reports"
   383→                              onClick={() => setReportsPanel({ agent, open: true })}
   384→                            >
   385→                              <Eye className="h-3.5 w-3.5" />
   386→                            </Button>
   387→                            <Button
   388→                              variant="ghost"
   389→                              size="sm"
   390→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-amber"
   391→                              title="Remote Wipe Device"
   392→                              onClick={() => setConfirmAction({ type: 'REMOTE_WIPE', agent })}
   393→                              disabled={actionMutation.isPending}
   394→                            >
   395→                              <Wrench className="h-3.5 w-3.5" />
   396→                            </Button>
   397→                            <Button
   398→                              variant="ghost"
   399→                              size="sm"
   400→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-rose"
   401→                              title="Remove Agent"
   402→                              onClick={() => setConfirmAction({ type: 'DELETE', agent })}
   403→                              disabled={actionMutation.isPending || agent.role === 'SUPER_ADMIN'}
   404→                            >
   405→                              <Trash2 className="h-3.5 w-3.5" />
   406→                            </Button>
   407→                          </div>
   408→                        </TableCell>
   409→                      </motion.tr>
   410→                    ))}
   411→                  </AnimatePresence>
   412→                </TableBody>
   413→              </Table>
   414→            )}
   415→          </div>
   416→        </CardContent>
   417→      </Card>
   418→
   419→      {/* ===== ADD AGENT DIALOG ===== */}
   420→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   421→        <DialogContent className="sm:max-w-md">
   422→          <DialogHeader>
   423→            <DialogTitle className="flex items-center gap-2">
   424→              <UserCheck className="h-5 w-5 text-emerald" />
   425→              Add New Agent
   426→            </DialogTitle>
   427→            <DialogDescription>
   428→              Create a new user account. They will be able to login from the main screen using their email.
   429→            </DialogDescription>
   430→          </DialogHeader>
   431→          <div className="space-y-4 py-2">
   432→            <div className="space-y-2">
   433→              <label className="text-xs font-medium text-muted-foreground">Full Name</label>
   434→              <Input
   435→                placeholder="e.g. Adebayo Johnson"
   436→                value={newName}
   437→                onChange={(e) => setNewName(e.target.value)}
   438→                className="h-9 text-sm"
   439→              />
   440→            </div>
   441→            <div className="space-y-2">
   442→              <label className="text-xs font-medium text-muted-foreground">Email Address</label>
   443→              <Input
   444→                placeholder="e.g. agent@nigeriaelectionwatch.org"
   445→                type="email"
   446→                value={newEmail}
   447→                onChange={(e) => setNewEmail(e.target.value)}
   448→                className="h-9 text-sm"
   449→              />
   450→            </div>
   451→            <div className="space-y-2">
   452→              <label className="text-xs font-medium text-muted-foreground">Role</label>
   453→              <Select value={newRole} onValueChange={setNewRole}>
   454→                <SelectTrigger className="h-9 text-sm">
   455→                  <SelectValue />
   456→                </SelectTrigger>
   457→                <SelectContent>
   458→                  <SelectItem value="FIELD_AGENT">Field Agent</SelectItem>
   459→                  <SelectItem value="ANALYST">Analyst</SelectItem>
   460→                  <SelectItem value="TRUST_SAFETY">Trust &amp; Safety</SelectItem>
   461→                  <SelectItem value="TENANT_ADMIN">Tenant Admin</SelectItem>
   462→                </SelectContent>
   463→              </Select>
   464→            </div>
   465→          </div>
   466→          <DialogFooter>
   467→            <Button variant="outline" onClick={() => setAddOpen(false)} disabled={addMutation.isPending}>
   468→              Cancel
   469→            </Button>
   470→            <Button
   471→              onClick={handleAddAgent}
   472→              disabled={addMutation.isPending || !newName.trim() || !newEmail.trim()}
   473→              className="bg-emerald hover:bg-emerald/90 text-emerald-950"
   474→            >
   475→              {addMutation.isPending ? (
   476→                <><Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> Adding...</>
   477→              ) : (
   478→                <><UserCheck className="h-4 w-4 mr-1.5" /> Add Agent</>
   479→              )}
   480→            </Button>
   481→          </DialogFooter>
   482→        </DialogContent>
   483→      </Dialog>
   484→
   485→      {/* ===== CONFIRM ACTION DIALOG ===== */}
   486→      <AlertDialog open={!!confirmAction.type} onOpenChange={(open) => !open && setConfirmAction({ type: '', agent: null })}>
   487→        <AlertDialogContent>
   488→          <AlertDialogHeader>
   489→            <AlertDialogTitle className={cn(
   490→              confirmAction.type === 'DELETE' ? 'text-rose' : 'text-amber'
   491→            )}>
   492→              {confirmAction.type === 'DELETE' ? 'Remove Agent' :
   493→               confirmAction.type === 'REMOTE_WIPE' ? 'Remote Wipe Device' :
   494→               'Confirm Action'}
   495→            </AlertDialogTitle>
   496→            <AlertDialogDescription>
   497→              {confirmAction.type === 'DELETE' && (
   498→                <>Are you sure you want to permanently remove <strong>{confirmAction.agent?.name}</strong>? This will delete their account and all associated audit logs. This action cannot be undone.</>
   499→              )}
   500→              {confirmAction.type === 'REMOTE_WIPE' && (
   501→                <>Send a remote wipe command to <strong>{confirmAction.agent?.name}</strong>&apos;s device? This will clear all cached data, credentials, and session tokens on the device. The agent will be set to offline.</>
   502→              )}
   503→            </AlertDialogDescription>
   504→          </AlertDialogHeader>
   505→          <AlertDialogFooter>
   506→            <AlertDialogCancel disabled={actionMutation.isPending}>Cancel</AlertDialogCancel>
   507→            <AlertDialogAction
   508→              onClick={handleConfirmAction}
   509→              disabled={actionMutation.isPending}
   510→              className={cn(
   511→                confirmAction.type === 'DELETE'
   512→                  ? 'bg-rose hover:bg-rose/90 text-white'
   513→                  : 'bg-amber hover:bg-amber/90 text-amber-950'
   514→              )}
   515→            >
   516→              {actionMutation.isPending ? (
   517→                <><Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> Processing...</>
   518→              ) : confirmAction.type === 'DELETE' ? (
   519→                <><Trash2 className="h-4 w-4 mr-1.5" /> Remove Permanently</>
   520→              ) : (
   521→                <><Wrench className="h-4 w-4 mr-1.5" /> Send Remote Wipe</>
   522→              )}
   523→            </AlertDialogAction>
   524→          </AlertDialogFooter>
   525→        </AlertDialogContent>
   526→      </AlertDialog>
   527→
   528→      {/* ===== REPORTS SLIDE PANEL ===== */}
   529→      <Dialog open={reportsPanel.open} onOpenChange={(open) => setReportsPanel({ ...reportsPanel, open })}>
   530→        <DialogContent className="sm:max-w-2xl max-h-[80vh] overflow-hidden flex flex-col">
   531→          <DialogHeader>
   532→            <DialogTitle className="flex items-center gap-2">
   533→              <FileText className="h-5 w-5 text-amber" />
   534→              Reports by {reportsPanel.agent?.name}
   535→            </DialogTitle>
   536→            <DialogDescription>
   537→              Viewing all incident reports submitted by this agent
   538→            </DialogDescription>
   539→          </DialogHeader>
   540→          <AgentReportsPanel agentId={reportsPanel.agent?.id || ''} />
   541→        </DialogContent>
   542→      </Dialog>
   543→    </div>
   544→  );
   545→}

   547→// Sub-component: fetches and displays reports for a specific agent
   548→function AgentReportsPanel({ agentId }: { agentId: string }) {
   549→  const { data, isLoading } = useQuery<{ incidents: Array<{
   550→    id: string; type: string; severity: string; status: string;
   551→    description: string; submittedAt: string;
   552→    pollingUnit: { name: string; state: string; lga: string } | null;
   553→  }>; total: number }>({
   554→    queryKey: ['agent-reports', agentId],
   555→    queryFn: () => fetch(`/api/incidents?reporterId=${agentId}&limit=50`).then(r => r.json()),
   556→    enabled: !!agentId,
   557→  });
   558→
   559→  if (isLoading) {
   560→    return (
   561→      <div className="flex items-center justify-center py-12">
   562→        <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   563→      </div>
   564→    );
   565→  }
   566→
   567→  const incidents = data?.incidents || [];
   568→
   569→  if (incidents.length === 0) {
   570→    return (
   571→      <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   572→        <FileText className="h-8 w-8 mb-2 opacity-40" />
   573→        <p className="text-sm">No reports submitted yet</p>
   574→      </div>
   575→    );
   576→  }
   577→
   578→  const severityColors: Record<string, string> = {
   579→    LOW: 'border-emerald/30 text-emerald bg-emerald/10',
   580→    MEDIUM: 'border-amber/30 text-amber bg-amber/10',
   581→    HIGH: 'border-orange/30 text-orange bg-orange/10',
   582→    CRITICAL: 'border-rose/30 text-rose bg-rose/10',
   583→  };
   584→
   585→  const statusColors: Record<string, string> = {
   586→    PENDING: 'text-muted-foreground',
   587→    REVIEWED: 'text-cyan',
   588→    ESCALATED: 'text-amber',
   589→    DISMISSED: 'text-muted-foreground/50',
   590→    QUARANTINED: 'text-rose',
   591→  };
   592→
   593→  return (
   594→    <div className="flex-1 overflow-y-auto space-y-2 pr-1">
   595→      <p className="text-xs text-muted-foreground mb-2">{data?.total || 0} total reports</p>
   596→      {incidents.map(inc => (
   597→        <div key={inc.id} className="rounded-lg border border-border bg-card/40 p-3 space-y-1.5">
   598→          <div className="flex items-center justify-between gap-2">
   599→            <div className="flex items-center gap-2">
   600→              <Badge variant="outline" className={cn('text-[10px] h-5', severityColors[inc.severity] || '')}>
   601→                {inc.severity}
   602→              </Badge>
   603→              <Badge variant="outline" className="text-[10px] h-5 border-border text-muted-foreground">
   604→                {inc.type.replace(/_/g, ' ')}
   605→              </Badge>
   606→            </div>
   607→            <div className="flex items-center gap-2">
   608→              <span className={cn('text-[10px] font-medium', statusColors[inc.status] || '')}>
   609→                {inc.status}
   610→              </span>
   611→            </div>
   612→          </div>
   613→          <p className="text-xs text-foreground/80 leading-relaxed">{inc.description}</p>
   614→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   615→            {inc.pollingUnit && (
   616→              <span className="flex items-center gap-1">
   617→                <span className="w-1 h-1 rounded-full bg-muted-foreground/40" />
   618→                {inc.pollingUnit.name}, {inc.pollingUnit.state}
   619→              </span>
   620→            )}
   621→            <span>{new Date(inc.submittedAt).toLocaleString()}</span>
   622→          </div>
   623→        </div>
   624→      ))}
   625→    </div>
   626→  );
