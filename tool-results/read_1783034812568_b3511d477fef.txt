   136→
   137→const RISK_CELL_COLORS: Record<string, { bg: string; text: string }> = {
   138→  LOW: { bg: 'rgba(16,185,129,0.20)', text: '#10b981' },
   139→  MEDIUM: { bg: 'rgba(245,158,11,0.30)', text: '#f59e0b' },
   140→  HIGH: { bg: 'rgba(239,68,68,0.30)', text: '#ef4444' },
   141→  CRITICAL: { bg: 'rgba(220,38,38,0.50)', text: '#dc2626' },
   142→};
   143→
   144→function riskCellColor(level: string) {
   145→  return RISK_CELL_COLORS[level] ?? RISK_CELL_COLORS.LOW;
   146→}
   147→
   148→function riskBarColor(score: number): string {
   149→  if (score >= 0.8) return '#ef4444';
   150→  if (score >= 0.6) return '#f97316';
   151→  if (score >= 0.3) return '#f59e0b';
   152→  return '#10b981';
   153→}
   154→
   155→function riskBadgeVariant(level: string): string {
   156→  switch (level) {
   157→    case 'CRITICAL':
   158→      return 'bg-rose-600/20 text-rose-400 border-rose-600/30';
   159→    case 'HIGH':
   160→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   161→    case 'MEDIUM':
   162→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   163→    default:
   164→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   165→  }
   166→}
   167→
   168→function statusBadgeVariant(status: string): string {
   169→  switch (status) {
   170→    case 'RUNNING':
   171→      return 'bg-cyan-500/20 text-cyan-400 border-cyan-500/30';
   172→    case 'COMPLETED':
   173→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   174→    case 'PENDING':
   175→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   176→    case 'FAILED':
   177→      return 'bg-rose-500/20 text-rose-400 border-rose-500/30';
   178→    default:
   179→      return 'bg-muted text-muted-foreground border-border';
   180→  }
   181→}
   182→
   183→function formatDate(dateStr: string): string {
   184→  const d = new Date(dateStr);
   185→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   186→}
   187→
   188→function formatDay(dateStr: string): string {
   189→  const d = new Date(dateStr);
   190→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
   191→}

   211→export function FlashpointWargame() {
   212→  const { tenantId } = useDashboardStore();
   213→  const queryClient = useQueryClient();
   214→
   215→  // Wargame state
   216→  const [selectedScenarioId, setSelectedScenarioId] = useState<string | null>(null);
   217→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   218→  const [newTitle, setNewTitle] = useState('');
   219→  const [newDescription, setNewDescription] = useState('');
   220→  const [newRole, setNewRole] = useState('RED_TEAM');
   221→  const [newParams, setNewParams] = useState('{\n  \n}');
   222→  const [completeScore, setCompleteScore] = useState('');
   223→
   224→  /* ---- Data fetching ---- */
   225→  const { data, isLoading, refetch } = useQuery<FlashpointData>({
   226→    queryKey: ['flashpoint', tenantId],
   227→    queryFn: () =>
   228→      fetch(`/api/flashpoint?tenantId=${tenantId}`).then((r) => r.json()),
   229→    refetchInterval: 20000,
   230→  });
   231→
   232→  const mutation = useMutation({
   233→    mutationFn: (body: any) =>
   234→      fetch('/api/flashpoint?tenantId=' + tenantId, {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify(body),
   238→      }).then((r) => r.json()),
   239→    onSuccess: () => {
   240→      queryClient.invalidateQueries({ queryKey: ['flashpoint', tenantId] });
   241→    },
   242→    onError: () => {
   243→      toast.error('Flashpoint operation failed. Please try again.');
   244→    },
   245→  });
   246→
   247→  /* ---- Derived data ---- */
   248→  const stats = data?.stats;
   249→  const forecasts = data?.forecasts ?? [];
   250→  const scenarios = data?.scenarios ?? [];
   251→  const heatmapData = data?.heatmapData ?? [];
   252→  const topFactors = data?.topContributingFactors ?? [];
   253→
   254→  // Unique states sorted alphabetically
   255→  const uniqueStates = useMemo(() => {
   256→    const set = new Set(heatmapData.map((d) => d.state));
   257→    return Array.from(set).sort();
   258→  }, [heatmapData]);
   259→
   260→  // Unique dates sorted
   261→  const uniqueDates = useMemo(() => {
   262→    const set = new Set(heatmapData.map((d) => d.date));
   263→    return Array.from(set).sort();
   264→  }, [heatmapData]);
   265→
   266→  // Build lookup map for heatmap cells
   267→  const heatmapMap = useMemo(() => {
   268→    const map = new Map<string, (typeof heatmapData)[number]>();
   269→    for (const d of heatmapData) {
   270→      map.set(`${d.state}|${d.date}`, d);
   271→    }
   272→    return map;
   273→  }, [heatmapData]);
   274→
   275→  // State risk averages for bar chart
   276→  const stateRisks = useMemo(() => {
   277→    const map = new Map<string, { overall: number; level: string; count: number }>();
   278→    for (const f of forecasts) {
   279→      const existing = map.get(f.state);
   280→      if (existing) {
   281→        existing.overall = (existing.overall * existing.count + f.riskScores.overall) / (existing.count + 1);
   282→        existing.count += 1;
   283→        // Upgrade risk level if higher
   284→        const order = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
   285→        if (order.indexOf(f.riskLevel) > order.indexOf(existing.level)) {
   286→          existing.level = f.riskLevel;
   287→        }
   288→      } else {
   289→        map.set(f.state, { overall: f.riskScores.overall, level: f.riskLevel, count: 1 });
   290→      }
   291→    }
   292→    return Array.from(map.entries())
   293→      .map(([state, v]) => ({ state, ...v }))
   294→      .sort((a, b) => b.overall - a.overall);
   295→  }, [forecasts]);
   296→
   297→  // Top factor
   298→  const topFactor = topFactors.length > 0 ? topFactors[0].factor : '—';
   299→
   300→  // Selected scenario
   301→  const selectedScenario = scenarios.find((s) => s.id === selectedScenarioId) ?? null;
   302→
   303→  // Max factor count for bar scaling
   304→  const maxFactorCount = Math.max(...topFactors.map((f) => f.count), 1);
   305→
   306→  /* ---- Handlers ---- */
   307→  const handleCreateScenario = () => {
   308→    if (!newTitle.trim()) {
   309→      toast.error('Title is required');
   310→      return;
   311→    }
   312→    let parsedParams: any = {};
   313→    try {
   314→      parsedParams = newParams.trim() ? JSON.parse(newParams) : {};
   315→    } catch {
   316→      toast.error('Invalid JSON in parameters');
   317→      return;
   318→    }
   319→    mutation.mutate(
   320→      { action: 'RUN_WARGAME', title: newTitle, description: newDescription, playerRole: newRole, parameters: parsedParams },
   321→      {
   322→        onSuccess: () => {
   323→          toast.success('Scenario created successfully');
   324→          setCreateDialogOpen(false);
   325→          setNewTitle('');
   326→          setNewDescription('');
   327→          setNewRole('RED_TEAM');
   328→          setNewParams('{\n  \n}');
   329→        },
   330→        onError: () => toast.error('Failed to create scenario'),
   331→      },
   332→    );
   333→  };
   334→
   335→  const handleCompleteScenario = () => {
   336→    if (!selectedScenarioId || !completeScore) {
   337→      toast.error('Score is required');
   338→      return;
   339→    }
   340→    const score = parseFloat(completeScore);
   341→    if (isNaN(score) || score < 0 || score > 100) {
   342→      toast.error('Score must be between 0 and 100');
   343→      return;
   344→    }
   345→    mutation.mutate(
   346→      { action: 'COMPLETE_WARGAME', scenarioId: selectedScenarioId, score },
   347→      {
   348→        onSuccess: () => {
   349→          toast.success('Scenario completed');
   350→          setCompleteScore('');
   351→          setSelectedScenarioId(null);
   352→        },
   353→        onError: () => toast.error('Failed to complete scenario'),
   354→      },
   355→    );
   356→  };
   357→
   358→  /* ---- Loading ---- */
   359→  if (isLoading) {
   360→    return (
   361→      <div className="h-full flex flex-col items-center justify-center gap-3">
   362→        <Loader2 className="size-8 text-emerald-500 animate-spin" />
   363→        <p className="text-sm text-muted-foreground">Loading flashpoint data…</p>
   364→      </div>
   365→    );
   366→  }
   367→
   368→  return (
   369→    <div className="h-full flex flex-col gap-4">
   370→      <Tabs defaultValue="forecast" className="h-full flex flex-col gap-3">
   371→        <TabsList>
   372→          <TabsTrigger value="forecast" className="gap-1.5">
   373→            <Flame className="size-4" />
   374→            Risk Forecast
   375→          </TabsTrigger>
   376→          <TabsTrigger value="wargame" className="gap-1.5">
   377→            <Swords className="size-4" />
   378→            Wargame Simulator
   379→          </TabsTrigger>
   380→        </TabsList>
   381→
   382→        {/* ================================================================ */}
   383→        {/* TAB 1 — Risk Forecast                                           */}
   384→        {/* ================================================================ */}
   385→        <TabsContent value="forecast" className="flex-1 min-h-0 flex flex-col">
   386→          <motion.div
   387→            variants={staggerContainer}
   388→            initial="initial"
   389→            animate="animate"
   390→            className="flex flex-col gap-4 flex-1 min-h-0"
   391→          >
   392→            {/* --- Stats Row --- */}
   393→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   394→              <motion.div variants={fadeIn}>
   395→                <Card className="h-full">
   396→                  <CardContent className="p-4 flex flex-col gap-1">
   397→                    <div className="flex items-center gap-2 text-muted-foreground">
   398→                      <Activity className="size-4" />
   399→                      <span className="text-xs font-medium">Total Forecasts</span>
   400→                    </div>
   401→                    <p className="text-2xl font-bold">{stats?.totalForecasts ?? 0}</p>
   402→                  </CardContent>
   403→                </Card>
   404→              </motion.div>
   405→
   406→              <motion.div variants={fadeIn}>
   407→                <Card className="h-full border-red-500/20">
   408→                  <CardContent className="p-4 flex flex-col gap-1">
   409→                    <div className="flex items-center gap-2 text-red-400">
   410→                      <AlertTriangle className="size-4" />
   411→                      <span className="text-xs font-medium">High Risk States</span>
   412→                    </div>
   413→                    <p className="text-2xl font-bold text-red-400">{stats?.highRiskStates?.length ?? 0}</p>
   414→                  </CardContent>
   415→                </Card>
   416→              </motion.div>
   417→
   418→              <motion.div variants={fadeIn}>
   419→                <Card className="h-full border-rose-500/20">
   420→                  <CardContent className="p-4 flex flex-col gap-1">
   421→                    <div className="flex items-center gap-2 text-rose-400">
   422→                      <ShieldAlert className="size-4" />
   423→                      <span className="text-xs font-medium">Critical</span>
   424→                    </div>
   425→                    <p className="text-2xl font-bold text-rose-400">{stats?.byRiskLevel?.CRITICAL ?? 0}</p>
   426→                  </CardContent>
   427→                </Card>
   428→              </motion.div>
   429→
   430→              <motion.div variants={fadeIn}>
   431→                <Card className="h-full">
   432→                  <CardContent className="p-4 flex flex-col gap-1">
   433→                    <div className="flex items-center gap-2 text-cyan-400">
   434→                      <TrendingUp className="size-4" />
   435→                      <span className="text-xs font-medium">Avg Confidence</span>
   436→                    </div>
   437→                    <p className="text-2xl font-bold text-cyan-400">
   438→                      {stats?.avgConfidence != null ? `${(stats.avgConfidence * 100).toFixed(0)}%` : '—'}
   439→                    </p>
   440→                  </CardContent>
   441→                </Card>
   442→              </motion.div>
   443→
   444→              <motion.div variants={fadeIn}>
   445→                <Card className="h-full">
   446→                  <CardContent className="p-4 flex flex-col gap-1">
   447→                    <div className="flex items-center gap-2 text-amber-400">
   448→                      <Zap className="size-4" />
   449→                      <span className="text-xs font-medium">Top Factor</span>
   450→                    </div>
   451→                    <p className="text-sm font-semibold truncate">{topFactor}</p>
   452→                  </CardContent>
   453→                </Card>
   454→              </motion.div>
   455→            </div>
   456→
   457→            {/* --- Main Two Column Layout --- */}
   458→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 min-h-0">
   459→              {/* LEFT: 7-Day Heatmap */}
   460→              <motion.div variants={fadeIn}>
   461→                <Card className="h-full flex flex-col">
   462→                  <CardContent className="p-4 flex flex-col gap-3 flex-1 min-h-0">
   463→                    <div className="flex items-center justify-between">
   464→                      <h3 className="text-sm font-semibold flex items-center gap-2">
   465→                        <CalendarDays className="size-4 text-emerald-400" />
   466→                        7-Day Risk Heatmap
   467→                      </h3>
   468→                      <Button variant="ghost" size="sm" onClick={() => refetch()} className="h-7 px-2 text-xs">
   469→                        <Crosshair className="size-3 mr-1" />
   470→                        Refresh
   471→                      </Button>
   472→                    </div>
   473→
   474→                    {uniqueStates.length === 0 || uniqueDates.length === 0 ? (
   475→                      <div className="flex-1 flex items-center justify-center text-muted-foreground text-sm">
   476→                        No forecast data available
   477→                      </div>
   478→                    ) : (
   479→                      <ScrollArea className="flex-1">
   480→                        <div className="min-w-fit">
   481→                          {/* Header row: empty + dates */}
   482→                          <div
   483→                            className="grid gap-[2px] mb-[2px]"
   484→                            style={{
   485→                              gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   486→                            }}
   487→                          >
   488→                            <div className="text-[10px] text-muted-foreground font-medium px-1" />
   489→                            {uniqueDates.map((date) => (
   490→                              <div
   491→                                key={date}
   492→                                className="text-[10px] text-muted-foreground font-medium text-center truncate px-1"
   493→                              >
   494→                                {formatDay(date)}
   495→                              </div>
   496→                            ))}
   497→                          </div>
   498→
   499→                          {/* Data rows: state + cells */}
   500→                          {uniqueStates.map((state) => (
   501→                            <div
   502→                              key={state}
   503→                              className="grid gap-[2px] mb-[2px]"
   504→                              style={{
   505→                                gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   506→                              }}
   507→                            >
   508→                              <div className="flex items-center text-xs font-medium truncate px-1 text-foreground/80">
   509→                                {state}
   510→                              </div>
   511→                              {uniqueDates.map((date) => {
   512→                                const cell = heatmapMap.get(`${state}|${date}`);
   513→                                const colors = cell ? riskCellColor(cell.riskLevel) : { bg: 'rgba(255,255,255,0.05)', text: '#666' };
   514→                                return (
   515→                                  <TooltipProvider key={`${state}|${date}`}>
   516→                                    <Tooltip>
   517→                                      <TooltipTrigger asChild>
   518→                                        <div
   519→                                          className="flex items-center justify-center rounded-sm text-[11px] font-semibold cursor-default transition-transform hover:scale-110 h-8"
   520→                                          style={{
   521→                                            backgroundColor: colors.bg,
   522→                                            color: colors.text,
   523→                                          }}
   524→                                        >
   525→                                          {cell ? cell.overall.toFixed(1) : '—'}
   526→                                        </div>
   527→                                      </TooltipTrigger>
   528→                                      {cell && (
   529→                                        <TooltipContent side="top" className="bg-popover text-popover-foreground border">
   530→                                          <div className="text-xs space-y-1 min-w-[140px]">
   531→                                            <div className="font-semibold">{state} — {formatDate(date)}</div>
   532→                                            <Separator className="my-1" />
   533→                                            <div className="flex justify-between">
   534→                                              <span className="text-muted-foreground">Violence</span>
   535→                                              <span className="font-mono">{cell.violence.toFixed(2)}</span>
   536→                                            </div>
   537→                                            <div className="flex justify-between">
   538→                                              <span className="text-muted-foreground">Intimidation</span>
   539→                                              <span className="font-mono">{cell.intimidation.toFixed(2)}</span>
   540→                                            </div>
   541→                                            <div className="flex justify-between">
   542→                                              <span className="text-muted-foreground">Logistics</span>
   543→                                              <span className="font-mono">{cell.logistics.toFixed(2)}</span>
   544→                                            </div>
   545→                                            <Separator className="my-1" />
   546→                                            <div className="flex justify-between font-semibold">
   547→                                              <span>Overall</span>
   548→                                              <span style={{ color: colors.text }}>{cell.overall.toFixed(2)}</span>
   549→                                            </div>
   550→                                            <div className="flex justify-between">
   551→                                              <span>Level</span>
   552→                                              <Badge className={cn('text-[10px] h-4', riskBadgeVariant(cell.riskLevel))}>
   553→                                                {cell.riskLevel}
   554→                                              </Badge>
   555→                                            </div>
   556→                                          </div>
   557→                                        </TooltipContent>
   558→                                      )}
   559→                                    </Tooltip>
   560→                                  </TooltipProvider>
   561→                                );
   562→                              })}
   563→                            </div>
   564→                          ))}
   565→                        </div>
   566→                      </ScrollArea>
   567→                    )}
