   122→function generateNigerianNumbers(count: number): string[] {
   123→  const prefixes = ['0703', '0803', '0806', '0809', '0810', '0811', '0812', '0813', '0814', '0815', '0816', '0817', '0818', '0902', '0903', '0905', '0906', '0907', '0908', '0909', '0913', '0915', '0916'];
   124→  const numbers: string[] = [];
   125→  for (let i = 0; i < count; i++) {
   126→    const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
   127→    const suffix = String(Math.floor(Math.random() * 10_000_000)).padStart(7, '0');
   128→    numbers.push(`${prefix}${suffix}`);
   129→  }
   130→  return numbers;
   131→}
   132→
   133→function formatNumber(n: number): string {
   134→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   135→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   136→  return String(n);
   137→}
   138→
   139→function formatDate(d: string | null) {
   140→  if (!d) return '—';
   141→  return new Date(d).toLocaleDateString('en-NG', {
   142→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   143→  });
   144→}
   145→
   146→function segmentLabel(seg: string) {
   147→  return SEGMENT_OPTIONS.find(s => s.value === seg)?.label ?? seg;
   148→}
