   130→// ── Helpers ──────────────────────────────────────────────────────────
   131→function formatRelativeTime(d: string | Date | null): string {
   132→  if (!d) return 'Never';
   133→  const date = new Date(d);
   134→  const now = Date.now();
   135→  const diff = Math.floor((now - date.getTime()) / 60000);
   136→  if (diff < 1) return 'Just now';
   137→  if (diff < 60) return `${diff}m ago`;
   138→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   139→  return date.toLocaleDateString('en-NG', { day: '2-digit', month: 'short' });
   140→}
   141→
   142→function formatDeadlineCountdown(deadline: string): { text: string; isOverdue: boolean } {
   143→  const diff = new Date(deadline).getTime() - Date.now();
   144→  const mins = Math.floor(Math.abs(diff) / 60000);
   145→  if (diff > 0) {
   146→    return { text: `${mins}m left`, isOverdue: false };
   147→  }
   148→  return { text: `OVERDUE by ${mins}m`, isOverdue: true };
   149→}

   151→function batteryIcon(level: number | null) {
   152→  if (level === null) return <Battery className="h-3.5 w-3.5 text-muted-foreground" />;
   153→  if (level > 60) return <BatteryFull className="h-3.5 w-3.5 text-emerald" />;
   154→  if (level > 20) return <BatteryWarning className="h-3.5 w-3.5 text-amber" />;
   155→  return <BatteryLow className="h-3.5 w-3.5 text-rose" />;
   156→}
   157→
   158→function batteryColor(level: number | null) {
   159→  if (level === null) return 'text-muted-foreground';
   160→  if (level > 60) return 'text-emerald';
   161→  if (level > 20) return 'text-amber';
   162→  return 'text-rose';
   163→}

   165→function statusBadge(status: string) {
   166→  switch (status) {
   167→    case 'CHECKED_IN':
   168→      return <Badge className="border-emerald/30 bg-emerald/10 text-emerald text-[10px]">CHECKED IN</Badge>;
   169→    case 'CHECKED_OUT':
   170→      return <Badge className="border-cyan/30 bg-cyan/10 text-cyan text-[10px]">CHECKED OUT</Badge>;
   171→    case 'SOS_TRIGGERED':
   172→      return <Badge className="border-rose/30 bg-rose/10 text-rose text-[10px] glow-rose animate-pulse-dot">SOS</Badge>;
   173→    case 'EXPIRED':
   174→      return <Badge className="border-amber/30 bg-amber/10 text-amber text-[10px]">EXPIRED</Badge>;
   175→    default:
   176→      return <Badge variant="outline" className="text-[10px]">{status}</Badge>;
   177→  }
   178→}

   180→function escalationBadge(level: number) {
   181→  if (level === 0) return null;

   184→  return <Badge className="border-rose/30 bg-rose/10 text-rose text-[10px] animate-pulse-dot glow-rose">EMERGENCY</Badge>;

   706→function KpiCard({ label, value, icon, accent, delay }: {
   707→  label: string; value: number; icon: React.ReactNode; accent: 'emerald' | 'rose' | 'amber' | 'cyan'; delay: number;
   708→}) {
   709→  const glowClass = accent === 'rose' ? 'glow-rose' : accent === 'amber' ? 'glow-amber' : accent === 'emerald' ? 'glow-emerald' : '';
   710→  return (
   711→    <motion.div variants={item} transition={{ delay: delay * 0.05 }}>
   712→      <Card className={cn('bg-card/40 border-border rounded-xl', glowClass, value > 0 && (accent === 'rose' || accent === 'amber') ? '' : '')}>
   713→        <CardContent className="p-4 flex items-center gap-3">
   714→          <div className={cn(
   715→            'flex items-center justify-center h-9 w-9 rounded-lg shrink-0',
   716→            accent === 'emerald' && 'bg-emerald/10',
   717→            accent === 'rose' && 'bg-rose/10',
   718→            accent === 'amber' && 'bg-amber/10',
   719→            accent === 'cyan' && 'bg-cyan/10',
   720→          )}>
   721→            {icon}
   722→          </div>
   723→          <div>
   724→            <div className="text-xl font-bold leading-none">{value}</div>
   725→            <div className="text-[11px] text-muted-foreground mt-0.5">{label}</div>
   726→          </div>
   727→        </CardContent>
   728→      </Card>
   729→    </motion.div>
   730→  );
   731→}
