     1→'use client';

     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Input } from '@/components/ui/input';
     9→import { Textarea } from '@/components/ui/textarea';
    10→import { Separator } from '@/components/ui/separator';

    12→import {
    13→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    14→} from '@/components/ui/select';
    15→import { useDashboardStore } from '@/store/dashboard';
    16→import {
    17→  Send, MapPin, Camera, Mic, AlertTriangle, CheckCircle2,
    18→  Radio, Loader2, ShieldCheck, Vote, FileWarning, Plus, X,
    19→  TrendingUp, BarChart3, Users,
    20→} from 'lucide-react';

    32→  ],
    33→  STATE: [
    34→    { code: 'APC', name: 'APC', color: '#008751' },
    35→    { code: 'PDP', name: 'PDP', color: '#CE1126' },
    36→    { code: 'LP', name: 'LP', color: '#2196F3' },

    39→  ],
    40→  LOCAL: [

    43→    { code: 'LP', name: 'LP', color: '#2196F3' },
    44→  ],
    45→};
    46→
    47→const INCIDENT_TYPES = [
    48→  { value: 'VIOLENCE', label: 'Violence', severity: 'HIGH', color: 'rose' },
    49→  { value: 'INTIMIDATION', label: 'Voter Intimidation', severity: 'HIGH', color: 'rose' },
    50→  { value: 'BALLOT_STUFFING', label: 'Ballot Stuffing', severity: 'CRITICAL', color: 'rose' },
    51→  { value: 'BRIBERY', label: 'Bribery / Vote Buying', severity: 'HIGH', color: 'amber' },
    52→  { value: 'UNDERAGE_VOTING', label: 'Underage Voting', severity: 'MEDIUM', color: 'amber' },
    53→  { value: 'MULTIPLE_VOTING', label: 'Multiple Voting', severity: 'HIGH', color: 'amber' },
    54→  { value: 'SNATCHED_BALLOT', label: 'Snatched Ballot Box', severity: 'CRITICAL', color: 'rose' },
    55→  { value: 'IMPEDIMENT', label: 'Impediment to Voting', severity: 'MEDIUM', color: 'amber' },
    56→  { value: 'LOGISTICS', label: 'Logistics / Materials Issue', severity: 'LOW', color: 'cyan' },
    57→  { value: 'BVAS_FAILURE', label: 'BVAS / Tech Failure', severity: 'MEDIUM', color: 'amber' },
    58→  { value: 'OBSERVATION', label: 'General Observation', severity: 'LOW', color: 'muted' },
    59→];
    60→
    61→type Tab = 'results' | 'statistics' | 'incident';
    62→
    63→interface PartyVote {
    64→  party: string;
    65→  votes: string; // string for input, parsed on submit
    66→}
    67→
    68→export function SubmitReport() {
    69→  const { user, electionTier, tenantId } = useDashboardStore();
    70→  const queryClient = useQueryClient();
    71→
    72→  const [activeTab, setActiveTab] = useState<Tab>('results');
    73→  const [selectedPU, setSelectedPU] = useState('');
    74→  const [submitting, setSubmitting] = useState(false);
    75→  const [submitted, setSubmitted] = useState(false);
    76→  const [submittedType, setSubmittedType] = useState('');
    77→
    78→  // ---- RESULTS STATE ----
    79→  const [accredited, setAccredited] = useState('');
    80→  const [validVotes, setValidVotes] = useState('');
    81→  const [rejectedBallots, setRejectedBallots] = useState('');
    82→  const [partyVotes, setPartyVotes] = useState<PartyVote[]>(
    83→    (PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL).map(p => ({ party: p.code, votes: '' }))
    84→  );
    85→
    86→  // ---- STATISTICS STATE ----
    87→  const [bvasUsed, setBvasUsed] = useState(true);
    88→  const [materialsOnTime, setMaterialsOnTime] = useState(true);
    89→  const [securityPresent, setSecurityPresent] = useState(true);
    90→  const [violenceOccurred, setViolenceOccurred] = useState(false);
    91→  const [statsNotes, setStatsNotes] = useState('');
    92→
    93→  // ---- INCIDENT STATE ----
    94→  const [incType, setIncType] = useState('');
    95→  const [incSeverity, setIncSeverity] = useState('MEDIUM');
    96→  const [incDescription, setIncDescription] = useState('');
    97→
    98→  // Fetch polling units for the agent (show first 10 as options)
    99→  const { data: puData } = useQuery({
   100→    queryKey: ['my-polling-units'],
   101→    queryFn: () => fetch(`/api/dashboard?tenantId=${tenantId}`).then(r => r.json()).then(d => d.pollingUnits?.slice(0, 15) || []),
   102→  });
   103→
   104→  // Fetch agent's report counts for live stats
   105→  const { data: reportCounts } = useQuery({
   106→    queryKey: ['my-report-counts', user?.id],
   107→    queryFn: () => fetch(`/api/reports?reporterId=${user!.id}`).then(r => r.json()).then(d => d.counts),
   108→    enabled: !!user?.id,
   109→    refetchInterval: 10000,
   110→  });
   111→
   112→  const pollingUnits = puData || [];
   113→  const counts = reportCounts || { totalResults: 0, totalIncidents: 0, resultsToday: 0, incidentsToday: 0 };
   114→
   115→  const parties = PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL;
   116→
   117→  // Add a party vote row
   118→  const addPartyRow = () => {
   119→    const existingCodes = partyVotes.map(p => p.party);
   120→    const missing = parties.find(p => !existingCodes.includes(p.code));
   121→    if (missing) {
   122→      setPartyVotes(prev => [...prev, { party: missing.code, votes: '' }]);
   123→    }
   124→  };
   125→
   126→  const removePartyRow = (idx: number) => {
   127→    if (partyVotes.length > 2) {
   128→      setPartyVotes(prev => prev.filter((_, i) => i !== idx));
   129→    }
   130→  };
   131→
   132→  const updatePartyVote = (idx: number, field: 'party' | 'votes', value: string) => {
   133→    setPartyVotes(prev => prev.map((p, i) => i === idx ? { ...p, [field]: value } : p));
   134→  };
   135→
   136→  // Auto-calculate totals
   137→  const totalVotesCast = partyVotes.reduce((s, p) => s + (parseInt(p.votes) || 0), 0);
   138→  const accreditedNum = parseInt(accredited) || 0;
   139→  const validVotesNum = parseInt(validVotes) || 0;
   140→  const rejectedNum = parseInt(rejectedBallots) || 0;
   141→
   142→  // ---- SUBMIT HANDLERS ----
   143→
   144→  const resultMutation = useMutation({
   145→    mutationFn: (body: Record<string, unknown>) =>
   146→      fetch('/api/results', {
   147→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   148→      }).then(r => r.json()),
   149→    onSuccess: (data) => {
   150→      if (data.error) { toast.error(data.error); return; }
   151→      toast.success('Election results submitted successfully!');
   152→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   153→      queryClient.invalidateQueries({ queryKey: ['situation-room'] });
   154→      queryClient.invalidateQueries({ queryKey: ['my-reports'] });
   155→      queryClient.invalidateQueries({ queryKey: ['my-report-counts'] });
   156→      setSubmitted(true); setSubmittedType('results');
   157→      setTimeout(() => setSubmitted(false), 4000);
   158→    },
   159→    onError: () => toast.error('Failed to submit results'),
   160→  });
   161→
   162→  const incidentMutation = useMutation({
   163→    mutationFn: (body: Record<string, unknown>) =>
   164→      fetch('/api/incidents', {
   165→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   166→      }).then(r => r.json()),
   167→    onSuccess: (data) => {
   168→      if (data.error) { toast.error(data.error); return; }
   169→      toast.success('Incident report submitted!');
   170→      if (data.incident?.gpsAnomaly) {
   171→        toast.warning('GPS anomaly detected — report quarantined for review');
   172→      }
   173→      queryClient.invalidateQueries({ queryKey: ['incidents'] });
   174→      queryClient.invalidateQueries({ queryKey: ['alerts'] });
   175→      queryClient.invalidateQueries({ queryKey: ['my-reports'] });
   176→      queryClient.invalidateQueries({ queryKey: ['my-report-counts'] });
   177→      setSubmitted(true); setSubmittedType('incident');
   178→      setIncDescription(''); setIncType('');
   179→      setTimeout(() => setSubmitted(false), 4000);
   180→    },
   181→    onError: () => toast.error('Failed to submit incident'),
   182→  });
   183→
   184→  const handleSubmitResults = () => {
   185→    if (!user?.id || !selectedPU) {
   186→      toast.error('Please select a polling unit');
   187→      return;
   188→    }
   189→    if (partyVotes.every(p => !p.votes)) {
   190→      toast.error('Enter at least one party vote count');
   191→      return;
   192→    }
   193→
   194→    const parsedPartyResults = partyVotes
   195→      .filter(p => p.votes)
   196→      .map(p => {
   197→        const partyInfo = parties.find(pt => pt.code === p.party);
   198→        return { party: p.party, name: partyInfo?.name || p.party, votes: parseInt(p.votes) || 0, color: partyInfo?.color || '#888' };
   199→      });
   200→
   201→    setSubmitting(true);
   202→    resultMutation.mutate({
   203→      reporterId: user.id,
   204→      pollingUnitId: selectedPU,
   205→      accreditedVoters: accreditedNum,
   206→      totalValidVotes: validVotesNum,
   207→      rejectedBallots: rejectedNum,
   208→      totalVotesCast,
   209→      partyResults: parsedPartyResults,
   210→      bvasUsed, materialsArrivedOnTime: materialsOnTime,
   211→      securityPresent, violenceOccurred, notes: statsNotes,
   212→    }, { onSettled: () => setSubmitting(false) });
   213→  };
   214→
   215→  const handleSubmitIncident = () => {
   216→    if (!user?.id) return;
   217→    if (!incType) { toast.error('Select an incident type'); return; }
   218→    if (!incDescription.trim()) { toast.error('Describe the incident'); return; }
   219→
   220→    const typeConfig = INCIDENT_TYPES.find(t => t.value === incType);
   221→    setSubmitting(true);
   222→    incidentMutation.mutate({
   223→      reporterId: user.id,
   224→      pollingUnitId: selectedPU || undefined,
   225→      type: incType,
   226→      severity: typeConfig?.severity || incSeverity,
   227→      description: incDescription.trim(),
   228→    }, { onSettled: () => setSubmitting(false) });
   229→  };
   230→
   231→  const selectedPUData = pollingUnits.find(p => p.id === selectedPU);
   232→
   233→  return (
   234→    <div className="h-full overflow-y-auto p-4 sm:p-6">
   235→      <div className="max-w-2xl mx-auto space-y-4">
   236→        {/* Header */}
   237→        <div>
   238→          <h2 className="text-lg font-semibold flex items-center gap-2">
   239→            <Send className="h-5 w-5 text-emerald" />
   240→            Field Report
   241→          </h2>
   242→          <p className="text-sm text-muted-foreground mt-0.5">
   243→            Submit election results, polling unit statistics, or report incidents and infractions.
   244→          </p>
   245→        </div>
   246→
   247→        {/* Agent info + PU selector */}
   248→        <Card className="border-emerald/20 bg-emerald/5">
   249→          <CardContent className="p-3.5 flex items-center gap-3">
   250→            <div className="w-10 h-10 rounded-full bg-emerald/20 flex items-center justify-center text-sm font-bold text-emerald">
   251→              {user?.name.split(' ').map(n => n[0]).join('')}
   252→            </div>
   253→            <div className="flex-1 min-w-0">
   254→              <p className="text-sm font-medium">{user?.name}</p>
   255→              <div className="flex items-center gap-1.5 mt-0.5">
   256→                <ShieldCheck className="h-3 w-3 text-emerald" />
   257→                <span className="text-[10px] text-muted-foreground">In-App Capture &middot; C2PA enabled</span>
   258→              </div>
   259→            </div>
   260→          </CardContent>
   261→        </Card>
   262→
   263→        {/* Polling Unit Selector */}
   264→        <div className="space-y-1.5">
   265→          <label className="text-xs font-medium flex items-center gap-1.5">
   266→            <MapPin className="h-3.5 w-3.5 text-emerald" />
   267→            Select Polling Unit
   268→          </label>
   269→          <Select value={selectedPU} onValueChange={setSelectedPU}>
   270→            <SelectTrigger className="h-10">
   271→              <SelectValue placeholder="Choose your assigned polling unit..." />
   272→            </SelectTrigger>
   273→            <SelectContent>
   274→              {pollingUnits.map(pu => (
   275→                <SelectItem key={pu.id} value={pu.id}>
   276→                  <span className="flex items-center gap-2">
   277→                    <span className="text-muted-foreground text-[10px]">{pu.code}</span>
   278→                    <span>{pu.name}</span>
   279→                    <span className="text-muted-foreground text-[10px]">{pu.state}/{pu.lga}</span>
   280→                    {pu.status === 'FLAGGED' && <Badge variant="outline" className="text-[9px] h-4 border-amber/30 text-amber ml-1">FLAGGED</Badge>}
   281→                  </span>
   282→                </SelectItem>
   283→              ))}
   284→            </SelectContent>
   285→          </Select>
   286→          {selectedPUData && (
   287→            <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   288→              <Badge variant="outline" className="text-[10px] h-5 border-emerald/30 text-emerald">
   289→                <span className="w-1.5 h-1.5 rounded-full bg-emerald mr-1" />IN GEOFENCE
   290→              </Badge>
   291→              <span>{selectedPUData.registered} registered voters</span>
   292→              <span>{selectedPUData.state} / {selectedPUData.lga}</span>
   293→            </div>
   294→          )}
   295→        </div>
   296→
   297→        {/* Tab selector */}
   298→        <div className="grid grid-cols-3 gap-1 p-1 rounded-lg bg-card/60 border border-border">
   299→          {[
   300→            { id: 'results' as Tab, label: 'Election Results', icon: <Vote className="h-4 w-4" /> },
   301→            { id: 'statistics' as Tab, label: 'Statistics', icon: <BarChart3 className="h-4 w-4" /> },
   302→            { id: 'incident' as Tab, label: 'Incident / Infraction', icon: <FileWarning className="h-4 w-4" /> },
   303→          ].map(tab => (
   304→            <button
   305→              key={tab.id}
   306→              onClick={() => setActiveTab(tab.id)}
   307→              className={cn(
   308→                'flex items-center justify-center gap-1.5 px-3 py-2.5 rounded-md text-xs font-medium transition-all',
   309→                activeTab === tab.id
   310→                  ? 'bg-emerald/15 text-emerald shadow-sm'
   311→                  : 'text-muted-foreground hover:text-foreground hover:bg-card'
   312→              )}
   313→            >
   314→              {tab.icon}
   315→              <span className="hidden sm:inline">{tab.label}</span>
   316→            </button>
   317→          ))}
   318→        </div>
   319→
   320→        {/* Tab content */}
   321→        <AnimatePresence mode="wait">
   322→          <motion.div
   323→            key={activeTab}
   324→            initial={{ opacity: 0, y: 4 }}
   325→            animate={{ opacity: 1, y: 0 }}
   326→            exit={{ opacity: 0, y: -4 }}
   327→            transition={{ duration: 0.15 }}
   328→          >
   329→            {/* ===== ELECTION RESULTS TAB ===== */}
   330→            {activeTab === 'results' && (
   331→              <Card>
   332→                <CardContent className="p-4 space-y-4">
   333→                  <div>
   334→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   335→                      <Vote className="h-4 w-4 text-emerald" />
   336→                      Election Results Form
   337→                    </h3>
   338→                    <p className="text-[11px] text-muted-foreground mt-0.5">
   339→                      Enter the official vote counts from Form EC8A at your polling unit.
   340→                    </p>
   341→                  </div>
   342→
   343→                  {/* Summary fields */}
   344→                  <div className="grid grid-cols-2 gap-3">
   345→                    <div className="space-y-1.5">
   346→                      <label className="text-[11px] font-medium text-muted-foreground">Accredited Voters</label>
   347→                      <Input
   348→                        type="number" min="0"
   349→                        placeholder="0"
   350→                        value={accredited}
   351→                        onChange={(e) => setAccredited(e.target.value)}
   352→                        className="h-10 text-sm tabular-nums"
   353→                      />
   354→                    </div>
   355→                    <div className="space-y-1.5">
   356→                      <label className="text-[11px] font-medium text-muted-foreground">Rejected Ballots</label>
   357→                      <Input
   358→                        type="number" min="0"
   359→                        placeholder="0"
   360→                        value={rejectedBallots}
   361→                        onChange={(e) => setRejectedBallots(e.target.value)}
   362→                        className="h-10 text-sm tabular-nums"
   363→                      />
   364→                    </div>
   365→                  </div>
   366→
   367→                  <Separator />
   368→
   369→                  {/* Party vote entries */}
   370→                  <div className="space-y-2">
   371→                    <div className="flex items-center justify-between">
   372→                      <label className="text-xs font-medium">Vote Count by Party</label>
   373→                      <Button variant="ghost" size="sm" className="h-6 text-[10px] gap-1" onClick={addPartyRow}>
   374→                        <Plus className="h-3 w-3" /> Add Party
   375→                      </Button>
   376→                    </div>
   377→
   378→                    <div className="space-y-2">
   379→                      {partyVotes.map((pv, idx) => {
   380→                        const partyInfo = parties.find(p => p.code === pv.party);
   381→                        return (
   382→                          <div key={idx} className="flex items-center gap-2">
   383→                            <div
   384→                              className="w-3 h-8 rounded-sm shrink-0"
   385→                              style={{ backgroundColor: partyInfo?.color || '#555' }}
   386→                            />
   387→                            <div className="flex-1 min-w-0">
   388→                              <Select value={pv.party} onValueChange={(v) => updatePartyVote(idx, 'party', v)}>
   389→                                <SelectTrigger className="h-8 text-xs">
   390→                                  <SelectValue />
   391→                                </SelectTrigger>
   392→                                <SelectContent>
   393→                                  {parties.map(p => (
   394→                                    <SelectItem key={p.code} value={p.code}>
   395→                                      <span className="flex items-center gap-2">
   396→                                        <span className="w-2 h-2 rounded-sm" style={{ backgroundColor: p.color }} />
   397→                                        <span className="font-medium">{p.code}</span>
   398→                                        <span className="text-muted-foreground text-[10px] hidden sm:inline">{p.name}</span>
   399→                                      </span>
   400→                                    </SelectItem>
   401→                                  ))}
   402→                                </SelectContent>
   403→                              </Select>
   404→                            </div>
   405→                            <Input
   406→                              type="number" min="0" placeholder="0"
   407→                              value={pv.votes}
   408→                              onChange={(e) => updatePartyVote(idx, 'votes', e.target.value)}
   409→                              className="w-28 h-8 text-sm tabular-nums text-right"
   410→                            />
   411→                            {partyVotes.length > 2 && (
   412→                              <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-rose" onClick={() => removePartyRow(idx)}>
   413→                                <X className="h-3.5 w-3.5" />
   414→                              </Button>
   415→                            )}
   416→                          </div>
   417→                        );
   418→                      })}
   419→                    </div>
   420→                  </div>
   421→
   422→                  {/* Live totals */}
   423→                  <div className="rounded-lg bg-background border border-border p-3 space-y-2">
   424→                    <div className="flex items-center gap-2 text-xs font-medium">
   425→                      <TrendingUp className="h-3.5 w-3.5 text-emerald" />
   426→                      Live Totals
   427→                    </div>
   428→                    <div className="grid grid-cols-3 gap-2 text-center">
   429→                      <div>
   430→                        <p className="text-lg font-bold tabular-nums text-emerald">{totalVotesCast.toLocaleString()}</p>
   431→                        <p className="text-[10px] text-muted-foreground">Total Cast</p>
   432→                      </div>
   433→                      <div>
   434→                        <p className="text-lg font-bold tabular-nums text-cyan">{rejectedNum}</p>
   435→                        <p className="text-[10px] text-muted-foreground">Rejected</p>
   436→                      </div>
   437→                      <div>
   438→                        <p className="text-lg font-bold tabular-nums text-amber">{accreditedNum > 0 ? Math.round((totalVotesCast / accreditedNum) * 10000) / 100 : 0}%</p>
   439→                        <p className="text-[10px] text-muted-foreground">Turnout</p>
   440→                      </div>
   441→                    </div>
   442→                    {/* Simple bar chart */}
   443→                    {partyVotes.some(p => p.votes) && (
   444→                      <div className="flex h-3 rounded-full overflow-hidden bg-secondary mt-1">
   445→                        {partyVotes.filter(p => p.votes).map((pv) => {
   446→                          const partyInfo = parties.find(p => p.code === pv.party);
   447→                          const pct = totalVotesCast > 0 ? (parseInt(pv.votes) || 0) / totalVotesCast * 100 : 0;
   448→                          return (
   449→                            <div
   450→                              key={pv.party}
   451→                              className="h-full transition-all duration-500"
   452→                              style={{ width: `${pct}%`, backgroundColor: partyInfo?.color || '#555' }}
   453→                              title={`${pv.party}: ${pv.votes}`}
   454→                            />
   455→                          );
   456→                        })}
   457→                      </div>
   458→                    )}
   459→                  </div>
   460→
   461→                  <Button
   462→                    onClick={handleSubmitResults}
   463→                    disabled={submitting || !selectedPU || partyVotes.every(p => !p.votes)}
   464→                    className="w-full h-11 bg-emerald hover:bg-emerald/90 text-emerald-950 font-semibold gap-2"
   465→                  >
   466→                    {submitting && submittedType === 'results' ? (
   467→                      <><Loader2 className="h-4 w-4 animate-spin" /> Submitting...</>
   468→                    ) : submitted && submittedType === 'results' ? (
   469→                      <><CheckCircle2 className="h-4 w-4" /> Results Submitted!</>
   470→                    ) : (
