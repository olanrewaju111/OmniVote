     1→'use client';
     2→
     3→import { Card, CardContent } from '@/components/ui/card';
     4→import { Badge } from '@/components/ui/badge';
     5→import { Button } from '@/components/ui/button';
     6→import {
     7→  Dialog, DialogContent, DialogDescription, DialogFooter,
     8→  DialogHeader, DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import {
    11→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    12→} from '@/components/ui/select';
    13→import { Input } from '@/components/ui/input';
    14→import {
    15→  Building2, Users, Settings, Shield, Vote, Loader2, MapPin, Save,
    16→  RotateCcw, Plus, Trash2, UserPlus, Mail, ChevronRight, Globe,
    17→  Eye, UserCheck, Radio, AlertTriangle, Pencil,
    18→} from 'lucide-react';
    19→import { useState, useEffect, useCallback } from 'react';
    20→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
    21→import { cn } from '@/lib/utils';
    22→import { toast } from 'sonner';
    23→import { fetchJson } from '@/lib/api';
    24→import { useDashboardStore, type UserRole } from '@/store/dashboard';
    25→
    26→// ---- Shared types & constants ----
    27→const REGION_PRESETS = [
    28→  { label: 'Nigeria (full)', minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0 },
    29→  { label: 'Lagos', minLat: 6.3, maxLat: 6.7, minLng: 3.2, maxLng: 3.5 },
    30→  { label: 'Abuja FCT', minLat: 8.8, maxLat: 9.2, minLng: 7.2, maxLng: 7.7 },
    31→  { label: 'Kano', minLat: 11.8, maxLat: 12.4, minLng: 8.3, maxLng: 8.9 },
    32→  { label: 'Rivers', minLat: 4.5, maxLat: 5.2, minLng: 6.5, maxLng: 7.1 },
    33→  { label: 'Southeast', minLat: 4.8, maxLat: 7.0, minLng: 6.8, maxLng: 8.2 },
    34→  { label: 'Southwest', minLat: 6.0, maxLat: 9.0, minLng: 2.5, maxLng: 5.0 },
    35→  { label: 'Northwest', minLat: 10.0, maxLat: 14.0, minLng: 3.0, maxLng: 9.0 },
    36→  { label: 'North-Central', minLat: 7.0, maxLat: 10.5, minLng: 3.0, maxLng: 9.5 },
    37→  { label: 'South-South', minLat: 4.3, maxLat: 6.5, minLng: 5.0, maxLng: 8.5 },
    38→  { label: 'Northeast', minLat: 8.0, maxLat: 14.0, minLng: 10.0, maxLng: 15.0 },
    39→];
    40→
    41→interface MapBoundsData {
    42→  minLat: number; maxLat: number; minLng: number; maxLng: number; label: string;
    43→}
    44→
    45→const VALID_ROLES: UserRole[] = ['SUPER_ADMIN', 'TENANT_ADMIN', 'ANALYST', 'TRUST_SAFETY', 'FIELD_AGENT'];
    46→
    47→const ROLE_BADGE: Record<string, string> = {
    48→  SUPER_ADMIN: 'border-emerald/30 text-emerald bg-emerald/10',
    49→  TENANT_ADMIN: 'border-cyan/30 text-cyan bg-cyan/10',
    50→  ANALYST: 'border-amber/30 text-amber bg-amber/10',
    51→  TRUST_SAFETY: 'border-rose/30 text-rose bg-rose/10',
    52→  FIELD_AGENT: 'border-border text-muted-foreground bg-card',
    53→};
    54→
    55→const ROLE_ICONS: Record<string, React.ReactNode> = {
    56→  SUPER_ADMIN: <Shield className="h-3.5 w-3.5" />,
    57→  TENANT_ADMIN: <Users className="h-3.5 w-3.5" />,
    58→  ANALYST: <Eye className="h-3.5 w-3.5" />,
    59→  TRUST_SAFETY: <UserCheck className="h-3.5 w-3.5" />,
    60→  FIELD_AGENT: <Radio className="h-3.5 w-3.5" />,
    61→};
    62→
    63→// ---- Main Component ----
    64→export function TenantManagement() {
    65→  const { user, tenantId } = useDashboardStore();
    66→  const queryClient = useQueryClient();
    67→  const isSuperAdmin = user?.role === 'SUPER_ADMIN';
    68→  const isTenantAdmin = user?.role === 'TENANT_ADMIN';
    69→  const isAdmin = isSuperAdmin || isTenantAdmin;
    70→
    71→  // ===================== SUPER_ADMIN: Platform Tenants =====================
    72→  const { data: allTenants, isLoading: tenantsLoading } = useQuery({
    73→    queryKey: ['all-tenants'],
    74→    queryFn: async () => {
    75→      const d = await fetchJson<{ tenants?: unknown[] }>('/api/tenants');
    76→      return d.tenants || [];
    77→    },
    78→    enabled: isSuperAdmin,
    79→  });
    80→
    81→  // ===================== TENANT_ADMIN: Own tenant settings =====================
    82→  const { data: settings, isLoading: settingsLoading } = useQuery({
    83→    queryKey: ['tenant-settings', tenantId],
    84→    queryFn: async () => {
    85→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`);
    86→    },
    87→    enabled: !!tenantId && isAdmin,
    88→  });
    89→
    90→  // ===================== Common: Tenant Users =====================
    91→  const { data: tenantUsersData, isLoading: usersLoading } = useQuery({
    92→    queryKey: ['tenant-users', tenantId],
    93→    queryFn: async () => {
    94→      const d = await fetchJson<{ users?: unknown[] }>(`/api/tenants/users?tenantId=${tenantId}`);
    95→      return d.users || [];
    96→    },
    97→    enabled: !!tenantId && isAdmin,
    98→  });
    99→
   100→  const currentBounds: MapBoundsData | null = settings?.mapBounds || null;
   101→  const tenantUsers = tenantUsersData || [];
   102→
   103→  // ===================== State =====================
   104→  // Map config dialog
   105→  const [mapConfigOpen, setMapConfigOpen] = useState(false);
   106→  const [mapLabel, setMapLabel] = useState('');
   107→  const [minLat, setMinLat] = useState('');
   108→  const [maxLat, setMaxLat] = useState('');
   109→  const [minLng, setMinLng] = useState('');
   110→  const [maxLng, setMaxLng] = useState('');
   111→
   112→  // Create tenant dialog
   113→  const [createTenantOpen, setCreateTenantOpen] = useState(false);
   114→  const [newTenantName, setNewTenantName] = useState('');
   115→  const [newTenantSlug, setNewTenantSlug] = useState('');
   116→  const [newTenantColor, setNewTenantColor] = useState('#10b981');
   117→  const [newAdminName, setNewAdminName] = useState('');
   118→  const [newAdminEmail, setNewAdminEmail] = useState('');
   119→
   120→  // Add user dialog
   121→  const [addUserOpen, setAddUserOpen] = useState(false);
   122→  const [newUserName, setNewUserName] = useState('');
   123→  const [newUserEmail, setNewUserEmail] = useState('');
   124→  const [newUserRole, setNewUserRole] = useState<string>('FIELD_AGENT');
   125→
   126→  // User role change dialog
   127→  const [roleChangeOpen, setRoleChangeOpen] = useState(false);
   128→  const [roleChangeUser, setRoleChangeUser] = useState<{ id: string; name: string; role: string } | null>(null);
   129→  const [newRole, setNewRole] = useState('');
   130→
   131→  // Delete confirmation
   132→  const [deleteConfirm, setDeleteConfirm] = useState<{ type: 'tenant' | 'user'; item: { id: string; name: string } } | null>(null);
   133→
   134→  // Active sub-tab for SUPER_ADMIN
   135→  const [subTab, setSubTab] = useState<'tenants' | 'my-tenant' | 'map'>('tenants');
   136→
   137→  // ===================== Effects =====================
   138→  useEffect(() => {
   139→    if (mapConfigOpen && currentBounds) {
   140→      setMapLabel(currentBounds.label || '');
   141→      setMinLat(String(currentBounds.minLat));
   142→      setMaxLat(String(currentBounds.maxLat));
   143→      setMinLng(String(currentBounds.minLng));
   144→      setMaxLng(String(currentBounds.maxLng));
   145→    } else if (mapConfigOpen) {
   146→      setMapLabel('');
   147→      setMinLat(''); setMaxLat(''); setMinLng(''); setMaxLng('');
   148→    }
   149→  }, [mapConfigOpen, currentBounds]);
   150→
   151→  // Auto-derive slug from tenant name
   152→  useEffect(() => {
   153→    if (createTenantOpen && newTenantName && !newTenantSlug) {
   154→      setNewTenantSlug(newTenantName.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, ''));
   155→    }
   156→  }, [createTenantOpen, newTenantName, newTenantSlug]);
   157→
   158→  // ===================== Mutations =====================
   159→  // Save map bounds
   160→  const saveMapMutation = useMutation({
   161→    mutationFn: async (bounds: MapBoundsData) => {
   162→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`, {
   163→        method: 'PUT',
   164→        headers: { 'Content-Type': 'application/json' },
   165→        body: JSON.stringify({ mapBounds: bounds }),
   166→      });
   167→    },
   168→    onSuccess: () => {
   169→      queryClient.invalidateQueries({ queryKey: ['tenant-settings', tenantId] });
   170→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   171→      setMapConfigOpen(false);
   172→      toast.success('Map area saved. The map will update automatically.');
   173→    },
   174→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to save map configuration'),
   175→  });
   176→
   177→  // Create tenant
   178→  const createTenantMutation = useMutation({
   179→    mutationFn: (data: { name: string; slug: string; primaryColor: string; adminName: string; adminEmail: string }) =>
   180→      fetchJson('/api/tenants', {
   181→        method: 'POST',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify(data),
   184→      }),
   185→    onSuccess: (data) => {
   186→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   187→      setCreateTenantOpen(false);
   188→      setNewTenantName(''); setNewTenantSlug(''); setNewTenantColor('#10b981');
   189→      setNewAdminName(''); setNewAdminEmail('');
   190→      toast.success(`Tenant "${data.tenant.name}" created. Admin: ${data.admin.email}`);
   191→    },
   192→    onError: (err) => toast.error(err?.message || 'Failed to create tenant'),
   193→  });
   194→
   195→  // Delete tenant
   196→  const deleteTenantMutation = useMutation({
   197→    mutationFn: (id: string) =>
   198→      fetchJson(`/api/tenants?id=${id}`, { method: 'DELETE' }),
   199→    onSuccess: () => {
   200→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   201→      setDeleteConfirm(null);
   202→      toast.success('Tenant deleted successfully');
   203→    },
   204→    onError: (err) => toast.error(err?.message || 'Failed to delete tenant'),
   205→  });
   206→
   207→  // Toggle tenant active
   208→  const toggleTenantMutation = useMutation({
   209→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   210→      fetchJson('/api/tenants', {
   211→        method: 'PUT',
   212→        headers: { 'Content-Type': 'application/json' },
   213→        body: JSON.stringify({ id, isActive }),
   214→      }),
   215→    onSuccess: () => {
   216→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   217→      toast.success('Tenant status updated');
   218→    },
   219→    onError: (err) => toast.error(err?.message || 'Failed to update tenant'),
   220→  });
   221→
   222→  // Add user
   223→  const addUserMutation = useMutation({
   224→    mutationFn: (data: { tenantId: string; name: string; email: string; role: string }) =>
   225→      fetchJson('/api/tenants/users', {
   226→        method: 'POST',
   227→        headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify(data),
   229→      }),
   230→    onSuccess: () => {
   231→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   232→      setAddUserOpen(false);
   233→      setNewUserName(''); setNewUserEmail(''); setNewUserRole('FIELD_AGENT');
   234→      toast.success('User added successfully');
   235→    },
   236→    onError: (err) => toast.error(err?.message || 'Failed to add user'),
   237→  });
   238→
   239→  // Change user role
   240→  const changeRoleMutation = useMutation({
   241→    mutationFn: ({ id, role }: { id: string; role: string }) =>
   242→      fetchJson('/api/tenants/users', {
   243→        method: 'PATCH',
   244→        headers: { 'Content-Type': 'application/json' },
   245→        body: JSON.stringify({ id, role }),
   246→      }),
   247→    onSuccess: () => {
   248→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   249→      setRoleChangeOpen(false);
   250→      setRoleChangeUser(null);
   251→      toast.success('User role updated');
   252→    },
   253→    onError: (err) => toast.error(err?.message || 'Failed to update role'),
   254→  });
   255→
   256→  // Remove user
   257→  const deleteUserMutation = useMutation({
   258→    mutationFn: (id: string) =>
   259→      fetchJson(`/api/tenants/users?id=${id}`, { method: 'DELETE' }),
   260→    onSuccess: () => {
   261→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   262→      setDeleteConfirm(null);
   263→      toast.success('User removed');
   264→    },
   265→    onError: (err) => toast.error(err?.message || 'Failed to remove user'),
   266→  });
   267→
   268→  // ===================== Handlers =====================
   269→  const handleSaveMap = () => {
   270→    const minLatN = parseFloat(minLat);
   271→    const maxLatN = parseFloat(maxLat);
   272→    const minLngN = parseFloat(minLng);
   273→    const maxLngN = parseFloat(maxLng);
   274→    if ([minLatN, maxLatN, minLngN, maxLngN].some(isNaN)) { toast.error('All coordinates must be valid numbers'); return; }
   275→    if (minLatN >= maxLatN || minLngN >= maxLngN) { toast.error('minLat must be < maxLat, and minLng must be < maxLng'); return; }
   276→    saveMapMutation.mutate({ minLat: minLatN, maxLat: maxLatN, minLng: minLngN, maxLng: maxLngN, label: mapLabel || 'Custom Area' });
   277→  };
   278→
   279→  const handlePreset = (preset: typeof REGION_PRESETS[number]) => {
   280→    setMapLabel(preset.label);
   281→    setMinLat(String(preset.minLat));
   282→    setMaxLat(String(preset.maxLat));
   283→    setMinLng(String(preset.minLng));
   284→    setMaxLng(String(preset.maxLng));
   285→  };
   286→
   287→  const handleResetMap = () => {
   288→    saveMapMutation.mutate({ minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0, label: 'Nigeria' });
   289→  };
   290→
   291→  const handleCreateTenant = () => {
   292→    if (!newTenantName || !newTenantSlug || !newAdminName || !newAdminEmail) {
   293→      toast.error('All fields are required'); return;
   294→    }
   295→    createTenantMutation.mutate({
   296→      name: newTenantName, slug: newTenantSlug, primaryColor: newTenantColor,
   297→      adminName: newAdminName, adminEmail: newAdminEmail,
   298→    });
   299→  };
   300→
   301→  const handleAddUser = () => {
   302→    if (!newUserName || !newUserEmail || !newUserRole) { toast.error('All fields are required'); return; }
   303→    addUserMutation.mutate({ tenantId: tenantId!, name: newUserName, email: newUserEmail, role: newUserRole });
   304→  };
   305→
   306→  const handleRoleChange = () => {
   307→    if (!roleChangeUser || !newRole) return;
   308→    changeRoleMutation.mutate({ id: roleChangeUser.id, role: newRole });
   309→  };
   310→
   311→  // ===================== Restricted access =====================
   312→  if (!isAdmin) {
   313→    return (
   314→      <div className="h-full flex items-center justify-center p-6">
   315→        <div className="text-center space-y-2">
   316→          <Shield className="h-8 w-8 text-muted-foreground/30 mx-auto" />
   317→          <p className="text-sm text-muted-foreground">Access restricted to Super Admin and Tenant Admin</p>
   318→        </div>
   319→      </div>
   320→    );
   321→  }
   322→
   323→  const isLoading = isSuperAdmin ? tenantsLoading : settingsLoading;
   324→
   325→  // ===================== Render =====================
   326→  return (
   327→    <div className="h-full flex flex-col overflow-hidden">
   328→      {/* Header */}
   329→      <div className="px-4 sm:px-6 pt-4 pb-3 shrink-0">
   330→        <div className="flex items-center justify-between flex-wrap gap-3">
   331→          <div>
   332→            <h2 className="text-lg font-semibold flex items-center gap-2">
   333→              <Building2 className="h-5 w-5 text-emerald" />
   334→              {isSuperAdmin ? 'Platform Management' : 'Organization Settings'}
   335→            </h2>
   336→            <p className="text-sm text-muted-foreground mt-0.5">
   337→              {isSuperAdmin
   338→                ? 'Create and manage tenant organizations, their users, and configurations'
   339→                : 'Manage your organization settings, users, and map configuration'}
   340→            </p>
   341→          </div>
   342→          {isSuperAdmin && (
   343→            <Button
   344→              size="sm"
   345→              className="gap-1.5 text-xs bg-emerald hover:bg-emerald/90 text-emerald-950 h-8"
   346→              onClick={() => setCreateTenantOpen(true)}
   347→            >
   348→              <Plus className="h-3.5 w-3.5" /> New Tenant
   349→            </Button>
   350→          )}
   351→        </div>
   352→
   353→        {/* Sub-tabs for SUPER_ADMIN */}
   354→        {isSuperAdmin && (
   355→          <div className="flex items-center gap-1 mt-3 bg-muted/30 rounded-lg p-0.5 w-fit">
   356→            {([
   357→              { id: 'tenants' as const, label: 'All Tenants', icon: <Globe className="h-3.5 w-3.5" /> },
   358→              { id: 'my-tenant' as const, label: 'My Organization', icon: <Building2 className="h-3.5 w-3.5" /> },
   359→              { id: 'map' as const, label: 'Map Config', icon: <MapPin className="h-3.5 w-3.5" /> },
   360→            ]).map(tab => (
   361→              <button
   362→                key={tab.id}
   363→                onClick={() => setSubTab(tab.id)}
   364→                className={cn(
   365→                  'flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium transition-colors',
   366→                  subTab === tab.id ? 'bg-card shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   367→                )}
   368→              >
   369→                {tab.icon} {tab.label}
   370→              </button>
   371→            ))}
   372→          </div>
   373→        )}
   374→      </div>
   375→
   376→      {/* Scrollable content */}
   377→      <div className="flex-1 overflow-y-auto px-4 sm:px-6 pb-6 space-y-4 min-h-0">
   378→
   379→        {isLoading ? (
   380→          <div className="flex-1 flex items-center justify-center py-20">
   381→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   382→          </div>
   383→        ) : isSuperAdmin ? (
   384→          /* ========== SUPER_ADMIN VIEWS ========== */
   385→          <>
   386→            {subTab === 'tenants' && (
   387→              <SuperAdminTenantsView
   388→                tenants={allTenants || []}
   389→                onToggle={(id, isActive) => toggleTenantMutation.mutate({ id, isActive })}
   390→                onDelete={(item) => setDeleteConfirm({ type: 'tenant', item })}
   391→                onManageUsers={() => setSubTab('my-tenant')}
   392→              />
   393→            )}
   394→            {subTab === 'my-tenant' && (
   395→              <TenantUsersView
   396→                settings={settings}
   397→                tenantUsers={tenantUsers}
   398→                userRole={user?.role || ''}
   399→                onOpenUserDialog={() => setAddUserOpen(true)}
   400→                onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   401→                onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   402→                onOpenMapConfig={() => { setSubTab('map'); setTimeout(() => setMapConfigOpen(true), 100); }}
   403→              />
   404→            )}
   405→            {subTab === 'map' && (
   406→              <MapConfigSection
   407→                currentBounds={currentBounds}
   408→                onConfigure={() => setMapConfigOpen(true)}
   409→              />
   410→            )}
   411→          </>
   412→        ) : (
   413→          /* ========== TENANT_ADMIN VIEW ========== */
   414→          <>
   415→            <TenantUsersView
   416→              settings={settings}
   417→              tenantUsers={tenantUsers}
   418→              userRole={user?.role || ''}
   419→              onOpenUserDialog={() => setAddUserOpen(true)}
   420→              onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   421→              onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   422→              onOpenMapConfig={() => setMapConfigOpen(true)}
   423→            />
   424→            <MapConfigSection
   425→              currentBounds={currentBounds}
   426→              onConfigure={() => setMapConfigOpen(true)}
   427→            />
   428→            <InfoNotices />
   429→          </>
   430→        )}
   431→      </div>
   432→
   433→      {/* ===== MAP CONFIG DIALOG ===== */}
   434→      <Dialog open={mapConfigOpen} onOpenChange={setMapConfigOpen}>
   435→        <DialogContent className="sm:max-w-lg">
   436→          <DialogHeader>
   437→            <DialogTitle className="flex items-center gap-2">
   438→              <MapPin className="h-5 w-5 text-cyan" />
   439→              Configure Map Area
   440→            </DialogTitle>
   441→            <DialogDescription>
   442→              Set the geographic bounding box for the Polling Unit Map. Use a preset or enter custom WGS84 coordinates.
   443→            </DialogDescription>
   444→          </DialogHeader>
   445→          <div className="space-y-4 py-2">
   446→            <div className="space-y-1.5">
   447→              <label className="text-xs font-medium text-muted-foreground">Area Label</label>
   448→              <Input placeholder="e.g. Lagos, Southeast, Kano Central" value={mapLabel} onChange={(e) => setMapLabel(e.target.value)} className="h-9 text-sm" />
   449→            </div>
   450→            <div className="space-y-1.5">
   451→              <label className="text-xs font-medium text-muted-foreground">Quick Preset</label>
   452→              <Select onValueChange={(v) => { const preset = REGION_PRESETS.find(p => p.label === v); if (preset) handlePreset(preset); }}>
   453→                <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select a region preset..." /></SelectTrigger>
   454→                <SelectContent>
   455→                  {REGION_PRESETS.map(p => (
   456→                    <SelectItem key={p.label} value={p.label} className="text-xs">
   457→                      {p.label}
   458→                      <span className="text-muted-foreground ml-2">({p.minLat}, {p.maxLat}, {p.minLng}, {p.maxLng})</span>
   459→                    </SelectItem>
   460→                  ))}
   461→                </SelectContent>
   462→              </Select>
   463→            </div>
   464→            <div className="grid grid-cols-2 gap-3">
   465→              <div className="space-y-1.5">
   466→                <label className="text-xs font-medium text-muted-foreground">South (minLat)</label>
   467→                <Input type="number" step="0.1" placeholder="4.0" value={minLat} onChange={(e) => setMinLat(e.target.value)} className="h-9 text-sm" />
