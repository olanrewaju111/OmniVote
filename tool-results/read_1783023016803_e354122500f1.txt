   149→
   150→function StatCard({
   151→  icon, label, value, color, sub,
   152→}: {
   153→  icon: React.ReactNode; label: string; value: string | number; color: string; sub?: string;
   154→}) {
   155→  const colorMap: Record<string, string> = {
   156→    emerald: 'text-emerald', cyan: 'text-cyan', amber: 'text-amber',
   157→    rose: 'text-rose', purple: 'text-purple', foreground: 'text-foreground',
   158→  };
   159→  return (
   160→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.25 }}>
   161→      <Card className="border-border bg-card/40">
   162→        <CardContent className="p-3 sm:p-4">
   163→          <div className="flex items-center gap-1.5 mb-1.5">
   164→            <span className={colorMap[color] || 'text-muted-foreground'}>{icon}</span>
   165→            <span className="text-[10px] text-muted-foreground uppercase tracking-wider">{label}</span>
   166→          </div>
   167→          <p className={cn('text-lg font-bold tabular-nums', colorMap[color] || 'text-foreground')}>{value}</p>
   168→          {sub && <p className="text-[10px] text-muted-foreground/60 mt-0.5">{sub}</p>}
   169→        </CardContent>
   170→      </Card>
   171→    </motion.div>
   172→  );
   173→}

   179→function ColorBar({ value, max = 100, colorClass }: { value: number; max?: number; colorClass: string }) {
   180→  const pct = Math.min((value / max) * 100, 100);
   181→  return (
   182→    <div className="h-2 bg-secondary rounded-full overflow-hidden w-full min-w-[80px]">
   183→      <motion.div
   184→        initial={{ width: 0 }}
   185→        animate={{ width: `${pct}%` }}
   186→        transition={{ duration: 0.6, ease: 'easeOut' }}
   187→        className={cn('h-full rounded-full', colorClass)}
   188→      />
   189→    </div>
   190→  );
   191→}

   235→    mutationFn: (body: Record<string, unknown>) =>

   237→        method: 'POST',
   238→        headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify(body),
   240→      }).then((r) => r.json()),

   264→  );

   268→    setHpName('');
   269→    setHpState('');
   270→    setHpLga('');
   271→    setHpTrapType('');

   273→  };
   274→
   275→  const handleAddResultRow = () => {
   276→    setHpExpectedResults((prev) => [...prev, { party: '', votes: '' }]);
   277→  };
   278→
   279→  const handleRemoveResultRow = (idx: number) => {
   280→    setHpExpectedResults((prev) => prev.filter((_, i) => i !== idx));
   281→  };
   282→
   283→  const handleResultChange = (idx: number, field: 'party' | 'votes', value: string) => {
   284→    setHpExpectedResults((prev) =>
   285→      prev.map((r, i) => (i === idx ? { ...r, [field]: value } : r)),
   286→    );
   287→  };
   288→
   289→  const handleSubmitHoneypot = () => {
   290→    if (!hpName.trim() || !hpState.trim() || !hpTrapType) {
   291→      toast.error('Please fill in name, state, and trap type');
   292→      return;
   293→    }
   294→    const validResults = hpExpectedResults
   295→      .filter((r) => r.party.trim() && r.votes.trim())
   296→      .map((r) => ({ party: r.party.trim(), votes: Number(r.votes) }));
   297→    mutation.mutate(
   298→      {
   299→        action: 'CREATE_HONEYPOT',
   300→        name: hpName.trim(),
   301→        state: hpState.trim(),
   302→        lga: hpLga.trim() || null,
   303→        trapType: hpTrapType,
   304→        expectedResults: validResults,
   305→      },
   306→      {
   307→        onSuccess: () => {
   308→          setHoneypotDialogOpen(false);
   309→          resetHoneypotForm();
   310→        },
   311→      },
   312→    );
   313→  };
   314→
   315→  // ── Accessibility dialog handlers ──
   316→  const resetAccessibilityForm = () => {
   317→    setArPollingUnitId('');
   318→    setArFeatures({
   319→      rampAccess: false, brailleBallots: false, signLanguage: false,
   320→      assistiveTech: false, accessibleToilet: false,
   321→    });
   322→    setArBarriers([]);
   323→    setArPwdServed('');
   324→    setArPwdTurnedAway('');
   325→    setArScore([50]);
   326→  };
   327→

   329→    setArBarriers((prev) =>
   330→      prev.includes(barrier) ? prev.filter((b) => b !== barrier) : [...prev, barrier],
   331→    );
   332→  };
   333→
   334→  const toggleFeature = (key: keyof typeof arFeatures) => {
   335→    setArFeatures((prev) => ({ ...prev, [key]: !prev[key] }));
   336→  };
   337→
   338→  const handleSubmitAccessibility = () => {
   339→    if (!arPollingUnitId.trim()) {
   340→      toast.error('Please enter a polling unit ID');
   341→      return;
   342→    }
   343→    mutation.mutate(
   344→      {
   345→        action: 'CREATE_ACCESSIBILITY_REPORT',
   346→        pollingUnitId: arPollingUnitId.trim(),
   347→        features: arFeatures,
   348→        barrierTypes: arBarriers,
   349→        pwdVotersServed: Number(arPwdServed) || 0,
   350→        pwdVotersTurnedAway: Number(arPwdTurnedAway) || 0,
   351→        overallScore: arScore[0],
   352→      },
   353→      {
   354→        onSuccess: () => {
   355→          setAccessibilityDialogOpen(false);
   356→          resetAccessibilityForm();
   357→        },
   358→      },
   359→    );
   360→  };
   361→
   362→  // ── Render ──
   363→  return (
   364→    <div className="h-full flex flex-col p-4 sm:p-6 gap-4 overflow-hidden">
   365→      {/* Header */}
   366→      <div className="flex items-center justify-between flex-wrap gap-3 shrink-0">
   367→        <div>
   368→          <h2 className="text-lg font-semibold flex items-center gap-2">
   369→            <Radar className="h-5 w-5 text-purple" />
   370→            Honeypot Stations, Biometrics & PWD Tracking
   371→          </h2>
   372→          <p className="text-sm text-muted-foreground mt-0.5">
   373→            Decoy traps, agent biometric verification, and accessibility compliance
   374→          </p>
   375→        </div>
   376→        <Button
   377→          variant="outline"
   378→          size="sm"
   379→          onClick={() => refetch()}
   380→          className="gap-1.5 text-xs"
   381→        >
   382→          <Activity className="h-3.5 w-3.5" />
   383→          Refresh
   384→        </Button>
   385→      </div>
   386→
   387→      {isLoading ? (
   388→        <div className="flex-1 flex items-center justify-center">
   389→          <Loader2 className="h-6 w-6 animate-spin text-purple" />
   390→        </div>

   395→              <ShieldAlert className="h-3.5 w-3.5" />
