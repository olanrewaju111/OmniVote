    99→function formatDate(d: string | null | undefined): string {
   100→  if (!d) return '—';
   101→  return new Date(d).toLocaleDateString('en-NG', {
   102→    day: '2-digit',
   103→    month: 'short',
   104→    hour: '2-digit',
   105→    minute: '2-digit',
   106→  });
   107→}

   199→/* ------------------------------------------------------------------ */
   200→/*  Circular Gauge                                                     */
   201→/* ------------------------------------------------------------------ */
   202→
   203→function SecurityGauge({ score }: { score: number }) {
   204→  const radius = 54;
   205→  const circumference = 2 * Math.PI * radius;
   206→  const offset = circumference - (score / 100) * circumference;
   207→  const color = scoreColor(score);
   208→  const stroke = scoreStroke(score);
   209→
   210→  return (
   211→    <div className="relative flex items-center justify-center">
   212→      <svg width="140" height="140" viewBox="0 0 140 140" className="-rotate-90">
   213→        <circle
   214→          cx="70" cy="70" r={radius}
   215→          fill="none" strokeWidth="8"
   216→          className="stroke-muted/30"
   217→        />
   218→        <motion.circle
   219→          cx="70" cy="70" r={radius}
   220→          fill="none" strokeWidth="8"
   221→          strokeLinecap="round"
   222→          className={stroke}
   223→          initial={{ strokeDashoffset: circumference }}
   224→          animate={{ strokeDashoffset: offset }}
   225→          transition={{ duration: 1.2, ease: 'easeOut' }}
   226→          strokeDasharray={circumference}
   227→          strokeDashoffset={offset}
   228→        />
   229→      </svg>
   230→      <div className="absolute flex flex-col items-center">
   231→        <motion.span
   232→          className={cn('text-3xl font-bold tabular-nums', color)}
   233→          initial={{ opacity: 0, scale: 0.8 }}
   234→          animate={{ opacity: 1, scale: 1 }}
   235→          transition={{ delay: 0.3, duration: 0.5 }}
   236→        >
   237→          {score}
   238→        </motion.span>
   239→        <span className="text-[10px] text-muted-foreground">/ 100</span>
   240→      </div>
   241→    </div>
   242→  );
   243→}

   438→
   439→function EventLogTab({ data, resolveMutation }: {
   440→  data: SecurityData;
   441→  resolveMutation: { mutate: (vars: { eventId: string; resolvedById: string | null }) => void; isPending: boolean };
   442→}) {
   443→  const [severityFilter, setSeverityFilter] = useState<string>('ALL');
   444→  const [typeFilter, setTypeFilter] = useState<string>('ALL');
   445→  const [expandedId, setExpandedId] = useState<string | null>(null);
   446→
   447→  const eventTypes = useMemo(() => {
   448→    const types = new Set(data.events.map(e => e.eventType));
   449→    return Array.from(types).sort();
   450→  }, [data.events]);
   451→
   452→  const filtered = useMemo(() => {
   453→    return data.events.filter(e => {
   454→      if (severityFilter !== 'ALL' && e.severity !== severityFilter) return false;
   455→      if (typeFilter !== 'ALL' && e.eventType !== typeFilter) return false;
   456→      return true;
   457→    });
   458→  }, [data.events, severityFilter, typeFilter]);
   459→
   460→  const currentUser = useDashboardStore(s => s.user);
   461→
   462→  const handleResolve = (eventId: string) => {
   463→    resolveMutation.mutate({
   464→      eventId,
   465→      resolvedById: currentUser?.id || null,
   466→    });
   467→  };
   468→
   469→  return (
   470→    <div className="flex flex-col gap-3 flex-1 overflow-hidden">
   471→      {/* Filter row */}
   472→      <div className="flex items-center gap-3 shrink-0 flex-wrap">
   473→        <Select value={severityFilter} onValueChange={setSeverityFilter}>
   474→          <SelectTrigger className="w-[160px] h-8 text-xs">
   475→            <SelectValue placeholder="Severity" />
   476→          </SelectTrigger>
   477→          <SelectContent>
   478→            <SelectItem value="ALL">All Severities</SelectItem>
   479→            <SelectItem value="INFO">Info</SelectItem>
   480→            <SelectItem value="WARNING">Warning</SelectItem>
   481→            <SelectItem value="CRITICAL">Critical</SelectItem>
   482→          </SelectContent>
   483→        </Select>
   484→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   485→          <SelectTrigger className="w-[180px] h-8 text-xs">
   486→            <SelectValue placeholder="Event Type" />
   487→          </SelectTrigger>
   488→          <SelectContent>
   489→            <SelectItem value="ALL">All Types</SelectItem>
   490→            {eventTypes.map(t => (
   491→              <SelectItem key={t} value={t}>{t.replace(/_/g, ' ')}</SelectItem>
   492→            ))}
   493→          </SelectContent>
   494→        </Select>
   495→        <span className="text-[10px] text-muted-foreground ml-auto">
   496→          {filtered.length} of {data.events.length} events
   497→        </span>
   498→      </div>
   499→
   500→      {/* Table */}
   501→      <div className="flex-1 overflow-y-auto rounded-lg border border-border">
   502→        <Table>
   503→          <TableHeader>
   504→            <TableRow className="hover:bg-transparent">
   505→              <TableHead className="text-[10px] h-8 w-28">Time</TableHead>
   506→              <TableHead className="text-[10px] h-8 w-36">Type</TableHead>
   507→              <TableHead className="text-[10px] h-8 w-24">Severity</TableHead>
   508→              <TableHead className="text-[10px] h-8">Description</TableHead>
   509→              <TableHead className="text-[10px] h-8 w-28">IP Address</TableHead>
   510→              <TableHead className="text-[10px] h-8 w-28">User</TableHead>
   511→              <TableHead className="text-[10px] h-8 w-20">Status</TableHead>
   512→              <TableHead className="text-[10px] h-8 w-16" />
   513→            </TableRow>
   514→          </TableHeader>
   515→          <TableBody>
   516→            {filtered.map(event => (
   517→              <>
   518→                <TableRow
   519→                  key={event.id}
   520→                  className={cn(
   521→                    'cursor-pointer transition-colors',
   522→                    expandedId === event.id && 'bg-card/60',
   523→                  )}
   524→                  onClick={() => setExpandedId(expandedId === event.id ? null : event.id)}
   525→                >
   526→                  <TableCell className="text-[10px] py-2 text-muted-foreground">
   527→                    {formatDate(event.createdAt)}
   528→                  </TableCell>
   529→                  <TableCell className="text-xs py-2">
   530→                    <div className="flex items-center gap-1.5">
   531→                      {eventTypeIcon(event.eventType, 'h-3.5 w-3.5')}
   532→                      <span>{event.eventType.replace(/_/g, ' ')}</span>

   880→}) {
   881→  const policies = data.policies;
   882→  const [encryptionEnabled, setEncryptionEnabled] = useState(policies?.encryptionEnabled ?? false);
   883→  const [twoFactorEnabled, setTwoFactorEnabled] = useState(policies?.twoFactorEnabled ?? false);
   884→  const [sessionTimeout, setSessionTimeout] = useState(policies?.sessionTimeoutMin ?? 30);
   885→  const [dataRetention, setDataRetention] = useState(policies?.dataRetentionDays ?? 365);
   886→  const [auditLogRetention, setAuditLogRetention] = useState(policies?.auditLogRetentionDays ?? 90);
   887→  const [ipList, setIpList] = useState<string[]>(policies?.ipWhitelist ?? []);
   888→  const [newIp, setNewIp] = useState('');
   889→
   890→  const handleAddIp = () => {
   891→    const trimmed = newIp.trim();
   892→    if (!trimmed) return;
   893→    if (ipList.includes(trimmed)) {
   894→      toast.error('IP address already in whitelist');
   895→      return;
   896→    }
   897→    const ipRegex = /^(\d{1,3}\.){3}\d{1,3}(\/\d{1,2})?$/;
   898→    if (!ipRegex.test(trimmed)) {
   899→      toast.error('Invalid IP address format');
   900→      return;
   901→    }
   902→    setIpList(prev => [...prev, trimmed]);
   903→    setNewIp('');
   904→  };
   905→
   906→  const handleRemoveIp = (ip: string) => {
   907→    setIpList(prev => prev.filter(i => i !== ip));
   908→  };
   909→
   910→  const handleSave = () => {
   911→    policyMutation.mutate({
   912→      action: 'UPDATE_POLICY',
   913→      encryptionEnabled,
   914→      twoFactorEnabled,
   915→      sessionTimeoutMin: sessionTimeout,
   916→      ipWhitelist: JSON.stringify(ipList),
   917→      dataRetentionDays: dataRetention,
   918→    });
   919→  };
   920→
   921→  const hasChanges = policies && (
   922→    encryptionEnabled !== policies.encryptionEnabled ||
   923→    twoFactorEnabled !== policies.twoFactorEnabled ||
   924→    sessionTimeout !== policies.sessionTimeoutMin ||
   925→    dataRetention !== policies.dataRetentionDays ||
   926→    auditLogRetention !== policies.auditLogRetentionDays ||
   927→    JSON.stringify(ipList) !== JSON.stringify(policies.ipWhitelist)
   928→  );
   929→
   930→  if (!policies) {
   931→    return (
   932→      <div className="flex-1 flex items-center justify-center text-xs text-muted-foreground">
   933→        No policy data available
   934→      </div>
   935→    );
   936→  }
   937→
   938→  return (
   939→    <div className="space-y-4 overflow-y-auto flex-1 pr-1">
   940→      {/* Encryption & 2FA */}
   941→      <Card className="bg-card/40 border-border">
   942→        <CardContent className="p-4 space-y-4">
   943→          <h3 className="text-xs font-semibold flex items-center gap-2 text-muted-foreground">
   944→            <Shield className="h-3.5 w-3.5" />
   945→            Authentication & Encryption
   946→          </h3>
   947→
   948→          <div className="flex items-center justify-between">
   949→            <div className="flex items-center gap-3">
   950→              <Lock className="h-4 w-4 text-cyan" />
   951→              <div>
   952→                <Label className="text-xs font-medium">Encryption at Rest (AES-256)</Label>
   953→                <p className="text-[10px] text-muted-foreground">Encrypt all sensitive data stored in the database</p>
   954→              </div>
   955→            </div>
   956→            <Switch
   957→              checked={encryptionEnabled}
   958→              onCheckedChange={setEncryptionEnabled}
   959→            />
   960→          </div>
   961→
   962→          <Separator className="bg-border" />
   963→
   964→          <div className="flex items-center justify-between">
   965→            <div className="flex items-center gap-3">
   966→              <ShieldCheck className="h-4 w-4 text-emerald" />
   967→              <div>
   968→                <Label className="text-xs font-medium">Two-Factor Authentication</Label>
   969→                <p className="text-[10px] text-muted-foreground">Require 2FA for all user accounts in this tenant</p>
   970→              </div>
   971→            </div>
   972→            <Switch
   973→              checked={twoFactorEnabled}
   974→              onCheckedChange={setTwoFactorEnabled}
   975→            />
   976→          </div>
   977→
   978→          <Separator className="bg-border" />
   979→
   980→          <div className="flex items-center justify-between">
   981→            <div className="flex items-center gap-3">
