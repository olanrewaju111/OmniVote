   128→// ─── Sankey Layout Engine ────────────────────────────────────────────────────
   129→
   130→function buildSankeyLayout(
   131→  sankeyData: PvtData['sankeyData'],
   132→  svgW: number,
   133→  svgH: number,
   134→): {
   135→  nodeRects: Record<string, { x: number; y: number; w: number; h: number; label: string; color: string; isLeft: boolean }>;
   136→  linkPaths: SankeyLinkPath[];
   137→} {
   138→  const { nodes, links } = sankeyData;
   139→  if (!nodes.length || !links.length) return { nodeRects: {}, linkPaths: [] };
   140→
   141→  const sourceIds = new Set(links.map((l) => l.source));
   142→  const targetIds = new Set(links.map((l) => l.target));
   143→  const leftNodes = nodes.filter((n) => sourceIds.has(n.id));
   144→  const rightNodes = nodes.filter((n) => targetIds.has(n.id));
   145→
   146→  // Total flow per node
   147→  const nodeTotals: Record<string, number> = {};
   148→  for (const link of links) {
   149→    nodeTotals[link.source] = (nodeTotals[link.source] || 0) + link.value;
   150→    nodeTotals[link.target] = (nodeTotals[link.target] || 0) + link.value;
   151→  }
   152→
   153→  // Scale: total flow on one side (they should be equal)
   154→  const totalLeftFlow = leftNodes.reduce((s, n) => s + (nodeTotals[n.id] || 0), 0);
   155→  const totalRightFlow = rightNodes.reduce((s, n) => s + (nodeTotals[n.id] || 0), 0);
   156→  const maxFlow = Math.max(totalLeftFlow, totalRightFlow, 1);
   157→
   158→  const padTop = 16;
   159→  const padBot = 16;
   160→  const availH = svgH - padTop - padBot;
   161→  const nodeW = 14;
   162→  const labelLeftW = 130;
   163→  const labelRightW = 130;
   164→  const leftX = labelLeftW;
   165→  const rightX = svgW - labelRightW - nodeW;
   166→
   167→  const scale = availH / maxFlow;
   168→
   169→  const nodeRects: Record<
   170→    string,
   171→    { x: number; y: number; w: number; h: number; label: string; color: string; isLeft: boolean }
   172→  > = {};
   173→
   174→  // Position left nodes
   175→  let yOff = padTop;
   176→  for (const node of leftNodes) {
   177→    const h = Math.max(8, (nodeTotals[node.id] || 0) * scale);
   178→    nodeRects[node.id] = { x: leftX, y: yOff, w: nodeW, h, label: node.label, color: node.color, isLeft: true };
   179→    yOff += h + 3;
   180→  }
   181→
   182→  // Position right nodes
   183→  yOff = padTop;
   184→  for (const node of rightNodes) {
   185→    const h = Math.max(8, (nodeTotals[node.id] || 0) * scale);
   186→    nodeRects[node.id] = {
   187→      x: rightX,
   188→      y: yOff,
   189→      w: nodeW,
   190→      h,
   191→      label: node.label,
   192→      color: node.color,
   193→      isLeft: false,
   194→    };
   195→    yOff += h + 3;
   196→  }
   197→
   198→  // Build link paths — track vertical offsets per node
   199→  const linkOffsets: Record<string, number> = {};
   200→  const nodeColorMap: Record<string, string> = {};
   201→  for (const n of nodes) nodeColorMap[n.id] = n.color;
   202→
   203→  const linkPaths: SankeyLinkPath[] = [];
   204→  for (const link of links) {
   205→    const src = nodeRects[link.source];
   206→    const tgt = nodeRects[link.target];
   207→    if (!src || !tgt) continue;
   208→
   209→    const linkH = Math.max(1.5, link.value * scale);
   210→    const srcOff = linkOffsets[link.source] || 0;
   211→    const tgtOff = linkOffsets[link.target] || 0;
   212→
   213→    linkOffsets[link.source] = srcOff + linkH;
   214→    linkOffsets[link.target] = tgtOff + linkH;
   215→
   216→    const x1 = src.x + src.w;
   217→    const y1 = src.y + srcOff + linkH / 2;
   218→    const x2 = tgt.x;
   219→    const y2 = tgt.y + tgtOff + linkH / 2;
   220→    const cx = (x1 + x2) / 2;
   221→
   222→    const path = `M ${x1} ${y1} C ${cx} ${y1}, ${cx} ${y2}, ${x2} ${y2}`;
   223→
   224→    linkPaths.push({
   225→      ...link,
   226→      color: nodeColorMap[link.source] || '#6B7280',
   227→      path,
   228→      x1,
   229→      y1,
   230→      x2,
   231→      y2,
   232→      linkH,
   233→    });
   234→  }
   235→
   236→  return { nodeRects, linkPaths };
   237→}

   280→function SankeyDiagram({ data }: { data: PvtData['sankeyData'] }) {
   281→  const [tooltip, setTooltip] = useState<{
   282→    x: number;
   283→    y: number;
   284→    source: string;
   285→    target: string;
   286→    value: number;
   287→  } | null>(null);
   288→
   289→  const svgW = 680;
   290→  const svgH = 380;
   291→
   292→  const { nodeRects, linkPaths } = useMemo(() => buildSankeyLayout(data, svgW, svgH), [data]);
   293→
   294→  if (!linkPaths.length) {
   295→    return (
   296→      <div className="flex h-full items-center justify-center text-sm text-muted-foreground">
   297→        No Sankey data available
   298→      </div>
   299→    );
   300→  }
   301→
   302→  const leftNodes = Object.values(nodeRects).filter((n) => n.isLeft);
   303→  const rightNodes = Object.values(nodeRects).filter((n) => !n.isLeft);
   304→
   305→  return (
   306→    <div className="relative h-full w-full">
   307→      <svg
   308→        viewBox={`0 0 ${svgW} ${svgH}`}
   309→        className="h-full w-full"
   310→        preserveAspectRatio="xMidYMid meet"
   311→      >
   312→        {/* Links */}
   313→        {linkPaths.map((lp, i) => (
   314→          <path
   315→            key={`${lp.source}-${lp.target}-${i}`}
   316→            d={lp.path}
   317→            fill="none"
   318→            stroke={lp.color}
   319→            strokeWidth={lp.linkH}
   320→            strokeOpacity={tooltip && tooltip.source === lp.source && tooltip.target === lp.target ? 0.8 : 0.4}
   321→            className="cursor-pointer transition-[stroke-opacity] duration-200"
   322→            onMouseEnter={(e) => {
   323→              const rect = (e.currentTarget.closest('svg') as SVGSVGElement).getBoundingClientRect();
   324→              const svgEl = e.currentTarget.closest('svg') as SVGSVGElement;
   325→              const pt = svgEl.createSVGPoint();
   326→              pt.x = e.clientX - rect.left;
   327→              pt.y = e.clientY - rect.top;
   328→              const ctm = svgEl.getScreenCTM();
   329→              if (ctm) {
   330→                const svgPt = pt.matrixTransform(ctm.inverse());
   331→                setTooltip({ x: svgPt.x, y: svgPt.y, source: lp.source, target: lp.target, value: lp.value });
   332→              }
   333→            }}
   334→            onMouseLeave={() => setTooltip(null)}
   335→          />
   336→        ))}
   337→
   338→        {/* Left node labels */}
   339→        {leftNodes.map((n) => (
   340→          <g key={n.label}>
   341→            <rect x={n.x} y={n.y} width={n.w} height={n.h} rx={3} fill={n.color} opacity={0.85} />
   342→            <text
   343→              x={n.x - 8}
   344→              y={n.y + n.h / 2}
   345→              textAnchor="end"
   346→              dominantBaseline="central"
   347→              className="fill-foreground text-[11px] font-medium"
   348→            >
   349→              {n.label}
   350→            </text>
   351→          </g>
   352→        ))}
   353→
   354→        {/* Right node labels */}
   355→        {rightNodes.map((n) => (
   356→          <g key={n.label}>
   357→            <rect x={n.x} y={n.y} width={n.w} height={n.h} rx={3} fill={n.color} opacity={0.85} />
   358→            <text
   359→              x={n.x + n.w + 8}
   360→              y={n.y + n.h / 2}
   361→              textAnchor="start"
   362→              dominantBaseline="central"
   363→              className="fill-foreground text-[11px] font-medium"
   364→            >
   365→              {n.label}
   366→            </text>
   367→          </g>
   368→        ))}
   369→
   370→        {/* Tooltip */}
   371→        {tooltip && (
   372→          <g>
   373→            <rect
   374→              x={tooltip.x - 90}
   375→              y={tooltip.y - 36}
   376→              width={180}
   377→              height={30}
   378→              rx={6}
   379→              className="fill-popover stroke-border"
   380→              strokeWidth={1}
   381→            />
   382→            <text
   383→              x={tooltip.x}
   384→              y={tooltip.y - 16}
   385→              textAnchor="middle"
   386→              dominantBaseline="central"
   387→              className="fill-foreground text-[10px]"
   388→            >
   389→              {tooltip.source} → {tooltip.target}: {tooltip.value.toLocaleString()}
   390→            </text>
   391→          </g>
   392→        )}
   393→      </svg>
   394→    </div>
   395→  );
   396→}

   398→function PartyBarChart({ partyTotals }: { partyTotals: PvtData['partyTotals'] }) {
   399→  const maxVotes = useMemo(() => {
   400→    if (!partyTotals.length) return 1;
   401→    return Math.max(...partyTotals.map((p) => p.votes), 1);
   402→  }, [partyTotals]);
   403→
   404→  if (!partyTotals.length) {
   405→    return (
   406→      <div className="flex h-full items-center justify-center text-sm text-muted-foreground">
   407→        No party data available
   408→      </div>
   409→    );
   410→  }
   411→
   412→  return (
   413→    <div className="space-y-2.5 px-1">
   414→      {partyTotals.map((pt, idx) => {
   415→        const pct = (pt.votes / maxVotes) * 100;
   416→        const color = getPartyColor(pt.party);
   417→        return (
   418→          <div key={pt.party} className="flex items-center gap-3">
   419→            <span className="w-12 shrink-0 text-right text-xs font-semibold text-foreground">
   420→              {pt.party}
   421→            </span>
   422→            <div className="relative h-5 flex-1 rounded bg-muted">
   423→              <motion.div
   424→                className="absolute inset-y-0 left-0 rounded"
   425→                style={{ backgroundColor: color }}
   426→                initial={{ width: 0 }}
   427→                animate={{ width: `${pct}%` }}
   428→                transition={{ duration: 0.7, delay: idx * 0.08, ease: 'easeOut' }}
   429→              />
   430→            </div>
   431→            <span className="w-16 shrink-0 text-right text-xs tabular-nums text-muted-foreground">
   432→              {pt.votes.toLocaleString()}
   433→            </span>
   434→          </div>
   435→        );
   436→      })}
   437→    </div>
   438→  );
   439→}

   441→function DeltaBadge({ pct }: { pct: number }) {
   442→  const abs = Math.abs(pct);
   443→  const colorClass =
   444→    abs > 10
   445→      ? 'text-rose-400 bg-rose-500/15'
   446→      : abs >= 5
   447→        ? 'text-amber-400 bg-amber-500/15'
   448→        : 'text-emerald-400 bg-emerald-500/15';
   449→  return (
   450→    <span className={cn('inline-flex items-center rounded px-1.5 py-0.5 text-xs font-medium', colorClass)}>
   451→      {pct > 0 ? '+' : ''}
   452→      {pct.toFixed(1)}%
   453→    </span>
   454→  );
   455→}
