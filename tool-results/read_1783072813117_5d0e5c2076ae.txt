   151→function StatCard({
   152→  icon, label, value, color, sub,
   153→}: {
   154→  icon: React.ReactNode; label: string; value: string | number; color: string; sub?: string;
   155→}) {
   156→  const colorMap: Record<string, string> = {
   157→    emerald: 'text-emerald', cyan: 'text-cyan', amber: 'text-amber',
   158→    rose: 'text-rose', purple: 'text-purple', foreground: 'text-foreground',
   159→  };
   160→  return (
   161→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.25 }}>
   162→      <Card className="border-border bg-card/40">
   163→        <CardContent className="p-3 sm:p-4">
   164→          <div className="flex items-center gap-1.5 mb-1.5">
   165→            <span className={colorMap[color] || 'text-muted-foreground'}>{icon}</span>
   166→            <span className="text-[10px] text-muted-foreground uppercase tracking-wider">{label}</span>
   167→          </div>
   168→          <p className={cn('text-lg font-bold tabular-nums', colorMap[color] || 'text-foreground')}>{value}</p>
   169→          {sub && <p className="text-[10px] text-muted-foreground/60 mt-0.5">{sub}</p>}
   170→        </CardContent>
   171→      </Card>
   172→    </motion.div>
   173→  );
   174→}

   179→
   180→function ColorBar({ value, max = 100, color }: { value: number; max?: number; color: string }) {
   181→  const pct = Math.min((value / max) * 100, 100);
   182→  return (
   183→    <div className="h-2 bg-secondary rounded-full overflow-hidden w-full min-w-[80px]">
   184→      <motion.div
   185→        initial={{ width: 0 }}
   186→        animate={{ width: `${pct}%` }}
   187→        transition={{ duration: 0.6, ease: 'easeOut' }}
   188→        className="h-full rounded-full"
   189→        style={{ backgroundColor: color }}
   190→      />
   191→    </div>
   192→  );
   193→}

   212→  const [hpExpectedResults, setHpExpectedResults] = useState<Array<{ party: string; votes: string }>>([

   230→    queryFn: () =>

   237→    mutationFn: (body: Record<string, unknown>) =>
   238→      fetchJson(`/api/honeypot?tenantId=${tenantId}`, {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify(body),
   242→      }),
   243→    onSuccess: () => {
   244→      queryClient.invalidateQueries({ queryKey: ['honeypot', tenantId] });

   248→      toast.error('Failed to complete action');

   252→  const isMutating = mutation.isPending;
   253→
   254→  // ── Derived data ──
   255→  const honeypots = data?.honeypots || [];
   256→  const accessibility = data?.accessibility || [];

   262→      [...(biometricSummary?.agents || [])].sort(

   267→

   269→  const resetHoneypotForm = () => {
   270→    setHpName('');
   271→    setHpState('');
   272→    setHpLga('');
   273→    setHpTrapType('');
   274→    setHpExpectedResults([{ party: '', votes: '' }]);
   275→  };
   276→
   277→  const handleAddResultRow = () => {
   278→    setHpExpectedResults((prev) => [...prev, { party: '', votes: '' }]);
   279→  };
   280→
   281→  const handleRemoveResultRow = (idx: number) => {
   282→    setHpExpectedResults((prev) => prev.filter((_, i) => i !== idx));
   283→  };
   284→
   285→  const handleResultChange = (idx: number, field: 'party' | 'votes', value: string) => {
   286→    setHpExpectedResults((prev) =>
   287→      prev.map((r, i) => (i === idx ? { ...r, [field]: value } : r)),
   288→    );
   289→  };
   290→
   291→  const handleSubmitHoneypot = () => {
   292→    if (!hpName.trim() || !hpState.trim() || !hpTrapType) {
   293→      toast.error('Please fill in name, state, and trap type');
   294→      return;
   295→    }
   296→    const validResults = hpExpectedResults
   297→      .filter((r) => r.party.trim() && r.votes.trim())
   298→      .map((r) => ({ party: r.party.trim(), votes: Number(r.votes) }));
   299→    mutation.mutate(
   300→      {
   301→        action: 'CREATE_HONEYPOT',
   302→        name: hpName.trim(),
   303→        state: hpState.trim(),
   304→        lga: hpLga.trim() || null,
   305→        trapType: hpTrapType,
   306→        expectedResults: validResults,
   307→      },
   308→      {
   309→        onSuccess: () => {
   310→          setHoneypotDialogOpen(false);
   311→          resetHoneypotForm();
   312→        },
   313→      },
   314→    );
   315→  };
   316→
   317→  // ── Accessibility dialog handlers ──
   318→  const resetAccessibilityForm = () => {
   319→    setArPollingUnitId('');
   320→    setArFeatures({
   321→      rampAccess: false, brailleBallots: false, signLanguage: false,
   322→      assistiveTech: false, accessibleToilet: false,
   323→    });
   324→    setArBarriers([]);
   325→    setArPwdServed('');
   326→    setArPwdTurnedAway('');
   327→    setArScore([50]);
   328→  };
   329→
   330→  const toggleBarrier = (barrier: string) => {
   331→    setArBarriers((prev) =>
   332→      prev.includes(barrier) ? prev.filter((b) => b !== barrier) : [...prev, barrier],
   333→    );
   334→  };
   335→
   336→  const toggleFeature = (key: keyof typeof arFeatures) => {
   337→    setArFeatures((prev) => ({ ...prev, [key]: !prev[key] }));
   338→  };
   339→
   340→  const handleSubmitAccessibility = () => {
   341→    if (!arPollingUnitId.trim()) {
   342→      toast.error('Please enter a polling unit ID');
   343→      return;
   344→    }
   345→    mutation.mutate(
   346→      {
   347→        action: 'CREATE_ACCESSIBILITY_REPORT',
   348→        pollingUnitId: arPollingUnitId.trim(),
   349→        features: arFeatures,
   350→        barrierTypes: arBarriers,
   351→        pwdVotersServed: Number(arPwdServed) || 0,
   352→        pwdVotersTurnedAway: Number(arPwdTurnedAway) || 0,
   353→        overallScore: arScore[0],
   354→      },
   355→      {
   356→        onSuccess: () => {
   357→          setAccessibilityDialogOpen(false);
   358→          resetAccessibilityForm();
   359→        },
   360→      },
   361→    );
   362→  };
   363→
   364→  // ── Render ──
   365→  return (
   366→    <div className="h-full flex flex-col p-4 sm:p-6 gap-4 overflow-hidden">
   367→      {/* Header */}
   368→      <div className="flex items-center justify-between flex-wrap gap-3 shrink-0">
   369→        <div>
   370→          <h2 className="text-lg font-semibold flex items-center gap-2">
   371→            <Radar className="h-5 w-5 text-purple" />
   372→            Honeypot Stations, Biometrics & PWD Tracking
   373→          </h2>
   374→          <p className="text-sm text-muted-foreground mt-0.5">
   375→            Decoy traps, agent biometric verification, and accessibility compliance
   376→          </p>
   377→        </div>
   378→        <Button
   379→          variant="outline"
   380→          size="sm"
   381→          onClick={() => refetch()}
   382→          className="gap-1.5 text-xs"
   383→        >
   384→          <Activity className="h-3.5 w-3.5" />
   385→          Refresh
   386→        </Button>
   387→      </div>
   388→
   389→      {isLoading ? (
   390→        <div className="flex-1 flex items-center justify-center">
   391→          <Loader2 className="h-6 w-6 animate-spin text-purple" />
   392→        </div>
   393→      ) : (
   394→        <Tabs defaultValue="honeypot" className="flex-1 min-h-0 flex flex-col">
   395→          <TabsList className="shrink-0">
   396→            <TabsTrigger value="honeypot" className="gap-1.5 text-xs">
   397→              <ShieldAlert className="h-3.5 w-3.5" />
   398→              Honeypot Stations
