   110→function statusConfig(status: string) {
   111→  switch (status) {
   112→    case 'DRAFT':
   113→      return { label: 'Draft', cls: 'border-zinc-500/30 bg-zinc-500/10 text-zinc-400' };
   114→    case 'REVIEWED':
   115→      return { label: 'Reviewed', cls: 'border-amber-500/30 bg-amber-500/10 text-amber-400' };
   116→    case 'CERTIFIED':
   117→      return { label: 'Certified', cls: 'border-emerald-500/30 bg-emerald-500/10 text-emerald-400' };
   118→    case 'DISMISSED':
   119→      return { label: 'Dismissed', cls: 'border-rose-500/30 bg-rose-500/10 text-rose-400' };
   120→    default:
   121→      return { label: status, cls: 'border-border bg-card text-muted-foreground' };
   122→  }
   123→}
   124→
   125→function confidenceColor(val: number) {
   126→  if (val >= 80) return 'bg-emerald-500';
   127→  if (val >= 60) return 'bg-amber-500';
   128→  return 'bg-rose-500';
   129→}

   131→function formatBytes(bytes: number) {
   132→  if (bytes < 1024) return `${bytes} B`;
   133→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   134→  return `${(bytes / 1048576).toFixed(1)} MB`;
   135→}
   136→
   137→function formatDuration(ms: number) {
   138→  if (ms < 1000) return `${ms}ms`;
   139→  return `${(ms / 1000).toFixed(1)}s`;
   140→}
   141→
   142→function timeAgo(dateStr: string | null) {
   143→  if (!dateStr) return 'N/A';
   144→  const d = new Date(dateStr);
   145→  const now = new Date();
   146→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   147→  if (diff < 1) return 'Just now';
   148→  if (diff < 60) return `${diff}m ago`;
   149→  if (diff < 1440) return `${Math.floor(diff / 60)}h ${diff % 60}m ago`;
   150→  return `${Math.floor(diff / 1440)}d ago`;
   151→}
   152→
   153→function formatDate(dateStr: string) {
   154→  return new Date(dateStr).toLocaleDateString('en-US', {
   155→    month: 'short', day: 'numeric', year: 'numeric',
   156→  });
   157→}
