     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { useDashboardStore } from '@/store/dashboard';
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table';
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter,
    16→  DialogHeader, DialogTitle,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    20→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  Search, Users, UserCheck, UserX, Shield, ShieldAlert, Wrench,
    27→  Loader2, FileText, Trash2, Eye, ToggleLeft, X, Pencil,
    28→} from 'lucide-react';
    29→import { cn } from '@/lib/utils';
    30→import { fetchJson } from '@/lib/api';
    31→import { toast } from 'sonner';
    32→
    33→interface AgentUser {
    34→  id: string;
    35→  email: string;
    36→  name: string;
    37→  role: string;
    38→  isOnline: boolean;
    39→  lastSeenAt: string | null;
    40→  createdAt: string;
    41→  _count?: { incidents: number; auditLogs: number };
    42→}
    43→
    44→interface ReportsSlideOver {
    45→  agent: AgentUser | null;
    46→  open: boolean;
    47→}
    48→
    49→export function AgentRoster() {
    50→  const queryClient = useQueryClient();
    51→  const { tenantId } = useDashboardStore();
    52→  const [search, setSearch] = useState('');
    53→  const [searchInput, setSearchInput] = useState('');
    54→  const [roleFilter, setRoleFilter] = useState('ALL');
    55→  const [addOpen, setAddOpen] = useState(false);
    56→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    57→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    58→  const [editOpen, setEditOpen] = useState(false);
    59→  const [editAgent, setEditAgent] = useState<AgentUser | null>(null);
    60→  const [editName, setEditName] = useState('');
    61→  const [editPhone, setEditPhone] = useState('');
    62→  const [editRole, setEditRole] = useState('');
    63→
    64→  // Add agent form state
    65→  const [newName, setNewName] = useState('');
    66→  const [newEmail, setNewEmail] = useState('');
    67→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    68→
    69→  // Fetch agents from proper API
    70→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    71→    queryKey: ['agents', search, roleFilter, tenantId],
    72→    queryFn: () => {
    73→      const params = new URLSearchParams();
    74→      if (search) params.set('search', search);
    75→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    76→      if (tenantId) params.set('tenantId', tenantId);
    77→      return fetchJson(`/api/agents?${params}`);
    78→    },
    79→  });
    80→
    81→  // Add agent mutation
    82→  const addMutation = useMutation({
    83→    mutationFn: (body: { name: string; email: string; role: string }) =>
    84→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
    85→        method: 'POST',
    86→        headers: { 'Content-Type': 'application/json' },
    87→        body: JSON.stringify(body),
    88→      }),
    89→    onSuccess: (data) => {
    90→      if (data.error) {
    91→        toast.error(data.error);
    92→        return;
    93→      }
    94→      toast.success(`Agent "${data.user.name}" added successfully`);
    95→      queryClient.invalidateQueries({ queryKey: ['agents'] });
    96→      setAddOpen(false);
    97→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
    98→    },
    99→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to add agent'),
   100→  });
   101→
   102→  // Action mutation (toggle online, remote wipe, change role, delete)
   103→  const actionMutation = useMutation({
   104→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
   105→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
   106→        method: 'PATCH',
   107→        headers: { 'Content-Type': 'application/json' },
   108→        body: JSON.stringify(body),
   109→      }),
   110→    onSuccess: (data, variables) => {
   111→      if (data.error) {
   112→        toast.error(data.error);
   113→        return;
   114→      }
   115→      if (variables.action === 'DELETE') {
   116→        toast.success(data.message || 'Agent removed');
   117→      } else if (variables.action === 'REMOTE_WIPE') {
   118→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   119→      } else if (variables.action === 'CHANGE_ROLE') {
   120→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   121→      } else {
   122→        toast.success('Status updated');
   123→      }
   124→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   125→      setConfirmAction({ type: '', agent: null });
   126→    },
   127→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Action failed'),
   128→  });
   129→
   130→  const users = data?.users || [];
   131→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   132→  const online = fieldAgents.filter(u => u.isOnline).length;
   133→
   134→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   135→    acc[u.role] = (acc[u.role] || 0) + 1;
   136→    return acc;
   137→  }, {});
   138→
   139→  // Search with debounce via the query key
   140→  const handleSearch = useCallback(() => {
   141→    setSearch(searchInput);
   142→  }, [searchInput]);
   143→
   144→  // Edit agent mutation
   145→  const editMutation = useMutation({
   146→    mutationFn: (body: { id: string; name?: string; phone?: string; role?: string }) =>
   147→      fetchJson('/api/tenants/users', {
   148→        method: 'PATCH',
   149→        headers: { 'Content-Type': 'application/json' },
   150→        body: JSON.stringify(body),
   151→      }),
   152→    onSuccess: () => {
   153→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   154→      toast.success('Agent details updated');
   155→      setEditOpen(false);
   156→      setEditAgent(null);
   157→    },
   158→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to update agent'),
   159→  });
   160→
   161→  const openEditDialog = (agent: AgentUser) => {
   162→    setEditAgent(agent);
   163→    setEditName(agent.name);
   164→    setEditPhone('');
   165→    setEditRole(agent.role);
   166→    setEditOpen(true);
   167→  };
   168→
   169→  const handleEditSave = () => {
   170→    if (!editAgent) return;
   171→    if (!editName.trim()) { toast.error('Name is required'); return; }
   172→    editMutation.mutate({
   173→      id: editAgent.id,
   174→      name: editName.trim(),
   175→      phone: editPhone.trim() || undefined,
   176→      role: editRole,
   177→    });
   178→  };
   179→
   180→  // Handle confirm action
   181→  const handleConfirmAction = () => {
   182→    if (!confirmAction.agent) return;
   183→    if (confirmAction.type === 'DELETE') {
   184→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   185→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   186→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   187→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   188→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   189→    }
   190→  };
   191→
   192→  // Handle add agent
   193→  const handleAddAgent = () => {
   194→    if (!newName.trim() || !newEmail.trim()) {
   195→      toast.error('Name and email are required');
   196→      return;
   197→    }
   198→    if (!newEmail.includes('@')) {
   199→      toast.error('Please enter a valid email');
   200→      return;
   201→    }
   202→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   203→  };
   204→
   205→  return (
   206→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   207→      {/* Header */}
   208→      <div className="flex items-center justify-between flex-wrap gap-3">
   209→        <div>
   210→          <h2 className="text-lg font-semibold flex items-center gap-2">
   211→            <Users className="h-5 w-5 text-emerald" />
   212→            Agent Roster
   213→          </h2>
   214→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   215→        </div>
   216→        <Button
   217→          onClick={() => setAddOpen(true)}
   218→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   219→        >
   220→          <UserCheck className="h-4 w-4" />
   221→          Add Agent
   222→        </Button>
   223→      </div>
   224→
   225→      {/* Summary cards */}
   226→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   227→        <Card className="border-border bg-card/40">
   228→          <CardContent className="p-3.5">
   229→            <div className="flex items-center gap-2 mb-1">
   230→              <Users className="h-4 w-4 text-emerald" />
   231→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   232→            </div>
   233→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   234→          </CardContent>
   235→        </Card>
   236→        <Card className="border-emerald/20 bg-emerald/5">
   237→          <CardContent className="p-3.5">
   238→            <div className="flex items-center gap-2 mb-1">
   239→              <UserCheck className="h-4 w-4 text-emerald" />
   240→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   241→            </div>
   242→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   243→          </CardContent>
   244→        </Card>
   245→        <Card className="border-amber/20 bg-amber/5">
   246→          <CardContent className="p-3.5">
   247→            <div className="flex items-center gap-2 mb-1">
   248→              <UserX className="h-4 w-4 text-amber" />
   249→              <span className="text-[11px] text-muted-foreground">Offline</span>
   250→            </div>
   251→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   252→          </CardContent>
   253→        </Card>
   254→        <Card className="border-border bg-card/40">
   255→          <CardContent className="p-3.5">
   256→            <div className="flex items-center gap-2 mb-1">
   257→              <Shield className="h-4 w-4 text-cyan" />
   258→              <span className="text-[11px] text-muted-foreground">All Users</span>
   259→            </div>
   260→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   261→          </CardContent>
   262→        </Card>
   263→      </div>
   264→
   265→      {/* Role filter + distribution badges */}
   266→      <div className="flex items-center justify-between flex-wrap gap-3">
   267→        <div className="flex items-center gap-2 flex-wrap">
   268→          {Object.entries(roleCounts).map(([role, count]) => (
   269→            <button
   270→              key={role}
   271→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   272→              className="cursor-pointer"
   273→            >
   274→              <Badge
   275→                variant="outline"
   276→                className={cn(
   277→                  'text-[11px] h-6 transition-colors',
   278→                  roleFilter === role ? 'bg-foreground/10' : '',
   279→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   280→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   281→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   282→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   283→                  'border-border text-muted-foreground'
   284→                )}
   285→              >
   286→                {role.replace(/_/g, ' ')}: {count}
   287→              </Badge>
   288→            </button>
   289→          ))}
   290→        </div>
   291→        {roleFilter !== 'ALL' && (
   292→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   293→            <X className="h-3 w-3" /> Clear filter
   294→          </button>
   295→        )}
   296→      </div>
   297→
   298→      {/* Agent table */}
   299→      <Card>
   300→        <CardHeader className="pb-2 pt-4 px-4">
   301→          <div className="flex items-center justify-between gap-3">
   302→            <CardTitle className="text-sm">
   303→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   304→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   305→            </CardTitle>
   306→            <div className="flex items-center gap-2">
   307→              <div className="relative">
   308→                <label htmlFor="agent-search" className="sr-only">Search agents</label>
   309→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   310→                <Input
   311→                  id="agent-search"
   312→                  placeholder="Search agents..."
   313→                  value={searchInput}
   314→                  onChange={(e) => setSearchInput(e.target.value)}
   315→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   316→                  className="pl-8 h-8 w-48 text-xs"
   317→                />
   318→              </div>
   319→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   320→                Search
   321→              </Button>
   322→            </div>
   323→          </div>
   324→        </CardHeader>
   325→        <CardContent className="px-0 pb-0">
   326→          <div className="max-h-[480px] overflow-y-auto">
   327→            {isLoading ? (
   328→              <div className="flex items-center justify-center py-16">
   329→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   330→              </div>
   331→            ) : users.length === 0 ? (
   332→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   333→                <Users className="h-8 w-8 mb-2 opacity-40" />
   334→                <p className="text-sm">No users found</p>
   335→              </div>
   336→            ) : (
   337→              <Table>
   338→                <TableHeader>
   339→                  <TableRow className="border-border hover:bg-transparent">
   340→                    <TableHead className="text-[11px] h-9">User</TableHead>
   341→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   342→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   343→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   344→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   345→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   346→                  </TableRow>
   347→                </TableHeader>
   348→                <TableBody>
   349→                  <AnimatePresence>
   350→                    {users.map(agent => (
   351→                      <motion.tr
   352→                        key={agent.id}
   353→                        initial={{ opacity: 0, y: 4 }}
   354→                        animate={{ opacity: 1, y: 0 }}
   355→                        exit={{ opacity: 0, y: -4 }}
   356→                        className="border-border hover:bg-card/60 transition-colors"
   357→                      >
   358→                        <TableCell className="py-2.5">
   359→                          <div className="flex items-center gap-2.5">
   360→                            <div className="relative">
   361→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   362→                                {agent.name.split(' ').map(n => n[0]).join('')}
   363→                              </div>
   364→                              <span className={cn(
   365→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   366→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   367→                              )} />
   368→                            </div>
   369→                            <div className="min-w-0">
   370→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   371→                              {agent.lastSeenAt && (
   372→                                <span className="text-[10px] text-muted-foreground/60">
   373→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   374→                                </span>
   375→                              )}
   376→                            </div>
   377→                          </div>
   378→                        </TableCell>
   379→                        <TableCell className="py-2.5">
   380→                          <Badge
   381→                            variant="outline"
   382→                            className={cn(
   383→                              'text-[10px] h-5',
   384→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   385→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   386→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   387→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   388→                              'border-border text-muted-foreground'
   389→                            )}
   390→                          >
   391→                            {agent.role.replace(/_/g, ' ')}
   392→                          </Badge>
   393→                        </TableCell>
   394→                        <TableCell className="py-2.5">
   395→                          <button
   396→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   397→                            className="cursor-pointer"
   398→                            disabled={actionMutation.isPending}
   399→                          >
   400→                            <Badge
   401→                              variant="outline"
   402→                              className={cn(
   403→                                'text-[10px] h-5 transition-colors cursor-pointer',
   404→                                agent.isOnline
   405→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   406→                                  : 'border-border text-muted-foreground hover:bg-card'
   407→                              )}
   408→                            >
   409→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   410→                              {agent.isOnline ? 'Online' : 'Offline'}
   411→                            </Badge>
   412→                          </button>
   413→                        </TableCell>
   414→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   415→                        <TableCell className="py-2.5 hidden md:table-cell">
   416→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   417→                            {agent._count?.incidents ?? 0}
   418→                          </span>
   419→                        </TableCell>
   420→                        <TableCell className="py-2.5 text-right">
   421→                          <div className="flex items-center justify-end gap-0.5">
   422→                            <Button
   423→                              variant="ghost"
   424→                              size="sm"
   425→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-cyan"
   426→                              title="Edit Details"
   427→                              aria-label={`Edit details for ${agent.name}`}
   428→                              onClick={() => openEditDialog(agent)}
   429→                              disabled={agent.role === 'SUPER_ADMIN'}
   430→                            >
   431→                              <Pencil className="h-3.5 w-3.5" />
   432→                            </Button>
   433→                            <Button
   434→                              variant="ghost"
   435→                              size="sm"
   436→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   437→                              title="View Reports"
   438→                              aria-label={`View reports for ${agent.name}`}
   439→                              onClick={() => setReportsPanel({ agent, open: true })}
   440→                            >
   441→                              <Eye className="h-3.5 w-3.5" />
   442→                            </Button>
   443→                            <Button
   444→                              variant="ghost"
   445→                              size="sm"
   446→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-amber"
   447→                              title="Remote Wipe Device"
   448→                              aria-label={`Remote wipe device for ${agent.name}`}
   449→                              onClick={() => setConfirmAction({ type: 'REMOTE_WIPE', agent })}
   450→                              disabled={actionMutation.isPending}
   451→                            >
   452→                              <Wrench className="h-3.5 w-3.5" />
   453→                            </Button>
   454→                            <Button
   455→                              variant="ghost"
   456→                              size="sm"
   457→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-rose"
   458→                              title="Remove Agent"
   459→                              aria-label={`Remove agent ${agent.name}`}
   460→                              onClick={() => setConfirmAction({ type: 'DELETE', agent })}
   461→                              disabled={actionMutation.isPending || agent.role === 'SUPER_ADMIN'}
   462→                            >
   463→                              <Trash2 className="h-3.5 w-3.5" />
   464→                            </Button>
   465→                          </div>
   466→                        </TableCell>
   467→                      </motion.tr>
   468→                    ))}
   469→                  </AnimatePresence>
   470→                </TableBody>
   471→              </Table>
   472→            )}
   473→          </div>
   474→        </CardContent>
   475→      </Card>
   476→
   477→      {/* ===== ADD AGENT DIALOG ===== */}
   478→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   479→        <DialogContent className="sm:max-w-md">
   480→          <DialogHeader>
   481→            <DialogTitle className="flex items-center gap-2">
   482→              <UserCheck className="h-5 w-5 text-emerald" />
   483→              Add New Agent
   484→            </DialogTitle>
   485→            <DialogDescription>
   486→              Create a new user account. They will be able to login from the main screen using their email.
   487→            </DialogDescription>
   488→          </DialogHeader>
   489→          <div className="space-y-4 py-2">
   490→            <div className="space-y-2">
   491→              <label className="text-xs font-medium text-muted-foreground">Full Name</label>
   492→              <Input
   493→                placeholder="e.g. Adebayo Johnson"
   494→                value={newName}
   495→                onChange={(e) => setNewName(e.target.value)}
   496→                className="h-9 text-sm"
   497→              />
   498→            </div>
   499→            <div className="space-y-2">
   500→              <label className="text-xs font-medium text-muted-foreground">Email Address</label>
   501→              <Input
   502→                placeholder="e.g. agent@nigeriaelectionwatch.org"
   503→                type="email"
   504→                value={newEmail}

   652→        </DialogContent>

   655→      {/* ===== REPORTS SLIDE PANEL ===== */}

   661→              Reports by {reportsPanel.agent?.name}

   663→            <DialogDescription>

   665→            </DialogDescription>
   666→          </DialogHeader>
   667→          <AgentReportsPanel agentId={reportsPanel.agent?.id || ''} />
   668→        </DialogContent>
   669→      </Dialog>
   670→    </div>
   671→  );

   674→// Sub-component: fetches and displays reports for a specific agent
   675→function AgentReportsPanel({ agentId }: { agentId: string }) {
   676→  const { data, isLoading } = useQuery<{ incidents: Array<{
   677→    id: string; type: string; severity: string; status: string;
   678→    description: string; submittedAt: string;
   679→    pollingUnit: { name: string; state: string; lga: string } | null;
   680→  }>; total: number }>({
   681→    queryKey: ['agent-reports', agentId],
   682→    queryFn: () => fetchJson(`/api/incidents?reporterId=${agentId}&limit=50`),
   683→    enabled: !!agentId,
   684→  });
   685→
   686→  if (isLoading) {
   687→    return (
   688→      <div className="flex items-center justify-center py-12">
   689→        <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   690→      </div>
   691→    );
   692→  }
   693→
   694→  const incidents = data?.incidents || [];
   695→
   696→  if (incidents.length === 0) {
   697→    return (
   698→      <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   699→        <FileText className="h-8 w-8 mb-2 opacity-40" />
   700→        <p className="text-sm">No reports submitted yet</p>
   701→      </div>
   702→    );
   703→  }
   704→
   705→  const severityColors: Record<string, string> = {
   706→    LOW: 'border-emerald/30 text-emerald bg-emerald/10',
   707→    MEDIUM: 'border-amber/30 text-amber bg-amber/10',
   708→    HIGH: 'border-orange/30 text-orange bg-orange/10',
   709→    CRITICAL: 'border-rose/30 text-rose bg-rose/10',
   710→  };
   711→
   712→  const statusColors: Record<string, string> = {
   713→    PENDING: 'text-muted-foreground',
   714→    REVIEWED: 'text-cyan',
   715→    ESCALATED: 'text-amber',
   716→    DISMISSED: 'text-muted-foreground/50',
   717→    QUARANTINED: 'text-rose',
   718→  };
   719→
   720→  return (
   721→    <div className="flex-1 overflow-y-auto space-y-2 pr-1">
   722→      <p className="text-xs text-muted-foreground mb-2">{data?.total || 0} total reports</p>
   723→      {incidents.map(inc => (
   724→        <div key={inc.id} className="rounded-lg border border-border bg-card/40 p-3 space-y-1.5">
   725→          <div className="flex items-center justify-between gap-2">
   726→            <div className="flex items-center gap-2">
   727→              <Badge variant="outline" className={cn('text-[10px] h-5', severityColors[inc.severity] || '')}>
   728→                {inc.severity}
   729→              </Badge>
   730→              <Badge variant="outline" className="text-[10px] h-5 border-border text-muted-foreground">
   731→                {inc.type.replace(/_/g, ' ')}
   732→              </Badge>
   733→            </div>
   734→            <div className="flex items-center gap-2">
   735→              <span className={cn('text-[10px] font-medium', statusColors[inc.status] || '')}>
   736→                {inc.status}
   737→              </span>
   738→            </div>
   739→          </div>
   740→          <p className="text-xs text-foreground/80 leading-relaxed">{inc.description}</p>
   741→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   742→            {inc.pollingUnit && (
   743→              <span className="flex items-center gap-1">
   744→                <span className="w-1 h-1 rounded-full bg-muted-foreground/40" />
   745→                {inc.pollingUnit.name}, {inc.pollingUnit.state}
   746→              </span>
   747→            )}
   748→            <span>{new Date(inc.submittedAt).toLocaleString()}</span>
   749→          </div>
   750→        </div>
   751→      ))}
   752→    </div>
   753→  );
