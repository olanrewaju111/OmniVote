     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Input } from '@/components/ui/input';
     9→import { Textarea } from '@/components/ui/textarea';
    10→import { Separator } from '@/components/ui/separator';
    11→import { Switch } from '@/components/ui/switch';
    12→import {
    13→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    14→} from '@/components/ui/select';
    15→import { useDashboardStore } from '@/store/dashboard';
    16→import {
    17→  Send, MapPin, Camera, Mic, AlertTriangle, CheckCircle2,
    18→  Radio, Loader2, ShieldCheck, Vote, FileWarning, Plus, X,
    19→  TrendingUp, BarChart3, Users,
    20→} from 'lucide-react';
    21→import { motion, AnimatePresence } from 'framer-motion';
    22→import { cn } from '@/lib/utils';

    25→

    28→  PRESIDENTIAL: [

    30→    { code: 'PDP', name: 'Peoples Democratic Party', color: '#CE1126' },

    32→    { code: 'NNPP', name: 'New Nigeria Peoples Party', color: '#FF9800' },
    33→  ],
    34→  STATE: [
    35→    { code: 'APC', name: 'APC', color: '#008751' },
    36→    { code: 'PDP', name: 'PDP', color: '#CE1126' },
    37→    { code: 'LP', name: 'LP', color: '#2196F3' },
    38→    { code: 'NNPP', name: 'NNPP', color: '#FF9800' },
    39→    { code: 'ADC', name: 'ADC', color: '#9C27B0' },
    40→  ],
    41→  LOCAL: [
    42→    { code: 'APC', name: 'APC', color: '#008751' },
    43→    { code: 'PDP', name: 'PDP', color: '#CE1126' },
    44→    { code: 'LP', name: 'LP', color: '#2196F3' },
    45→  ],
    46→};
    47→
    48→const INCIDENT_TYPES = [
    49→  { value: 'VIOLENCE', label: 'Violence', severity: 'HIGH', color: 'rose' },
    50→  { value: 'INTIMIDATION', label: 'Voter Intimidation', severity: 'HIGH', color: 'rose' },
    51→  { value: 'BALLOT_STUFFING', label: 'Ballot Stuffing', severity: 'CRITICAL', color: 'rose' },
    52→  { value: 'BRIBERY', label: 'Bribery / Vote Buying', severity: 'HIGH', color: 'amber' },
    53→  { value: 'UNDERAGE_VOTING', label: 'Underage Voting', severity: 'MEDIUM', color: 'amber' },
    54→  { value: 'MULTIPLE_VOTING', label: 'Multiple Voting', severity: 'HIGH', color: 'amber' },
    55→  { value: 'SNATCHED_BALLOT', label: 'Snatched Ballot Box', severity: 'CRITICAL', color: 'rose' },
    56→  { value: 'IMPEDIMENT', label: 'Impediment to Voting', severity: 'MEDIUM', color: 'amber' },
    57→  { value: 'LOGISTICS', label: 'Logistics / Materials Issue', severity: 'LOW', color: 'cyan' },
    58→  { value: 'BVAS_FAILURE', label: 'BVAS / Tech Failure', severity: 'MEDIUM', color: 'amber' },
    59→  { value: 'OBSERVATION', label: 'General Observation', severity: 'LOW', color: 'muted' },
    60→];
    61→
    62→type Tab = 'results' | 'statistics' | 'incident';
    63→
    64→interface PartyVote {
    65→  party: string;
    66→  votes: string; // string for input, parsed on submit
    67→}
    68→
    69→export function SubmitReport() {
    70→  const { user, electionTier, tenantId } = useDashboardStore();
    71→  const queryClient = useQueryClient();
    72→
    73→  const [activeTab, setActiveTab] = useState<Tab>('results');
    74→  const [selectedPU, setSelectedPU] = useState('');
    75→  const [submitting, setSubmitting] = useState(false);
    76→  const [submitted, setSubmitted] = useState(false);
    77→  const [submittedType, setSubmittedType] = useState('');
    78→
    79→  // ---- RESULTS STATE ----
    80→  const [accredited, setAccredited] = useState('');
    81→  const [validVotes, setValidVotes] = useState('');
    82→  const [rejectedBallots, setRejectedBallots] = useState('');
    83→  const [partyVotes, setPartyVotes] = useState<PartyVote[]>(
    84→    (PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL).map(p => ({ party: p.code, votes: '' }))
    85→  );
    86→
    87→  // ---- STATISTICS STATE ----
    88→  const [bvasUsed, setBvasUsed] = useState(true);
    89→  const [materialsOnTime, setMaterialsOnTime] = useState(true);
    90→  const [securityPresent, setSecurityPresent] = useState(true);
    91→  const [violenceOccurred, setViolenceOccurred] = useState(false);
    92→  const [statsNotes, setStatsNotes] = useState('');
    93→
    94→  // ---- INCIDENT STATE ----
    95→  const [incType, setIncType] = useState('');
    96→  const [incSeverity, setIncSeverity] = useState('MEDIUM');
    97→  const [incDescription, setIncDescription] = useState('');
    98→
    99→  // Fetch polling units for the agent (show first 10 as options)
   100→  const { data: puData } = useQuery({
   101→    queryKey: ['my-polling-units'],
   102→    queryFn: () => fetchJson(`/api/dashboard?tenantId=${tenantId}`).then(d => d.pollingUnits?.slice(0, 15) || []),
   103→  });
   104→
   105→  // Fetch agent's report counts for live stats
   106→  const { data: reportCounts } = useQuery({
   107→    queryKey: ['my-report-counts', user?.id],
   108→    queryFn: () => fetchJson(`/api/reports?reporterId=${user!.id}`).then(d => d.counts),
   109→    enabled: !!user?.id,
   110→    refetchInterval: 10000,
   111→  });
   112→
   113→  const pollingUnits = puData || [];
   114→  const counts = reportCounts || { totalResults: 0, totalIncidents: 0, resultsToday: 0, incidentsToday: 0 };
   115→
   116→  const parties = PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL;
   117→
   118→  // Add a party vote row
   119→  const addPartyRow = () => {
   120→    const existingCodes = partyVotes.map(p => p.party);
   121→    const missing = parties.find(p => !existingCodes.includes(p.code));
   122→    if (missing) {
   123→      setPartyVotes(prev => [...prev, { party: missing.code, votes: '' }]);
   124→    }
   125→  };
   126→
   127→  const removePartyRow = (idx: number) => {
   128→    if (partyVotes.length > 2) {
   129→      setPartyVotes(prev => prev.filter((_, i) => i !== idx));
   130→    }
   131→  };
   132→
   133→  const updatePartyVote = (idx: number, field: 'party' | 'votes', value: string) => {
   134→    setPartyVotes(prev => prev.map((p, i) => i === idx ? { ...p, [field]: value } : p));
   135→  };
   136→
   137→  // Auto-calculate totals
   138→  const totalVotesCast = partyVotes.reduce((s, p) => s + (parseInt(p.votes) || 0), 0);
   139→  const accreditedNum = parseInt(accredited) || 0;
   140→  const validVotesNum = parseInt(validVotes) || 0;
   141→  const rejectedNum = parseInt(rejectedBallots) || 0;
   142→
   143→  // ---- SUBMIT HANDLERS ----
   144→
   145→  const resultMutation = useMutation({
   146→    mutationFn: (body: Record<string, unknown>) =>
   147→      fetchJson('/api/results', {
   148→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   149→      }),
   150→    onSuccess: (data) => {
   151→      if (data.error) { toast.error(data.error); return; }
   152→      toast.success('Election results submitted successfully!');
   153→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   154→      queryClient.invalidateQueries({ queryKey: ['situation-room'] });
   155→      queryClient.invalidateQueries({ queryKey: ['my-reports'] });
   156→      queryClient.invalidateQueries({ queryKey: ['my-report-counts'] });
   157→      setSubmitted(true); setSubmittedType('results');
   158→      setTimeout(() => setSubmitted(false), 4000);
   159→    },
   160→    onError: () => toast.error('Failed to submit results'),
   161→  });
   162→
   163→  const incidentMutation = useMutation({
   164→    mutationFn: (body: Record<string, unknown>) =>
   165→      fetchJson('/api/incidents', {
   166→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   167→      }),
   168→    onSuccess: (data) => {
   169→      if (data.error) { toast.error(data.error); return; }
   170→      toast.success('Incident report submitted!');
   171→      if (data.incident?.gpsAnomaly) {
   172→        toast.warning('GPS anomaly detected — report quarantined for review');
   173→      }
   174→      queryClient.invalidateQueries({ queryKey: ['incidents'] });
   175→      queryClient.invalidateQueries({ queryKey: ['alerts'] });
   176→      queryClient.invalidateQueries({ queryKey: ['my-reports'] });
   177→      queryClient.invalidateQueries({ queryKey: ['my-report-counts'] });
   178→      setSubmitted(true); setSubmittedType('incident');
   179→      setIncDescription(''); setIncType('');
   180→      setTimeout(() => setSubmitted(false), 4000);
   181→    },
   182→    onError: () => toast.error('Failed to submit incident'),
   183→  });
   184→
   185→  const handleSubmitResults = () => {
   186→    if (!user?.id || !selectedPU) {
   187→      toast.error('Please select a polling unit');
   188→      return;
   189→    }
   190→    if (partyVotes.every(p => !p.votes)) {
   191→      toast.error('Enter at least one party vote count');
   192→      return;
   193→    }
   194→
   195→    const parsedPartyResults = partyVotes
   196→      .filter(p => p.votes)
   197→      .map(p => {
   198→        const partyInfo = parties.find(pt => pt.code === p.party);
   199→        return { party: p.party, name: partyInfo?.name || p.party, votes: parseInt(p.votes) || 0, color: partyInfo?.color || '#888' };
   200→      });
   201→
   202→    setSubmitting(true);
   203→    resultMutation.mutate({
   204→      reporterId: user.id,
   205→      pollingUnitId: selectedPU,
   206→      accreditedVoters: accreditedNum,
   207→      totalValidVotes: validVotesNum,
   208→      rejectedBallots: rejectedNum,
   209→      totalVotesCast,
   210→      partyResults: parsedPartyResults,
   211→      bvasUsed, materialsArrivedOnTime: materialsOnTime,
   212→      securityPresent, violenceOccurred, notes: statsNotes,
   213→    }, { onSettled: () => setSubmitting(false) });
   214→  };
   215→
   216→  const handleSubmitIncident = () => {
   217→    if (!user?.id) return;
   218→    if (!incType) { toast.error('Select an incident type'); return; }
   219→    if (!incDescription.trim()) { toast.error('Describe the incident'); return; }
   220→
   221→    const typeConfig = INCIDENT_TYPES.find(t => t.value === incType);
   222→    setSubmitting(true);
   223→    incidentMutation.mutate({
   224→      reporterId: user.id,
   225→      pollingUnitId: selectedPU || undefined,
   226→      type: incType,
   227→      severity: typeConfig?.severity || incSeverity,
   228→      description: incDescription.trim(),
   229→    }, { onSettled: () => setSubmitting(false) });
   230→  };
   231→
   232→  const selectedPUData = pollingUnits.find(p => p.id === selectedPU);
   233→
   234→  return (
   235→    <div className="h-full overflow-y-auto p-4 sm:p-6">
   236→      <div className="max-w-2xl mx-auto space-y-4">
   237→        {/* Header */}
   238→        <div>
   239→          <h2 className="text-lg font-semibold flex items-center gap-2">
   240→            <Send className="h-5 w-5 text-emerald" />
   241→            Field Report
   242→          </h2>
   243→          <p className="text-sm text-muted-foreground mt-0.5">
   244→            Submit election results, polling unit statistics, or report incidents and infractions.
   245→          </p>
   246→        </div>
   247→
   248→        {/* Agent info + PU selector */}
   249→        <Card className="border-emerald/20 bg-emerald/5">
   250→          <CardContent className="p-3.5 flex items-center gap-3">
   251→            <div className="w-10 h-10 rounded-full bg-emerald/20 flex items-center justify-center text-sm font-bold text-emerald">
   252→              {user?.name.split(' ').map(n => n[0]).join('')}
   253→            </div>
   254→            <div className="flex-1 min-w-0">
   255→              <p className="text-sm font-medium">{user?.name}</p>
   256→              <div className="flex items-center gap-1.5 mt-0.5">
   257→                <ShieldCheck className="h-3 w-3 text-emerald" />
   258→                <span className="text-[10px] text-muted-foreground">In-App Capture &middot; C2PA enabled</span>
   259→              </div>
   260→            </div>
   261→          </CardContent>
   262→        </Card>
   263→
   264→        {/* Polling Unit Selector */}
   265→        <div className="space-y-1.5">
   266→          <label htmlFor="pu-select" className="text-xs font-medium flex items-center gap-1.5">
   267→            <MapPin className="h-3.5 w-3.5 text-emerald" />
   268→            Select Polling Unit
   269→          </label>
   270→          <Select value={selectedPU} onValueChange={setSelectedPU}>
   271→            <SelectTrigger id="pu-select" className="h-10">
   272→              <SelectValue placeholder="Choose your assigned polling unit..." />
   273→            </SelectTrigger>
   274→            <SelectContent>
   275→              {pollingUnits.map(pu => (
   276→                <SelectItem key={pu.id} value={pu.id}>
   277→                  <span className="flex items-center gap-2">
   278→                    <span className="text-muted-foreground text-[10px]">{pu.code}</span>
   279→                    <span>{pu.name}</span>
   280→                    <span className="text-muted-foreground text-[10px]">{pu.state}/{pu.lga}</span>
   281→                    {pu.status === 'FLAGGED' && <Badge variant="outline" className="text-[9px] h-4 border-amber/30 text-amber ml-1">FLAGGED</Badge>}
   282→                  </span>
   283→                </SelectItem>
   284→              ))}
   285→            </SelectContent>
   286→          </Select>
   287→          {selectedPUData && (
   288→            <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   289→              <Badge variant="outline" className="text-[10px] h-5 border-emerald/30 text-emerald">
   290→                <span className="w-1.5 h-1.5 rounded-full bg-emerald mr-1" />IN GEOFENCE
   291→              </Badge>
   292→              <span>{selectedPUData.registered} registered voters</span>
   293→              <span>{selectedPUData.state} / {selectedPUData.lga}</span>
   294→            </div>
   295→          )}
   296→        </div>
   297→
   298→        {/* Tab selector */}
   299→        <div className="grid grid-cols-3 gap-1 p-1 rounded-lg bg-card/60 border border-border">
   300→          {[
   301→            { id: 'results' as Tab, label: 'Election Results', icon: <Vote className="h-4 w-4" /> },
   302→            { id: 'statistics' as Tab, label: 'Statistics', icon: <BarChart3 className="h-4 w-4" /> },
   303→            { id: 'incident' as Tab, label: 'Incident / Infraction', icon: <FileWarning className="h-4 w-4" /> },
   304→          ].map(tab => (
   305→            <button
   306→              key={tab.id}
   307→              onClick={() => setActiveTab(tab.id)}
   308→              className={cn(
   309→                'flex items-center justify-center gap-1.5 px-3 py-2.5 rounded-md text-xs font-medium transition-all',
   310→                activeTab === tab.id
   311→                  ? 'bg-emerald/15 text-emerald shadow-sm'
   312→                  : 'text-muted-foreground hover:text-foreground hover:bg-card'
   313→              )}
   314→            >
   315→              {tab.icon}
   316→              <span className="hidden sm:inline">{tab.label}</span>
   317→            </button>
   318→          ))}
   319→        </div>
   320→
   321→        {/* Tab content */}
   322→        <AnimatePresence mode="wait">
   323→          <motion.div
   324→            key={activeTab}
   325→            initial={{ opacity: 0, y: 4 }}
   326→            animate={{ opacity: 1, y: 0 }}
   327→            exit={{ opacity: 0, y: -4 }}
   328→            transition={{ duration: 0.15 }}
   329→          >
   330→            {/* ===== ELECTION RESULTS TAB ===== */}
   331→            {activeTab === 'results' && (
   332→              <Card>
   333→                <CardContent className="p-4 space-y-4">
   334→                  <div>
   335→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   336→                      <Vote className="h-4 w-4 text-emerald" />
   337→                      Election Results Form
   338→                    </h3>
   339→                    <p className="text-[11px] text-muted-foreground mt-0.5">
   340→                      Enter the official vote counts from Form EC8A at your polling unit.
   341→                    </p>
   342→                  </div>
   343→
   344→                  {/* Summary fields */}
   345→                  <div className="grid grid-cols-2 gap-3">
   346→                    <div className="space-y-1.5">
   347→                      <label htmlFor="accredited-voters" className="text-[11px] font-medium text-muted-foreground">Accredited Voters</label>
   348→                      <Input
   349→                        id="accredited-voters"
   350→                        type="number" min="0"
   351→                        placeholder="0"
   352→                        value={accredited}
   353→                        onChange={(e) => setAccredited(e.target.value)}
   354→                        className="h-10 text-sm tabular-nums"
   355→                      />
   356→                    </div>
   357→                    <div className="space-y-1.5">
   358→                      <label htmlFor="rejected-ballots" className="text-[11px] font-medium text-muted-foreground">Rejected Ballots</label>
   359→                      <Input
   360→                        id="rejected-ballots"
   361→                        type="number" min="0"
   362→                        placeholder="0"
   363→                        value={rejectedBallots}
   364→                        onChange={(e) => setRejectedBallots(e.target.value)}
   365→                        className="h-10 text-sm tabular-nums"
   366→                      />
   367→                    </div>
   368→                  </div>
   369→
   370→                  <Separator />
   371→
   372→                  {/* Party vote entries */}
   373→                  <div className="space-y-2">
   374→                    <div className="flex items-center justify-between">
   375→                      <label className="text-xs font-medium">Vote Count by Party</label>
   376→                      <Button variant="ghost" size="sm" className="h-6 text-[10px] gap-1" onClick={addPartyRow}>
   377→                        <Plus className="h-3 w-3" /> Add Party
   378→                      </Button>
   379→                    </div>
   380→
   381→                    <div className="space-y-2">
   382→                      {partyVotes.map((pv, idx) => {
   383→                        const partyInfo = parties.find(p => p.code === pv.party);
   384→                        return (
   385→                          <div key={idx} className="flex items-center gap-2">
   386→                            <div
   387→                              className="w-3 h-8 rounded-sm shrink-0"
   388→                              style={{ backgroundColor: partyInfo?.color || '#555' }}
   389→                            />
   390→                            <div className="flex-1 min-w-0">
   391→                              <Select value={pv.party} onValueChange={(v) => updatePartyVote(idx, 'party', v)}>
   392→                                <SelectTrigger aria-label={`Select party for row ${idx + 1}`} className="h-8 text-xs">
   393→                                  <SelectValue />
   394→                                </SelectTrigger>
   395→                                <SelectContent>
   396→                                  {parties.map(p => (
   397→                                    <SelectItem key={p.code} value={p.code}>
   398→                                      <span className="flex items-center gap-2">
   399→                                        <span className="w-2 h-2 rounded-sm" style={{ backgroundColor: p.color }} />
   400→                                        <span className="font-medium">{p.code}</span>
   401→                                        <span className="text-muted-foreground text-[10px] hidden sm:inline">{p.name}</span>
   402→                                      </span>
   403→                                    </SelectItem>
   404→                                  ))}
   405→                                </SelectContent>
   406→                              </Select>
   407→                            </div>
   408→                            <Input
   409→                              type="number" min="0" placeholder="0"
   410→                              aria-label={`Votes for ${pv.party || 'party'} row ${idx + 1}`}
   411→                              value={pv.votes}
   412→                              onChange={(e) => updatePartyVote(idx, 'votes', e.target.value)}
   413→                              className="w-28 h-8 text-sm tabular-nums text-right"
   414→                            />
   415→                            {partyVotes.length > 2 && (
   416→                              <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-rose" aria-label="Remove party row" onClick={() => removePartyRow(idx)}>
   417→                                <X className="h-3.5 w-3.5" />
   418→                              </Button>
   419→                            )}
   420→                          </div>
   421→                        );
   422→                      })}
   423→                    </div>
   424→                  </div>
   425→
   426→                  {/* Live totals */}
   427→                  <div className="rounded-lg bg-background border border-border p-3 space-y-2">
   428→                    <div className="flex items-center gap-2 text-xs font-medium">
   429→                      <TrendingUp className="h-3.5 w-3.5 text-emerald" />
   430→                      Live Totals
   431→                    </div>
   432→                    <div className="grid grid-cols-3 gap-2 text-center">
   433→                      <div>
   434→                        <p className="text-lg font-bold tabular-nums text-emerald">{totalVotesCast.toLocaleString()}</p>
   435→                        <p className="text-[10px] text-muted-foreground">Total Cast</p>
   436→                      </div>
   437→                      <div>
   438→                        <p className="text-lg font-bold tabular-nums text-cyan">{rejectedNum}</p>
   439→                        <p className="text-[10px] text-muted-foreground">Rejected</p>
   440→                      </div>
   441→                      <div>
   442→                        <p className="text-lg font-bold tabular-nums text-amber">{accreditedNum > 0 ? Math.round((totalVotesCast / accreditedNum) * 10000) / 100 : 0}%</p>
   443→                        <p className="text-[10px] text-muted-foreground">Turnout</p>
   444→                      </div>
   445→                    </div>
   446→                    {/* Simple bar chart */}
   447→                    {partyVotes.some(p => p.votes) && (
   448→                      <div className="flex h-3 rounded-full overflow-hidden bg-secondary mt-1">
   449→                        {partyVotes.filter(p => p.votes).map((pv) => {
   450→                          const partyInfo = parties.find(p => p.code === pv.party);
   451→                          const pct = totalVotesCast > 0 ? (parseInt(pv.votes) || 0) / totalVotesCast * 100 : 0;
   452→                          return (
   453→                            <div
   454→                              key={pv.party}
   455→                              className="h-full transition-all duration-500"
   456→                              style={{ width: `${pct}%`, backgroundColor: partyInfo?.color || '#555' }}
   457→                              title={`${pv.party}: ${pv.votes}`}
   458→                            />
   459→                          );
   460→                        })}
   461→                      </div>
   462→                    )}
   463→                  </div>
   464→
   465→                  <Button
   466→                    onClick={handleSubmitResults}
   467→                    disabled={submitting || !selectedPU || partyVotes.every(p => !p.votes)}
   468→                    className="w-full h-11 bg-emerald hover:bg-emerald/90 text-emerald-950 font-semibold gap-2"
   469→                  >
