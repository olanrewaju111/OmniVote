   150→
   151→function StatCard({
   152→  icon, label, value, color, sub,
   153→}: {
   154→  icon: React.ReactNode; label: string; value: string | number; color: string; sub?: string;
   155→}) {
   156→  const colorMap: Record<string, string> = {
   157→    emerald: 'text-emerald', cyan: 'text-cyan', amber: 'text-amber',
   158→    rose: 'text-rose', purple: 'text-purple', foreground: 'text-foreground',
   159→  };
   160→  return (
   161→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.25 }}>
   162→      <Card className="border-border bg-card/40">
   163→        <CardContent className="p-3 sm:p-4">
   164→          <div className="flex items-center gap-1.5 mb-1.5">
   165→            <span className={colorMap[color] || 'text-muted-foreground'}>{icon}</span>
   166→            <span className="text-[10px] text-muted-foreground uppercase tracking-wider">{label}</span>
   167→          </div>
   168→          <p className={cn('text-lg font-bold tabular-nums', colorMap[color] || 'text-foreground')}>{value}</p>
   169→          {sub && <p className="text-[10px] text-muted-foreground/60 mt-0.5">{sub}</p>}
   170→        </CardContent>
   171→      </Card>
   172→    </motion.div>
   173→  );
   174→}

   179→
   180→function ColorBar({ value, max = 100, color }: { value: number; max?: number; color: string }) {
   181→  const pct = Math.min((value / max) * 100, 100);
   182→  return (
   183→    <div className="h-2 bg-secondary rounded-full overflow-hidden w-full min-w-[80px]">
   184→      <motion.div
   185→        initial={{ width: 0 }}
   186→        animate={{ width: `${pct}%` }}
   187→        transition={{ duration: 0.6, ease: 'easeOut' }}
   188→        className="h-full rounded-full"
   189→        style={{ backgroundColor: color }}
   190→      />
   191→    </div>
   192→  );
   193→}

   199→export function HoneypotBiometrics() {
   200→  const queryClient = useQueryClient();
   201→  const { tenantId } = useDashboardStore();
   202→
   203→  // Dialog states
   204→  const [honeypotDialogOpen, setHoneypotDialogOpen] = useState(false);
   205→  const [accessibilityDialogOpen, setAccessibilityDialogOpen] = useState(false);
   206→
   207→  // ── Honeypot form state ──
   208→  const [hpName, setHpName] = useState('');
   209→  const [hpState, setHpState] = useState('');
   210→  const [hpLga, setHpLga] = useState('');
   211→  const [hpTrapType, setHpTrapType] = useState('');
   212→  const [hpExpectedResults, setHpExpectedResults] = useState<Array<{ party: string; votes: string }>>([
   213→    { party: '', votes: '' },
   214→  ]);
   215→
   216→  // ── Accessibility form state ──
   217→  const [arPollingUnitId, setArPollingUnitId] = useState('');
   218→  const [arFeatures, setArFeatures] = useState({
   219→    rampAccess: false, brailleBallots: false, signLanguage: false,
   220→    assistiveTech: false, accessibleToilet: false,
   221→  });
   222→  const [arBarriers, setArBarriers] = useState<string[]>([]);
   223→  const [arPwdServed, setArPwdServed] = useState('');
   224→  const [arPwdTurnedAway, setArPwdTurnedAway] = useState('');
   225→  const [arScore, setArScore] = useState([50]);
   226→
   227→  // ── Data fetching ──
   228→  const { data, isLoading, isError, refetch } = useQuery<HoneypotData>({
   229→    queryKey: ['honeypot', tenantId],
   230→    queryFn: () =>
   231→      fetchJson(`/api/honeypot?tenantId=${tenantId}`),
   232→    refetchInterval: 15000,
   233→  });
   234→
   235→  // ── Mutation ──
   236→  const mutation = useMutation({
   237→    mutationFn: (body: Record<string, unknown>) =>
   238→      fetchJson(`/api/honeypot?tenantId=${tenantId}`, {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify(body),
   242→      }),
   243→    onSuccess: () => {
   244→      queryClient.invalidateQueries({ queryKey: ['honeypot', tenantId] });
   245→      toast.success('Action completed successfully');
   246→    },
   247→    onError: () => {
   248→      toast.error('Failed to complete action');
   249→    },
   250→  });
   251→
   252→  const isMutating = mutation.isPending;
   253→
   254→  // ── Derived data ──
   255→  const honeypots = data?.honeypots || [];
   256→  const accessibility = data?.accessibility || [];
   257→  const stats = data?.stats;
   258→  const biometricSummary = data?.biometricSummary;
   259→  const trapEffectiveness = data?.trapEffectiveness;
   260→  const biometricAgents = useMemo(
   261→    () =>
   262→      [...(biometricSummary?.agents || [])].sort(
   263→        (a, b) => b.biometricRiskScore - a.biometricRiskScore,
   264→      ),
   265→    [biometricSummary],
   266→  );
   267→
   268→  // ── Honeypot dialog handlers ──
   269→  const resetHoneypotForm = () => {
   270→    setHpName('');
   271→    setHpState('');
   272→    setHpLga('');
   273→    setHpTrapType('');
   274→    setHpExpectedResults([{ party: '', votes: '' }]);
   275→  };
   276→
   277→  const handleAddResultRow = () => {
   278→    setHpExpectedResults((prev) => [...prev, { party: '', votes: '' }]);
   279→  };
   280→
   281→  const handleRemoveResultRow = (idx: number) => {
   282→    setHpExpectedResults((prev) => prev.filter((_, i) => i !== idx));
   283→  };
   284→
   285→  const handleResultChange = (idx: number, field: 'party' | 'votes', value: string) => {
   286→    setHpExpectedResults((prev) =>
   287→      prev.map((r, i) => (i === idx ? { ...r, [field]: value } : r)),
   288→    );
   289→  };
   290→
   291→  const handleSubmitHoneypot = () => {
   292→    if (!hpName.trim() || !hpState.trim() || !hpTrapType) {
   293→      toast.error('Please fill in name, state, and trap type');
   294→      return;
   295→    }
   296→    const validResults = hpExpectedResults
   297→      .filter((r) => r.party.trim() && r.votes.trim())
   298→      .map((r) => ({ party: r.party.trim(), votes: Number(r.votes) }));
   299→    mutation.mutate(
   300→      {
   301→        action: 'CREATE_HONEYPOT',
   302→        name: hpName.trim(),
   303→        state: hpState.trim(),
   304→        lga: hpLga.trim() || null,
   305→        trapType: hpTrapType,
   306→        expectedResults: validResults,
   307→      },
   308→      {
   309→        onSuccess: () => {
   310→          setHoneypotDialogOpen(false);
   311→          resetHoneypotForm();
   312→        },
   313→      },
   314→    );
   315→  };
   316→
   317→  // ── Accessibility dialog handlers ──
   318→  const resetAccessibilityForm = () => {
   319→    setArPollingUnitId('');
   320→    setArFeatures({
   321→      rampAccess: false, brailleBallots: false, signLanguage: false,
   322→      assistiveTech: false, accessibleToilet: false,
   323→    });
   324→    setArBarriers([]);
   325→    setArPwdServed('');
   326→    setArPwdTurnedAway('');
   327→    setArScore([50]);
   328→  };
   329→
   330→  const toggleBarrier = (barrier: string) => {
   331→    setArBarriers((prev) =>
   332→      prev.includes(barrier) ? prev.filter((b) => b !== barrier) : [...prev, barrier],
   333→    );
   334→  };
   335→
   336→  const toggleFeature = (key: keyof typeof arFeatures) => {
   337→    setArFeatures((prev) => ({ ...prev, [key]: !prev[key] }));
   338→  };
   339→
   340→  const handleSubmitAccessibility = () => {
   341→    if (!arPollingUnitId.trim()) {
   342→      toast.error('Please enter a polling unit ID');
   343→      return;
   344→    }
   345→    mutation.mutate(
   346→      {
   347→        action: 'CREATE_ACCESSIBILITY_REPORT',
   348→        pollingUnitId: arPollingUnitId.trim(),
   349→        features: arFeatures,
   350→        barrierTypes: arBarriers,
   351→        pwdVotersServed: Number(arPwdServed) || 0,
   352→        pwdVotersTurnedAway: Number(arPwdTurnedAway) || 0,
   353→        overallScore: arScore[0],
   354→      },
   355→      {
   356→        onSuccess: () => {
   357→          setAccessibilityDialogOpen(false);
   358→          resetAccessibilityForm();
   359→        },
   360→      },
   361→    );
   362→  };
   363→
   364→  // ── Render ──
   365→  return (
   366→    <div className="h-full flex flex-col p-4 sm:p-6 gap-4 overflow-hidden">
   367→      {/* Header */}
   368→      <div className="flex items-center justify-between flex-wrap gap-3 shrink-0">
   369→        <div>
   370→          <h2 className="text-lg font-semibold flex items-center gap-2">
   371→            <Radar className="h-5 w-5 text-purple" />
   372→            Honeypot Stations, Biometrics & PWD Tracking
   373→          </h2>
   374→          <p className="text-sm text-muted-foreground mt-0.5">
   375→            Decoy traps, agent biometric verification, and accessibility compliance
   376→          </p>
   377→        </div>
   378→        <Button
   379→          variant="outline"
   380→          size="sm"
   381→          onClick={() => refetch()}
   382→          className="gap-1.5 text-xs"
   383→        >
   384→          <Activity className="h-3.5 w-3.5" />
   385→          Refresh
   386→        </Button>
   387→      </div>
   388→
   389→      {isError ? (
   390→        <div className="flex-1 flex flex-col items-center justify-center min-h-[200px] text-center p-6">
   391→          <AlertCircle className="h-10 w-10 text-destructive mb-3" />
   392→          <p className="text-sm text-muted-foreground">Failed to load data. Please try again.</p>
   393→          <Button variant="outline" size="sm" className="mt-3" onClick={() => window.location.reload()}>
   394→            Retry
   395→          </Button>
   396→        </div>
   397→      ) : isLoading ? (
   398→        <div className="flex-1 flex items-center justify-center">
   399→          <Loader2 className="h-6 w-6 animate-spin text-purple" />
   400→        </div>
   401→      ) : (
   402→        <Tabs defaultValue="honeypot" className="flex-1 min-h-0 flex flex-col">
   403→          <TabsList className="shrink-0">
   404→            <TabsTrigger value="honeypot" className="gap-1.5 text-xs">
   405→              <ShieldAlert className="h-3.5 w-3.5" />
   406→              Honeypot Stations
   407→            </TabsTrigger>
   408→            <TabsTrigger value="biometrics" className="gap-1.5 text-xs">
   409→              <Fingerprint className="h-3.5 w-3.5" />
   410→              Biometrics
   411→            </TabsTrigger>
   412→            <TabsTrigger value="pwd" className="gap-1.5 text-xs">
   413→              <Accessibility className="h-3.5 w-3.5" />
   414→              PWD / Accessibility
   415→            </TabsTrigger>
   416→          </TabsList>
   417→
   418→          {/* ═══════════ TAB 1: Honeypot Stations ═══════════ */}
   419→          <TabsContent value="honeypot" className="flex-1 min-h-0 flex flex-col gap-4 mt-3">
   420→            {/* Stats Row */}
   421→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-2.5 shrink-0">

   423→                icon={<ShieldAlert className="h-3.5 w-3.5" />}
   424→                label="Total Traps"
   425→                value={stats?.totalHoneypots ?? 0}
   426→                color="purple"
   427→              />
   428→              <StatCard
   429→                icon={<CircleDot className="h-3.5 w-3.5" />}
   430→                label="Active Traps"
   431→                value={stats?.activeHoneypots ?? 0}
   432→                color="emerald"
   433→              />
   434→              <StatCard
   435→                icon={<TrendingDown className="h-3.5 w-3.5" />}
   436→                label="Deviations Detected"
   437→                value={stats?.deviationsDetected ?? 0}
   438→                color="rose"
   439→              />
   440→              <StatCard
   441→                icon={<Siren className="h-3.5 w-3.5" />}
   442→                label="Alerts Triggered"
   443→                value={stats?.alertsTriggered ?? 0}
   444→                color="rose"
   445→                sub="Critical"

   454→            </div>

   456→            {/* Create button */}

   463→                <Plus className="h-3.5 w-3.5" />
   464→                Create Honeypot
   465→              </Button>
   466→            </div>
   467→
   468→            {/* Honeypot cards grid */}
   469→            <ScrollArea className="flex-1 min-h-0">
   470→              {honeypots.length === 0 ? (
   471→                <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   472→                  <ShieldAlert className="h-8 w-8 mb-2 opacity-40" />
   473→                  <p className="text-sm">No honeypot stations deployed</p>
   474→                </div>
   475→              ) : (
   476→                <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 pr-4 pb-4">
   477→                  {honeypots.map((hp, idx) => (
   478→                    <motion.div
   479→                      key={hp.id}
   480→                      initial={{ opacity: 0, y: 8 }}
   481→                      animate={{ opacity: 1, y: 0 }}
   482→                      transition={{ delay: idx * 0.04, duration: 0.25 }}
   483→                    >
   484→                      <Card
   485→                        className={cn(
   486→                          'border transition-colors',
   487→                          hp.alertTriggered
   488→                            ? 'border-rose/50 bg-rose/5'
   489→                            : hp.deviationDetected
   490→                              ? 'border-amber/40 bg-amber/5'
   491→                              : 'border-border bg-card/40',
   492→                        )}
   493→                      >
   494→                        <CardContent className="p-4 space-y-3">
   495→                          {/* Top: name, status, type */}
   496→                          <div className="flex items-start justify-between gap-2">
   497→                            <div className="flex-1 min-w-0">
   498→                              <div className="flex items-center gap-2 flex-wrap">
   499→                                <span className="text-sm font-semibold truncate">{hp.name}</span>
   500→                                {hp.alertTriggered && (
   501→                                  <Badge className="bg-rose text-white text-[10px] h-5 px-1.5 gap-1">

   509→                              </p>
   510→                            </div>
   511→                            <div className="flex items-center gap-2 shrink-0">
   512→                              <Badge
   513→                                variant="outline"
   514→                                className={cn(
   515→                                  'text-[10px] h-5 px-1.5',
   516→                                  TRAP_TYPE_CONFIG[hp.trapType]?.bg,
   517→                                  TRAP_TYPE_CONFIG[hp.trapType]?.color,
   518→                                  TRAP_TYPE_CONFIG[hp.trapType]?.border,
   519→                                )}
   520→                              >
   521→                                {TRAP_TYPE_CONFIG[hp.trapType]?.label || hp.trapType}
   522→                              </Badge>
   523→                              <div className="flex items-center gap-1">
   524→                                {hp.isActive ? (
   525→                                  <>
   526→                                    <span className="h-2 w-2 rounded-full bg-emerald animate-pulse" />
   527→                                    <span className="text-[10px] text-emerald">Active</span>
   528→                                  </>
   529→                                ) : (
   530→                                  <>
   531→                                    <span className="h-2 w-2 rounded-full bg-muted-foreground/40" />
   532→                                    <span className="text-[10px] text-muted-foreground">Inactive</span>
   533→                                  </>
   534→                                )}

   536→                            </div>
