   138→function formatTime(date: string | Date) {
   139→  const d = new Date(date);
   140→  const now = new Date();
   141→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   142→  if (diff < 1) return 'Just now';
   143→  if (diff < 60) return `${diff}m ago`;
   144→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   145→  return `${Math.floor(diff / 1440)}d ago`;
   146→}
   147→
   148→function formatNumber(n: number): string {
   149→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   150→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   151→  return String(n);
   152→}
