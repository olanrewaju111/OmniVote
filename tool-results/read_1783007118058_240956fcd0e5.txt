   151→// ─── Helpers ─────────────────────────────────────────────────────────────────
   152→
   153→function formatTime(date: string) {
   154→  const d = new Date(date);
   155→  const now = new Date();
   156→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   157→  if (diff < 1) return 'Just now';
   158→  if (diff < 60) return `${diff}m ago`;
   159→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   160→  return d.toLocaleDateString('en-NG', { day: '2-digit', month: 'short', year: 'numeric' });
   161→}
   162→
   163→function formatNumber(n: number | null | undefined): string {
   164→  if (n == null) return '—';
   165→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k`;
   166→  return n.toLocaleString();
   167→}
   168→
   169→function getWeekCount(events: CampaignEvent[]): number {
   170→  const now = new Date();
   171→  const weekAgo = new Date(now.getTime() - 7 * 86400000);
   172→  return events.filter(e => new Date(e.eventDate) >= weekAgo).length;
   173→}
