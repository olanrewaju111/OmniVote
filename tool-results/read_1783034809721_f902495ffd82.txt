     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { useDashboardStore } from '@/store/dashboard';
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table';
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter,
    16→  DialogHeader, DialogTitle,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    20→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  Search, Users, UserCheck, UserX, Shield, ShieldAlert, Wrench,
    27→  Loader2, FileText, Trash2, Eye, ToggleLeft, X, Pencil,
    28→} from 'lucide-react';
    29→import { cn } from '@/lib/utils';
    30→import { toast } from 'sonner';
    31→
    32→interface AgentUser {
    33→  id: string;
    34→  email: string;
    35→  name: string;
    36→  role: string;
    37→  isOnline: boolean;
    38→  lastSeenAt: string | null;
    39→  createdAt: string;
    40→  _count?: { incidents: number; auditLogs: number };
    41→}
    42→
    43→interface ReportsSlideOver {
    44→  agent: AgentUser | null;
    45→  open: boolean;
    46→}
    47→
    48→export function AgentRoster() {
    49→  const queryClient = useQueryClient();
    50→  const { tenantId } = useDashboardStore();
    51→  const [search, setSearch] = useState('');
    52→  const [searchInput, setSearchInput] = useState('');
    53→  const [roleFilter, setRoleFilter] = useState('ALL');
    54→  const [addOpen, setAddOpen] = useState(false);
    55→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    56→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    57→  const [editOpen, setEditOpen] = useState(false);
    58→  const [editAgent, setEditAgent] = useState<AgentUser | null>(null);
    59→  const [editName, setEditName] = useState('');
    60→  const [editPhone, setEditPhone] = useState('');
    61→  const [editRole, setEditRole] = useState('');
    62→
    63→  // Add agent form state
    64→  const [newName, setNewName] = useState('');
    65→  const [newEmail, setNewEmail] = useState('');
    66→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    67→
    68→  // Fetch agents from proper API
    69→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    70→    queryKey: ['agents', search, roleFilter, tenantId],
    71→    queryFn: () => {
    72→      const params = new URLSearchParams();
    73→      if (search) params.set('search', search);
    74→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    75→      if (tenantId) params.set('tenantId', tenantId);
    76→      return fetch(`/api/agents?${params}`).then(r => r.json());
    77→    },
    78→  });
    79→
    80→  // Add agent mutation
    81→  const addMutation = useMutation({
    82→    mutationFn: (body: { name: string; email: string; role: string }) =>
    83→      fetch(`/api/agents?tenantId=${tenantId}`, {
    84→        method: 'POST',
    85→        headers: { 'Content-Type': 'application/json' },
    86→        body: JSON.stringify(body),
    87→      }).then(async r => {
    88→        const data = await r.json();
    89→        if (!r.ok) throw new Error(data.error || 'Failed to add agent');
    90→        return data;
    91→      }),
    92→    onSuccess: (data) => {
    93→      if (data.error) {
    94→        toast.error(data.error);
    95→        return;
    96→      }
    97→      toast.success(`Agent "${data.user.name}" added successfully`);
    98→      queryClient.invalidateQueries({ queryKey: ['agents'] });
    99→      setAddOpen(false);
   100→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
   101→    },
   102→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to add agent'),
   103→  });
   104→
   105→  // Action mutation (toggle online, remote wipe, change role, delete)
   106→  const actionMutation = useMutation({
   107→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
   108→      fetch(`/api/agents?tenantId=${tenantId}`, {
   109→        method: 'PATCH',
   110→        headers: { 'Content-Type': 'application/json' },
   111→        body: JSON.stringify(body),
   112→      }).then(async r => {
   113→        const data = await r.json();
   114→        if (!r.ok) throw new Error(data.error || 'Action failed');
   115→        return data;
   116→      }),
   117→    onSuccess: (data, variables) => {
   118→      if (data.error) {
   119→        toast.error(data.error);
   120→        return;
   121→      }
   122→      if (variables.action === 'DELETE') {
   123→        toast.success(data.message || 'Agent removed');
   124→      } else if (variables.action === 'REMOTE_WIPE') {
   125→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   126→      } else if (variables.action === 'CHANGE_ROLE') {
   127→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   128→      } else {
   129→        toast.success('Status updated');
   130→      }
   131→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   132→      setConfirmAction({ type: '', agent: null });
   133→    },
   134→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Action failed'),
   135→  });
   136→
   137→  const users = data?.users || [];
   138→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   139→  const online = fieldAgents.filter(u => u.isOnline).length;
   140→
   141→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   142→    acc[u.role] = (acc[u.role] || 0) + 1;
   143→    return acc;
   144→  }, {});
   145→
   146→  // Search with debounce via the query key
   147→  const handleSearch = useCallback(() => {
   148→    setSearch(searchInput);
   149→  }, [searchInput]);
   150→
   151→  // Edit agent mutation
   152→  const editMutation = useMutation({
   153→    mutationFn: (body: { id: string; name?: string; phone?: string; role?: string }) =>
   154→      fetch('/api/tenants/users', {
   155→        method: 'PATCH',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify(body),
   158→      }).then(async r => {
   159→        const data = await r.json();
   160→        if (!r.ok) throw new Error(data.error || 'Failed to update agent');
   161→        return data;
   162→      }),
   163→    onSuccess: () => {
   164→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   165→      toast.success('Agent details updated');
   166→      setEditOpen(false);
   167→      setEditAgent(null);
   168→    },
   169→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to update agent'),
   170→  });
   171→
   172→  const openEditDialog = (agent: AgentUser) => {
   173→    setEditAgent(agent);
   174→    setEditName(agent.name);
   175→    setEditPhone('');
   176→    setEditRole(agent.role);
   177→    setEditOpen(true);
   178→  };
   179→
   180→  const handleEditSave = () => {
   181→    if (!editAgent) return;
   182→    if (!editName.trim()) { toast.error('Name is required'); return; }
   183→    editMutation.mutate({
   184→      id: editAgent.id,
   185→      name: editName.trim(),
   186→      phone: editPhone.trim() || undefined,
   187→      role: editRole,
   188→    });
   189→  };
   190→
   191→  // Handle confirm action
   192→  const handleConfirmAction = () => {
   193→    if (!confirmAction.agent) return;
   194→    if (confirmAction.type === 'DELETE') {
   195→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   196→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   197→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   198→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   199→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   200→    }
   201→  };
   202→
   203→  // Handle add agent
   204→  const handleAddAgent = () => {
   205→    if (!newName.trim() || !newEmail.trim()) {
   206→      toast.error('Name and email are required');
   207→      return;
   208→    }
   209→    if (!newEmail.includes('@')) {
   210→      toast.error('Please enter a valid email');
   211→      return;
   212→    }
   213→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   214→  };
   215→
   216→  return (
   217→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   218→      {/* Header */}
   219→      <div className="flex items-center justify-between flex-wrap gap-3">
   220→        <div>
   221→          <h2 className="text-lg font-semibold flex items-center gap-2">
   222→            <Users className="h-5 w-5 text-emerald" />
   223→            Agent Roster
   224→          </h2>
   225→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   226→        </div>
   227→        <Button
   228→          onClick={() => setAddOpen(true)}
   229→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   230→        >
   231→          <UserCheck className="h-4 w-4" />
   232→          Add Agent
   233→        </Button>
   234→      </div>
   235→
   236→      {/* Summary cards */}
   237→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   238→        <Card className="border-border bg-card/40">
   239→          <CardContent className="p-3.5">
   240→            <div className="flex items-center gap-2 mb-1">
   241→              <Users className="h-4 w-4 text-emerald" />
   242→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   243→            </div>
   244→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   245→          </CardContent>
   246→        </Card>
   247→        <Card className="border-emerald/20 bg-emerald/5">
   248→          <CardContent className="p-3.5">
   249→            <div className="flex items-center gap-2 mb-1">
   250→              <UserCheck className="h-4 w-4 text-emerald" />
   251→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   252→            </div>
   253→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   254→          </CardContent>
   255→        </Card>
   256→        <Card className="border-amber/20 bg-amber/5">
   257→          <CardContent className="p-3.5">
   258→            <div className="flex items-center gap-2 mb-1">
   259→              <UserX className="h-4 w-4 text-amber" />
   260→              <span className="text-[11px] text-muted-foreground">Offline</span>
   261→            </div>
   262→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   263→          </CardContent>
   264→        </Card>
   265→        <Card className="border-border bg-card/40">
   266→          <CardContent className="p-3.5">
   267→            <div className="flex items-center gap-2 mb-1">
   268→              <Shield className="h-4 w-4 text-cyan" />
   269→              <span className="text-[11px] text-muted-foreground">All Users</span>
   270→            </div>
   271→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   272→          </CardContent>
   273→        </Card>
   274→      </div>
   275→
   276→      {/* Role filter + distribution badges */}
   277→      <div className="flex items-center justify-between flex-wrap gap-3">
   278→        <div className="flex items-center gap-2 flex-wrap">
   279→          {Object.entries(roleCounts).map(([role, count]) => (
   280→            <button
   281→              key={role}
   282→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   283→              className="cursor-pointer"
   284→            >
   285→              <Badge
   286→                variant="outline"
   287→                className={cn(
   288→                  'text-[11px] h-6 transition-colors',
   289→                  roleFilter === role ? 'bg-foreground/10' : '',
   290→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   291→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   292→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   293→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   294→                  'border-border text-muted-foreground'
   295→                )}
   296→              >
   297→                {role.replace(/_/g, ' ')}: {count}
   298→              </Badge>
   299→            </button>
   300→          ))}
   301→        </div>
   302→        {roleFilter !== 'ALL' && (
   303→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   304→            <X className="h-3 w-3" /> Clear filter
   305→          </button>
   306→        )}
   307→      </div>
   308→
   309→      {/* Agent table */}
   310→      <Card>
   311→        <CardHeader className="pb-2 pt-4 px-4">
   312→          <div className="flex items-center justify-between gap-3">
   313→            <CardTitle className="text-sm">
   314→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   315→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   316→            </CardTitle>
   317→            <div className="flex items-center gap-2">
   318→              <div className="relative">
   319→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   320→                <Input
   321→                  placeholder="Search agents..."
   322→                  value={searchInput}
   323→                  onChange={(e) => setSearchInput(e.target.value)}
   324→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   325→                  className="pl-8 h-8 w-48 text-xs"
   326→                />
   327→              </div>
   328→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   329→                Search
   330→              </Button>
   331→            </div>
   332→          </div>
   333→        </CardHeader>
   334→        <CardContent className="px-0 pb-0">
   335→          <div className="max-h-[480px] overflow-y-auto">
   336→            {isLoading ? (
   337→              <div className="flex items-center justify-center py-16">
   338→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   339→              </div>
   340→            ) : users.length === 0 ? (
   341→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   342→                <Users className="h-8 w-8 mb-2 opacity-40" />
   343→                <p className="text-sm">No users found</p>
   344→              </div>
   345→            ) : (
   346→              <Table>
   347→                <TableHeader>
   348→                  <TableRow className="border-border hover:bg-transparent">
   349→                    <TableHead className="text-[11px] h-9">User</TableHead>
   350→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   351→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   352→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   353→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   354→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   355→                  </TableRow>
   356→                </TableHeader>
   357→                <TableBody>
   358→                  <AnimatePresence>
   359→                    {users.map(agent => (
   360→                      <motion.tr
   361→                        key={agent.id}
   362→                        initial={{ opacity: 0, y: 4 }}
   363→                        animate={{ opacity: 1, y: 0 }}
   364→                        exit={{ opacity: 0, y: -4 }}
   365→                        className="border-border hover:bg-card/60 transition-colors"
   366→                      >
   367→                        <TableCell className="py-2.5">
   368→                          <div className="flex items-center gap-2.5">
   369→                            <div className="relative">
   370→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   371→                                {agent.name.split(' ').map(n => n[0]).join('')}
   372→                              </div>
   373→                              <span className={cn(
   374→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   375→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   376→                              )} />
   377→                            </div>
   378→                            <div className="min-w-0">
   379→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   380→                              {agent.lastSeenAt && (
   381→                                <span className="text-[10px] text-muted-foreground/60">
   382→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   383→                                </span>
   384→                              )}
   385→                            </div>
   386→                          </div>
   387→                        </TableCell>
   388→                        <TableCell className="py-2.5">
   389→                          <Badge
   390→                            variant="outline"
   391→                            className={cn(
   392→                              'text-[10px] h-5',
   393→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   394→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   395→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   396→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   397→                              'border-border text-muted-foreground'
   398→                            )}
   399→                          >
   400→                            {agent.role.replace(/_/g, ' ')}
   401→                          </Badge>
   402→                        </TableCell>
   403→                        <TableCell className="py-2.5">
   404→                          <button
   405→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   406→                            className="cursor-pointer"
   407→                            disabled={actionMutation.isPending}
   408→                          >
   409→                            <Badge
   410→                              variant="outline"
   411→                              className={cn(
   412→                                'text-[10px] h-5 transition-colors cursor-pointer',
   413→                                agent.isOnline
   414→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   415→                                  : 'border-border text-muted-foreground hover:bg-card'
   416→                              )}
   417→                            >
   418→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   419→                              {agent.isOnline ? 'Online' : 'Offline'}
   420→                            </Badge>
   421→                          </button>
   422→                        </TableCell>
   423→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   424→                        <TableCell className="py-2.5 hidden md:table-cell">
   425→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   426→                            {agent._count?.incidents ?? 0}
   427→                          </span>
   428→                        </TableCell>
   429→                        <TableCell className="py-2.5 text-right">
   430→                          <div className="flex items-center justify-end gap-0.5">
   431→                            <Button
   432→                              variant="ghost"
   433→                              size="sm"
   434→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-cyan"
   435→                              title="Edit Details"
   436→                              onClick={() => openEditDialog(agent)}
   437→                              disabled={agent.role === 'SUPER_ADMIN'}
   438→                            >
   439→                              <Pencil className="h-3.5 w-3.5" />
   440→                            </Button>
   441→                            <Button
   442→                              variant="ghost"
   443→                              size="sm"
   444→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   445→                              title="View Reports"
   446→                              onClick={() => setReportsPanel({ agent, open: true })}
   447→                            >
   448→                              <Eye className="h-3.5 w-3.5" />
   449→                            </Button>
   450→                            <Button
   451→                              variant="ghost"
   452→                              size="sm"
   453→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-amber"
   454→                              title="Remote Wipe Device"
   455→                              onClick={() => setConfirmAction({ type: 'REMOTE_WIPE', agent })}
   456→                              disabled={actionMutation.isPending}
   457→                            >
   458→                              <Wrench className="h-3.5 w-3.5" />
   459→                            </Button>
   460→                            <Button
   461→                              variant="ghost"
   462→                              size="sm"
   463→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-rose"
   464→                              title="Remove Agent"
   465→                              onClick={() => setConfirmAction({ type: 'DELETE', agent })}
   466→                              disabled={actionMutation.isPending || agent.role === 'SUPER_ADMIN'}
   467→                            >
   468→                              <Trash2 className="h-3.5 w-3.5" />
   469→                            </Button>
   470→                          </div>
   471→                        </TableCell>
   472→                      </motion.tr>
   473→                    ))}
   474→                  </AnimatePresence>
   475→                </TableBody>
   476→              </Table>
   477→            )}
   478→          </div>
   479→        </CardContent>
   480→      </Card>
   481→
   482→      {/* ===== ADD AGENT DIALOG ===== */}
   483→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   484→        <DialogContent className="sm:max-w-md">
   485→          <DialogHeader>
   486→            <DialogTitle className="flex items-center gap-2">
   487→              <UserCheck className="h-5 w-5 text-emerald" />
   488→              Add New Agent
   489→            </DialogTitle>
   490→            <DialogDescription>
   491→              Create a new user account. They will be able to login from the main screen using their email.
   492→            </DialogDescription>
   493→          </DialogHeader>
   494→          <div className="space-y-4 py-2">
   495→            <div className="space-y-2">
   496→              <label className="text-xs font-medium text-muted-foreground">Full Name</label>
   497→              <Input
   498→                placeholder="e.g. Adebayo Johnson"
   499→                value={newName}
   500→                onChange={(e) => setNewName(e.target.value)}
   501→                className="h-9 text-sm"
   502→              />
   503→            </div>
   504→            <div className="space-y-2">
   505→              <label className="text-xs font-medium text-muted-foreground">Email Address</label>
   506→              <Input

   668→            <DialogDescription>

   670→            </DialogDescription>
   671→          </DialogHeader>
   672→          <AgentReportsPanel agentId={reportsPanel.agent?.id || ''} />
   673→        </DialogContent>
   674→      </Dialog>
   675→    </div>
   676→  );

   679→// Sub-component: fetches and displays reports for a specific agent
   680→function AgentReportsPanel({ agentId }: { agentId: string }) {
   681→  const { data, isLoading } = useQuery<{ incidents: Array<{
   682→    id: string; type: string; severity: string; status: string;
   683→    description: string; submittedAt: string;
   684→    pollingUnit: { name: string; state: string; lga: string } | null;
   685→  }>; total: number }>({
   686→    queryKey: ['agent-reports', agentId],
   687→    queryFn: () => fetch(`/api/incidents?reporterId=${agentId}&limit=50`).then(r => r.json()),
   688→    enabled: !!agentId,
   689→  });
   690→
   691→  if (isLoading) {
   692→    return (
   693→      <div className="flex items-center justify-center py-12">
   694→        <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   695→      </div>
   696→    );
   697→  }
   698→
   699→  const incidents = data?.incidents || [];
   700→
   701→  if (incidents.length === 0) {
   702→    return (
   703→      <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   704→        <FileText className="h-8 w-8 mb-2 opacity-40" />
   705→        <p className="text-sm">No reports submitted yet</p>
   706→      </div>
   707→    );
   708→  }
   709→
   710→  const severityColors: Record<string, string> = {
   711→    LOW: 'border-emerald/30 text-emerald bg-emerald/10',
   712→    MEDIUM: 'border-amber/30 text-amber bg-amber/10',
   713→    HIGH: 'border-orange/30 text-orange bg-orange/10',
   714→    CRITICAL: 'border-rose/30 text-rose bg-rose/10',
   715→  };
   716→
   717→  const statusColors: Record<string, string> = {
   718→    PENDING: 'text-muted-foreground',
   719→    REVIEWED: 'text-cyan',
   720→    ESCALATED: 'text-amber',
   721→    DISMISSED: 'text-muted-foreground/50',
   722→    QUARANTINED: 'text-rose',
   723→  };
   724→
   725→  return (
   726→    <div className="flex-1 overflow-y-auto space-y-2 pr-1">
   727→      <p className="text-xs text-muted-foreground mb-2">{data?.total || 0} total reports</p>
   728→      {incidents.map(inc => (
   729→        <div key={inc.id} className="rounded-lg border border-border bg-card/40 p-3 space-y-1.5">
   730→          <div className="flex items-center justify-between gap-2">
   731→            <div className="flex items-center gap-2">
   732→              <Badge variant="outline" className={cn('text-[10px] h-5', severityColors[inc.severity] || '')}>
   733→                {inc.severity}
   734→              </Badge>
   735→              <Badge variant="outline" className="text-[10px] h-5 border-border text-muted-foreground">
   736→                {inc.type.replace(/_/g, ' ')}
   737→              </Badge>
   738→            </div>
   739→            <div className="flex items-center gap-2">
   740→              <span className={cn('text-[10px] font-medium', statusColors[inc.status] || '')}>
   741→                {inc.status}
   742→              </span>
   743→            </div>
   744→          </div>
   745→          <p className="text-xs text-foreground/80 leading-relaxed">{inc.description}</p>
   746→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   747→            {inc.pollingUnit && (
   748→              <span className="flex items-center gap-1">
   749→                <span className="w-1 h-1 rounded-full bg-muted-foreground/40" />
   750→                {inc.pollingUnit.name}, {inc.pollingUnit.state}
   751→              </span>
   752→            )}
   753→            <span>{new Date(inc.submittedAt).toLocaleString()}</span>
   754→          </div>
   755→        </div>
   756→      ))}
   757→    </div>
   758→  );
