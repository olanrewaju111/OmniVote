    16→function randomInt(min: number, max: number) { return Math.floor(Math.random() * (max - min + 1)) + min; }
    17→function randomFloat(min: number, max: number, dec = 2) { return parseFloat((Math.random() * (max - min) + min).toFixed(dec)); }
    18→function pick<T>(arr: T[]): T { return arr[Math.floor(Math.random() * arr.length)]; }

    20→function generatePartyResults(total: number) {
    21→  let remaining = total;
    22→  const results: { party: string; votes: number }[] = [];
    23→  for (let i = 0; i < PARTIES.length - 1; i++) {
    24→    const share = randomInt(5, Math.floor(remaining * 0.5));
    25→    results.push({ party: PARTIES[i], votes: share });
    26→    remaining -= share;
    27→  }
    28→  results.push({ party: PARTIES[PARTIES.length - 1], votes: remaining });
    29→  return results.sort((a, b) => b.votes - a.votes);
    30→}

    88→async function seedComparisons() {
    89→  console.log('Seeding result comparisons...');
    90→
    91→  for (const tenant of TENANTS) {
    92→    const elections = await db.election.findMany({ where: { tenantId: tenant.id }, select: { id: true } });
    93→    if (!elections.length) continue;
    94→
    95→    const pvts = await db.pvtSubmission.findMany({
    96→      where: { tenantId: tenant.id, electionId: elections[0].id },
    97→      select: { id: true, pollingUnitId: true, partyResults: true, totalVotesCast: true },
    98→    });
    99→
   100→    const officials = await db.electionResult.findMany({
   101→      where: { tenantId: tenant.id },
   102→      select: { id: true, pollingUnitId: true, partyResults: true, totalVotesCast: true },
   103→    });
   104→
   105→    const officialByUnit = new Map(officials.map(o => [o.pollingUnitId, o]));
   106→    let created = 0;
   107→
   108→    for (const pvt of pvts) {
   109→      const official = officialByUnit.get(pvt.pollingUnitId);
   110→      if (!official) continue;
   111→
   112→      const pvtResults: { party: string; votes: number }[] = JSON.parse(pvt.partyResults || '[]');
   113→      const offResults: { party: string; votes: number }[] = JSON.parse(official.partyResults || '[]');
   114→
   115→      const partyDeltas = pvtResults.map(pr => {
   116→        const off = offResults.find(o => o.party === pr.party);
   117→        const offVotes = off?.votes || 0;
   118→        const delta = offVotes - pr.votes;
   119→        return { party: pr.party, pvtVotes: pr.votes, officialVotes: offVotes, delta, deltaPct: pr.votes > 0 ? Math.abs(delta / pr.votes) * 100 : 0 };
   120→      });
   121→
   122→      const totalDelta = official.totalVotesCast - pvt.totalVotesCast;
   123→      const deltaPct = pvt.totalVotesCast > 0 ? Math.abs(totalDelta / pvt.totalVotesCast) * 100 : 0;
   124→      const isAnomaly = deltaPct > 5;
   125→
   126→      await db.resultComparison.create({
   127→        data: {
   128→          tenantId: tenant.id,
   129→          electionId: elections[0].id,
   130→          pollingUnitId: pvt.pollingUnitId,
   131→          pvtSubmissionId: pvt.id,
   132→          officialResultId: official.id,
   133→          partyDeltas: JSON.stringify(partyDeltas),
   134→          totalPvtVotes: pvt.totalVotesCast,
   135→          totalOfficialVotes: official.totalVotesCast,
   136→          totalDelta,
   137→          deltaPct,
   138→          isAnomaly,
   139→          anomalyReason: isAnomaly ? (deltaPct > 10 ? 'exceeds_10pct' : 'exceeds_5pct') : null,
   140→        },
   141→      });
   142→      created++;
   143→    }
   144→    console.log(`  ${tenant.name}: ${created} comparisons`);
   145→  }
   146→}

   511→async function seedGeofence() {
   512→  console.log('Seeding geofence zones...');
   513→
   514→  const ZONES = [
   515→    { name: 'Lagos Island Central', state: 'Lagos', lga: 'Lagos Island', lat: 6.4541, lng: 3.3947, r: 500 },
   516→    { name: 'Kano Municipal HQ', state: 'Kano', lga: 'Kano Municipal', lat: 12.0022, lng: 8.5920, r: 800 },
   517→    { name: 'Enugu Urban Zone A', state: 'Enugu', lga: 'Enugu North', lat: 6.4424, lng: 7.4930, r: 600 },
   518→    { name: 'Abuja Central District', state: 'Abuja FCT', lga: 'AMAC', lat: 9.0579, lng: 7.4951, r: 1000 },
   519→    { name: 'Rivers Port Harcourt Metro', state: 'Rivers', lga: 'Port Harcourt', lat: 4.8156, lng: 7.0498, r: 700 },
   520→  ];
   521→
   522→  for (const tenant of TENANTS) {
   523→    const elections = await db.election.findMany({ where: { tenantId: tenant.id }, select: { id: true } });
   524→    if (!elections.length) continue;
   525→    const units = await db.pollingUnit.findMany({ where: { electionId: elections[0].id }, select: { id: true } });
   526→    const agents = await db.user.findMany({ where: { tenantId: tenant.id, role: 'FIELD_AGENT' }, select: { id: true } });
   527→
   528→    for (const z of ZONES) {
   529→      const zone = await db.geofenceZone.create({
   530→        data: {
   531→          tenantId: tenant.id,
   532→          name: z.name,
   533→          state: z.state,
   534→          lga: z.lga,
   535→          centerLat: z.lat,
   536→          centerLng: z.lng,
   537→          radiusMeters: z.r,
   538→          pollingUnitIds: JSON.stringify(units.slice(0, randomInt(3, 8)).map(u => u.id)),
   539→          assignedAgentIds: JSON.stringify(agents.slice(0, randomInt(2, 5)).map(a => a.id)),
   540→          isActive: true,
   541→          checkInIntervalMin: randomInt(30, 120),
   542→          maxMissedCheckIns: randomInt(2, 4),
   543→        },
   544→      });
   545→
   546→      // Create check-ins for agents in this zone
   547→      const zoneAgents = agents.slice(0, randomInt(2, 5));
   548→      for (let j = 0; j < zoneAgents.length; j++) {
   549→        const isInside = Math.random() > 0.2;

   551→          data: {

   557→            isInsideZone: isInside,

   560→            accuracyMeters: randomFloat(5, 50),
   561→            status: isInside ? 'CHECKED_IN' : (Math.random() > 0.9 ? 'SOS_TRIGGERED' : 'CHECKED_OUT'),
   562→            checkedInAt: new Date(Date.now() - randomInt(5, 120) * 60000),
   563→          },
   564→        });
   565→      }
   566→
   567→      // Create dead-man's switches for agents
   568→      for (let j = 0; j < Math.min(3, zoneAgents.length); j++) {

   570→        await db.deadMansSwitch.create({
   571→          data: {
   572→            tenantId: tenant.id,
   573→            agentId: zoneAgents[j].id,
   574→            geofenceZoneId: zone.id,
   575→            isActive: !isOverdue,
   576→            checkInDeadline: new Date(Date.now() + (isOverdue ? -randomInt(10, 60) : randomInt(20, 90)) * 60000),
   577→            lastCheckInAt: new Date(Date.now() - randomInt(30, 180) * 60000),
   578→            missedCheckIns: isOverdue ? randomInt(1, 3) : 0,
   579→            escalationLevel: isOverdue ? randomInt(1, 3) : 0,
   580→            autoSOSTriggered: isOverdue && Math.random() > 0.5,
   581→          },
   582→        });
   583→      }
   584→    }
   585→    console.log(`  ${tenant.name}: ${ZONES.length} zones with check-ins and switches`);
   586→  }
   587→}
