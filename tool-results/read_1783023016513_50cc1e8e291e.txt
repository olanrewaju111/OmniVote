   136→
   137→const RISK_CELL_COLORS: Record<string, { bg: string; text: string }> = {
   138→  LOW: { bg: 'rgba(16,185,129,0.20)', text: '#10b981' },
   139→  MEDIUM: { bg: 'rgba(245,158,11,0.30)', text: '#f59e0b' },
   140→  HIGH: { bg: 'rgba(239,68,68,0.30)', text: '#ef4444' },
   141→  CRITICAL: { bg: 'rgba(220,38,38,0.50)', text: '#dc2626' },
   142→};
   143→
   144→function riskCellColor(level: string) {
   145→  return RISK_CELL_COLORS[level] ?? RISK_CELL_COLORS.LOW;
   146→}
   147→
   148→function riskBarColor(score: number): string {
   149→  if (score >= 0.8) return '#ef4444';
   150→  if (score >= 0.6) return '#f97316';
   151→  if (score >= 0.3) return '#f59e0b';
   152→  return '#10b981';
   153→}
   154→
   155→function riskBadgeVariant(level: string): string {
   156→  switch (level) {
   157→    case 'CRITICAL':
   158→      return 'bg-rose-600/20 text-rose-400 border-rose-600/30';
   159→    case 'HIGH':
   160→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   161→    case 'MEDIUM':
   162→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   163→    default:
   164→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   165→  }
   166→}
   167→
   168→function statusBadgeVariant(status: string): string {
   169→  switch (status) {
   170→    case 'RUNNING':
   171→      return 'bg-cyan-500/20 text-cyan-400 border-cyan-500/30';
   172→    case 'COMPLETED':
   173→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   174→    case 'PENDING':
   175→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   176→    case 'FAILED':
   177→      return 'bg-rose-500/20 text-rose-400 border-rose-500/30';
   178→    default:
   179→      return 'bg-muted text-muted-foreground border-border';
   180→  }
   181→}
   182→
   183→function formatDate(dateStr: string): string {
   184→  const d = new Date(dateStr);
   185→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   186→}
   187→
   188→function formatDay(dateStr: string): string {
   189→  const d = new Date(dateStr);
   190→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
   191→}

   210→
   211→export function FlashpointWargame() {
   212→  const { tenantId } = useDashboardStore();
   213→  const queryClient = useQueryClient();
   214→
   215→  // Wargame state
   216→  const [selectedScenarioId, setSelectedScenarioId] = useState<string | null>(null);
   217→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   218→  const [newTitle, setNewTitle] = useState('');
   219→  const [newDescription, setNewDescription] = useState('');
   220→  const [newRole, setNewRole] = useState('RED_TEAM');
   221→  const [newParams, setNewParams] = useState('{\n  \n}');
   222→  const [completeScore, setCompleteScore] = useState('');
   223→
   224→  /* ---- Data fetching ---- */
   225→  const { data, isLoading, refetch } = useQuery<FlashpointData>({
   226→    queryKey: ['flashpoint', tenantId],
   227→    queryFn: () =>
   228→      fetch(`/api/flashpoint?tenantId=${tenantId}`).then((r) => r.json()),
   229→    refetchInterval: 20000,
   230→  });
   231→
   232→  const mutation = useMutation({
   233→    mutationFn: (body: any) =>
   234→      fetch('/api/flashpoint?tenantId=' + tenantId, {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify(body),
   238→      }).then((r) => r.json()),
   239→    onSuccess: () => {
   240→      queryClient.invalidateQueries({ queryKey: ['flashpoint', tenantId] });
   241→    },
   242→  });
   243→
   244→  /* ---- Derived data ---- */
   245→  const stats = data?.stats;
   246→  const forecasts = data?.forecasts ?? [];
   247→  const scenarios = data?.scenarios ?? [];
   248→  const heatmapData = data?.heatmapData ?? [];
   249→  const topFactors = data?.topContributingFactors ?? [];
   250→
   251→  // Unique states sorted alphabetically
   252→  const uniqueStates = useMemo(() => {
   253→    const set = new Set(heatmapData.map((d) => d.state));
   254→    return Array.from(set).sort();
   255→  }, [heatmapData]);
   256→
   257→  // Unique dates sorted
   258→  const uniqueDates = useMemo(() => {
   259→    const set = new Set(heatmapData.map((d) => d.date));
   260→    return Array.from(set).sort();
   261→  }, [heatmapData]);
   262→
   263→  // Build lookup map for heatmap cells
   264→  const heatmapMap = useMemo(() => {
   265→    const map = new Map<string, (typeof heatmapData)[number]>();
   266→    for (const d of heatmapData) {
   267→      map.set(`${d.state}|${d.date}`, d);
   268→    }
   269→    return map;
   270→  }, [heatmapData]);
   271→
   272→  // State risk averages for bar chart
   273→  const stateRisks = useMemo(() => {
   274→    const map = new Map<string, { overall: number; level: string; count: number }>();
   275→    for (const f of forecasts) {
   276→      const existing = map.get(f.state);
   277→      if (existing) {
   278→        existing.overall = (existing.overall * existing.count + f.riskScores.overall) / (existing.count + 1);
   279→        existing.count += 1;
   280→        // Upgrade risk level if higher
   281→        const order = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
   282→        if (order.indexOf(f.riskLevel) > order.indexOf(existing.level)) {
   283→          existing.level = f.riskLevel;
   284→        }
   285→      } else {
   286→        map.set(f.state, { overall: f.riskScores.overall, level: f.riskLevel, count: 1 });
   287→      }
   288→    }
   289→    return Array.from(map.entries())
   290→      .map(([state, v]) => ({ state, ...v }))
   291→      .sort((a, b) => b.overall - a.overall);
   292→  }, [forecasts]);
   293→
   294→  // Top factor
   295→  const topFactor = topFactors.length > 0 ? topFactors[0].factor : '—';
   296→
   297→  // Selected scenario
   298→  const selectedScenario = scenarios.find((s) => s.id === selectedScenarioId) ?? null;
   299→
   300→  // Max factor count for bar scaling
   301→  const maxFactorCount = Math.max(...topFactors.map((f) => f.count), 1);
   302→
   303→  /* ---- Handlers ---- */
   304→  const handleCreateScenario = () => {
   305→    if (!newTitle.trim()) {
   306→      toast.error('Title is required');
   307→      return;
   308→    }
   309→    let parsedParams: any = {};
   310→    try {
   311→      parsedParams = newParams.trim() ? JSON.parse(newParams) : {};
   312→    } catch {
   313→      toast.error('Invalid JSON in parameters');
   314→      return;
   315→    }
   316→    mutation.mutate(
   317→      { action: 'RUN_WARGAME', title: newTitle, description: newDescription, playerRole: newRole, parameters: parsedParams },
   318→      {
   319→        onSuccess: () => {
   320→          toast.success('Scenario created successfully');
   321→          setCreateDialogOpen(false);
   322→          setNewTitle('');
   323→          setNewDescription('');
   324→          setNewRole('RED_TEAM');
   325→          setNewParams('{\n  \n}');
   326→        },
   327→        onError: () => toast.error('Failed to create scenario'),
   328→      },
   329→    );
   330→  };
   331→
   332→  const handleCompleteScenario = () => {
   333→    if (!selectedScenarioId || !completeScore) {
   334→      toast.error('Score is required');
   335→      return;
   336→    }
   337→    const score = parseFloat(completeScore);
   338→    if (isNaN(score) || score < 0 || score > 100) {
   339→      toast.error('Score must be between 0 and 100');
   340→      return;
   341→    }
   342→    mutation.mutate(
   343→      { action: 'COMPLETE_WARGAME', scenarioId: selectedScenarioId, score },
   344→      {
   345→        onSuccess: () => {
   346→          toast.success('Scenario completed');
   347→          setCompleteScore('');
   348→          setSelectedScenarioId(null);
   349→        },
   350→        onError: () => toast.error('Failed to complete scenario'),
   351→      },
   352→    );
   353→  };
   354→
   355→  /* ---- Loading ---- */
   356→  if (isLoading) {
   357→    return (
   358→      <div className="h-full flex flex-col items-center justify-center gap-3">
   359→        <Loader2 className="size-8 text-emerald-500 animate-spin" />
   360→        <p className="text-sm text-muted-foreground">Loading flashpoint data…</p>
   361→      </div>
   362→    );
   363→  }
   364→
   365→  return (
   366→    <div className="h-full flex flex-col gap-4">
   367→      <Tabs defaultValue="forecast" className="h-full flex flex-col gap-3">
   368→        <TabsList>
   369→          <TabsTrigger value="forecast" className="gap-1.5">
   370→            <Flame className="size-4" />
   371→            Risk Forecast
   372→          </TabsTrigger>
   373→          <TabsTrigger value="wargame" className="gap-1.5">
   374→            <Swords className="size-4" />
   375→            Wargame Simulator
   376→          </TabsTrigger>
   377→        </TabsList>
   378→
   379→        {/* ================================================================ */}
   380→        {/* TAB 1 — Risk Forecast                                           */}
   381→        {/* ================================================================ */}
   382→        <TabsContent value="forecast" className="flex-1 min-h-0 overflow-auto">
   383→          <motion.div
   384→            variants={staggerContainer}
   385→            initial="initial"
   386→            animate="animate"
   387→            className="flex flex-col gap-4"
   388→          >
   389→            {/* --- Stats Row --- */}
   390→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   391→              <motion.div variants={fadeIn}>
   392→                <Card className="h-full">
   393→                  <CardContent className="p-4 flex flex-col gap-1">
   394→                    <div className="flex items-center gap-2 text-muted-foreground">
   395→                      <Activity className="size-4" />
   396→                      <span className="text-xs font-medium">Total Forecasts</span>
   397→                    </div>
   398→                    <p className="text-2xl font-bold">{stats?.totalForecasts ?? 0}</p>
   399→                  </CardContent>
   400→                </Card>
   401→              </motion.div>
   402→
   403→              <motion.div variants={fadeIn}>
   404→                <Card className="h-full border-red-500/20">
   405→                  <CardContent className="p-4 flex flex-col gap-1">
   406→                    <div className="flex items-center gap-2 text-red-400">
   407→                      <AlertTriangle className="size-4" />
   408→                      <span className="text-xs font-medium">High Risk States</span>
   409→                    </div>
   410→                    <p className="text-2xl font-bold text-red-400">{stats?.highRiskStates?.length ?? 0}</p>
   411→                  </CardContent>
   412→                </Card>
   413→              </motion.div>
   414→
   415→              <motion.div variants={fadeIn}>
   416→                <Card className="h-full border-rose-500/20">
   417→                  <CardContent className="p-4 flex flex-col gap-1">
   418→                    <div className="flex items-center gap-2 text-rose-400">
   419→                      <ShieldAlert className="size-4" />
   420→                      <span className="text-xs font-medium">Critical</span>
   421→                    </div>
   422→                    <p className="text-2xl font-bold text-rose-400">{stats?.byRiskLevel?.CRITICAL ?? 0}</p>
   423→                  </CardContent>
   424→                </Card>
   425→              </motion.div>
   426→
   427→              <motion.div variants={fadeIn}>
   428→                <Card className="h-full">
   429→                  <CardContent className="p-4 flex flex-col gap-1">
   430→                    <div className="flex items-center gap-2 text-cyan-400">
   431→                      <TrendingUp className="size-4" />
   432→                      <span className="text-xs font-medium">Avg Confidence</span>
   433→                    </div>
   434→                    <p className="text-2xl font-bold text-cyan-400">
   435→                      {stats?.avgConfidence != null ? `${(stats.avgConfidence * 100).toFixed(0)}%` : '—'}
   436→                    </p>
   437→                  </CardContent>
   438→                </Card>
   439→              </motion.div>
   440→
   441→              <motion.div variants={fadeIn}>
   442→                <Card className="h-full">
   443→                  <CardContent className="p-4 flex flex-col gap-1">
   444→                    <div className="flex items-center gap-2 text-amber-400">
   445→                      <Zap className="size-4" />
   446→                      <span className="text-xs font-medium">Top Factor</span>
   447→                    </div>
   448→                    <p className="text-sm font-semibold truncate">{topFactor}</p>
   449→                  </CardContent>
   450→                </Card>
   451→              </motion.div>
   452→            </div>
   453→
   454→            {/* --- Main Two Column Layout --- */}
   455→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 min-h-0">
   456→              {/* LEFT: 7-Day Heatmap */}
   457→              <motion.div variants={fadeIn}>
   458→                <Card className="h-full flex flex-col">
   459→                  <CardContent className="p-4 flex flex-col gap-3 flex-1 min-h-0">
   460→                    <div className="flex items-center justify-between">
   461→                      <h3 className="text-sm font-semibold flex items-center gap-2">
   462→                        <CalendarDays className="size-4 text-emerald-400" />
   463→                        7-Day Risk Heatmap
   464→                      </h3>
   465→                      <Button variant="ghost" size="sm" onClick={() => refetch()} className="h-7 px-2 text-xs">
   466→                        <Crosshair className="size-3 mr-1" />
   467→                        Refresh
   468→                      </Button>
   469→                    </div>
   470→
   471→                    {uniqueStates.length === 0 ? (
   472→                      <div className="flex-1 flex items-center justify-center text-muted-foreground text-sm">
   473→                        No heatmap data available
   474→                      </div>
   475→                    ) : (
   476→                      <ScrollArea className="flex-1">
   477→                        <div className="min-w-fit">
   478→                          {/* Header row: empty + dates */}
   479→                          <div
   480→                            className="grid gap-[2px] mb-[2px]"
   481→                            style={{
   482→                              gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   483→                            }}
   484→                          >
   485→                            <div className="text-[10px] text-muted-foreground font-medium px-1" />
   486→                            {uniqueDates.map((date) => (
   487→                              <div
   488→                                key={date}
   489→                                className="text-[10px] text-muted-foreground font-medium text-center truncate px-1"
   490→                              >
   491→                                {formatDay(date)}
   492→                              </div>
   493→                            ))}
   494→                          </div>
   495→
   496→                          {/* Data rows: state + cells */}
   497→                          {uniqueStates.map((state) => (
   498→                            <div
   499→                              key={state}
   500→                              className="grid gap-[2px] mb-[2px]"
   501→                              style={{
   502→                                gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   503→                              }}
   504→                            >
   505→                              <div className="flex items-center text-xs font-medium truncate px-1 text-foreground/80">
   506→                                {state}
   507→                              </div>
   508→                              {uniqueDates.map((date) => {
   509→                                const cell = heatmapMap.get(`${state}|${date}`);
   510→                                const colors = cell ? riskCellColor(cell.riskLevel) : { bg: 'rgba(255,255,255,0.05)', text: '#666' };
   511→                                return (
   512→                                  <TooltipProvider key={`${state}|${date}`}>
   513→                                    <Tooltip>
   514→                                      <TooltipTrigger asChild>
   515→                                        <div
   516→                                          className="flex items-center justify-center rounded-sm text-[11px] font-semibold cursor-default transition-transform hover:scale-110 h-8"
   517→                                          style={{
   518→                                            backgroundColor: colors.bg,
   519→                                            color: colors.text,
   520→                                          }}
   521→                                        >
   522→                                          {cell ? cell.overall.toFixed(1) : '—'}
   523→                                        </div>
   524→                                      </TooltipTrigger>
   525→                                      {cell && (
   526→                                        <TooltipContent side="top" className="bg-popover text-popover-foreground border">
   527→                                          <div className="text-xs space-y-1 min-w-[140px]">
   528→                                            <div className="font-semibold">{state} — {formatDate(date)}</div>
   529→                                            <Separator className="my-1" />
   530→                                            <div className="flex justify-between">
   531→                                              <span className="text-muted-foreground">Violence</span>
   532→                                              <span className="font-mono">{cell.violence.toFixed(2)}</span>
   533→                                            </div>
   534→                                            <div className="flex justify-between">
   535→                                              <span className="text-muted-foreground">Intimidation</span>
   536→                                              <span className="font-mono">{cell.intimidation.toFixed(2)}</span>
   537→                                            </div>
   538→                                            <div className="flex justify-between">
   539→                                              <span className="text-muted-foreground">Logistics</span>
   540→                                              <span className="font-mono">{cell.logistics.toFixed(2)}</span>
   541→                                            </div>
   542→                                            <Separator className="my-1" />
   543→                                            <div className="flex justify-between font-semibold">
   544→                                              <span>Overall</span>
   545→                                              <span style={{ color: colors.text }}>{cell.overall.toFixed(2)}</span>
   546→                                            </div>
   547→                                            <div className="flex justify-between">
   548→                                              <span>Level</span>
   549→                                              <Badge className={cn('text-[10px] h-4', riskBadgeVariant(cell.riskLevel))}>
   550→                                                {cell.riskLevel}
   551→                                              </Badge>
   552→                                            </div>
   553→                                          </div>
   554→                                        </TooltipContent>
   555→                                      )}
   556→                                    </Tooltip>
   557→                                  </TooltipProvider>
   558→                                );
   559→                              })}
   560→                            </div>
   561→                          ))}
   562→                        </div>
   563→                      </ScrollArea>
   564→                    )}
   565→
