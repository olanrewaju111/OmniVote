   129→// ── Helpers ──────────────────────────────────────────────────────────
   130→function formatRelativeTime(d: string | Date | null): string {
   131→  if (!d) return 'Never';
   132→  const date = new Date(d);
   133→  const now = Date.now();
   134→  const diff = Math.floor((now - date.getTime()) / 60000);
   135→  if (diff < 1) return 'Just now';
   136→  if (diff < 60) return `${diff}m ago`;
   137→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   138→  return date.toLocaleDateString('en-NG', { day: '2-digit', month: 'short' });
   139→}
   140→
   141→function formatDeadlineCountdown(deadline: string): { text: string; isOverdue: boolean } {
   142→  const diff = new Date(deadline).getTime() - Date.now();
   143→  const mins = Math.floor(Math.abs(diff) / 60000);
   144→  if (diff > 0) {
   145→    return { text: `${mins}m left`, isOverdue: false };
   146→  }
   147→  return { text: `OVERDUE by ${mins}m`, isOverdue: true };
   148→}

   150→function batteryIcon(level: number | null) {
   151→  if (level === null) return <Battery className="h-3.5 w-3.5 text-muted-foreground" />;
   152→  if (level > 60) return <BatteryFull className="h-3.5 w-3.5 text-emerald" />;
   153→  if (level > 20) return <BatteryWarning className="h-3.5 w-3.5 text-amber" />;
   154→  return <BatteryLow className="h-3.5 w-3.5 text-rose" />;
   155→}

   157→function batteryColor(level: number | null) {
   158→  if (level === null) return 'text-muted-foreground';
   159→  if (level > 60) return 'text-emerald';
   160→  if (level > 20) return 'text-amber';
   161→  return 'text-rose';
   162→}

   164→function statusBadge(status: string) {
   165→  switch (status) {
   166→    case 'CHECKED_IN':
   167→      return <Badge className="border-emerald/30 bg-emerald/10 text-emerald text-[10px]">CHECKED IN</Badge>;
   168→    case 'CHECKED_OUT':
   169→      return <Badge className="border-cyan/30 bg-cyan/10 text-cyan text-[10px]">CHECKED OUT</Badge>;
   170→    case 'SOS_TRIGGERED':
   171→      return <Badge className="border-rose/30 bg-rose/10 text-rose text-[10px] glow-rose animate-pulse-dot">SOS</Badge>;
   172→    case 'EXPIRED':
   173→      return <Badge className="border-amber/30 bg-amber/10 text-amber text-[10px]">EXPIRED</Badge>;
   174→    default:
   175→      return <Badge variant="outline" className="text-[10px]">{status}</Badge>;
   176→  }
   177→}

   756→// ── KPI Card Sub-component ───────────────────────────────────────────
   757→function KpiCard({ label, value, icon, accent, delay }: {
   758→  label: string; value: number; icon: React.ReactNode; accent: 'emerald' | 'rose' | 'amber' | 'cyan'; delay: number;
   759→}) {
   760→  const glowClass = accent === 'rose' ? 'glow-rose' : accent === 'amber' ? 'glow-amber' : accent === 'emerald' ? 'glow-emerald' : '';
   761→  return (
   762→    <motion.div variants={item} transition={{ delay: delay * 0.05 }}>
   763→      <Card className={cn('bg-card/40 border-border rounded-xl', glowClass, value > 0 && (accent === 'rose' || accent === 'amber') ? '' : '')}>
   764→        <CardContent className="p-4 flex items-center gap-3">
   765→          <div className={cn(
   766→            'flex items-center justify-center h-9 w-9 rounded-lg shrink-0',
   767→            accent === 'emerald' && 'bg-emerald/10',
   768→            accent === 'rose' && 'bg-rose/10',
   769→            accent === 'amber' && 'bg-amber/10',
   770→            accent === 'cyan' && 'bg-cyan/10',
   771→          )}>
   772→            {icon}
   773→          </div>
   774→          <div>
   775→            <div className="text-xl font-bold leading-none">{value}</div>
   776→            <div className="text-[11px] text-muted-foreground mt-0.5">{label}</div>
   777→          </div>
   778→        </CardContent>
   779→      </Card>
   780→    </motion.div>
   781→  );
