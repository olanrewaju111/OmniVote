   168→function generateNigerianNumbers(count: number): string[] {
   169→  const prefixes = ['0703', '0803', '0806', '0809', '0810', '0811', '0812', '0813', '0814', '0815', '0816', '0817', '0818', '0902', '0903', '0905', '0906', '0907', '0908', '0909', '0913', '0915', '0916'];
   170→  const numbers: string[] = [];
   171→  for (let i = 0; i < count; i++) {
   172→    const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
   173→    const suffix = String(Math.floor(Math.random() * 10_000_000)).padStart(7, '0');
   174→    numbers.push(`${prefix}${suffix}`);
   175→  }
   176→  return numbers;
   177→}
   178→
   179→function formatNumber(n: number): string {
   180→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   181→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   182→  return String(n);
   183→}
   184→
   185→function formatDate(d: string | null) {
   186→  if (!d) return '—';
   187→  return new Date(d).toLocaleDateString('en-NG', {
   188→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   189→  });
   190→}
   191→
   192→function segmentLabel(seg: string) {
   193→  return SEGMENT_OPTIONS.find(s => s.value === seg)?.label ?? seg;
   194→}
