   152→// ─── Helpers ─────────────────────────────────────────────────────────────────
   153→
   154→function formatTime(date: string) {
   155→  const d = new Date(date);
   156→  const now = new Date();
   157→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   158→  if (diff < 1) return 'Just now';
   159→  if (diff < 60) return `${diff}m ago`;
   160→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   161→  return d.toLocaleDateString('en-NG', { day: '2-digit', month: 'short', year: 'numeric' });
   162→}
   163→
   164→function formatNumber(n: number | null | undefined): string {
   165→  if (n == null) return '—';
   166→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k`;
   167→  return n.toLocaleString();
   168→}
   169→
   170→function getWeekCount(events: CampaignEvent[]): number {
   171→  const now = new Date();
   172→  const weekAgo = new Date(now.getTime() - 7 * 86400000);
   173→  return events.filter(e => new Date(e.eventDate) >= weekAgo).length;
   174→}
