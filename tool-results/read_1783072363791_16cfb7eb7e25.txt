   111→function statusConfig(status: string) {
   112→  switch (status) {
   113→    case 'DRAFT':
   114→      return { label: 'Draft', cls: 'border-zinc-500/30 bg-zinc-500/10 text-zinc-400' };
   115→    case 'REVIEWED':

   117→    case 'CERTIFIED':
   118→      return { label: 'Certified', cls: 'border-emerald-500/30 bg-emerald-500/10 text-emerald-400' };
   119→    case 'DISMISSED':
   120→      return { label: 'Dismissed', cls: 'border-rose-500/30 bg-rose-500/10 text-rose-400' };
   121→    default:
   122→      return { label: status, cls: 'border-border bg-card text-muted-foreground' };

   126→function confidenceColor(val: number) {
   127→  if (val >= 80) return 'bg-emerald-500';
   128→  if (val >= 60) return 'bg-amber-500';
   129→  return 'bg-rose-500';
   130→}

   132→function formatBytes(bytes: number) {
   133→  if (bytes < 1024) return `${bytes} B`;
   134→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   135→  return `${(bytes / 1048576).toFixed(1)} MB`;
   136→}
   137→
   138→function formatDuration(ms: number) {
   139→  if (ms < 1000) return `${ms}ms`;
   140→  return `${(ms / 1000).toFixed(1)}s`;
   141→}
   142→
   143→function timeAgo(dateStr: string | null) {
   144→  if (!dateStr) return 'N/A';
   145→  const d = new Date(dateStr);
   146→  const now = new Date();
   147→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   148→  if (diff < 1) return 'Just now';
   149→  if (diff < 60) return `${diff}m ago`;
   150→  if (diff < 1440) return `${Math.floor(diff / 60)}h ${diff % 60}m ago`;
   151→  return `${Math.floor(diff / 1440)}d ago`;
   152→}
   153→
   154→function formatDate(dateStr: string) {
   155→  return new Date(dateStr).toLocaleDateString('en-US', {
   156→    month: 'short', day: 'numeric', year: 'numeric',
   157→  });
   158→}
