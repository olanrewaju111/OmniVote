     1→'use client';
     2→
     3→import { Card, CardContent } from '@/components/ui/card';
     4→import { Badge } from '@/components/ui/badge';
     5→import { Button } from '@/components/ui/button';
     6→import {
     7→  Dialog, DialogContent, DialogDescription, DialogFooter,
     8→  DialogHeader, DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import {
    11→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    12→} from '@/components/ui/select';
    13→import { Input } from '@/components/ui/input';
    14→import {
    15→  Building2, Users, Settings, Shield, Vote, Loader2, MapPin, Save,
    16→  RotateCcw, Plus, Trash2, UserPlus, Mail, ChevronRight, Globe,
    17→  Eye, UserCheck, Radio, AlertTriangle, Pencil,
    18→} from 'lucide-react';
    19→import { useState, useEffect, useCallback } from 'react';
    20→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
    21→import { cn } from '@/lib/utils';
    22→import { toast } from 'sonner';
    23→import { useDashboardStore, type UserRole } from '@/store/dashboard';
    24→
    25→// ---- Shared types & constants ----
    26→const REGION_PRESETS = [
    27→  { label: 'Nigeria (full)', minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0 },
    28→  { label: 'Lagos', minLat: 6.3, maxLat: 6.7, minLng: 3.2, maxLng: 3.5 },
    29→  { label: 'Abuja FCT', minLat: 8.8, maxLat: 9.2, minLng: 7.2, maxLng: 7.7 },
    30→  { label: 'Kano', minLat: 11.8, maxLat: 12.4, minLng: 8.3, maxLng: 8.9 },
    31→  { label: 'Rivers', minLat: 4.5, maxLat: 5.2, minLng: 6.5, maxLng: 7.1 },
    32→  { label: 'Southeast', minLat: 4.8, maxLat: 7.0, minLng: 6.8, maxLng: 8.2 },

    36→  { label: 'South-South', minLat: 4.3, maxLat: 6.5, minLng: 5.0, maxLng: 8.5 },
    37→  { label: 'Northeast', minLat: 8.0, maxLat: 14.0, minLng: 10.0, maxLng: 15.0 },
    38→];
    39→
    40→interface MapBoundsData {
    41→  minLat: number; maxLat: number; minLng: number; maxLng: number; label: string;
    42→}
    43→
    44→const VALID_ROLES: UserRole[] = ['SUPER_ADMIN', 'TENANT_ADMIN', 'ANALYST', 'TRUST_SAFETY', 'FIELD_AGENT'];
    45→
    46→const ROLE_BADGE: Record<string, string> = {
    47→  SUPER_ADMIN: 'border-emerald/30 text-emerald bg-emerald/10',
    48→  TENANT_ADMIN: 'border-cyan/30 text-cyan bg-cyan/10',
    49→  ANALYST: 'border-amber/30 text-amber bg-amber/10',
    50→  TRUST_SAFETY: 'border-rose/30 text-rose bg-rose/10',
    51→  FIELD_AGENT: 'border-border text-muted-foreground bg-card',
    52→};
    53→
    54→const ROLE_ICONS: Record<string, React.ReactNode> = {
    55→  SUPER_ADMIN: <Shield className="h-3.5 w-3.5" />,
    56→  TENANT_ADMIN: <Users className="h-3.5 w-3.5" />,
    57→  ANALYST: <Eye className="h-3.5 w-3.5" />,
    58→  TRUST_SAFETY: <UserCheck className="h-3.5 w-3.5" />,
    59→  FIELD_AGENT: <Radio className="h-3.5 w-3.5" />,
    60→};
    61→
    62→// ---- Main Component ----
    63→export function TenantManagement() {
    64→  const { user, tenantId } = useDashboardStore();
    65→  const queryClient = useQueryClient();
    66→  const isSuperAdmin = user?.role === 'SUPER_ADMIN';
    67→  const isTenantAdmin = user?.role === 'TENANT_ADMIN';
    68→  const isAdmin = isSuperAdmin || isTenantAdmin;
    69→
    70→  // ===================== SUPER_ADMIN: Platform Tenants =====================
    71→  const { data: allTenants, isLoading: tenantsLoading } = useQuery({
    72→    queryKey: ['all-tenants'],
    73→    queryFn: async () => {
    74→      const res = await fetch('/api/tenants');
    75→      const d = await res.json();
    76→      if (!res.ok) throw new Error(d.error || 'Failed to load tenants');
    77→      return d.tenants || [];
    78→    },
    79→    enabled: isSuperAdmin,
    80→  });
    81→
    82→  // ===================== TENANT_ADMIN: Own tenant settings =====================
    83→  const { data: settings, isLoading: settingsLoading } = useQuery({
    84→    queryKey: ['tenant-settings', tenantId],
    85→    queryFn: async () => {
    86→      const res = await fetch(`/api/tenant-settings?tenantId=${tenantId}`);
    87→      const d = await res.json();
    88→      if (!res.ok) throw new Error(d.error || 'Failed to load settings');
    89→      return d;
    90→    },
    91→    enabled: !!tenantId && isAdmin,
    92→  });
    93→
    94→  // ===================== Common: Tenant Users =====================
    95→  const { data: tenantUsersData, isLoading: usersLoading } = useQuery({
    96→    queryKey: ['tenant-users', tenantId],
    97→    queryFn: async () => {
    98→      const res = await fetch(`/api/tenants/users?tenantId=${tenantId}`);
    99→      const d = await res.json();
   100→      if (!res.ok) throw new Error(d.error || 'Failed to load users');
   101→      return d.users || [];
   102→    },
   103→    enabled: !!tenantId && isAdmin,
   104→  });
   105→
   106→  const currentBounds: MapBoundsData | null = settings?.mapBounds || null;
   107→  const tenantUsers = tenantUsersData || [];
   108→
   109→  // ===================== State =====================
   110→  // Map config dialog
   111→  const [mapConfigOpen, setMapConfigOpen] = useState(false);
   112→  const [mapLabel, setMapLabel] = useState('');
   113→  const [minLat, setMinLat] = useState('');
   114→  const [maxLat, setMaxLat] = useState('');
   115→  const [minLng, setMinLng] = useState('');
   116→  const [maxLng, setMaxLng] = useState('');
   117→
   118→  // Create tenant dialog
   119→  const [createTenantOpen, setCreateTenantOpen] = useState(false);
   120→  const [newTenantName, setNewTenantName] = useState('');
   121→  const [newTenantSlug, setNewTenantSlug] = useState('');
   122→  const [newTenantColor, setNewTenantColor] = useState('#10b981');
   123→  const [newAdminName, setNewAdminName] = useState('');
   124→  const [newAdminEmail, setNewAdminEmail] = useState('');
   125→
   126→  // Add user dialog
   127→  const [addUserOpen, setAddUserOpen] = useState(false);
   128→  const [newUserName, setNewUserName] = useState('');
   129→  const [newUserEmail, setNewUserEmail] = useState('');
   130→  const [newUserRole, setNewUserRole] = useState<string>('FIELD_AGENT');
   131→
   132→  // User role change dialog
   133→  const [roleChangeOpen, setRoleChangeOpen] = useState(false);
   134→  const [roleChangeUser, setRoleChangeUser] = useState<{ id: string; name: string; role: string } | null>(null);
   135→  const [newRole, setNewRole] = useState('');
   136→
   137→  // Delete confirmation
   138→  const [deleteConfirm, setDeleteConfirm] = useState<{ type: 'tenant' | 'user'; item: { id: string; name: string } } | null>(null);
   139→
   140→  // Active sub-tab for SUPER_ADMIN
   141→  const [subTab, setSubTab] = useState<'tenants' | 'my-tenant' | 'map'>('tenants');
   142→
   143→  // ===================== Effects =====================
   144→  useEffect(() => {
   145→    if (mapConfigOpen && currentBounds) {
   146→      setMapLabel(currentBounds.label || '');
   147→      setMinLat(String(currentBounds.minLat));
   148→      setMaxLat(String(currentBounds.maxLat));
   149→      setMinLng(String(currentBounds.minLng));
   150→      setMaxLng(String(currentBounds.maxLng));
   151→    } else if (mapConfigOpen) {
   152→      setMapLabel('');
   153→      setMinLat(''); setMaxLat(''); setMinLng(''); setMaxLng('');
   154→    }
   155→  }, [mapConfigOpen, currentBounds]);
   156→
   157→  // Auto-derive slug from tenant name
   158→  useEffect(() => {
   159→    if (createTenantOpen && newTenantName && !newTenantSlug) {
   160→      setNewTenantSlug(newTenantName.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, ''));
   161→    }
   162→  }, [createTenantOpen, newTenantName, newTenantSlug]);
   163→
   164→  // ===================== Mutations =====================
   165→  // Save map bounds
   166→  const saveMapMutation = useMutation({
   167→    mutationFn: async (bounds: MapBoundsData) => {
   168→      const res = await fetch(`/api/tenant-settings?tenantId=${tenantId}`, {
   169→        method: 'PUT',
   170→        headers: { 'Content-Type': 'application/json' },
   171→        body: JSON.stringify({ mapBounds: bounds }),
   172→      });
   173→      const data = await res.json();
   174→      if (!res.ok) throw new Error(data.error || 'Failed to save map configuration');
   175→      return data;
   176→    },
   177→    onSuccess: () => {
   178→      queryClient.invalidateQueries({ queryKey: ['tenant-settings', tenantId] });
   179→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   180→      setMapConfigOpen(false);
   181→      toast.success('Map area saved. The map will update automatically.');
   182→    },
   183→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to save map configuration'),
   184→  });
   185→
   186→  // Create tenant
   187→  const createTenantMutation = useMutation({
   188→    mutationFn: (data: { name: string; slug: string; primaryColor: string; adminName: string; adminEmail: string }) =>
   189→      fetch('/api/tenants', {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify(data),
   193→      }).then(r => {
   194→        if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); });
   195→        return r.json();
   196→      }),
   197→    onSuccess: (data) => {
   198→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   199→      setCreateTenantOpen(false);
   200→      setNewTenantName(''); setNewTenantSlug(''); setNewTenantColor('#10b981');
   201→      setNewAdminName(''); setNewAdminEmail('');
   202→      toast.success(`Tenant "${data.tenant.name}" created. Admin: ${data.admin.email}`);
   203→    },
   204→    onError: (err) => toast.error(err?.message || 'Failed to create tenant'),
   205→  });
   206→
   207→  // Delete tenant
   208→  const deleteTenantMutation = useMutation({
   209→    mutationFn: (id: string) =>
   210→      fetch(`/api/tenants?id=${id}`, { method: 'DELETE' })
   211→        .then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   212→    onSuccess: () => {
   213→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   214→      setDeleteConfirm(null);
   215→      toast.success('Tenant deleted successfully');
   216→    },
   217→    onError: (err) => toast.error(err?.message || 'Failed to delete tenant'),
   218→  });
   219→
   220→  // Toggle tenant active
   221→  const toggleTenantMutation = useMutation({
   222→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   223→      fetch('/api/tenants', {
   224→        method: 'PUT',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify({ id, isActive }),
   227→      }).then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   228→    onSuccess: () => {
   229→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   230→      toast.success('Tenant status updated');
   231→    },
   232→    onError: (err) => toast.error(err?.message || 'Failed to update tenant'),
   233→  });
   234→
   235→  // Add user
   236→  const addUserMutation = useMutation({
   237→    mutationFn: (data: { tenantId: string; name: string; email: string; role: string }) =>
   238→      fetch('/api/tenants/users', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify(data),
   242→      }).then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   243→    onSuccess: () => {
   244→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   245→      setAddUserOpen(false);
   246→      setNewUserName(''); setNewUserEmail(''); setNewUserRole('FIELD_AGENT');
   247→      toast.success('User added successfully');
   248→    },
   249→    onError: (err) => toast.error(err?.message || 'Failed to add user'),
   250→  });
   251→
   252→  // Change user role
   253→  const changeRoleMutation = useMutation({
   254→    mutationFn: ({ id, role }: { id: string; role: string }) =>
   255→      fetch('/api/tenants/users', {
   256→        method: 'PATCH',
   257→        headers: { 'Content-Type': 'application/json' },
   258→        body: JSON.stringify({ id, role }),
   259→      }).then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   260→    onSuccess: () => {
   261→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   262→      setRoleChangeOpen(false);
   263→      setRoleChangeUser(null);
   264→      toast.success('User role updated');
   265→    },
   266→    onError: (err) => toast.error(err?.message || 'Failed to update role'),
   267→  });
   268→
   269→  // Remove user
   270→  const deleteUserMutation = useMutation({
   271→    mutationFn: (id: string) =>
   272→      fetch(`/api/tenants/users?id=${id}`, { method: 'DELETE' })
   273→        .then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   274→    onSuccess: () => {
   275→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   276→      setDeleteConfirm(null);
   277→      toast.success('User removed');
   278→    },
   279→    onError: (err) => toast.error(err?.message || 'Failed to remove user'),
   280→  });
   281→
   282→  // ===================== Handlers =====================
   283→  const handleSaveMap = () => {
   284→    const minLatN = parseFloat(minLat);
   285→    const maxLatN = parseFloat(maxLat);
   286→    const minLngN = parseFloat(minLng);
   287→    const maxLngN = parseFloat(maxLng);
   288→    if ([minLatN, maxLatN, minLngN, maxLngN].some(isNaN)) { toast.error('All coordinates must be valid numbers'); return; }
   289→    if (minLatN >= maxLatN || minLngN >= maxLngN) { toast.error('minLat must be < maxLat, and minLng must be < maxLng'); return; }
   290→    saveMapMutation.mutate({ minLat: minLatN, maxLat: maxLatN, minLng: minLngN, maxLng: maxLngN, label: mapLabel || 'Custom Area' });
   291→  };
   292→
   293→  const handlePreset = (preset: typeof REGION_PRESETS[number]) => {
   294→    setMapLabel(preset.label);
   295→    setMinLat(String(preset.minLat));
   296→    setMaxLat(String(preset.maxLat));
   297→    setMinLng(String(preset.minLng));
   298→    setMaxLng(String(preset.maxLng));
   299→  };
   300→
   301→  const handleResetMap = () => {
   302→    saveMapMutation.mutate({ minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0, label: 'Nigeria' });
   303→  };
   304→
   305→  const handleCreateTenant = () => {
   306→    if (!newTenantName || !newTenantSlug || !newAdminName || !newAdminEmail) {
   307→      toast.error('All fields are required'); return;
   308→    }
   309→    createTenantMutation.mutate({
   310→      name: newTenantName, slug: newTenantSlug, primaryColor: newTenantColor,
   311→      adminName: newAdminName, adminEmail: newAdminEmail,
   312→    });
   313→  };
   314→
   315→  const handleAddUser = () => {
   316→    if (!newUserName || !newUserEmail || !newUserRole) { toast.error('All fields are required'); return; }
   317→    addUserMutation.mutate({ tenantId: tenantId!, name: newUserName, email: newUserEmail, role: newUserRole });
   318→  };
   319→
   320→  const handleRoleChange = () => {
   321→    if (!roleChangeUser || !newRole) return;
   322→    changeRoleMutation.mutate({ id: roleChangeUser.id, role: newRole });
   323→  };
   324→
   325→  // ===================== Restricted access =====================
   326→  if (!isAdmin) {
   327→    return (
   328→      <div className="h-full flex items-center justify-center p-6">
   329→        <div className="text-center space-y-2">
   330→          <Shield className="h-8 w-8 text-muted-foreground/30 mx-auto" />
   331→          <p className="text-sm text-muted-foreground">Access restricted to Super Admin and Tenant Admin</p>
   332→        </div>
   333→      </div>
   334→    );
   335→  }
   336→
   337→  const isLoading = isSuperAdmin ? tenantsLoading : settingsLoading;
   338→
   339→  // ===================== Render =====================
   340→  return (
   341→    <div className="h-full flex flex-col overflow-hidden">
   342→      {/* Header */}
   343→      <div className="px-4 sm:px-6 pt-4 pb-3 shrink-0">
   344→        <div className="flex items-center justify-between flex-wrap gap-3">
   345→          <div>
   346→            <h2 className="text-lg font-semibold flex items-center gap-2">
   347→              <Building2 className="h-5 w-5 text-emerald" />
   348→              {isSuperAdmin ? 'Platform Management' : 'Organization Settings'}
   349→            </h2>
   350→            <p className="text-sm text-muted-foreground mt-0.5">
   351→              {isSuperAdmin
   352→                ? 'Create and manage tenant organizations, their users, and configurations'
   353→                : 'Manage your organization settings, users, and map configuration'}
   354→            </p>
   355→          </div>
   356→          {isSuperAdmin && (
   357→            <Button
   358→              size="sm"
   359→              className="gap-1.5 text-xs bg-emerald hover:bg-emerald/90 text-emerald-950 h-8"
   360→              onClick={() => setCreateTenantOpen(true)}
   361→            >
   362→              <Plus className="h-3.5 w-3.5" /> New Tenant
   363→            </Button>
   364→          )}
   365→        </div>
   366→
   367→        {/* Sub-tabs for SUPER_ADMIN */}
   368→        {isSuperAdmin && (
   369→          <div className="flex items-center gap-1 mt-3 bg-muted/30 rounded-lg p-0.5 w-fit">
   370→            {([
   371→              { id: 'tenants' as const, label: 'All Tenants', icon: <Globe className="h-3.5 w-3.5" /> },
   372→              { id: 'my-tenant' as const, label: 'My Organization', icon: <Building2 className="h-3.5 w-3.5" /> },
   373→              { id: 'map' as const, label: 'Map Config', icon: <MapPin className="h-3.5 w-3.5" /> },
   374→            ]).map(tab => (
   375→              <button
   376→                key={tab.id}
   377→                onClick={() => setSubTab(tab.id)}
   378→                className={cn(
   379→                  'flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium transition-colors',
   380→                  subTab === tab.id ? 'bg-card shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   381→                )}
   382→              >
   383→                {tab.icon} {tab.label}
   384→              </button>
   385→            ))}
   386→          </div>
   387→        )}
   388→      </div>
   389→
   390→      {/* Scrollable content */}
   391→      <div className="flex-1 overflow-y-auto px-4 sm:px-6 pb-6 space-y-4 min-h-0">
   392→
   393→        {isLoading ? (
   394→          <div className="flex-1 flex items-center justify-center py-20">
   395→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   396→          </div>
   397→        ) : isSuperAdmin ? (
   398→          /* ========== SUPER_ADMIN VIEWS ========== */
   399→          <>
   400→            {subTab === 'tenants' && (
   401→              <SuperAdminTenantsView
   402→                tenants={allTenants || []}
   403→                onToggle={(id, isActive) => toggleTenantMutation.mutate({ id, isActive })}
   404→                onDelete={(item) => setDeleteConfirm({ type: 'tenant', item })}
   405→                onManageUsers={() => setSubTab('my-tenant')}
   406→              />
   407→            )}
   408→            {subTab === 'my-tenant' && (
   409→              <TenantUsersView
   410→                settings={settings}
   411→                tenantUsers={tenantUsers}
   412→                userRole={user?.role || ''}
   413→                onOpenUserDialog={() => setAddUserOpen(true)}
   414→                onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   415→                onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   416→                onOpenMapConfig={() => { setSubTab('map'); setTimeout(() => setMapConfigOpen(true), 100); }}
   417→              />
   418→            )}
   419→            {subTab === 'map' && (
   420→              <MapConfigSection
   421→                currentBounds={currentBounds}
   422→                onConfigure={() => setMapConfigOpen(true)}
   423→              />
   424→            )}
   425→          </>
   426→        ) : (
   427→          /* ========== TENANT_ADMIN VIEW ========== */
   428→          <>
   429→            <TenantUsersView
   430→              settings={settings}
   431→              tenantUsers={tenantUsers}
   432→              userRole={user?.role || ''}
   433→              onOpenUserDialog={() => setAddUserOpen(true)}
   434→              onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   435→              onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   436→              onOpenMapConfig={() => setMapConfigOpen(true)}
   437→            />
   438→            <MapConfigSection
   439→              currentBounds={currentBounds}
   440→              onConfigure={() => setMapConfigOpen(true)}
   441→            />
   442→            <InfoNotices />
   443→          </>
   444→        )}
   445→      </div>
   446→
   447→      {/* ===== MAP CONFIG DIALOG ===== */}
   448→      <Dialog open={mapConfigOpen} onOpenChange={setMapConfigOpen}>
   449→        <DialogContent className="sm:max-w-lg">
   450→          <DialogHeader>
   451→            <DialogTitle className="flex items-center gap-2">
   452→              <MapPin className="h-5 w-5 text-cyan" />
   453→              Configure Map Area
   454→            </DialogTitle>
   455→            <DialogDescription>
   456→              Set the geographic bounding box for the Polling Unit Map. Use a preset or enter custom WGS84 coordinates.
   457→            </DialogDescription>
   458→          </DialogHeader>
   459→          <div className="space-y-4 py-2">
   460→            <div className="space-y-1.5">
   461→              <label className="text-xs font-medium text-muted-foreground">Area Label</label>
   462→              <Input placeholder="e.g. Lagos, Southeast, Kano Central" value={mapLabel} onChange={(e) => setMapLabel(e.target.value)} className="h-9 text-sm" />
   463→            </div>
