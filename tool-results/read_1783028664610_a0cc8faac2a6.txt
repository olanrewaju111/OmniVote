     1→'use client';
     2→
     3→import { Card, CardContent } from '@/components/ui/card';
     4→import { Badge } from '@/components/ui/badge';
     5→import { Button } from '@/components/ui/button';
     6→import {
     7→  Dialog, DialogContent, DialogDescription, DialogFooter,
     8→  DialogHeader, DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import {
    11→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    12→} from '@/components/ui/select';
    13→import { Input } from '@/components/ui/input';
    14→import {
    15→  Building2, Users, Settings, Shield, Vote, Loader2, MapPin, Save,
    16→  RotateCcw, Plus, Trash2, UserPlus, Mail, ChevronRight, Globe,
    17→  Eye, UserCheck, Radio, AlertTriangle, Pencil,
    18→} from 'lucide-react';
    19→import { useState, useEffect, useCallback } from 'react';
    20→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
    21→import { cn } from '@/lib/utils';
    22→import { toast } from 'sonner';
    23→import { useDashboardStore, type UserRole } from '@/store/dashboard';
    24→
    25→// ---- Shared types & constants ----
    26→const REGION_PRESETS = [
    27→  { label: 'Nigeria (full)', minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0 },
    28→  { label: 'Lagos', minLat: 6.3, maxLat: 6.7, minLng: 3.2, maxLng: 3.5 },
    29→  { label: 'Abuja FCT', minLat: 8.8, maxLat: 9.2, minLng: 7.2, maxLng: 7.7 },
    30→  { label: 'Kano', minLat: 11.8, maxLat: 12.4, minLng: 8.3, maxLng: 8.9 },
    31→  { label: 'Rivers', minLat: 4.5, maxLat: 5.2, minLng: 6.5, maxLng: 7.1 },
    32→  { label: 'Southeast', minLat: 4.8, maxLat: 7.0, minLng: 6.8, maxLng: 8.2 },
    33→  { label: 'Southwest', minLat: 6.0, maxLat: 9.0, minLng: 2.5, maxLng: 5.0 },
    34→  { label: 'Northwest', minLat: 10.0, maxLat: 14.0, minLng: 3.0, maxLng: 9.0 },
    35→  { label: 'North-Central', minLat: 7.0, maxLat: 10.5, minLng: 3.0, maxLng: 9.5 },
    36→  { label: 'South-South', minLat: 4.3, maxLat: 6.5, minLng: 5.0, maxLng: 8.5 },
    37→  { label: 'Northeast', minLat: 8.0, maxLat: 14.0, minLng: 10.0, maxLng: 15.0 },
    38→];
    39→
    40→interface MapBoundsData {
    41→  minLat: number; maxLat: number; minLng: number; maxLng: number; label: string;
    42→}
    43→
    44→const VALID_ROLES: UserRole[] = ['SUPER_ADMIN', 'TENANT_ADMIN', 'ANALYST', 'TRUST_SAFETY', 'FIELD_AGENT'];
    45→
    46→const ROLE_BADGE: Record<string, string> = {
    47→  SUPER_ADMIN: 'border-emerald/30 text-emerald bg-emerald/10',
    48→  TENANT_ADMIN: 'border-cyan/30 text-cyan bg-cyan/10',
    49→  ANALYST: 'border-amber/30 text-amber bg-amber/10',
    50→  TRUST_SAFETY: 'border-rose/30 text-rose bg-rose/10',
    51→  FIELD_AGENT: 'border-border text-muted-foreground bg-card',
    52→};
    53→
    54→const ROLE_ICONS: Record<string, React.ReactNode> = {
    55→  SUPER_ADMIN: <Shield className="h-3.5 w-3.5" />,
    56→  TENANT_ADMIN: <Users className="h-3.5 w-3.5" />,
    57→  ANALYST: <Eye className="h-3.5 w-3.5" />,
    58→  TRUST_SAFETY: <UserCheck className="h-3.5 w-3.5" />,
    59→  FIELD_AGENT: <Radio className="h-3.5 w-3.5" />,
    60→};
    61→
    62→// ---- Main Component ----
    63→export function TenantManagement() {
    64→  const { user, tenantId } = useDashboardStore();
    65→  const queryClient = useQueryClient();
    66→  const isSuperAdmin = user?.role === 'SUPER_ADMIN';
    67→  const isTenantAdmin = user?.role === 'TENANT_ADMIN';
    68→  const isAdmin = isSuperAdmin || isTenantAdmin;
    69→
    70→  // ===================== SUPER_ADMIN: Platform Tenants =====================
    71→  const { data: allTenants, isLoading: tenantsLoading } = useQuery({
    72→    queryKey: ['all-tenants'],
    73→    queryFn: () => fetch('/api/tenants').then(r => r.json()).then(d => d.tenants || []),
    74→    enabled: isSuperAdmin,
    75→  });
    76→
    77→  // ===================== TENANT_ADMIN: Own tenant settings =====================
    78→  const { data: settings, isLoading: settingsLoading } = useQuery({
    79→    queryKey: ['tenant-settings', tenantId],
    80→    queryFn: () => fetch(`/api/tenant-settings?tenantId=${tenantId}`).then(r => r.json()),
    81→    enabled: !!tenantId && isAdmin,
    82→  });
    83→
    84→  // ===================== Common: Tenant Users =====================
    85→  const { data: tenantUsersData, isLoading: usersLoading } = useQuery({
    86→    queryKey: ['tenant-users', tenantId],
    87→    queryFn: () => fetch(`/api/tenants/users?tenantId=${tenantId}`).then(r => r.json()).then(d => d.users || []),
    88→    enabled: !!tenantId && isAdmin,
    89→  });
    90→
    91→  const currentBounds: MapBoundsData | null = settings?.mapBounds || null;
    92→  const tenantUsers = tenantUsersData || [];
    93→
    94→  // ===================== State =====================
    95→  // Map config dialog
    96→  const [mapConfigOpen, setMapConfigOpen] = useState(false);
    97→  const [mapLabel, setMapLabel] = useState('');
    98→  const [minLat, setMinLat] = useState('');
    99→  const [maxLat, setMaxLat] = useState('');
   100→  const [minLng, setMinLng] = useState('');
   101→  const [maxLng, setMaxLng] = useState('');
   102→
   103→  // Create tenant dialog
   104→  const [createTenantOpen, setCreateTenantOpen] = useState(false);
   105→  const [newTenantName, setNewTenantName] = useState('');
   106→  const [newTenantSlug, setNewTenantSlug] = useState('');
   107→  const [newTenantColor, setNewTenantColor] = useState('#10b981');
   108→  const [newAdminName, setNewAdminName] = useState('');
   109→  const [newAdminEmail, setNewAdminEmail] = useState('');
   110→
   111→  // Add user dialog
   112→  const [addUserOpen, setAddUserOpen] = useState(false);
   113→  const [newUserName, setNewUserName] = useState('');
   114→  const [newUserEmail, setNewUserEmail] = useState('');
   115→  const [newUserRole, setNewUserRole] = useState<string>('FIELD_AGENT');
   116→
   117→  // User role change dialog
   118→  const [roleChangeOpen, setRoleChangeOpen] = useState(false);
   119→  const [roleChangeUser, setRoleChangeUser] = useState<{ id: string; name: string; role: string } | null>(null);
   120→  const [newRole, setNewRole] = useState('');
   121→
   122→  // Delete confirmation
   123→  const [deleteConfirm, setDeleteConfirm] = useState<{ type: 'tenant' | 'user'; item: { id: string; name: string } } | null>(null);
   124→
   125→  // Active sub-tab for SUPER_ADMIN
   126→  const [subTab, setSubTab] = useState<'tenants' | 'my-tenant' | 'map'>('tenants');
   127→
   128→  // ===================== Effects =====================
   129→  useEffect(() => {
   130→    if (mapConfigOpen && currentBounds) {
   131→      setMapLabel(currentBounds.label || '');
   132→      setMinLat(String(currentBounds.minLat));
   133→      setMaxLat(String(currentBounds.maxLat));
   134→      setMinLng(String(currentBounds.minLng));
   135→      setMaxLng(String(currentBounds.maxLng));
   136→    } else if (mapConfigOpen) {
   137→      setMapLabel('');
   138→      setMinLat(''); setMaxLat(''); setMinLng(''); setMaxLng('');
   139→    }
   140→  }, [mapConfigOpen, currentBounds]);
   141→
   142→  // Auto-derive slug from tenant name
   143→  useEffect(() => {
   144→    if (createTenantOpen && newTenantName && !newTenantSlug) {
   145→      setNewTenantSlug(newTenantName.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, ''));
   146→    }
   147→  }, [createTenantOpen, newTenantName, newTenantSlug]);
   148→
   149→  // ===================== Mutations =====================
   150→  // Save map bounds
   151→  const saveMapMutation = useMutation({
   152→    mutationFn: (bounds: MapBoundsData) =>
   153→      fetch('/api/tenant-settings', {
   154→        method: 'PUT',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({ mapBounds: bounds }),
   157→      }).then(r => r.json()),
   158→    onSuccess: () => {
   159→      queryClient.invalidateQueries({ queryKey: ['tenant-settings'] });
   160→      queryClient.invalidateQueries({ queryKey: ['dashboard'] }); // auto-refresh map
   161→      setMapConfigOpen(false);
   162→      toast.success('Map area saved. The map will update automatically.');
   163→    },
   164→    onError: (err) => toast.error(err?.message || 'Failed to save map configuration'),
   165→  });
   166→
   167→  // Create tenant
   168→  const createTenantMutation = useMutation({
   169→    mutationFn: (data: { name: string; slug: string; primaryColor: string; adminName: string; adminEmail: string }) =>
   170→      fetch('/api/tenants', {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify(data),
   174→      }).then(r => {
   175→        if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); });
   176→        return r.json();
   177→      }),
   178→    onSuccess: (data) => {
   179→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   180→      setCreateTenantOpen(false);
   181→      setNewTenantName(''); setNewTenantSlug(''); setNewTenantColor('#10b981');
   182→      setNewAdminName(''); setNewAdminEmail('');
   183→      toast.success(`Tenant "${data.tenant.name}" created. Admin: ${data.admin.email}`);
   184→    },
   185→    onError: (err) => toast.error(err?.message || 'Failed to create tenant'),
   186→  });
   187→
   188→  // Delete tenant
   189→  const deleteTenantMutation = useMutation({
   190→    mutationFn: (id: string) =>
   191→      fetch(`/api/tenants?id=${id}`, { method: 'DELETE' })
   192→        .then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   193→    onSuccess: () => {
   194→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   195→      setDeleteConfirm(null);
   196→      toast.success('Tenant deleted successfully');
   197→    },
   198→    onError: (err) => toast.error(err?.message || 'Failed to delete tenant'),
   199→  });
   200→
   201→  // Toggle tenant active
   202→  const toggleTenantMutation = useMutation({
   203→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   204→      fetch('/api/tenants', {
   205→        method: 'PUT',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify({ id, isActive }),
   208→      }).then(r => r.json()),
   209→    onSuccess: () => {
   210→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   211→      toast.success('Tenant status updated');
   212→    },
   213→    onError: (err) => toast.error(err?.message || 'Failed to update tenant'),
   214→  });
   215→
   216→  // Add user
   217→  const addUserMutation = useMutation({
   218→    mutationFn: (data: { tenantId: string; name: string; email: string; role: string }) =>
   219→      fetch('/api/tenants/users', {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify(data),
   223→      }).then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   224→    onSuccess: () => {
   225→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   226→      setAddUserOpen(false);
   227→      setNewUserName(''); setNewUserEmail(''); setNewUserRole('FIELD_AGENT');
   228→      toast.success('User added successfully');
   229→    },
   230→    onError: (err) => toast.error(err?.message || 'Failed to add user'),
   231→  });
   232→
   233→  // Change user role
   234→  const changeRoleMutation = useMutation({
   235→    mutationFn: ({ id, role }: { id: string; role: string }) =>
   236→      fetch('/api/tenants/users', {
   237→        method: 'PATCH',
   238→        headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify({ id, role }),
   240→      }).then(r => r.json()),
   241→    onSuccess: () => {
   242→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   243→      setRoleChangeOpen(false);
   244→      setRoleChangeUser(null);
   245→      toast.success('User role updated');
   246→    },
   247→    onError: (err) => toast.error(err?.message || 'Failed to update role'),
   248→  });
   249→
   250→  // Remove user
   251→  const deleteUserMutation = useMutation({
   252→    mutationFn: (id: string) =>
   253→      fetch(`/api/tenants/users?id=${id}`, { method: 'DELETE' })
   254→        .then(r => { if (!r.ok) return r.json().then(e => { throw new Error(e.error || 'Failed'); }); return r.json(); }),
   255→    onSuccess: () => {
   256→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   257→      setDeleteConfirm(null);
   258→      toast.success('User removed');
   259→    },
   260→    onError: (err) => toast.error(err?.message || 'Failed to remove user'),
   261→  });
   262→
   263→  // ===================== Handlers =====================
   264→  const handleSaveMap = () => {
   265→    const minLatN = parseFloat(minLat);
   266→    const maxLatN = parseFloat(maxLat);
   267→    const minLngN = parseFloat(minLng);
   268→    const maxLngN = parseFloat(maxLng);
   269→    if ([minLatN, maxLatN, minLngN, maxLngN].some(isNaN)) { toast.error('All coordinates must be valid numbers'); return; }
   270→    if (minLatN >= maxLatN || minLngN >= maxLngN) { toast.error('minLat must be < maxLat, and minLng must be < maxLng'); return; }
   271→    saveMapMutation.mutate({ minLat: minLatN, maxLat: maxLatN, minLng: minLngN, maxLng: maxLngN, label: mapLabel || 'Custom Area' });
   272→  };
   273→
   274→  const handlePreset = (preset: typeof REGION_PRESETS[number]) => {
   275→    setMapLabel(preset.label);
   276→    setMinLat(String(preset.minLat));
   277→    setMaxLat(String(preset.maxLat));
   278→    setMinLng(String(preset.minLng));
   279→    setMaxLng(String(preset.maxLng));
   280→  };
   281→
   282→  const handleResetMap = () => {
   283→    saveMapMutation.mutate({ minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0, label: 'Nigeria' });
   284→  };
   285→
   286→  const handleCreateTenant = () => {
   287→    if (!newTenantName || !newTenantSlug || !newAdminName || !newAdminEmail) {
   288→      toast.error('All fields are required'); return;
   289→    }
   290→    createTenantMutation.mutate({
   291→      name: newTenantName, slug: newTenantSlug, primaryColor: newTenantColor,
   292→      adminName: newAdminName, adminEmail: newAdminEmail,
   293→    });
   294→  };
   295→
   296→  const handleAddUser = () => {
   297→    if (!newUserName || !newUserEmail || !newUserRole) { toast.error('All fields are required'); return; }
   298→    addUserMutation.mutate({ tenantId: tenantId!, name: newUserName, email: newUserEmail, role: newUserRole });
   299→  };
   300→
   301→  const handleRoleChange = () => {
   302→    if (!roleChangeUser || !newRole) return;
   303→    changeRoleMutation.mutate({ id: roleChangeUser.id, role: newRole });
   304→  };
   305→
   306→  // ===================== Restricted access =====================
   307→  if (!isAdmin) {
   308→    return (
   309→      <div className="h-full flex items-center justify-center p-6">
   310→        <div className="text-center space-y-2">
   311→          <Shield className="h-8 w-8 text-muted-foreground/30 mx-auto" />
   312→          <p className="text-sm text-muted-foreground">Access restricted to Super Admin and Tenant Admin</p>
   313→        </div>
   314→      </div>
   315→    );
   316→  }
   317→
   318→  const isLoading = isSuperAdmin ? tenantsLoading : settingsLoading;
   319→
   320→  // ===================== Render =====================
   321→  return (
   322→    <div className="h-full flex flex-col overflow-hidden">
   323→      {/* Header */}
   324→      <div className="px-4 sm:px-6 pt-4 pb-3 shrink-0">
   325→        <div className="flex items-center justify-between flex-wrap gap-3">
   326→          <div>
   327→            <h2 className="text-lg font-semibold flex items-center gap-2">
   328→              <Building2 className="h-5 w-5 text-emerald" />
   329→              {isSuperAdmin ? 'Platform Management' : 'Organization Settings'}
   330→            </h2>
   331→            <p className="text-sm text-muted-foreground mt-0.5">
   332→              {isSuperAdmin
   333→                ? 'Create and manage tenant organizations, their users, and configurations'
   334→                : 'Manage your organization settings, users, and map configuration'}
   335→            </p>
   336→          </div>
   337→          {isSuperAdmin && (
   338→            <Button
   339→              size="sm"
   340→              className="gap-1.5 text-xs bg-emerald hover:bg-emerald/90 text-emerald-950 h-8"
   341→              onClick={() => setCreateTenantOpen(true)}
   342→            >
   343→              <Plus className="h-3.5 w-3.5" /> New Tenant
   344→            </Button>
   345→          )}
   346→        </div>
   347→
   348→        {/* Sub-tabs for SUPER_ADMIN */}
   349→        {isSuperAdmin && (
   350→          <div className="flex items-center gap-1 mt-3 bg-muted/30 rounded-lg p-0.5 w-fit">
   351→            {([
   352→              { id: 'tenants' as const, label: 'All Tenants', icon: <Globe className="h-3.5 w-3.5" /> },
   353→              { id: 'my-tenant' as const, label: 'My Organization', icon: <Building2 className="h-3.5 w-3.5" /> },
   354→              { id: 'map' as const, label: 'Map Config', icon: <MapPin className="h-3.5 w-3.5" /> },
   355→            ]).map(tab => (
   356→              <button
   357→                key={tab.id}
   358→                onClick={() => setSubTab(tab.id)}
   359→                className={cn(
   360→                  'flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium transition-colors',
   361→                  subTab === tab.id ? 'bg-card shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   362→                )}
   363→              >
   364→                {tab.icon} {tab.label}
   365→              </button>
   366→            ))}
   367→          </div>
   368→        )}
   369→      </div>
   370→
   371→      {/* Scrollable content */}
   372→      <div className="flex-1 overflow-y-auto px-4 sm:px-6 pb-6 space-y-4 min-h-0">
   373→
   374→        {isLoading ? (
   375→          <div className="flex-1 flex items-center justify-center py-20">
   376→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   377→          </div>
   378→        ) : isSuperAdmin ? (
   379→          /* ========== SUPER_ADMIN VIEWS ========== */
   380→          <>
   381→            {subTab === 'tenants' && (
   382→              <SuperAdminTenantsView
   383→                tenants={allTenants || []}
   384→                onToggle={(id, isActive) => toggleTenantMutation.mutate({ id, isActive })}
   385→                onDelete={(item) => setDeleteConfirm({ type: 'tenant', item })}
   386→                onManageUsers={() => setSubTab('my-tenant')}
   387→              />
   388→            )}
   389→            {subTab === 'my-tenant' && (
   390→              <TenantUsersView
   391→                settings={settings}
   392→                tenantUsers={tenantUsers}
   393→                userRole={user!.role}
   394→                onOpenUserDialog={() => setAddUserOpen(true)}
   395→                onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   396→                onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   397→                onOpenMapConfig={() => { setSubTab('map'); setTimeout(() => setMapConfigOpen(true), 100); }}
   398→              />
   399→            )}
   400→            {subTab === 'map' && (
   401→              <MapConfigSection
   402→                currentBounds={currentBounds}
   403→                onConfigure={() => setMapConfigOpen(true)}
   404→              />
   405→            )}
   406→          </>
   407→        ) : (
   408→          /* ========== TENANT_ADMIN VIEW ========== */
   409→          <>
   410→            <TenantUsersView
   411→              settings={settings}
   412→              tenantUsers={tenantUsers}
   413→              userRole={user!.role}
   414→              onOpenUserDialog={() => setAddUserOpen(true)}
   415→              onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   416→              onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   417→              onOpenMapConfig={() => setMapConfigOpen(true)}
   418→            />
   419→            <MapConfigSection
   420→              currentBounds={currentBounds}
   421→              onConfigure={() => setMapConfigOpen(true)}
   422→            />
   423→            <InfoNotices />
   424→          </>
   425→        )}
   426→      </div>
   427→
   428→      {/* ===== MAP CONFIG DIALOG ===== */}
   429→      <Dialog open={mapConfigOpen} onOpenChange={setMapConfigOpen}>
   430→        <DialogContent className="sm:max-w-lg">
   431→          <DialogHeader>
   432→            <DialogTitle className="flex items-center gap-2">
   433→              <MapPin className="h-5 w-5 text-cyan" />
   434→              Configure Map Area
   435→            </DialogTitle>
   436→            <DialogDescription>
   437→              Set the geographic bounding box for the Polling Unit Map. Use a preset or enter custom WGS84 coordinates.
   438→            </DialogDescription>
   439→          </DialogHeader>
   440→          <div className="space-y-4 py-2">
   441→            <div className="space-y-1.5">
   442→              <label className="text-xs font-medium text-muted-foreground">Area Label</label>
   443→              <Input placeholder="e.g. Lagos, Southeast, Kano Central" value={mapLabel} onChange={(e) => setMapLabel(e.target.value)} className="h-9 text-sm" />
   444→            </div>
   445→            <div className="space-y-1.5">
   446→              <label className="text-xs font-medium text-muted-foreground">Quick Preset</label>
   447→              <Select onValueChange={(v) => { const preset = REGION_PRESETS.find(p => p.label === v); if (preset) handlePreset(preset); }}>
   448→                <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select a region preset..." /></SelectTrigger>
   449→                <SelectContent>
   450→                  {REGION_PRESETS.map(p => (
   451→                    <SelectItem key={p.label} value={p.label} className="text-xs">
   452→                      {p.label}
   453→                      <span className="text-muted-foreground ml-2">({p.minLat}, {p.maxLat}, {p.minLng}, {p.maxLng})</span>
   454→                    </SelectItem>
   455→                  ))}
   456→                </SelectContent>
   457→              </Select>
   458→            </div>
