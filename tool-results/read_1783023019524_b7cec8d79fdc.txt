   131→   131→function formatBytes(bytes: number) {
   132→   132→  if (bytes < 1024) return `${bytes} B`;
   133→   133→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   134→   134→  return `${(bytes / 1048576).toFixed(1)} MB`;
   135→   135→}
   136→   136→
   137→   137→function formatDuration(ms: number) {
   138→   138→  if (ms < 1000) return `${ms}ms`;
   139→   139→  return `${(ms / 1000).toFixed(1)}s`;
   140→   140→}
   141→   141→
   142→   142→function timeAgo(dateStr: string | null) {
   143→   143→  if (!dateStr) return 'N/A';
   144→   144→  const d = new Date(dateStr);
   145→   145→  const now = new Date();
   146→   146→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   147→   147→  if (diff < 1) return 'Just now';
   148→   148→  if (diff < 60) return `${diff}m ago`;
   149→   149→  if (diff < 1440) return `${Math.floor(diff / 60)}h ${diff % 60}m ago`;
   150→   150→  return `${Math.floor(diff / 1440)}d ago`;
   151→   151→}
   152→   152→
   153→   153→function formatDate(dateStr: string) {
   154→   154→  return new Date(dateStr).toLocaleDateString('en-US', {
   155→   155→    month: 'short', day: 'numeric', year: 'numeric',
   156→   156→  });
   157→   157→}
