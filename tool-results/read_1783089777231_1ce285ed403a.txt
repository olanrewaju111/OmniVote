    32→    32→
    33→    33→interface AgentUser {
    34→    34→  id: string;
    35→    35→  email: string;
    36→    36→  name: string;
    37→    37→  role: string;
    38→    38→  isOnline: boolean;
    39→    39→  lastSeenAt: string | null;

    48→    48→
    49→    49→export function AgentRoster() {
    50→    50→  const queryClient = useQueryClient();
    51→    51→  const { tenantId } = useDashboardStore();
    52→    52→  const [search, setSearch] = useState('');
    53→    53→  const [searchInput, setSearchInput] = useState('');
    54→    54→  const [roleFilter, setRoleFilter] = useState('ALL');
    55→    55→  const [addOpen, setAddOpen] = useState(false);
    56→    56→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    57→    57→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    58→    58→  const [editOpen, setEditOpen] = useState(false);
    59→    59→  const [editAgent, setEditAgent] = useState<AgentUser | null>(null);
    60→    60→  const [editName, setEditName] = useState('');
    61→    61→  const [editPhone, setEditPhone] = useState('');
    62→    62→  const [editRole, setEditRole] = useState('');
    63→    63→
    64→    64→  // Add agent form state
    65→    65→  const [newName, setNewName] = useState('');
    66→    66→  const [newEmail, setNewEmail] = useState('');
    67→    67→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    68→    68→
    69→    69→  // Fetch agents from proper API
    70→    70→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    71→    71→    queryKey: ['agents', search, roleFilter, tenantId],
    72→    72→    queryFn: () => {
    73→    73→      const params = new URLSearchParams();
    74→    74→      if (search) params.set('search', search);
    75→    75→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    76→    76→      if (tenantId) params.set('tenantId', tenantId);
    77→    77→      return fetchJson(`/api/agents?${params}`);
    78→    78→    },
    79→    79→  });
    80→    80→
    81→    81→  // Add agent mutation
    82→    82→  const addMutation = useMutation({
    83→    83→    mutationFn: (body: { name: string; email: string; role: string }) =>
    84→    84→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
    85→    85→        method: 'POST',
    86→    86→        headers: { 'Content-Type': 'application/json' },
    87→    87→        body: JSON.stringify(body),
    88→    88→      }),
    89→    89→    onSuccess: (data) => {
    90→    90→      if (data.error) {
    91→    91→        toast.error(data.error);
    92→    92→        return;
    93→    93→      }
    94→    94→      toast.success(`Agent "${data.user.name}" added successfully`);
    95→    95→      queryClient.invalidateQueries({ queryKey: ['agents'] });
    96→    96→      setAddOpen(false);
    97→    97→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
    98→    98→    },
    99→    99→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to add agent'),
   100→   100→  });

   103→   103→  const actionMutation = useMutation({
   104→   104→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
   105→   105→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
   106→   106→        method: 'PATCH',
   107→   107→        headers: { 'Content-Type': 'application/json' },
   108→   108→        body: JSON.stringify(body),
   109→   109→      }),
   110→   110→    onSuccess: (data, variables) => {
   111→   111→      if (data.error) {
   112→   112→        toast.error(data.error);
   113→   113→        return;
   114→   114→      }
   115→   115→      if (variables.action === 'DELETE') {
   116→   116→        toast.success(data.message || 'Agent removed');
   117→   117→      } else if (variables.action === 'REMOTE_WIPE') {
   118→   118→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   119→   119→      } else if (variables.action === 'CHANGE_ROLE') {
   120→   120→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   121→   121→      } else {
   122→   122→        toast.success('Status updated');
   123→   123→      }
   124→   124→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   125→   125→      setConfirmAction({ type: '', agent: null });
   126→   126→    },
   127→   127→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Action failed'),
   128→   128→  });
   129→   129→
   130→   130→  const users = data?.users || [];
   131→   131→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   132→   132→  const online = fieldAgents.filter(u => u.isOnline).length;
   133→   133→
   134→   134→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   135→   135→    acc[u.role] = (acc[u.role] || 0) + 1;
   136→   136→    return acc;
   137→   137→  }, {});
   138→   138→
   139→   139→  // Search with debounce via the query key
   140→   140→  const handleSearch = useCallback(() => {
   141→   141→    setSearch(searchInput);
   142→   142→  }, [searchInput]);
   143→   143→
   144→   144→  // Edit agent mutation
   145→   145→  const editMutation = useMutation({
   146→   146→    mutationFn: (body: { id: string; name?: string; phone?: string; role?: string }) =>
   147→   147→      fetchJson('/api/tenants/users', {
   148→   148→        method: 'PATCH',
   149→   149→        headers: { 'Content-Type': 'application/json' },
   150→   150→        body: JSON.stringify(body),
   151→   151→      }),
   152→   152→    onSuccess: () => {
   153→   153→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   154→   154→      toast.success('Agent details updated');
   155→   155→      setEditOpen(false);
   156→   156→      setEditAgent(null);
   157→   157→    },
   158→   158→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to update agent'),
   159→   159→  });

   161→   161→  const openEditDialog = (agent: AgentUser) => {
   162→   162→    setEditAgent(agent);
   163→   163→    setEditName(agent.name);
   164→   164→    setEditPhone('');
   165→   165→    setEditRole(agent.role);
   166→   166→    setEditOpen(true);
   167→   167→  };
   168→   168→
   169→   169→  const handleEditSave = () => {
   170→   170→    if (!editAgent) return;
   171→   171→    if (!editName.trim()) { toast.error('Name is required'); return; }
   172→   172→    editMutation.mutate({
   173→   173→      id: editAgent.id,
   174→   174→      name: editName.trim(),
   175→   175→      phone: editPhone.trim() || undefined,
   176→   176→      role: editRole,
   177→   177→    });
   178→   178→  };
   179→   179→
   180→   180→  // Handle confirm action
   181→   181→  const handleConfirmAction = () => {
   182→   182→    if (!confirmAction.agent) return;
   183→   183→    if (confirmAction.type === 'DELETE') {
   184→   184→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   185→   185→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   186→   186→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   187→   187→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   188→   188→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   189→   189→    }
   190→   190→  };
   191→   191→
   192→   192→  // Handle add agent
   193→   193→  const handleAddAgent = () => {
   194→   194→    if (!newName.trim() || !newEmail.trim()) {
   195→   195→      toast.error('Name and email are required');
   196→   196→      return;
   197→   197→    }
   198→   198→    if (!newEmail.includes('@')) {
   199→   199→      toast.error('Please enter a valid email');
   200→   200→      return;
   201→   201→    }
   202→   202→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   203→   203→  };
   204→   204→
   205→   205→  return (
   206→   206→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   207→   207→      {/* Header */}
   208→   208→      <div className="flex items-center justify-between flex-wrap gap-3">
   209→   209→        <div>
   210→   210→          <h2 className="text-lg font-semibold flex items-center gap-2">
   211→   211→            <Users className="h-5 w-5 text-emerald" />
   212→   212→            Agent Roster
   213→   213→          </h2>
   214→   214→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   215→   215→        </div>
   216→   216→        <Button
   217→   217→          onClick={() => setAddOpen(true)}
   218→   218→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   219→   219→        >
   220→   220→          <UserCheck className="h-4 w-4" />
   221→   221→          Add Agent
   222→   222→        </Button>
   223→   223→      </div>
   224→   224→
   225→   225→      {/* Summary cards */}
   226→   226→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   227→   227→        <Card className="border-border bg-card/40">
   228→   228→          <CardContent className="p-3.5">
   229→   229→            <div className="flex items-center gap-2 mb-1">
   230→   230→              <Users className="h-4 w-4 text-emerald" />
   231→   231→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   232→   232→            </div>
   233→   233→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   234→   234→          </CardContent>
   235→   235→        </Card>
   236→   236→        <Card className="border-emerald/20 bg-emerald/5">
   237→   237→          <CardContent className="p-3.5">
   238→   238→            <div className="flex items-center gap-2 mb-1">
   239→   239→              <UserCheck className="h-4 w-4 text-emerald" />
   240→   240→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   241→   241→            </div>
   242→   242→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   243→   243→          </CardContent>
   244→   244→        </Card>
   245→   245→        <Card className="border-amber/20 bg-amber/5">
   246→   246→          <CardContent className="p-3.5">
   247→   247→            <div className="flex items-center gap-2 mb-1">
   248→   248→              <UserX className="h-4 w-4 text-amber" />
   249→   249→              <span className="text-[11px] text-muted-foreground">Offline</span>
   250→   250→            </div>
   251→   251→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   252→   252→          </CardContent>
   253→   253→        </Card>
   254→   254→        <Card className="border-border bg-card/40">
   255→   255→          <CardContent className="p-3.5">
   256→   256→            <div className="flex items-center gap-2 mb-1">
   257→   257→              <Shield className="h-4 w-4 text-cyan" />
   258→   258→              <span className="text-[11px] text-muted-foreground">All Users</span>
   259→   259→            </div>
   260→   260→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   261→   261→          </CardContent>
   262→   262→        </Card>
   263→   263→      </div>
   264→   264→
   265→   265→      {/* Role filter + distribution badges */}
   266→   266→      <div className="flex items-center justify-between flex-wrap gap-3">
   267→   267→        <div className="flex items-center gap-2 flex-wrap">
   268→   268→          {Object.entries(roleCounts).map(([role, count]) => (
   269→   269→            <button
   270→   270→              key={role}
   271→   271→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   272→   272→              className="cursor-pointer"
   273→   273→            >
   274→   274→              <Badge
   275→   275→                variant="outline"
   276→   276→                className={cn(
   277→   277→                  'text-[11px] h-6 transition-colors',
   278→   278→                  roleFilter === role ? 'bg-foreground/10' : '',
   279→   279→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   280→   280→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   281→   281→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   282→   282→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   283→   283→                  'border-border text-muted-foreground'
   284→   284→                )}
   285→   285→              >
   286→   286→                {role.replace(/_/g, ' ')}: {count}
   287→   287→              </Badge>
   288→   288→            </button>
   289→   289→          ))}
   290→   290→        </div>
   291→   291→        {roleFilter !== 'ALL' && (
   292→   292→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   293→   293→            <X className="h-3 w-3" /> Clear filter
   294→   294→          </button>
   295→   295→        )}
   296→   296→      </div>
   297→   297→
   298→   298→      {/* Agent table */}
   299→   299→      <Card>
   300→   300→        <CardHeader className="pb-2 pt-4 px-4">
   301→   301→          <div className="flex items-center justify-between gap-3">
   302→   302→            <CardTitle className="text-sm">
   303→   303→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   304→   304→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   305→   305→            </CardTitle>
   306→   306→            <div className="flex items-center gap-2">
   307→   307→              <div className="relative">
   308→   308→                <label htmlFor="agent-search" className="sr-only">Search agents</label>
   309→   309→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   310→   310→                <Input
   311→   311→                  id="agent-search"
   312→   312→                  placeholder="Search agents..."
   313→   313→                  value={searchInput}
   314→   314→                  onChange={(e) => setSearchInput(e.target.value)}
   315→   315→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   316→   316→                  className="pl-8 h-8 w-48 text-xs"
   317→   317→                />
   318→   318→              </div>
   319→   319→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   320→   320→                Search
   321→   321→              </Button>
   322→   322→            </div>
   323→   323→          </div>
   324→   324→        </CardHeader>
   325→   325→        <CardContent className="px-0 pb-0">
   326→   326→          <div className="max-h-[480px] overflow-y-auto">
   327→   327→            {isLoading ? (
   328→   328→              <div className="flex items-center justify-center py-16">
   329→   329→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   330→   330→              </div>
   331→   331→            ) : users.length === 0 ? (
   332→   332→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   333→   333→                <Users className="h-8 w-8 mb-2 opacity-40" />
   334→   334→                <p className="text-sm">No users found</p>
   335→   335→              </div>
   336→   336→            ) : (
   337→   337→              <Table>
   338→   338→                <TableHeader>
   339→   339→                  <TableRow className="border-border hover:bg-transparent">
   340→   340→                    <TableHead className="text-[11px] h-9">User</TableHead>
   341→   341→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   342→   342→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   343→   343→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   344→   344→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   345→   345→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   346→   346→                  </TableRow>
   347→   347→                </TableHeader>
   348→   348→                <TableBody>
   349→   349→                  <AnimatePresence>
   350→   350→                    {users.map(agent => (
   351→   351→                      <motion.tr
   352→   352→                        key={agent.id}
   353→   353→                        initial={{ opacity: 0, y: 4 }}
   354→   354→                        animate={{ opacity: 1, y: 0 }}
   355→   355→                        exit={{ opacity: 0, y: -4 }}
   356→   356→                        className="border-border hover:bg-card/60 transition-colors"
   357→   357→                      >
   358→   358→                        <TableCell className="py-2.5">
   359→   359→                          <div className="flex items-center gap-2.5">
   360→   360→                            <div className="relative">
   361→   361→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   362→   362→                                {agent.name.split(' ').map(n => n[0]).join('')}
   363→   363→                              </div>
   364→   364→                              <span className={cn(
   365→   365→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   366→   366→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   367→   367→                              )} />
   368→   368→                            </div>
   369→   369→                            <div className="min-w-0">
   370→   370→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   371→   371→                              {agent.lastSeenAt && (
   372→   372→                                <span className="text-[10px] text-muted-foreground/60">
   373→   373→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   374→   374→                                </span>
   375→   375→                              )}
   376→   376→                            </div>
   377→   377→                          </div>
   378→   378→                        </TableCell>
   379→   379→                        <TableCell className="py-2.5">
   380→   380→                          <Badge
   381→   381→                            variant="outline"
   382→   382→                            className={cn(
   383→   383→                              'text-[10px] h-5',
   384→   384→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   385→   385→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   386→   386→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   387→   387→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   388→   388→                              'border-border text-muted-foreground'
   389→   389→                            )}
   390→   390→                          >
   391→   391→                            {agent.role.replace(/_/g, ' ')}
   392→   392→                          </Badge>
   393→   393→                        </TableCell>
   394→   394→                        <TableCell className="py-2.5">
   395→   395→                          <button
   396→   396→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   397→   397→                            className="cursor-pointer"
   398→   398→                            disabled={actionMutation.isPending}
   399→   399→                          >
   400→   400→                            <Badge
   401→   401→                              variant="outline"
   402→   402→                              className={cn(
   403→   403→                                'text-[10px] h-5 transition-colors cursor-pointer',
   404→   404→                                agent.isOnline
   405→   405→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   406→   406→                                  : 'border-border text-muted-foreground hover:bg-card'
   407→   407→                              )}
   408→   408→                            >
   409→   409→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   410→   410→                              {agent.isOnline ? 'Online' : 'Offline'}
   411→   411→                            </Badge>
   412→   412→                          </button>
   413→   413→                        </TableCell>
   414→   414→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   415→   415→                        <TableCell className="py-2.5 hidden md:table-cell">
   416→   416→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   417→   417→                            {agent._count?.incidents ?? 0}
   418→   418→                          </span>
   419→   419→                        </TableCell>
