   121→function generateNigerianNumbers(count: number): string[] {
   122→  const prefixes = ['0703', '0803', '0806', '0809', '0810', '0811', '0812', '0813', '0814', '0815', '0816', '0817', '0818', '0902', '0903', '0905', '0906', '0907', '0908', '0909', '0913', '0915', '0916'];
   123→  const numbers: string[] = [];
   124→  for (let i = 0; i < count; i++) {
   125→    const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
   126→    const suffix = String(Math.floor(Math.random() * 10_000_000)).padStart(7, '0');
   127→    numbers.push(`${prefix}${suffix}`);
   128→  }
   129→  return numbers;
   130→}
   131→
   132→function formatNumber(n: number): string {
   133→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   134→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   135→  return String(n);
   136→}
   137→
   138→function formatDate(d: string | null) {
   139→  if (!d) return '—';
   140→  return new Date(d).toLocaleDateString('en-NG', {
   141→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   142→  });
   143→}
   144→
   145→function segmentLabel(seg: string) {
   146→  return SEGMENT_OPTIONS.find(s => s.value === seg)?.label ?? seg;
   147→}
