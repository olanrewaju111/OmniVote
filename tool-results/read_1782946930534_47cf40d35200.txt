    62→    62→

    65→    65→  votes: string; // string for input, parsed on submit

    67→    67→
    68→    68→export function SubmitReport() {
    69→    69→  const { user, electionTier } = useDashboardStore();
    70→    70→  const queryClient = useQueryClient();
    71→    71→
    72→    72→  const [activeTab, setActiveTab] = useState<Tab>('results');
    73→    73→  const [selectedPU, setSelectedPU] = useState('');
    74→    74→  const [submitting, setSubmitting] = useState(false);
    75→    75→  const [submitted, setSubmitted] = useState(false);
    76→    76→  const [submittedType, setSubmittedType] = useState('');
    77→    77→
    78→    78→  // ---- RESULTS STATE ----
    79→    79→  const [accredited, setAccredited] = useState('');
    80→    80→  const [validVotes, setValidVotes] = useState('');
    81→    81→  const [rejectedBallots, setRejectedBallots] = useState('');
    82→    82→  const [partyVotes, setPartyVotes] = useState<PartyVote[]>(
    83→    83→    (PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL).map(p => ({ party: p.code, votes: '' }))
    84→    84→  );
    85→    85→
    86→    86→  // ---- STATISTICS STATE ----
    87→    87→  const [bvasUsed, setBvasUsed] = useState(true);
    88→    88→  const [materialsOnTime, setMaterialsOnTime] = useState(true);
    89→    89→  const [securityPresent, setSecurityPresent] = useState(true);
    90→    90→  const [violenceOccurred, setViolenceOccurred] = useState(false);
    91→    91→  const [statsNotes, setStatsNotes] = useState('');
    92→    92→
    93→    93→  // ---- INCIDENT STATE ----
    94→    94→  const [incType, setIncType] = useState('');
    95→    95→  const [incSeverity, setIncSeverity] = useState('MEDIUM');
    96→    96→  const [incDescription, setIncDescription] = useState('');
    97→    97→
    98→    98→  // Fetch polling units for the agent (show first 10 as options)
    99→    99→  const { data: puData } = useQuery({
   100→   100→    queryKey: ['my-polling-units'],
   101→   101→    queryFn: () => fetch('/api/dashboard').then(r => r.json()).then(d => d.pollingUnits?.slice(0, 15) || []),
   102→   102→  });
   103→   103→
   104→   104→  const pollingUnits = puData || [];
   105→   105→
   106→   106→  const parties = PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL;
   107→   107→
   108→   108→  // Add a party vote row
   109→   109→  const addPartyRow = () => {
   110→   110→    const existingCodes = partyVotes.map(p => p.party);
   111→   111→    const missing = parties.find(p => !existingCodes.includes(p.code));
   112→   112→    if (missing) {
   113→   113→      setPartyVotes(prev => [...prev, { party: missing.code, votes: '' }]);
   114→   114→    }
   115→   115→  };
   116→   116→
   117→   117→  const removePartyRow = (idx: number) => {
   118→   118→    if (partyVotes.length > 2) {
   119→   119→      setPartyVotes(prev => prev.filter((_, i) => i !== idx));
   120→   120→    }
   121→   121→  };
   122→   122→
   123→   123→  const updatePartyVote = (idx: number, field: 'party' | 'votes', value: string) => {
   124→   124→    setPartyVotes(prev => prev.map((p, i) => i === idx ? { ...p, [field]: value } : p));
   125→   125→  };
   126→   126→
   127→   127→  // Auto-calculate totals
   128→   128→  const totalVotesCast = partyVotes.reduce((s, p) => s + (parseInt(p.votes) || 0), 0);
   129→   129→  const accreditedNum = parseInt(accredited) || 0;
   130→   130→  const validVotesNum = parseInt(validVotes) || 0;
   131→   131→  const rejectedNum = parseInt(rejectedBallots) || 0;
   132→   132→
   133→   133→  // ---- SUBMIT HANDLERS ----
   134→   134→
   135→   135→  const resultMutation = useMutation({
   136→   136→    mutationFn: (body: Record<string, unknown>) =>
   137→   137→      fetch('/api/results', {
   138→   138→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   139→   139→      }).then(r => r.json()),
   140→   140→    onSuccess: (data) => {
   141→   141→      if (data.error) { toast.error(data.error); return; }
   142→   142→      toast.success('Election results submitted successfully!');
   143→   143→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   144→   144→      queryClient.invalidateQueries({ queryKey: ['situation-room'] });
   145→   145→      setSubmitted(true); setSubmittedType('results');
   146→   146→      setTimeout(() => setSubmitted(false), 4000);
   147→   147→    },
   148→   148→    onError: () => toast.error('Failed to submit results'),
   149→   149→  });
   150→   150→
   151→   151→  const incidentMutation = useMutation({
   152→   152→    mutationFn: (body: Record<string, unknown>) =>
   153→   153→      fetch('/api/incidents', {
   154→   154→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   155→   155→      }).then(r => r.json()),
   156→   156→    onSuccess: (data) => {
   157→   157→      if (data.error) { toast.error(data.error); return; }
   158→   158→      toast.success('Incident report submitted!');
   159→   159→      if (data.incident?.gpsAnomaly) {
   160→   160→        toast.warning('GPS anomaly detected — report quarantined for review');
   161→   161→      }
   162→   162→      queryClient.invalidateQueries({ queryKey: ['incidents'] });
   163→   163→      queryClient.invalidateQueries({ queryKey: ['alerts'] });
   164→   164→      setSubmitted(true); setSubmittedType('incident');
   165→   165→      setIncDescription(''); setIncType('');
   166→   166→      setTimeout(() => setSubmitted(false), 4000);
   167→   167→    },
   168→   168→    onError: () => toast.error('Failed to submit incident'),
   169→   169→  });
   170→   170→
   171→   171→  const handleSubmitResults = () => {
   172→   172→    if (!user?.id || !selectedPU) {
   173→   173→      toast.error('Please select a polling unit');
   174→   174→      return;
   175→   175→    }
   176→   176→    if (partyVotes.every(p => !p.votes)) {
   177→   177→      toast.error('Enter at least one party vote count');
   178→   178→      return;
   179→   179→    }
   180→   180→
   181→   181→    const parsedPartyResults = partyVotes
   182→   182→      .filter(p => p.votes)
   183→   183→      .map(p => {
   184→   184→        const partyInfo = parties.find(pt => pt.code === p.party);
   185→   185→        return { party: p.party, name: partyInfo?.name || p.party, votes: parseInt(p.votes) || 0, color: partyInfo?.color || '#888' };
   186→   186→      });
   187→   187→
   188→   188→    setSubmitting(true);
   189→   189→    resultMutation.mutate({
   190→   190→      reporterId: user.id,
   191→   191→      pollingUnitId: selectedPU,
   192→   192→      accreditedVoters: accreditedNum,
   193→   193→      totalValidVotes: validVotesNum,
   194→   194→      rejectedBallots: rejectedNum,
   195→   195→      totalVotesCast,
   196→   196→      partyResults: parsedPartyResults,
   197→   197→      bvasUsed, materialsArrivedOnTime: materialsOnTime,
   198→   198→      securityPresent, violenceOccurred, notes: statsNotes,
   199→   199→    }, { onSettled: () => setSubmitting(false) });
   200→   200→  };
   201→   201→
   202→   202→  const handleSubmitIncident = () => {
   203→   203→    if (!user?.id) return;
   204→   204→    if (!incType) { toast.error('Select an incident type'); return; }
   205→   205→    if (!incDescription.trim()) { toast.error('Describe the incident'); return; }
   206→   206→
   207→   207→    const typeConfig = INCIDENT_TYPES.find(t => t.value === incType);
   208→   208→    setSubmitting(true);
   209→   209→    incidentMutation.mutate({
   210→   210→      reporterId: user.id,
   211→   211→      pollingUnitId: selectedPU || undefined,
   212→   212→      type: incType,
   213→   213→      severity: typeConfig?.severity || incSeverity,
   214→   214→      description: incDescription.trim(),
   215→   215→    }, { onSettled: () => setSubmitting(false) });
   216→   216→  };
   217→   217→
   218→   218→  const selectedPUData = pollingUnits.find(p => p.id === selectedPU);
   219→   219→
   220→   220→  return (
   221→   221→    <div className="h-full overflow-y-auto p-4 sm:p-6">
   222→   222→      <div className="max-w-2xl mx-auto space-y-4">
   223→   223→        {/* Header */}
   224→   224→        <div>
   225→   225→          <h2 className="text-lg font-semibold flex items-center gap-2">
   226→   226→            <Send className="h-5 w-5 text-emerald" />
   227→   227→            Field Report
   228→   228→          </h2>
   229→   229→          <p className="text-sm text-muted-foreground mt-0.5">
   230→   230→            Submit election results, polling unit statistics, or report incidents and infractions.
   231→   231→          </p>
   232→   232→        </div>
   233→   233→
   234→   234→        {/* Agent info + PU selector */}
   235→   235→        <Card className="border-emerald/20 bg-emerald/5">
   236→   236→          <CardContent className="p-3.5 flex items-center gap-3">
   237→   237→            <div className="w-10 h-10 rounded-full bg-emerald/20 flex items-center justify-center text-sm font-bold text-emerald">
   238→   238→              {user?.name.split(' ').map(n => n[0]).join('')}
   239→   239→            </div>
   240→   240→            <div className="flex-1 min-w-0">
   241→   241→              <p className="text-sm font-medium">{user?.name}</p>
   242→   242→              <div className="flex items-center gap-1.5 mt-0.5">
   243→   243→                <ShieldCheck className="h-3 w-3 text-emerald" />
   244→   244→                <span className="text-[10px] text-muted-foreground">In-App Capture &middot; C2PA enabled</span>
   245→   245→              </div>
   246→   246→            </div>
   247→   247→          </CardContent>
   248→   248→        </Card>
   249→   249→
   250→   250→        {/* Polling Unit Selector */}
   251→   251→        <div className="space-y-1.5">
   252→   252→          <label className="text-xs font-medium flex items-center gap-1.5">
   253→   253→            <MapPin className="h-3.5 w-3.5 text-emerald" />
   254→   254→            Select Polling Unit
   255→   255→          </label>
   256→   256→          <Select value={selectedPU} onValueChange={setSelectedPU}>
   257→   257→            <SelectTrigger className="h-10">
   258→   258→              <SelectValue placeholder="Choose your assigned polling unit..." />
   259→   259→            </SelectTrigger>
   260→   260→            <SelectContent>
   261→   261→              {pollingUnits.map(pu => (
   262→   262→                <SelectItem key={pu.id} value={pu.id}>
   263→   263→                  <span className="flex items-center gap-2">
   264→   264→                    <span className="text-muted-foreground text-[10px]">{pu.code}</span>
   265→   265→                    <span>{pu.name}</span>
   266→   266→                    <span className="text-muted-foreground text-[10px]">{pu.state}/{pu.lga}</span>
   267→   267→                    {pu.status === 'FLAGGED' && <Badge variant="outline" className="text-[9px] h-4 border-amber/30 text-amber ml-1">FLAGGED</Badge>}
   268→   268→                  </span>
   269→   269→                </SelectItem>
   270→   270→              ))}
   271→   271→            </SelectContent>
   272→   272→          </Select>
   273→   273→          {selectedPUData && (
   274→   274→            <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   275→   275→              <Badge variant="outline" className="text-[10px] h-5 border-emerald/30 text-emerald">
   276→   276→                <span className="w-1.5 h-1.5 rounded-full bg-emerald mr-1" />IN GEOFENCE
   277→   277→              </Badge>
   278→   278→              <span>{selectedPUData.registered} registered voters</span>
   279→   279→              <span>{selectedPUData.state} / {selectedPUData.lga}</span>
   280→   280→            </div>
   281→   281→          )}
   282→   282→        </div>
   283→   283→
   284→   284→        {/* Tab selector */}
   285→   285→        <div className="grid grid-cols-3 gap-1 p-1 rounded-lg bg-card/60 border border-border">
   286→   286→          {[
   287→   287→            { id: 'results' as Tab, label: 'Election Results', icon: <Vote className="h-4 w-4" /> },
   288→   288→            { id: 'statistics' as Tab, label: 'Statistics', icon: <BarChart3 className="h-4 w-4" /> },
   289→   289→            { id: 'incident' as Tab, label: 'Incident / Infraction', icon: <FileWarning className="h-4 w-4" /> },
   290→   290→          ].map(tab => (
   291→   291→            <button
   292→   292→              key={tab.id}
   293→   293→              onClick={() => setActiveTab(tab.id)}
   294→   294→              className={cn(
   295→   295→                'flex items-center justify-center gap-1.5 px-3 py-2.5 rounded-md text-xs font-medium transition-all',
   296→   296→                activeTab === tab.id
   297→   297→                  ? 'bg-emerald/15 text-emerald shadow-sm'
   298→   298→                  : 'text-muted-foreground hover:text-foreground hover:bg-card'
   299→   299→              )}
   300→   300→            >
   301→   301→              {tab.icon}
   302→   302→              <span className="hidden sm:inline">{tab.label}</span>
   303→   303→            </button>
   304→   304→          ))}
   305→   305→        </div>
   306→   306→
   307→   307→        {/* Tab content */}
   308→   308→        <AnimatePresence mode="wait">
   309→   309→          <motion.div
   310→   310→            key={activeTab}
   311→   311→            initial={{ opacity: 0, y: 4 }}
   312→   312→            animate={{ opacity: 1, y: 0 }}
   313→   313→            exit={{ opacity: 0, y: -4 }}
   314→   314→            transition={{ duration: 0.15 }}
   315→   315→          >
   316→   316→            {/* ===== ELECTION RESULTS TAB ===== */}
   317→   317→            {activeTab === 'results' && (
   318→   318→              <Card>
   319→   319→                <CardContent className="p-4 space-y-4">
   320→   320→                  <div>
   321→   321→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   322→   322→                      <Vote className="h-4 w-4 text-emerald" />
   323→   323→                      Election Results Form
   324→   324→                    </h3>
   325→   325→                    <p className="text-[11px] text-muted-foreground mt-0.5">
   326→   326→                      Enter the official vote counts from Form EC8A at your polling unit.
   327→   327→                    </p>
   328→   328→                  </div>
   329→   329→
   330→   330→                  {/* Summary fields */}
   331→   331→                  <div className="grid grid-cols-2 gap-3">
   332→   332→                    <div className="space-y-1.5">
   333→   333→                      <label className="text-[11px] font-medium text-muted-foreground">Accredited Voters</label>
   334→   334→                      <Input
   335→   335→                        type="number" min="0"
   336→   336→                        placeholder="0"
   337→   337→                        value={accredited}
   338→   338→                        onChange={(e) => setAccredited(e.target.value)}
   339→   339→                        className="h-10 text-sm tabular-nums"
   340→   340→                      />
   341→   341→                    </div>
   342→   342→                    <div className="space-y-1.5">
   343→   343→                      <label className="text-[11px] font-medium text-muted-foreground">Rejected Ballots</label>
   344→   344→                      <Input
   345→   345→                        type="number" min="0"
   346→   346→                        placeholder="0"
   347→   347→                        value={rejectedBallots}
   348→   348→                        onChange={(e) => setRejectedBallots(e.target.value)}
   349→   349→                        className="h-10 text-sm tabular-nums"
   350→   350→                      />
   351→   351→                    </div>
   352→   352→                  </div>
   353→   353→
   354→   354→                  <Separator />
   355→   355→
   356→   356→                  {/* Party vote entries */}
   357→   357→                  <div className="space-y-2">
   358→   358→                    <div className="flex items-center justify-between">
   359→   359→                      <label className="text-xs font-medium">Vote Count by Party</label>
   360→   360→                      <Button variant="ghost" size="sm" className="h-6 text-[10px] gap-1" onClick={addPartyRow}>
   361→   361→                        <Plus className="h-3 w-3" /> Add Party
   362→   362→                      </Button>
   363→   363→                    </div>
   364→   364→
   365→   365→                    <div className="space-y-2">
   366→   366→                      {partyVotes.map((pv, idx) => {
   367→   367→                        const partyInfo = parties.find(p => p.code === pv.party);
   368→   368→                        return (
   369→   369→                          <div key={idx} className="flex items-center gap-2">
   370→   370→                            <div
   371→   371→                              className="w-3 h-8 rounded-sm shrink-0"
   372→   372→                              style={{ backgroundColor: partyInfo?.color || '#555' }}
   373→   373→                            />
   374→   374→                            <div className="flex-1 min-w-0">
   375→   375→                              <Select value={pv.party} onValueChange={(v) => updatePartyVote(idx, 'party', v)}>
   376→   376→                                <SelectTrigger className="h-8 text-xs">
   377→   377→                                  <SelectValue />
   378→   378→                                </SelectTrigger>
   379→   379→                                <SelectContent>
   380→   380→                                  {parties.map(p => (
   381→   381→                                    <SelectItem key={p.code} value={p.code}>
   382→   382→                                      <span className="flex items-center gap-2">
   383→   383→                                        <span className="w-2 h-2 rounded-sm" style={{ backgroundColor: p.color }} />
   384→   384→                                        <span className="font-medium">{p.code}</span>
   385→   385→                                        <span className="text-muted-foreground text-[10px] hidden sm:inline">{p.name}</span>
   386→   386→                                      </span>
   387→   387→                                    </SelectItem>
   388→   388→                                  ))}
   389→   389→                                </SelectContent>
   390→   390→                              </Select>
   391→   391→                            </div>
   392→   392→                            <Input
   393→   393→                              type="number" min="0" placeholder="0"
   394→   394→                              value={pv.votes}
   395→   395→                              onChange={(e) => updatePartyVote(idx, 'votes', e.target.value)}
   396→   396→                              className="w-28 h-8 text-sm tabular-nums text-right"
   397→   397→                            />
   398→   398→                            {partyVotes.length > 2 && (
   399→   399→                              <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-rose" onClick={() => removePartyRow(idx)}>
   400→   400→                                <X className="h-3.5 w-3.5" />
   401→   401→                              </Button>
   402→   402→                            )}
   403→   403→                          </div>
   404→   404→                        );
   405→   405→                      })}
