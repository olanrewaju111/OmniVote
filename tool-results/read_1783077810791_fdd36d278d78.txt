     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { useDashboardStore } from '@/store/dashboard';
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table';
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter,
    16→  DialogHeader, DialogTitle,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    20→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  Search, Users, UserCheck, UserX, Shield, ShieldAlert, Wrench,
    27→  Loader2, FileText, Trash2, Eye, ToggleLeft, X, Pencil,
    28→} from 'lucide-react';
    29→import { cn } from '@/lib/utils';
    30→import { fetchJson } from '@/lib/api';
    31→import { toast } from 'sonner';
    32→
    33→interface AgentUser {
    34→  id: string;
    35→  email: string;
    36→  name: string;
    37→  role: string;
    38→  isOnline: boolean;
    39→  lastSeenAt: string | null;
    40→  createdAt: string;
    41→  _count?: { incidents: number; auditLogs: number };
    42→}
    43→
    44→interface ReportsSlideOver {
    45→  agent: AgentUser | null;
    46→  open: boolean;
    47→}
    48→
    49→export function AgentRoster() {
    50→  const queryClient = useQueryClient();
    51→  const { tenantId } = useDashboardStore();
    52→  const [search, setSearch] = useState('');
    53→  const [searchInput, setSearchInput] = useState('');
    54→  const [roleFilter, setRoleFilter] = useState('ALL');
    55→  const [addOpen, setAddOpen] = useState(false);
    56→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    57→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    58→  const [editOpen, setEditOpen] = useState(false);
    59→  const [editAgent, setEditAgent] = useState<AgentUser | null>(null);
    60→  const [editName, setEditName] = useState('');
    61→  const [editPhone, setEditPhone] = useState('');
    62→  const [editRole, setEditRole] = useState('');
    63→
    64→  // Add agent form state
    65→  const [newName, setNewName] = useState('');
    66→  const [newEmail, setNewEmail] = useState('');
    67→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    68→
    69→  // Fetch agents from proper API
    70→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    71→    queryKey: ['agents', search, roleFilter, tenantId],
    72→    queryFn: () => {
    73→      const params = new URLSearchParams();
    74→      if (search) params.set('search', search);
    75→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    76→      if (tenantId) params.set('tenantId', tenantId);
    77→      return fetchJson(`/api/agents?${params}`);
    78→    },
    79→  });
    80→
    81→  // Add agent mutation
    82→  const addMutation = useMutation({
    83→    mutationFn: (body: { name: string; email: string; role: string }) =>
    84→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
    85→        method: 'POST',
    86→        headers: { 'Content-Type': 'application/json' },
    87→        body: JSON.stringify(body),
    88→      }),
    89→    onSuccess: (data) => {
    90→      if (data.error) {
    91→        toast.error(data.error);
    92→        return;
    93→      }
    94→      toast.success(`Agent "${data.user.name}" added successfully`);
    95→      queryClient.invalidateQueries({ queryKey: ['agents'] });
    96→      setAddOpen(false);
    97→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
    98→    },
    99→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to add agent'),
   100→  });
   101→
   102→  // Action mutation (toggle online, remote wipe, change role, delete)
   103→  const actionMutation = useMutation({
   104→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
   105→      fetchJson(`/api/agents?tenantId=${tenantId}`, {
   106→        method: 'PATCH',
   107→        headers: { 'Content-Type': 'application/json' },
   108→        body: JSON.stringify(body),
   109→      }),
   110→    onSuccess: (data, variables) => {
   111→      if (data.error) {
   112→        toast.error(data.error);
   113→        return;
   114→      }
   115→      if (variables.action === 'DELETE') {
   116→        toast.success(data.message || 'Agent removed');
   117→      } else if (variables.action === 'REMOTE_WIPE') {
   118→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   119→      } else if (variables.action === 'CHANGE_ROLE') {
   120→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   121→      } else {
   122→        toast.success('Status updated');
   123→      }
   124→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   125→      setConfirmAction({ type: '', agent: null });
   126→    },
   127→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Action failed'),
   128→  });
   129→
   130→  const users = data?.users || [];
   131→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   132→  const online = fieldAgents.filter(u => u.isOnline).length;
   133→
   134→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   135→    acc[u.role] = (acc[u.role] || 0) + 1;
   136→    return acc;
   137→  }, {});
   138→
   139→  // Search with debounce via the query key
   140→  const handleSearch = useCallback(() => {
   141→    setSearch(searchInput);
   142→  }, [searchInput]);
   143→
   144→  // Edit agent mutation
   145→  const editMutation = useMutation({
   146→    mutationFn: (body: { id: string; name?: string; phone?: string; role?: string }) =>
   147→      fetchJson('/api/tenants/users', {
   148→        method: 'PATCH',
   149→        headers: { 'Content-Type': 'application/json' },
   150→        body: JSON.stringify(body),
   151→      }),
   152→    onSuccess: () => {
   153→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   154→      toast.success('Agent details updated');
   155→      setEditOpen(false);
   156→      setEditAgent(null);
   157→    },
   158→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to update agent'),
   159→  });
   160→
   161→  const openEditDialog = (agent: AgentUser) => {
   162→    setEditAgent(agent);
   163→    setEditName(agent.name);
   164→    setEditPhone('');
   165→    setEditRole(agent.role);
   166→    setEditOpen(true);
   167→  };
   168→
   169→  const handleEditSave = () => {
   170→    if (!editAgent) return;
   171→    if (!editName.trim()) { toast.error('Name is required'); return; }
   172→    editMutation.mutate({
   173→      id: editAgent.id,
   174→      name: editName.trim(),
   175→      phone: editPhone.trim() || undefined,
   176→      role: editRole,
   177→    });
   178→  };
   179→
   180→  // Handle confirm action
   181→  const handleConfirmAction = () => {
   182→    if (!confirmAction.agent) return;
   183→    if (confirmAction.type === 'DELETE') {
   184→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   185→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   186→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   187→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   188→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   189→    }
   190→  };
   191→
   192→  // Handle add agent
   193→  const handleAddAgent = () => {
   194→    if (!newName.trim() || !newEmail.trim()) {
   195→      toast.error('Name and email are required');
   196→      return;
   197→    }
   198→    if (!newEmail.includes('@')) {
   199→      toast.error('Please enter a valid email');
   200→      return;
   201→    }
   202→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   203→  };
   204→
   205→  return (
   206→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   207→      {/* Header */}
   208→      <div className="flex items-center justify-between flex-wrap gap-3">
   209→        <div>
   210→          <h2 className="text-lg font-semibold flex items-center gap-2">
   211→            <Users className="h-5 w-5 text-emerald" />
   212→            Agent Roster
   213→          </h2>
   214→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   215→        </div>
   216→        <Button
   217→          onClick={() => setAddOpen(true)}
   218→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   219→        >
   220→          <UserCheck className="h-4 w-4" />
   221→          Add Agent
   222→        </Button>
   223→      </div>
   224→
   225→      {/* Summary cards */}
   226→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   227→        <Card className="border-border bg-card/40">
   228→          <CardContent className="p-3.5">
   229→            <div className="flex items-center gap-2 mb-1">
   230→              <Users className="h-4 w-4 text-emerald" />
   231→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   232→            </div>
   233→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   234→          </CardContent>
   235→        </Card>
   236→        <Card className="border-emerald/20 bg-emerald/5">
   237→          <CardContent className="p-3.5">
   238→            <div className="flex items-center gap-2 mb-1">
   239→              <UserCheck className="h-4 w-4 text-emerald" />
   240→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   241→            </div>
   242→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   243→          </CardContent>
   244→        </Card>
   245→        <Card className="border-amber/20 bg-amber/5">
   246→          <CardContent className="p-3.5">
   247→            <div className="flex items-center gap-2 mb-1">
   248→              <UserX className="h-4 w-4 text-amber" />
   249→              <span className="text-[11px] text-muted-foreground">Offline</span>
   250→            </div>
   251→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   252→          </CardContent>
   253→        </Card>
   254→        <Card className="border-border bg-card/40">
   255→          <CardContent className="p-3.5">
   256→            <div className="flex items-center gap-2 mb-1">
   257→              <Shield className="h-4 w-4 text-cyan" />
   258→              <span className="text-[11px] text-muted-foreground">All Users</span>
   259→            </div>
   260→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   261→          </CardContent>
   262→        </Card>
   263→      </div>
   264→
   265→      {/* Role filter + distribution badges */}
   266→      <div className="flex items-center justify-between flex-wrap gap-3">
   267→        <div className="flex items-center gap-2 flex-wrap">
   268→          {Object.entries(roleCounts).map(([role, count]) => (
   269→            <button
   270→              key={role}
   271→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   272→              className="cursor-pointer"
   273→            >
   274→              <Badge
   275→                variant="outline"
   276→                className={cn(
   277→                  'text-[11px] h-6 transition-colors',
   278→                  roleFilter === role ? 'bg-foreground/10' : '',
   279→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   280→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   281→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   282→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   283→                  'border-border text-muted-foreground'
   284→                )}
   285→              >
   286→                {role.replace(/_/g, ' ')}: {count}
   287→              </Badge>
   288→            </button>
   289→          ))}
   290→        </div>
   291→        {roleFilter !== 'ALL' && (
   292→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   293→            <X className="h-3 w-3" /> Clear filter
   294→          </button>
   295→        )}
   296→      </div>
   297→
   298→      {/* Agent table */}
   299→      <Card>
   300→        <CardHeader className="pb-2 pt-4 px-4">
   301→          <div className="flex items-center justify-between gap-3">
   302→            <CardTitle className="text-sm">
   303→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   304→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   305→            </CardTitle>
   306→            <div className="flex items-center gap-2">
   307→              <div className="relative">
   308→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   309→                <Input
   310→                  placeholder="Search agents..."
   311→                  value={searchInput}
   312→                  onChange={(e) => setSearchInput(e.target.value)}
   313→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   314→                  className="pl-8 h-8 w-48 text-xs"
   315→                />
   316→              </div>
   317→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   318→                Search
   319→              </Button>
   320→            </div>
   321→          </div>
   322→        </CardHeader>
   323→        <CardContent className="px-0 pb-0">
   324→          <div className="max-h-[480px] overflow-y-auto">
   325→            {isLoading ? (
   326→              <div className="flex items-center justify-center py-16">
   327→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   328→              </div>
   329→            ) : users.length === 0 ? (
   330→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   331→                <Users className="h-8 w-8 mb-2 opacity-40" />
   332→                <p className="text-sm">No users found</p>
   333→              </div>
   334→            ) : (
   335→              <Table>
   336→                <TableHeader>
   337→                  <TableRow className="border-border hover:bg-transparent">
   338→                    <TableHead className="text-[11px] h-9">User</TableHead>
   339→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   340→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   341→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   342→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   343→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   344→                  </TableRow>
   345→                </TableHeader>
   346→                <TableBody>
   347→                  <AnimatePresence>
   348→                    {users.map(agent => (
   349→                      <motion.tr
   350→                        key={agent.id}
   351→                        initial={{ opacity: 0, y: 4 }}
   352→                        animate={{ opacity: 1, y: 0 }}
   353→                        exit={{ opacity: 0, y: -4 }}
   354→                        className="border-border hover:bg-card/60 transition-colors"
   355→                      >
   356→                        <TableCell className="py-2.5">
   357→                          <div className="flex items-center gap-2.5">
   358→                            <div className="relative">
   359→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   360→                                {agent.name.split(' ').map(n => n[0]).join('')}
   361→                              </div>
   362→                              <span className={cn(
   363→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   364→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   365→                              )} />
   366→                            </div>
   367→                            <div className="min-w-0">
   368→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   369→                              {agent.lastSeenAt && (
   370→                                <span className="text-[10px] text-muted-foreground/60">
   371→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   372→                                </span>
   373→                              )}
   374→                            </div>
   375→                          </div>
   376→                        </TableCell>
   377→                        <TableCell className="py-2.5">
   378→                          <Badge
   379→                            variant="outline"
   380→                            className={cn(
   381→                              'text-[10px] h-5',
   382→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   383→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   384→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   385→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   386→                              'border-border text-muted-foreground'
   387→                            )}
   388→                          >
   389→                            {agent.role.replace(/_/g, ' ')}
   390→                          </Badge>
   391→                        </TableCell>
   392→                        <TableCell className="py-2.5">
   393→                          <button
   394→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   395→                            className="cursor-pointer"
   396→                            disabled={actionMutation.isPending}
   397→                          >
   398→                            <Badge
   399→                              variant="outline"
   400→                              className={cn(
   401→                                'text-[10px] h-5 transition-colors cursor-pointer',
   402→                                agent.isOnline
   403→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   404→                                  : 'border-border text-muted-foreground hover:bg-card'
   405→                              )}
   406→                            >
   407→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   408→                              {agent.isOnline ? 'Online' : 'Offline'}
   409→                            </Badge>
   410→                          </button>
   411→                        </TableCell>
   412→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   413→                        <TableCell className="py-2.5 hidden md:table-cell">
   414→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   415→                            {agent._count?.incidents ?? 0}
   416→                          </span>
   417→                        </TableCell>
   418→                        <TableCell className="py-2.5 text-right">
   419→                          <div className="flex items-center justify-end gap-0.5">
   420→                            <Button
   421→                              variant="ghost"
   422→                              size="sm"
   423→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-cyan"
   424→                              title="Edit Details"
   425→                              onClick={() => openEditDialog(agent)}
   426→                              disabled={agent.role === 'SUPER_ADMIN'}
   427→                            >
   428→                              <Pencil className="h-3.5 w-3.5" />
   429→                            </Button>
   430→                            <Button
   431→                              variant="ghost"
   432→                              size="sm"
   433→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   434→                              title="View Reports"
   435→                              onClick={() => setReportsPanel({ agent, open: true })}
   436→                            >
   437→                              <Eye className="h-3.5 w-3.5" />
   438→                            </Button>
   439→                            <Button
   440→                              variant="ghost"
   441→                              size="sm"
   442→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-amber"
   443→                              title="Remote Wipe Device"
   444→                              onClick={() => setConfirmAction({ type: 'REMOTE_WIPE', agent })}
   445→                              disabled={actionMutation.isPending}
   446→                            >
   447→                              <Wrench className="h-3.5 w-3.5" />
   448→                            </Button>
   449→                            <Button
   450→                              variant="ghost"
   451→                              size="sm"
   452→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-rose"
   453→                              title="Remove Agent"
   454→                              onClick={() => setConfirmAction({ type: 'DELETE', agent })}
   455→                              disabled={actionMutation.isPending || agent.role === 'SUPER_ADMIN'}
   456→                            >
   457→                              <Trash2 className="h-3.5 w-3.5" />
   458→                            </Button>
   459→                          </div>
   460→                        </TableCell>
   461→                      </motion.tr>
   462→                    ))}
   463→                  </AnimatePresence>
   464→                </TableBody>
   465→              </Table>
   466→            )}
   467→          </div>
   468→        </CardContent>
   469→      </Card>
   470→
   471→      {/* ===== ADD AGENT DIALOG ===== */}
   472→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   473→        <DialogContent className="sm:max-w-md">
   474→          <DialogHeader>
   475→            <DialogTitle className="flex items-center gap-2">
   476→              <UserCheck className="h-5 w-5 text-emerald" />
   477→              Add New Agent
   478→            </DialogTitle>
   479→            <DialogDescription>
   480→              Create a new user account. They will be able to login from the main screen using their email.
   481→            </DialogDescription>
   482→          </DialogHeader>
   483→          <div className="space-y-4 py-2">
   484→            <div className="space-y-2">
   485→              <label className="text-xs font-medium text-muted-foreground">Full Name</label>
   486→              <Input
   487→                placeholder="e.g. Adebayo Johnson"
   488→                value={newName}
   489→                onChange={(e) => setNewName(e.target.value)}
   490→                className="h-9 text-sm"
   491→              />
   492→            </div>
   493→            <div className="space-y-2">
   494→              <label className="text-xs font-medium text-muted-foreground">Email Address</label>
   495→              <Input
   496→                placeholder="e.g. agent@nigeriaelectionwatch.org"
   497→                type="email"
   498→                value={newEmail}
   499→                onChange={(e) => setNewEmail(e.target.value)}
   500→                className="h-9 text-sm"
   501→              />
   502→            </div>
   503→            <div className="space-y-2">
   504→              <label className="text-xs font-medium text-muted-foreground">Role</label>
   505→              <Select value={newRole} onValueChange={setNewRole}>

   656→            </DialogTitle>
   657→            <DialogDescription>
   658→              Viewing all incident reports submitted by this agent
   659→            </DialogDescription>
   660→          </DialogHeader>
   661→          <AgentReportsPanel agentId={reportsPanel.agent?.id || ''} />
   662→        </DialogContent>
   663→      </Dialog>
   664→    </div>
   665→  );

   668→// Sub-component: fetches and displays reports for a specific agent
   669→function AgentReportsPanel({ agentId }: { agentId: string }) {
   670→  const { data, isLoading } = useQuery<{ incidents: Array<{
   671→    id: string; type: string; severity: string; status: string;
   672→    description: string; submittedAt: string;
   673→    pollingUnit: { name: string; state: string; lga: string } | null;
   674→  }>; total: number }>({
   675→    queryKey: ['agent-reports', agentId],
   676→    queryFn: () => fetchJson(`/api/incidents?reporterId=${agentId}&limit=50`),
   677→    enabled: !!agentId,
   678→  });
   679→
   680→  if (isLoading) {
   681→    return (
   682→      <div className="flex items-center justify-center py-12">
   683→        <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   684→      </div>
   685→    );
   686→  }
   687→
   688→  const incidents = data?.incidents || [];
   689→
   690→  if (incidents.length === 0) {
   691→    return (
   692→      <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   693→        <FileText className="h-8 w-8 mb-2 opacity-40" />
   694→        <p className="text-sm">No reports submitted yet</p>
   695→      </div>
   696→    );
   697→  }
   698→
   699→  const severityColors: Record<string, string> = {
   700→    LOW: 'border-emerald/30 text-emerald bg-emerald/10',
   701→    MEDIUM: 'border-amber/30 text-amber bg-amber/10',
   702→    HIGH: 'border-orange/30 text-orange bg-orange/10',
   703→    CRITICAL: 'border-rose/30 text-rose bg-rose/10',
   704→  };
   705→
   706→  const statusColors: Record<string, string> = {
   707→    PENDING: 'text-muted-foreground',
   708→    REVIEWED: 'text-cyan',
   709→    ESCALATED: 'text-amber',
   710→    DISMISSED: 'text-muted-foreground/50',
   711→    QUARANTINED: 'text-rose',
   712→  };
   713→
   714→  return (
   715→    <div className="flex-1 overflow-y-auto space-y-2 pr-1">
   716→      <p className="text-xs text-muted-foreground mb-2">{data?.total || 0} total reports</p>
   717→      {incidents.map(inc => (
   718→        <div key={inc.id} className="rounded-lg border border-border bg-card/40 p-3 space-y-1.5">
   719→          <div className="flex items-center justify-between gap-2">
   720→            <div className="flex items-center gap-2">
   721→              <Badge variant="outline" className={cn('text-[10px] h-5', severityColors[inc.severity] || '')}>
   722→                {inc.severity}
   723→              </Badge>
   724→              <Badge variant="outline" className="text-[10px] h-5 border-border text-muted-foreground">
   725→                {inc.type.replace(/_/g, ' ')}
   726→              </Badge>
   727→            </div>
   728→            <div className="flex items-center gap-2">
   729→              <span className={cn('text-[10px] font-medium', statusColors[inc.status] || '')}>
   730→                {inc.status}
   731→              </span>
   732→            </div>
   733→          </div>
   734→          <p className="text-xs text-foreground/80 leading-relaxed">{inc.description}</p>
   735→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   736→            {inc.pollingUnit && (
   737→              <span className="flex items-center gap-1">
   738→                <span className="w-1 h-1 rounded-full bg-muted-foreground/40" />
   739→                {inc.pollingUnit.name}, {inc.pollingUnit.state}
   740→              </span>
   741→            )}
   742→            <span>{new Date(inc.submittedAt).toLocaleString()}</span>
   743→          </div>
   744→        </div>
   745→      ))}
   746→    </div>
   747→  );
