   149→function formatTime(date: string | Date) {
   150→  const d = new Date(date);
   151→  const now = new Date();
   152→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   153→  if (diff < 1) return 'Just now';
   154→  if (diff < 60) return `${diff}m ago`;
   155→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   156→  return `${Math.floor(diff / 1440)}d ago`;
   157→}
   158→
   159→function formatNumber(n: number): string {
   160→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   161→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   162→  return String(n);
   163→}
