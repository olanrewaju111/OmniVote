   137→function formatTime(date: string | Date) {
   138→  const d = new Date(date);
   139→  const now = new Date();
   140→  const diff = Math.floor((now.getTime() - d.getTime()) / 60000);
   141→  if (diff < 1) return 'Just now';
   142→  if (diff < 60) return `${diff}m ago`;
   143→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   144→  return `${Math.floor(diff / 1440)}d ago`;
   145→}
   146→
   147→function formatNumber(n: number): string {
   148→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   149→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   150→  return String(n);
   151→}
