     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { useDashboardStore } from '@/store/dashboard';
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table';
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter,
    16→  DialogHeader, DialogTitle,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    20→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  Search, Users, UserCheck, UserX, Shield, ShieldAlert, Wrench,
    27→  Loader2, FileText, Trash2, Eye, ToggleLeft, X, Pencil,
    28→} from 'lucide-react';
    29→import { cn } from '@/lib/utils';
    30→import { fetchJson } from '@/lib/api';
    31→import { toast } from 'sonner';
    32→
    33→interface AgentUser {
    34→  id: string;
    35→  email: string;
    36→  name: string;
    37→  role: string;
    38→  isOnline: boolean;
    39→  lastSeenAt: string | null;
    40→  createdAt: string;
    41→  _count?: { incidents: number; auditLogs: number };
    42→}
    43→
    44→interface ReportsSlideOver {
    45→  agent: AgentUser | null;
    46→  open: boolean;
    47→}
    48→
    49→export function AgentRoster() {
    50→  const queryClient = useQueryClient();
    51→  const { tenantId } = useDashboardStore();
    52→  const [search, setSearch] = useState('');
    53→  const [searchInput, setSearchInput] = useState('');
    54→  const [roleFilter, setRoleFilter] = useState('ALL');
    55→  const [addOpen, setAddOpen] = useState(false);
    56→  const [reportsPanel, setReportsPanel] = useState<ReportsSlideOver>({ agent: null, open: false });
    57→  const [confirmAction, setConfirmAction] = useState<{ type: string; agent: AgentUser | null }>({ type: '', agent: null });
    58→  const [editOpen, setEditOpen] = useState(false);
    59→  const [editAgent, setEditAgent] = useState<AgentUser | null>(null);
    60→  const [editName, setEditName] = useState('');
    61→  const [editPhone, setEditPhone] = useState('');
    62→  const [editRole, setEditRole] = useState('');
    63→
    64→  // Add agent form state
    65→  const [newName, setNewName] = useState('');
    66→  const [newEmail, setNewEmail] = useState('');
    67→  const [newRole, setNewRole] = useState('FIELD_AGENT');
    68→
    69→  // Fetch agents from proper API
    70→  const { data, isLoading, refetch } = useQuery<{ users: AgentUser[] }>({
    71→    queryKey: ['agents', search, roleFilter, tenantId],
    72→    queryFn: () => {
    73→      const params = new URLSearchParams();
    74→      if (search) params.set('search', search);
    75→      if (roleFilter !== 'ALL') params.set('role', roleFilter);
    76→      if (tenantId) params.set('tenantId', tenantId);
    77→      return fetchJson(`/api/agents?${params}`);
    78→    },
    79→  });
    80→
    81→  // Add agent mutation
    82→  const addMutation = useMutation({
    83→    mutationFn: (body: { name: string; email: string; role: string }) =>
    84→      fetch(`/api/agents?tenantId=${tenantId}`, {
    85→        method: 'POST',
    86→        headers: { 'Content-Type': 'application/json' },
    87→        body: JSON.stringify(body),
    88→      }).then(async r => {
    89→        const data = await r.json();
    90→        if (!r.ok) throw new Error(data.error || 'Failed to add agent');
    91→        return data;
    92→      }),
    93→    onSuccess: (data) => {
    94→      if (data.error) {
    95→        toast.error(data.error);
    96→        return;
    97→      }
    98→      toast.success(`Agent "${data.user.name}" added successfully`);
    99→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   100→      setAddOpen(false);
   101→      setNewName(''); setNewEmail(''); setNewRole('FIELD_AGENT');
   102→    },
   103→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to add agent'),
   104→  });
   105→
   106→  // Action mutation (toggle online, remote wipe, change role, delete)
   107→  const actionMutation = useMutation({
   108→    mutationFn: (body: { userId: string; action: string; newRole?: string }) =>
   109→      fetch(`/api/agents?tenantId=${tenantId}`, {
   110→        method: 'PATCH',
   111→        headers: { 'Content-Type': 'application/json' },
   112→        body: JSON.stringify(body),
   113→      }).then(async r => {
   114→        const data = await r.json();
   115→        if (!r.ok) throw new Error(data.error || 'Action failed');
   116→        return data;
   117→      }),
   118→    onSuccess: (data, variables) => {
   119→      if (data.error) {
   120→        toast.error(data.error);
   121→        return;
   122→      }
   123→      if (variables.action === 'DELETE') {
   124→        toast.success(data.message || 'Agent removed');
   125→      } else if (variables.action === 'REMOTE_WIPE') {
   126→        toast.success(`Remote wipe sent to ${confirmAction.agent?.name}. Device data cleared.`);
   127→      } else if (variables.action === 'CHANGE_ROLE') {
   128→        toast.success(`Role changed to ${variables.newRole?.replace(/_/g, ' ')}`);
   129→      } else {
   130→        toast.success('Status updated');
   131→      }
   132→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   133→      setConfirmAction({ type: '', agent: null });
   134→    },
   135→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Action failed'),
   136→  });
   137→
   138→  const users = data?.users || [];
   139→  const fieldAgents = users.filter(u => u.role === 'FIELD_AGENT');
   140→  const online = fieldAgents.filter(u => u.isOnline).length;
   141→
   142→  const roleCounts = users.reduce<Record<string, number>>((acc, u) => {
   143→    acc[u.role] = (acc[u.role] || 0) + 1;
   144→    return acc;
   145→  }, {});
   146→
   147→  // Search with debounce via the query key
   148→  const handleSearch = useCallback(() => {
   149→    setSearch(searchInput);
   150→  }, [searchInput]);
   151→
   152→  // Edit agent mutation
   153→  const editMutation = useMutation({
   154→    mutationFn: (body: { id: string; name?: string; phone?: string; role?: string }) =>
   155→      fetch('/api/tenants/users', {
   156→        method: 'PATCH',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify(body),
   159→      }).then(async r => {
   160→        const data = await r.json();
   161→        if (!r.ok) throw new Error(data.error || 'Failed to update agent');
   162→        return data;
   163→      }),
   164→    onSuccess: () => {
   165→      queryClient.invalidateQueries({ queryKey: ['agents'] });
   166→      toast.success('Agent details updated');
   167→      setEditOpen(false);
   168→      setEditAgent(null);
   169→    },
   170→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to update agent'),
   171→  });
   172→
   173→  const openEditDialog = (agent: AgentUser) => {
   174→    setEditAgent(agent);
   175→    setEditName(agent.name);
   176→    setEditPhone('');
   177→    setEditRole(agent.role);
   178→    setEditOpen(true);
   179→  };
   180→
   181→  const handleEditSave = () => {
   182→    if (!editAgent) return;
   183→    if (!editName.trim()) { toast.error('Name is required'); return; }
   184→    editMutation.mutate({
   185→      id: editAgent.id,
   186→      name: editName.trim(),
   187→      phone: editPhone.trim() || undefined,
   188→      role: editRole,
   189→    });
   190→  };
   191→
   192→  // Handle confirm action
   193→  const handleConfirmAction = () => {
   194→    if (!confirmAction.agent) return;
   195→    if (confirmAction.type === 'DELETE') {
   196→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'DELETE' });
   197→    } else if (confirmAction.type === 'REMOTE_WIPE') {
   198→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'REMOTE_WIPE' });
   199→    } else if (confirmAction.type === 'CHANGE_ROLE') {
   200→      actionMutation.mutate({ userId: confirmAction.agent.id, action: 'CHANGE_ROLE', newRole: 'ANALYST' });
   201→    }
   202→  };
   203→
   204→  // Handle add agent
   205→  const handleAddAgent = () => {
   206→    if (!newName.trim() || !newEmail.trim()) {
   207→      toast.error('Name and email are required');
   208→      return;
   209→    }
   210→    if (!newEmail.includes('@')) {
   211→      toast.error('Please enter a valid email');
   212→      return;
   213→    }
   214→    addMutation.mutate({ name: newName.trim(), email: newEmail.trim(), role: newRole });
   215→  };
   216→
   217→  return (
   218→    <div className="h-full overflow-y-auto p-4 sm:p-6 space-y-4">
   219→      {/* Header */}
   220→      <div className="flex items-center justify-between flex-wrap gap-3">
   221→        <div>
   222→          <h2 className="text-lg font-semibold flex items-center gap-2">
   223→            <Users className="h-5 w-5 text-emerald" />
   224→            Agent Roster
   225→          </h2>
   226→          <p className="text-sm text-muted-foreground mt-0.5">Manage field agents and organization users</p>
   227→        </div>
   228→        <Button
   229→          onClick={() => setAddOpen(true)}
   230→          className="bg-emerald hover:bg-emerald/90 text-emerald-950 text-sm gap-2"
   231→        >
   232→          <UserCheck className="h-4 w-4" />
   233→          Add Agent
   234→        </Button>
   235→      </div>
   236→
   237→      {/* Summary cards */}
   238→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   239→        <Card className="border-border bg-card/40">
   240→          <CardContent className="p-3.5">
   241→            <div className="flex items-center gap-2 mb-1">
   242→              <Users className="h-4 w-4 text-emerald" />
   243→              <span className="text-[11px] text-muted-foreground">Total Agents</span>
   244→            </div>
   245→            <p className="text-xl font-bold tabular-nums">{fieldAgents.length}</p>
   246→          </CardContent>
   247→        </Card>
   248→        <Card className="border-emerald/20 bg-emerald/5">
   249→          <CardContent className="p-3.5">
   250→            <div className="flex items-center gap-2 mb-1">
   251→              <UserCheck className="h-4 w-4 text-emerald" />
   252→              <span className="text-[11px] text-muted-foreground">Online Now</span>
   253→            </div>
   254→            <p className="text-xl font-bold tabular-nums text-emerald">{online}</p>
   255→          </CardContent>
   256→        </Card>
   257→        <Card className="border-amber/20 bg-amber/5">
   258→          <CardContent className="p-3.5">
   259→            <div className="flex items-center gap-2 mb-1">
   260→              <UserX className="h-4 w-4 text-amber" />
   261→              <span className="text-[11px] text-muted-foreground">Offline</span>
   262→            </div>
   263→            <p className="text-xl font-bold tabular-nums text-amber">{fieldAgents.length - online}</p>
   264→          </CardContent>
   265→        </Card>
   266→        <Card className="border-border bg-card/40">
   267→          <CardContent className="p-3.5">
   268→            <div className="flex items-center gap-2 mb-1">
   269→              <Shield className="h-4 w-4 text-cyan" />
   270→              <span className="text-[11px] text-muted-foreground">All Users</span>
   271→            </div>
   272→            <p className="text-xl font-bold tabular-nums">{users.length}</p>
   273→          </CardContent>
   274→        </Card>
   275→      </div>
   276→
   277→      {/* Role filter + distribution badges */}
   278→      <div className="flex items-center justify-between flex-wrap gap-3">
   279→        <div className="flex items-center gap-2 flex-wrap">
   280→          {Object.entries(roleCounts).map(([role, count]) => (
   281→            <button
   282→              key={role}
   283→              onClick={() => setRoleFilter(roleFilter === role ? 'ALL' : role)}
   284→              className="cursor-pointer"
   285→            >
   286→              <Badge
   287→                variant="outline"
   288→                className={cn(
   289→                  'text-[11px] h-6 transition-colors',
   290→                  roleFilter === role ? 'bg-foreground/10' : '',
   291→                  role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   292→                  role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   293→                  role === 'ANALYST' ? 'border-amber/30 text-amber' :
   294→                  role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   295→                  'border-border text-muted-foreground'
   296→                )}
   297→              >
   298→                {role.replace(/_/g, ' ')}: {count}
   299→              </Badge>
   300→            </button>
   301→          ))}
   302→        </div>
   303→        {roleFilter !== 'ALL' && (
   304→          <button onClick={() => setRoleFilter('ALL')} className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1">
   305→            <X className="h-3 w-3" /> Clear filter
   306→          </button>
   307→        )}
   308→      </div>
   309→
   310→      {/* Agent table */}
   311→      <Card>
   312→        <CardHeader className="pb-2 pt-4 px-4">
   313→          <div className="flex items-center justify-between gap-3">
   314→            <CardTitle className="text-sm">
   315→              {roleFilter === 'ALL' ? 'All Users' : roleFilter.replace(/_/g, ' ')}
   316→              <span className="text-muted-foreground font-normal ml-2">({users.length})</span>
   317→            </CardTitle>
   318→            <div className="flex items-center gap-2">
   319→              <div className="relative">
   320→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   321→                <Input
   322→                  placeholder="Search agents..."
   323→                  value={searchInput}
   324→                  onChange={(e) => setSearchInput(e.target.value)}
   325→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   326→                  className="pl-8 h-8 w-48 text-xs"
   327→                />
   328→              </div>
   329→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={handleSearch}>
   330→                Search
   331→              </Button>
   332→            </div>
   333→          </div>
   334→        </CardHeader>
   335→        <CardContent className="px-0 pb-0">
   336→          <div className="max-h-[480px] overflow-y-auto">
   337→            {isLoading ? (
   338→              <div className="flex items-center justify-center py-16">
   339→                <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   340→              </div>
   341→            ) : users.length === 0 ? (
   342→              <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   343→                <Users className="h-8 w-8 mb-2 opacity-40" />
   344→                <p className="text-sm">No users found</p>
   345→              </div>
   346→            ) : (
   347→              <Table>
   348→                <TableHeader>
   349→                  <TableRow className="border-border hover:bg-transparent">
   350→                    <TableHead className="text-[11px] h-9">User</TableHead>
   351→                    <TableHead className="text-[11px] h-9">Role</TableHead>
   352→                    <TableHead className="text-[11px] h-9">Status</TableHead>
   353→                    <TableHead className="text-[11px] h-9 hidden sm:table-cell">Email</TableHead>
   354→                    <TableHead className="text-[11px] h-9 hidden md:table-cell">Reports</TableHead>
   355→                    <TableHead className="text-[11px] h-9 text-right">Actions</TableHead>
   356→                  </TableRow>
   357→                </TableHeader>
   358→                <TableBody>
   359→                  <AnimatePresence>
   360→                    {users.map(agent => (
   361→                      <motion.tr
   362→                        key={agent.id}
   363→                        initial={{ opacity: 0, y: 4 }}
   364→                        animate={{ opacity: 1, y: 0 }}
   365→                        exit={{ opacity: 0, y: -4 }}
   366→                        className="border-border hover:bg-card/60 transition-colors"
   367→                      >
   368→                        <TableCell className="py-2.5">
   369→                          <div className="flex items-center gap-2.5">
   370→                            <div className="relative">
   371→                              <div className="w-7 h-7 rounded-full bg-secondary flex items-center justify-center text-[10px] font-bold">
   372→                                {agent.name.split(' ').map(n => n[0]).join('')}
   373→                              </div>
   374→                              <span className={cn(
   375→                                'absolute -bottom-0.5 -right-0.5 w-2 h-2 rounded-full border-2 border-card',
   376→                                agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/30'
   377→                              )} />
   378→                            </div>
   379→                            <div className="min-w-0">
   380→                              <span className="text-xs font-medium block truncate max-w-[140px]">{agent.name}</span>
   381→                              {agent.lastSeenAt && (
   382→                                <span className="text-[10px] text-muted-foreground/60">
   383→                                  Last: {new Date(agent.lastSeenAt).toLocaleDateString()}
   384→                                </span>
   385→                              )}
   386→                            </div>
   387→                          </div>
   388→                        </TableCell>
   389→                        <TableCell className="py-2.5">
   390→                          <Badge
   391→                            variant="outline"
   392→                            className={cn(
   393→                              'text-[10px] h-5',
   394→                              agent.role === 'SUPER_ADMIN' ? 'border-emerald/30 text-emerald' :
   395→                              agent.role === 'TENANT_ADMIN' ? 'border-cyan/30 text-cyan' :
   396→                              agent.role === 'ANALYST' ? 'border-amber/30 text-amber' :
   397→                              agent.role === 'TRUST_SAFETY' ? 'border-rose/30 text-rose' :
   398→                              'border-border text-muted-foreground'
   399→                            )}
   400→                          >
   401→                            {agent.role.replace(/_/g, ' ')}
   402→                          </Badge>
   403→                        </TableCell>
   404→                        <TableCell className="py-2.5">
   405→                          <button
   406→                            onClick={() => actionMutation.mutate({ userId: agent.id, action: 'TOGGLE_ONLINE' })}
   407→                            className="cursor-pointer"
   408→                            disabled={actionMutation.isPending}
   409→                          >
   410→                            <Badge
   411→                              variant="outline"
   412→                              className={cn(
   413→                                'text-[10px] h-5 transition-colors cursor-pointer',
   414→                                agent.isOnline
   415→                                  ? 'border-emerald/30 text-emerald bg-emerald/10 hover:bg-emerald/20'
   416→                                  : 'border-border text-muted-foreground hover:bg-card'
   417→                              )}
   418→                            >
   419→                              <span className={cn('w-1.5 h-1.5 rounded-full mr-1.5', agent.isOnline ? 'bg-emerald' : 'bg-muted-foreground/40')} />
   420→                              {agent.isOnline ? 'Online' : 'Offline'}
   421→                            </Badge>
   422→                          </button>
   423→                        </TableCell>
   424→                        <TableCell className="py-2.5 text-[11px] text-muted-foreground hidden sm:table-cell truncate max-w-[180px]">{agent.email}</TableCell>
   425→                        <TableCell className="py-2.5 hidden md:table-cell">
   426→                          <span className="text-[11px] text-muted-foreground tabular-nums">
   427→                            {agent._count?.incidents ?? 0}
   428→                          </span>
   429→                        </TableCell>
   430→                        <TableCell className="py-2.5 text-right">
   431→                          <div className="flex items-center justify-end gap-0.5">
   432→                            <Button
   433→                              variant="ghost"
   434→                              size="sm"
   435→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-cyan"
   436→                              title="Edit Details"
   437→                              onClick={() => openEditDialog(agent)}
   438→                              disabled={agent.role === 'SUPER_ADMIN'}
   439→                            >
   440→                              <Pencil className="h-3.5 w-3.5" />
   441→                            </Button>
   442→                            <Button
   443→                              variant="ghost"
   444→                              size="sm"
   445→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   446→                              title="View Reports"
   447→                              onClick={() => setReportsPanel({ agent, open: true })}
   448→                            >
   449→                              <Eye className="h-3.5 w-3.5" />
   450→                            </Button>
   451→                            <Button
   452→                              variant="ghost"
   453→                              size="sm"
   454→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-amber"
   455→                              title="Remote Wipe Device"
   456→                              onClick={() => setConfirmAction({ type: 'REMOTE_WIPE', agent })}
   457→                              disabled={actionMutation.isPending}
   458→                            >
   459→                              <Wrench className="h-3.5 w-3.5" />
   460→                            </Button>
   461→                            <Button
   462→                              variant="ghost"
   463→                              size="sm"
   464→                              className="h-7 w-7 p-0 text-muted-foreground hover:text-rose"
   465→                              title="Remove Agent"
   466→                              onClick={() => setConfirmAction({ type: 'DELETE', agent })}
   467→                              disabled={actionMutation.isPending || agent.role === 'SUPER_ADMIN'}
   468→                            >
   469→                              <Trash2 className="h-3.5 w-3.5" />
   470→                            </Button>
   471→                          </div>
   472→                        </TableCell>
   473→                      </motion.tr>
   474→                    ))}
   475→                  </AnimatePresence>
   476→                </TableBody>
   477→              </Table>
   478→            )}
   479→          </div>
   480→        </CardContent>
   481→      </Card>
   482→
   483→      {/* ===== ADD AGENT DIALOG ===== */}
   484→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   485→        <DialogContent className="sm:max-w-md">
   486→          <DialogHeader>
   487→            <DialogTitle className="flex items-center gap-2">
   488→              <UserCheck className="h-5 w-5 text-emerald" />
   489→              Add New Agent
   490→            </DialogTitle>
   491→            <DialogDescription>
   492→              Create a new user account. They will be able to login from the main screen using their email.
   493→            </DialogDescription>
   494→          </DialogHeader>
   495→          <div className="space-y-4 py-2">
   496→            <div className="space-y-2">
   497→              <label className="text-xs font-medium text-muted-foreground">Full Name</label>
   498→              <Input
   499→                placeholder="e.g. Adebayo Johnson"
   500→                value={newName}
   501→                onChange={(e) => setNewName(e.target.value)}
   502→                className="h-9 text-sm"
   503→              />
   504→            </div>
   505→            <div className="space-y-2">

   673→          <AgentReportsPanel agentId={reportsPanel.agent?.id || ''} />
   674→        </DialogContent>
   675→      </Dialog>
   676→    </div>
   677→  );

   680→// Sub-component: fetches and displays reports for a specific agent
   681→function AgentReportsPanel({ agentId }: { agentId: string }) {
   682→  const { data, isLoading } = useQuery<{ incidents: Array<{
   683→    id: string; type: string; severity: string; status: string;
   684→    description: string; submittedAt: string;
   685→    pollingUnit: { name: string; state: string; lga: string } | null;
   686→  }>; total: number }>({
   687→    queryKey: ['agent-reports', agentId],
   688→    queryFn: () => fetchJson(`/api/incidents?reporterId=${agentId}&limit=50`),
   689→    enabled: !!agentId,
   690→  });
   691→
   692→  if (isLoading) {
   693→    return (
   694→      <div className="flex items-center justify-center py-12">
   695→        <Loader2 className="h-5 w-5 animate-spin text-emerald" />
   696→      </div>
   697→    );
   698→  }
   699→
   700→  const incidents = data?.incidents || [];
   701→
   702→  if (incidents.length === 0) {
   703→    return (
   704→      <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   705→        <FileText className="h-8 w-8 mb-2 opacity-40" />
   706→        <p className="text-sm">No reports submitted yet</p>
   707→      </div>
   708→    );
   709→  }
   710→
   711→  const severityColors: Record<string, string> = {
   712→    LOW: 'border-emerald/30 text-emerald bg-emerald/10',
   713→    MEDIUM: 'border-amber/30 text-amber bg-amber/10',
   714→    HIGH: 'border-orange/30 text-orange bg-orange/10',
   715→    CRITICAL: 'border-rose/30 text-rose bg-rose/10',
   716→  };
   717→
   718→  const statusColors: Record<string, string> = {
   719→    PENDING: 'text-muted-foreground',
   720→    REVIEWED: 'text-cyan',
   721→    ESCALATED: 'text-amber',
   722→    DISMISSED: 'text-muted-foreground/50',
   723→    QUARANTINED: 'text-rose',
   724→  };
   725→
   726→  return (
   727→    <div className="flex-1 overflow-y-auto space-y-2 pr-1">
   728→      <p className="text-xs text-muted-foreground mb-2">{data?.total || 0} total reports</p>
   729→      {incidents.map(inc => (
   730→        <div key={inc.id} className="rounded-lg border border-border bg-card/40 p-3 space-y-1.5">
   731→          <div className="flex items-center justify-between gap-2">
   732→            <div className="flex items-center gap-2">
   733→              <Badge variant="outline" className={cn('text-[10px] h-5', severityColors[inc.severity] || '')}>
   734→                {inc.severity}
   735→              </Badge>
   736→              <Badge variant="outline" className="text-[10px] h-5 border-border text-muted-foreground">
   737→                {inc.type.replace(/_/g, ' ')}
   738→              </Badge>
   739→            </div>
   740→            <div className="flex items-center gap-2">
   741→              <span className={cn('text-[10px] font-medium', statusColors[inc.status] || '')}>
   742→                {inc.status}
   743→              </span>
   744→            </div>
   745→          </div>
   746→          <p className="text-xs text-foreground/80 leading-relaxed">{inc.description}</p>
   747→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   748→            {inc.pollingUnit && (
   749→              <span className="flex items-center gap-1">
   750→                <span className="w-1 h-1 rounded-full bg-muted-foreground/40" />
   751→                {inc.pollingUnit.name}, {inc.pollingUnit.state}
   752→              </span>
   753→            )}
   754→            <span>{new Date(inc.submittedAt).toLocaleString()}</span>
   755→          </div>
   756→        </div>
   757→      ))}
   758→    </div>
   759→  );
