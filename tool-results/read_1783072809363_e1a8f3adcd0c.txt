   137→
   138→const RISK_CELL_COLORS: Record<string, { bg: string; text: string }> = {
   139→  LOW: { bg: 'rgba(16,185,129,0.20)', text: '#10b981' },
   140→  MEDIUM: { bg: 'rgba(245,158,11,0.30)', text: '#f59e0b' },
   141→  HIGH: { bg: 'rgba(239,68,68,0.30)', text: '#ef4444' },
   142→  CRITICAL: { bg: 'rgba(220,38,38,0.50)', text: '#dc2626' },
   143→};
   144→
   145→function riskCellColor(level: string) {
   146→  return RISK_CELL_COLORS[level] ?? RISK_CELL_COLORS.LOW;
   147→}
   148→
   149→function riskBarColor(score: number): string {
   150→  if (score >= 0.8) return '#ef4444';
   151→  if (score >= 0.6) return '#f97316';
   152→  if (score >= 0.3) return '#f59e0b';
   153→  return '#10b981';
   154→}
   155→
   156→function riskBadgeVariant(level: string): string {
   157→  switch (level) {
   158→    case 'CRITICAL':
   159→      return 'bg-rose-600/20 text-rose-400 border-rose-600/30';
   160→    case 'HIGH':
   161→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   162→    case 'MEDIUM':
   163→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   164→    default:
   165→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   166→  }
   167→}
   168→
   169→function statusBadgeVariant(status: string): string {
   170→  switch (status) {
   171→    case 'RUNNING':
   172→      return 'bg-cyan-500/20 text-cyan-400 border-cyan-500/30';
   173→    case 'COMPLETED':
   174→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   175→    case 'PENDING':
   176→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   177→    case 'FAILED':
   178→      return 'bg-rose-500/20 text-rose-400 border-rose-500/30';
   179→    default:
   180→      return 'bg-muted text-muted-foreground border-border';
   181→  }
   182→}
   183→
   184→function formatDate(dateStr: string): string {
   185→  const d = new Date(dateStr);
   186→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   187→}
   188→
   189→function formatDay(dateStr: string): string {
   190→  const d = new Date(dateStr);
   191→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
   192→}

   212→export function FlashpointWargame() {
   213→  const { tenantId } = useDashboardStore();
   214→  const queryClient = useQueryClient();
   215→
   216→  // Wargame state
   217→  const [selectedScenarioId, setSelectedScenarioId] = useState<string | null>(null);
   218→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   219→  const [newTitle, setNewTitle] = useState('');
   220→  const [newDescription, setNewDescription] = useState('');
   221→  const [newRole, setNewRole] = useState('RED_TEAM');
   222→  const [newParams, setNewParams] = useState('{\n  \n}');
   223→  const [completeScore, setCompleteScore] = useState('');
   224→
   225→  /* ---- Data fetching ---- */
   226→  const { data, isLoading, refetch } = useQuery<FlashpointData>({
   227→    queryKey: ['flashpoint', tenantId],
   228→    queryFn: () =>
   229→      fetchJson(`/api/flashpoint?tenantId=${tenantId}`),
   230→    refetchInterval: 20000,
   231→  });
   232→
   233→  const mutation = useMutation({
   234→    mutationFn: (body: Record<string, unknown>) =>
   235→      fetchJson('/api/flashpoint?tenantId=' + tenantId, {
   236→        method: 'POST',
   237→        headers: { 'Content-Type': 'application/json' },
   238→        body: JSON.stringify(body),
   239→      }),
   240→    onSuccess: () => {
   241→      queryClient.invalidateQueries({ queryKey: ['flashpoint', tenantId] });
   242→    },
   243→    onError: () => {
   244→      toast.error('Flashpoint operation failed. Please try again.');
   245→    },
   246→  });
   247→
   248→  /* ---- Derived data ---- */
   249→  const stats = data?.stats;
   250→  const forecasts = data?.forecasts ?? [];
   251→  const scenarios = data?.scenarios ?? [];
   252→  const heatmapData = data?.heatmapData ?? [];
   253→  const topFactors = data?.topContributingFactors ?? [];
   254→
   255→  // Unique states sorted alphabetically
   256→  const uniqueStates = useMemo(() => {
   257→    const set = new Set(heatmapData.map((d) => d.state));
   258→    return Array.from(set).sort();
   259→  }, [heatmapData]);
   260→
   261→  // Unique dates sorted
   262→  const uniqueDates = useMemo(() => {
   263→    const set = new Set(heatmapData.map((d) => d.date));
   264→    return Array.from(set).sort();
   265→  }, [heatmapData]);
   266→
   267→  // Build lookup map for heatmap cells
   268→  const heatmapMap = useMemo(() => {
   269→    const map = new Map<string, (typeof heatmapData)[number]>();
   270→    for (const d of heatmapData) {
   271→      map.set(`${d.state}|${d.date}`, d);
   272→    }
   273→    return map;
   274→  }, [heatmapData]);
   275→
   276→  // State risk averages for bar chart
   277→  const stateRisks = useMemo(() => {
   278→    const map = new Map<string, { overall: number; level: string; count: number }>();
   279→    for (const f of forecasts) {
   280→      const existing = map.get(f.state);
   281→      if (existing) {
   282→        existing.overall = (existing.overall * existing.count + f.riskScores.overall) / (existing.count + 1);
   283→        existing.count += 1;
   284→        // Upgrade risk level if higher
   285→        const order = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
   286→        if (order.indexOf(f.riskLevel) > order.indexOf(existing.level)) {
   287→          existing.level = f.riskLevel;
   288→        }
   289→      } else {
   290→        map.set(f.state, { overall: f.riskScores.overall, level: f.riskLevel, count: 1 });
   291→      }
   292→    }
   293→    return Array.from(map.entries())
   294→      .map(([state, v]) => ({ state, ...v }))
   295→      .sort((a, b) => b.overall - a.overall);
   296→  }, [forecasts]);
   297→
   298→  // Top factor
   299→  const topFactor = topFactors.length > 0 ? topFactors[0].factor : '—';
   300→
   301→  // Selected scenario
   302→  const selectedScenario = scenarios.find((s) => s.id === selectedScenarioId) ?? null;
   303→
   304→  // Max factor count for bar scaling
   305→  const maxFactorCount = Math.max(...topFactors.map((f) => f.count), 1);
   306→
   307→  /* ---- Handlers ---- */
   308→  const handleCreateScenario = () => {
   309→    if (!newTitle.trim()) {
   310→      toast.error('Title is required');
   311→      return;
   312→    }
   313→    let parsedParams: Record<string, unknown> = {};
   314→    try {
   315→      parsedParams = newParams.trim() ? JSON.parse(newParams) : {};
   316→    } catch {
   317→      toast.error('Invalid JSON in parameters');
   318→      return;
   319→    }
   320→    mutation.mutate(
   321→      { action: 'RUN_WARGAME', title: newTitle, description: newDescription, playerRole: newRole, parameters: parsedParams },
   322→      {
   323→        onSuccess: () => {
   324→          toast.success('Scenario created successfully');
   325→          setCreateDialogOpen(false);
   326→          setNewTitle('');
   327→          setNewDescription('');
   328→          setNewRole('RED_TEAM');
   329→          setNewParams('{\n  \n}');
   330→        },
   331→        onError: () => toast.error('Failed to create scenario'),
   332→      },
   333→    );
   334→  };
   335→
   336→  const handleCompleteScenario = () => {
   337→    if (!selectedScenarioId || !completeScore) {
   338→      toast.error('Score is required');
   339→      return;
   340→    }
   341→    const score = parseFloat(completeScore);
   342→    if (isNaN(score) || score < 0 || score > 100) {
   343→      toast.error('Score must be between 0 and 100');
   344→      return;
   345→    }
   346→    mutation.mutate(
   347→      { action: 'COMPLETE_WARGAME', scenarioId: selectedScenarioId, score },
   348→      {
   349→        onSuccess: () => {
   350→          toast.success('Scenario completed');
   351→          setCompleteScore('');
   352→          setSelectedScenarioId(null);
   353→        },
   354→        onError: () => toast.error('Failed to complete scenario'),
   355→      },
   356→    );
   357→  };
   358→
   359→  /* ---- Loading ---- */
   360→  if (isLoading) {
   361→    return (
   362→      <div className="h-full flex flex-col items-center justify-center gap-3">
   363→        <Loader2 className="size-8 text-emerald-500 animate-spin" />
   364→        <p className="text-sm text-muted-foreground">Loading flashpoint data…</p>
   365→      </div>
   366→    );
   367→  }
   368→
   369→  return (
   370→    <div className="h-full flex flex-col gap-4">
   371→      <Tabs defaultValue="forecast" className="h-full flex flex-col gap-3">
   372→        <TabsList>
   373→          <TabsTrigger value="forecast" className="gap-1.5">
   374→            <Flame className="size-4" />
   375→            Risk Forecast
   376→          </TabsTrigger>
   377→          <TabsTrigger value="wargame" className="gap-1.5">
   378→            <Swords className="size-4" />
   379→            Wargame Simulator
   380→          </TabsTrigger>
   381→        </TabsList>
   382→
   383→        {/* ================================================================ */}
   384→        {/* TAB 1 — Risk Forecast                                           */}
   385→        {/* ================================================================ */}
   386→        <TabsContent value="forecast" className="flex-1 min-h-0 flex flex-col">
   387→          <motion.div
   388→            variants={staggerContainer}
   389→            initial="initial"
   390→            animate="animate"
   391→            className="flex flex-col gap-4 flex-1 min-h-0"
   392→          >
   393→            {/* --- Stats Row --- */}
   394→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   395→              <motion.div variants={fadeIn}>
   396→                <Card className="h-full">
   397→                  <CardContent className="p-4 flex flex-col gap-1">
   398→                    <div className="flex items-center gap-2 text-muted-foreground">
   399→                      <Activity className="size-4" />
   400→                      <span className="text-xs font-medium">Total Forecasts</span>
   401→                    </div>
   402→                    <p className="text-2xl font-bold">{stats?.totalForecasts ?? 0}</p>
   403→                  </CardContent>
   404→                </Card>
   405→              </motion.div>
   406→
   407→              <motion.div variants={fadeIn}>
   408→                <Card className="h-full border-red-500/20">
   409→                  <CardContent className="p-4 flex flex-col gap-1">
   410→                    <div className="flex items-center gap-2 text-red-400">
   411→                      <AlertTriangle className="size-4" />
   412→                      <span className="text-xs font-medium">High Risk States</span>
   413→                    </div>
   414→                    <p className="text-2xl font-bold text-red-400">{stats?.highRiskStates?.length ?? 0}</p>
   415→                  </CardContent>
   416→                </Card>
   417→              </motion.div>
   418→
   419→              <motion.div variants={fadeIn}>
   420→                <Card className="h-full border-rose-500/20">
   421→                  <CardContent className="p-4 flex flex-col gap-1">
   422→                    <div className="flex items-center gap-2 text-rose-400">
   423→                      <ShieldAlert className="size-4" />
   424→                      <span className="text-xs font-medium">Critical</span>
   425→                    </div>
   426→                    <p className="text-2xl font-bold text-rose-400">{stats?.byRiskLevel?.CRITICAL ?? 0}</p>
   427→                  </CardContent>
   428→                </Card>
   429→              </motion.div>
   430→
   431→              <motion.div variants={fadeIn}>
   432→                <Card className="h-full">
   433→                  <CardContent className="p-4 flex flex-col gap-1">
   434→                    <div className="flex items-center gap-2 text-cyan-400">
   435→                      <TrendingUp className="size-4" />
   436→                      <span className="text-xs font-medium">Avg Confidence</span>
   437→                    </div>
   438→                    <p className="text-2xl font-bold text-cyan-400">
   439→                      {stats?.avgConfidence != null ? `${(stats.avgConfidence * 100).toFixed(0)}%` : '—'}
   440→                    </p>
   441→                  </CardContent>
   442→                </Card>
   443→              </motion.div>
   444→
   445→              <motion.div variants={fadeIn}>
   446→                <Card className="h-full">
   447→                  <CardContent className="p-4 flex flex-col gap-1">
   448→                    <div className="flex items-center gap-2 text-amber-400">
   449→                      <Zap className="size-4" />
   450→                      <span className="text-xs font-medium">Top Factor</span>
   451→                    </div>
   452→                    <p className="text-sm font-semibold truncate">{topFactor}</p>
   453→                  </CardContent>
   454→                </Card>
   455→              </motion.div>
   456→            </div>
   457→
   458→            {/* --- Main Two Column Layout --- */}
   459→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 min-h-0">
   460→              {/* LEFT: 7-Day Heatmap */}
   461→              <motion.div variants={fadeIn}>
   462→                <Card className="h-full flex flex-col">
   463→                  <CardContent className="p-4 flex flex-col gap-3 flex-1 min-h-0">
   464→                    <div className="flex items-center justify-between">
   465→                      <h3 className="text-sm font-semibold flex items-center gap-2">
   466→                        <CalendarDays className="size-4 text-emerald-400" />
   467→                        7-Day Risk Heatmap
   468→                      </h3>
   469→                      <Button variant="ghost" size="sm" onClick={() => refetch()} className="h-7 px-2 text-xs">
   470→                        <Crosshair className="size-3 mr-1" />
   471→                        Refresh
   472→                      </Button>
   473→                    </div>
   474→
   475→                    {uniqueStates.length === 0 || uniqueDates.length === 0 ? (
   476→                      <div className="flex-1 flex items-center justify-center text-muted-foreground text-sm">
   477→                        No forecast data available
   478→                      </div>
   479→                    ) : (
   480→                      <ScrollArea className="flex-1">
   481→                        <div className="min-w-fit">
   482→                          {/* Header row: empty + dates */}
   483→                          <div
   484→                            className="grid gap-[2px] mb-[2px]"
   485→                            style={{
   486→                              gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   487→                            }}
   488→                          >
   489→                            <div className="text-[10px] text-muted-foreground font-medium px-1" />
   490→                            {uniqueDates.map((date) => (
   491→                              <div
   492→                                key={date}
   493→                                className="text-[10px] text-muted-foreground font-medium text-center truncate px-1"
   494→                              >
   495→                                {formatDay(date)}
   496→                              </div>
   497→                            ))}
   498→                          </div>
   499→
   500→                          {/* Data rows: state + cells */}
   501→                          {uniqueStates.map((state) => (
   502→                            <div
   503→                              key={state}
   504→                              className="grid gap-[2px] mb-[2px]"
   505→                              style={{
   506→                                gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   507→                              }}
   508→                            >
   509→                              <div className="flex items-center text-xs font-medium truncate px-1 text-foreground/80">
   510→                                {state}
   511→                              </div>
   512→                              {uniqueDates.map((date) => {
   513→                                const cell = heatmapMap.get(`${state}|${date}`);
   514→                                const colors = cell ? riskCellColor(cell.riskLevel) : { bg: 'rgba(255,255,255,0.05)', text: '#666' };
   515→                                return (
   516→                                  <TooltipProvider key={`${state}|${date}`}>
   517→                                    <Tooltip>
   518→                                      <TooltipTrigger asChild>
   519→                                        <div
   520→                                          className="flex items-center justify-center rounded-sm text-[11px] font-semibold cursor-default transition-transform hover:scale-110 h-8"
   521→                                          style={{
   522→                                            backgroundColor: colors.bg,
   523→                                            color: colors.text,
   524→                                          }}
   525→                                        >
   526→                                          {cell ? cell.overall.toFixed(1) : '—'}
   527→                                        </div>
   528→                                      </TooltipTrigger>
   529→                                      {cell && (
   530→                                        <TooltipContent side="top" className="bg-popover text-popover-foreground border">
   531→                                          <div className="text-xs space-y-1 min-w-[140px]">
   532→                                            <div className="font-semibold">{state} — {formatDate(date)}</div>
   533→                                            <Separator className="my-1" />
   534→                                            <div className="flex justify-between">
   535→                                              <span className="text-muted-foreground">Violence</span>
   536→                                              <span className="font-mono">{cell.violence.toFixed(2)}</span>
   537→                                            </div>
   538→                                            <div className="flex justify-between">
   539→                                              <span className="text-muted-foreground">Intimidation</span>
   540→                                              <span className="font-mono">{cell.intimidation.toFixed(2)}</span>
   541→                                            </div>
   542→                                            <div className="flex justify-between">
   543→                                              <span className="text-muted-foreground">Logistics</span>
   544→                                              <span className="font-mono">{cell.logistics.toFixed(2)}</span>
   545→                                            </div>
   546→                                            <Separator className="my-1" />
   547→                                            <div className="flex justify-between font-semibold">
   548→                                              <span>Overall</span>
   549→                                              <span style={{ color: colors.text }}>{cell.overall.toFixed(2)}</span>
   550→                                            </div>
   551→                                            <div className="flex justify-between">
   552→                                              <span>Level</span>
   553→                                              <Badge className={cn('text-[10px] h-4', riskBadgeVariant(cell.riskLevel))}>
   554→                                                {cell.riskLevel}
   555→                                              </Badge>
   556→                                            </div>
   557→                                          </div>
   558→                                        </TooltipContent>
   559→                                      )}
   560→                                    </Tooltip>
   561→                                  </TooltipProvider>
   562→                                );
   563→                              })}
   564→                            </div>
   565→                          ))}
   566→                        </div>
   567→                      </ScrollArea>
