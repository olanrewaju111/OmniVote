     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Input } from '@/components/ui/input';
     9→import { Textarea } from '@/components/ui/textarea';
    10→import { Separator } from '@/components/ui/separator';
    11→import { Switch } from '@/components/ui/switch';
    12→import {
    13→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    14→} from '@/components/ui/select';
    15→import { useDashboardStore } from '@/store/dashboard';
    16→import {
    17→  Send, MapPin, Camera, Mic, AlertTriangle, CheckCircle2,
    18→  Radio, Loader2, ShieldCheck, Vote, FileWarning, Plus, X,
    19→  TrendingUp, BarChart3, Users,
    20→} from 'lucide-react';
    21→import { motion, AnimatePresence } from 'framer-motion';
    22→import { cn } from '@/lib/utils';
    23→import { toast } from 'sonner';
    24→
    25→// Nigerian political parties for different election tiers
    26→const PARTIES_BY_TIER: Record<string, { code: string; name: string; color: string }[]> = {
    27→  PRESIDENTIAL: [
    28→    { code: 'APC', name: 'All Progressives Congress', color: '#008751' },
    29→    { code: 'PDP', name: 'Peoples Democratic Party', color: '#CE1126' },
    30→    { code: 'LP', name: 'Labour Party', color: '#2196F3' },
    31→    { code: 'NNPP', name: 'New Nigeria Peoples Party', color: '#FF9800' },
    32→  ],
    33→  STATE: [
    34→    { code: 'APC', name: 'APC', color: '#008751' },
    35→    { code: 'PDP', name: 'PDP', color: '#CE1126' },
    36→    { code: 'LP', name: 'LP', color: '#2196F3' },
    37→    { code: 'NNPP', name: 'NNPP', color: '#FF9800' },
    38→    { code: 'ADC', name: 'ADC', color: '#9C27B0' },
    39→  ],
    40→  LOCAL: [
    41→    { code: 'APC', name: 'APC', color: '#008751' },
    42→    { code: 'PDP', name: 'PDP', color: '#CE1126' },
    43→    { code: 'LP', name: 'LP', color: '#2196F3' },
    44→  ],
    45→};
    46→
    47→const INCIDENT_TYPES = [
    48→  { value: 'VIOLENCE', label: 'Violence', severity: 'HIGH', color: 'rose' },
    49→  { value: 'INTIMIDATION', label: 'Voter Intimidation', severity: 'HIGH', color: 'rose' },
    50→  { value: 'BALLOT_STUFFING', label: 'Ballot Stuffing', severity: 'CRITICAL', color: 'rose' },
    51→  { value: 'BRIBERY', label: 'Bribery / Vote Buying', severity: 'HIGH', color: 'amber' },
    52→  { value: 'UNDERAGE_VOTING', label: 'Underage Voting', severity: 'MEDIUM', color: 'amber' },
    53→  { value: 'MULTIPLE_VOTING', label: 'Multiple Voting', severity: 'HIGH', color: 'amber' },
    54→  { value: 'SNATCHED_BALLOT', label: 'Snatched Ballot Box', severity: 'CRITICAL', color: 'rose' },
    55→  { value: 'IMPEDIMENT', label: 'Impediment to Voting', severity: 'MEDIUM', color: 'amber' },
    56→  { value: 'LOGISTICS', label: 'Logistics / Materials Issue', severity: 'LOW', color: 'cyan' },
    57→  { value: 'BVAS_FAILURE', label: 'BVAS / Tech Failure', severity: 'MEDIUM', color: 'amber' },
    58→  { value: 'OBSERVATION', label: 'General Observation', severity: 'LOW', color: 'muted' },
    59→];
    60→
    61→type Tab = 'results' | 'statistics' | 'incident';
    62→
    63→interface PartyVote {
    64→  party: string;
    65→  votes: string; // string for input, parsed on submit
    66→}
    67→
    68→export function SubmitReport() {
    69→  const { user, electionTier } = useDashboardStore();
    70→  const queryClient = useQueryClient();
    71→
    72→  const [activeTab, setActiveTab] = useState<Tab>('results');
    73→  const [selectedPU, setSelectedPU] = useState('');
    74→  const [submitting, setSubmitting] = useState(false);
    75→  const [submitted, setSubmitted] = useState(false);
    76→  const [submittedType, setSubmittedType] = useState('');
    77→
    78→  // ---- RESULTS STATE ----
    79→  const [accredited, setAccredited] = useState('');
    80→  const [validVotes, setValidVotes] = useState('');
    81→  const [rejectedBallots, setRejectedBallots] = useState('');
    82→  const [partyVotes, setPartyVotes] = useState<PartyVote[]>(
    83→    (PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL).map(p => ({ party: p.code, votes: '' }))
    84→  );
    85→
    86→  // ---- STATISTICS STATE ----
    87→  const [bvasUsed, setBvasUsed] = useState(true);
    88→  const [materialsOnTime, setMaterialsOnTime] = useState(true);
    89→  const [securityPresent, setSecurityPresent] = useState(true);
    90→  const [violenceOccurred, setViolenceOccurred] = useState(false);
    91→  const [statsNotes, setStatsNotes] = useState('');
    92→
    93→  // ---- INCIDENT STATE ----
    94→  const [incType, setIncType] = useState('');
    95→  const [incSeverity, setIncSeverity] = useState('MEDIUM');
    96→  const [incDescription, setIncDescription] = useState('');
    97→
    98→  // Fetch polling units for the agent (show first 10 as options)
    99→  const { data: puData } = useQuery({
   100→    queryKey: ['my-polling-units'],
   101→    queryFn: () => fetch('/api/dashboard').then(r => r.json()).then(d => d.pollingUnits?.slice(0, 15) || []),
   102→  });
   103→
   104→  const pollingUnits = puData || [];
   105→
   106→  const parties = PARTIES_BY_TIER[electionTier] || PARTIES_BY_TIER.PRESIDENTIAL;
   107→
   108→  // Add a party vote row
   109→  const addPartyRow = () => {
   110→    const existingCodes = partyVotes.map(p => p.party);
   111→    const missing = parties.find(p => !existingCodes.includes(p.code));
   112→    if (missing) {
   113→      setPartyVotes(prev => [...prev, { party: missing.code, votes: '' }]);
   114→    }
   115→  };
   116→
   117→  const removePartyRow = (idx: number) => {
   118→    if (partyVotes.length > 2) {
   119→      setPartyVotes(prev => prev.filter((_, i) => i !== idx));
   120→    }
   121→  };
   122→
   123→  const updatePartyVote = (idx: number, field: 'party' | 'votes', value: string) => {
   124→    setPartyVotes(prev => prev.map((p, i) => i === idx ? { ...p, [field]: value } : p));
   125→  };
   126→
   127→  // Auto-calculate totals
   128→  const totalVotesCast = partyVotes.reduce((s, p) => s + (parseInt(p.votes) || 0), 0);
   129→  const accreditedNum = parseInt(accredited) || 0;
   130→  const validVotesNum = parseInt(validVotes) || 0;
   131→  const rejectedNum = parseInt(rejectedBallots) || 0;
   132→
   133→  // ---- SUBMIT HANDLERS ----
   134→
   135→  const resultMutation = useMutation({
   136→    mutationFn: (body: Record<string, unknown>) =>
   137→      fetch('/api/results', {
   138→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   139→      }).then(r => r.json()),
   140→    onSuccess: (data) => {
   141→      if (data.error) { toast.error(data.error); return; }
   142→      toast.success('Election results submitted successfully!');
   143→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   144→      queryClient.invalidateQueries({ queryKey: ['situation-room'] });
   145→      setSubmitted(true); setSubmittedType('results');
   146→      setTimeout(() => setSubmitted(false), 4000);
   147→    },
   148→    onError: () => toast.error('Failed to submit results'),
   149→  });
   150→
   151→  const incidentMutation = useMutation({
   152→    mutationFn: (body: Record<string, unknown>) =>
   153→      fetch('/api/incidents', {
   154→        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   155→      }).then(r => r.json()),
   156→    onSuccess: (data) => {
   157→      if (data.error) { toast.error(data.error); return; }
   158→      toast.success('Incident report submitted!');
   159→      if (data.incident?.gpsAnomaly) {
   160→        toast.warning('GPS anomaly detected — report quarantined for review');
   161→      }
   162→      queryClient.invalidateQueries({ queryKey: ['incidents'] });
   163→      queryClient.invalidateQueries({ queryKey: ['alerts'] });
   164→      setSubmitted(true); setSubmittedType('incident');
   165→      setIncDescription(''); setIncType('');
   166→      setTimeout(() => setSubmitted(false), 4000);
   167→    },
   168→    onError: () => toast.error('Failed to submit incident'),
   169→  });
   170→
   171→  const handleSubmitResults = () => {
   172→    if (!user?.id || !selectedPU) {
   173→      toast.error('Please select a polling unit');
   174→      return;
   175→    }
   176→    if (partyVotes.every(p => !p.votes)) {
   177→      toast.error('Enter at least one party vote count');
   178→      return;
   179→    }
   180→
   181→    const parsedPartyResults = partyVotes
   182→      .filter(p => p.votes)
   183→      .map(p => {
   184→        const partyInfo = parties.find(pt => pt.code === p.party);
   185→        return { party: p.party, name: partyInfo?.name || p.party, votes: parseInt(p.votes) || 0, color: partyInfo?.color || '#888' };
   186→      });
   187→
   188→    setSubmitting(true);
   189→    resultMutation.mutate({
   190→      reporterId: user.id,
   191→      pollingUnitId: selectedPU,
   192→      accreditedVoters: accreditedNum,
   193→      totalValidVotes: validVotesNum,
   194→      rejectedBallots: rejectedNum,
   195→      totalVotesCast,
   196→      partyResults: parsedPartyResults,
   197→      bvasUsed, materialsArrivedOnTime: materialsOnTime,
   198→      securityPresent, violenceOccurred, notes: statsNotes,
   199→    }, { onSettled: () => setSubmitting(false) });
   200→  };
   201→
   202→  const handleSubmitIncident = () => {
   203→    if (!user?.id) return;
   204→    if (!incType) { toast.error('Select an incident type'); return; }
   205→    if (!incDescription.trim()) { toast.error('Describe the incident'); return; }
   206→
   207→    const typeConfig = INCIDENT_TYPES.find(t => t.value === incType);
   208→    setSubmitting(true);
   209→    incidentMutation.mutate({
   210→      reporterId: user.id,
   211→      pollingUnitId: selectedPU || undefined,
   212→      type: incType,
   213→      severity: typeConfig?.severity || incSeverity,
   214→      description: incDescription.trim(),
   215→    }, { onSettled: () => setSubmitting(false) });
   216→  };
   217→
   218→  const selectedPUData = pollingUnits.find(p => p.id === selectedPU);
   219→
   220→  return (
   221→    <div className="h-full overflow-y-auto p-4 sm:p-6">
   222→      <div className="max-w-2xl mx-auto space-y-4">
   223→        {/* Header */}
   224→        <div>
   225→          <h2 className="text-lg font-semibold flex items-center gap-2">
   226→            <Send className="h-5 w-5 text-emerald" />
   227→            Field Report
   228→          </h2>
   229→          <p className="text-sm text-muted-foreground mt-0.5">
   230→            Submit election results, polling unit statistics, or report incidents and infractions.
   231→          </p>
   232→        </div>
   233→
   234→        {/* Agent info + PU selector */}
   235→        <Card className="border-emerald/20 bg-emerald/5">
   236→          <CardContent className="p-3.5 flex items-center gap-3">
   237→            <div className="w-10 h-10 rounded-full bg-emerald/20 flex items-center justify-center text-sm font-bold text-emerald">
   238→              {user?.name.split(' ').map(n => n[0]).join('')}
   239→            </div>
   240→            <div className="flex-1 min-w-0">
   241→              <p className="text-sm font-medium">{user?.name}</p>
   242→              <div className="flex items-center gap-1.5 mt-0.5">
   243→                <ShieldCheck className="h-3 w-3 text-emerald" />
   244→                <span className="text-[10px] text-muted-foreground">In-App Capture &middot; C2PA enabled</span>
   245→              </div>
   246→            </div>
   247→          </CardContent>
   248→        </Card>
   249→
   250→        {/* Polling Unit Selector */}
   251→        <div className="space-y-1.5">
   252→          <label className="text-xs font-medium flex items-center gap-1.5">
   253→            <MapPin className="h-3.5 w-3.5 text-emerald" />
   254→            Select Polling Unit
   255→          </label>
   256→          <Select value={selectedPU} onValueChange={setSelectedPU}>
   257→            <SelectTrigger className="h-10">
   258→              <SelectValue placeholder="Choose your assigned polling unit..." />
   259→            </SelectTrigger>
   260→            <SelectContent>
   261→              {pollingUnits.map(pu => (
   262→                <SelectItem key={pu.id} value={pu.id}>
   263→                  <span className="flex items-center gap-2">
   264→                    <span className="text-muted-foreground text-[10px]">{pu.code}</span>
   265→                    <span>{pu.name}</span>
   266→                    <span className="text-muted-foreground text-[10px]">{pu.state}/{pu.lga}</span>
   267→                    {pu.status === 'FLAGGED' && <Badge variant="outline" className="text-[9px] h-4 border-amber/30 text-amber ml-1">FLAGGED</Badge>}
   268→                  </span>
   269→                </SelectItem>
   270→              ))}
   271→            </SelectContent>
   272→          </Select>
   273→          {selectedPUData && (
   274→            <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   275→              <Badge variant="outline" className="text-[10px] h-5 border-emerald/30 text-emerald">
   276→                <span className="w-1.5 h-1.5 rounded-full bg-emerald mr-1" />IN GEOFENCE
   277→              </Badge>
   278→              <span>{selectedPUData.registered} registered voters</span>
   279→              <span>{selectedPUData.state} / {selectedPUData.lga}</span>
   280→            </div>
   281→          )}
   282→        </div>
   283→
   284→        {/* Tab selector */}
   285→        <div className="grid grid-cols-3 gap-1 p-1 rounded-lg bg-card/60 border border-border">
   286→          {[
   287→            { id: 'results' as Tab, label: 'Election Results', icon: <Vote className="h-4 w-4" /> },
   288→            { id: 'statistics' as Tab, label: 'Statistics', icon: <BarChart3 className="h-4 w-4" /> },
   289→            { id: 'incident' as Tab, label: 'Incident / Infraction', icon: <FileWarning className="h-4 w-4" /> },
   290→          ].map(tab => (
   291→            <button
   292→              key={tab.id}
   293→              onClick={() => setActiveTab(tab.id)}
   294→              className={cn(
   295→                'flex items-center justify-center gap-1.5 px-3 py-2.5 rounded-md text-xs font-medium transition-all',
   296→                activeTab === tab.id
   297→                  ? 'bg-emerald/15 text-emerald shadow-sm'
   298→                  : 'text-muted-foreground hover:text-foreground hover:bg-card'
   299→              )}
   300→            >
   301→              {tab.icon}
   302→              <span className="hidden sm:inline">{tab.label}</span>
   303→            </button>
   304→          ))}
   305→        </div>
   306→
   307→        {/* Tab content */}
   308→        <AnimatePresence mode="wait">
   309→          <motion.div
   310→            key={activeTab}
   311→            initial={{ opacity: 0, y: 4 }}
   312→            animate={{ opacity: 1, y: 0 }}
   313→            exit={{ opacity: 0, y: -4 }}
   314→            transition={{ duration: 0.15 }}
   315→          >
   316→            {/* ===== ELECTION RESULTS TAB ===== */}
   317→            {activeTab === 'results' && (
   318→              <Card>
   319→                <CardContent className="p-4 space-y-4">
   320→                  <div>
   321→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   322→                      <Vote className="h-4 w-4 text-emerald" />
   323→                      Election Results Form
   324→                    </h3>
   325→                    <p className="text-[11px] text-muted-foreground mt-0.5">
   326→                      Enter the official vote counts from Form EC8A at your polling unit.
   327→                    </p>
   328→                  </div>
   329→
   330→                  {/* Summary fields */}
   331→                  <div className="grid grid-cols-2 gap-3">
   332→                    <div className="space-y-1.5">
   333→                      <label className="text-[11px] font-medium text-muted-foreground">Accredited Voters</label>
   334→                      <Input
   335→                        type="number" min="0"
   336→                        placeholder="0"
   337→                        value={accredited}
   338→                        onChange={(e) => setAccredited(e.target.value)}
   339→                        className="h-10 text-sm tabular-nums"
   340→                      />
   341→                    </div>
   342→                    <div className="space-y-1.5">
   343→                      <label className="text-[11px] font-medium text-muted-foreground">Rejected Ballots</label>
   344→                      <Input
   345→                        type="number" min="0"
   346→                        placeholder="0"
   347→                        value={rejectedBallots}
   348→                        onChange={(e) => setRejectedBallots(e.target.value)}
   349→                        className="h-10 text-sm tabular-nums"
   350→                      />
   351→                    </div>
   352→                  </div>
   353→
   354→                  <Separator />
   355→
   356→                  {/* Party vote entries */}
   357→                  <div className="space-y-2">
   358→                    <div className="flex items-center justify-between">
   359→                      <label className="text-xs font-medium">Vote Count by Party</label>
   360→                      <Button variant="ghost" size="sm" className="h-6 text-[10px] gap-1" onClick={addPartyRow}>
   361→                        <Plus className="h-3 w-3" /> Add Party
   362→                      </Button>
   363→                    </div>
   364→
   365→                    <div className="space-y-2">
   366→                      {partyVotes.map((pv, idx) => {
   367→                        const partyInfo = parties.find(p => p.code === pv.party);
   368→                        return (
   369→                          <div key={idx} className="flex items-center gap-2">
   370→                            <div
   371→                              className="w-3 h-8 rounded-sm shrink-0"
   372→                              style={{ backgroundColor: partyInfo?.color || '#555' }}
   373→                            />
   374→                            <div className="flex-1 min-w-0">
   375→                              <Select value={pv.party} onValueChange={(v) => updatePartyVote(idx, 'party', v)}>
   376→                                <SelectTrigger className="h-8 text-xs">
   377→                                  <SelectValue />
   378→                                </SelectTrigger>
   379→                                <SelectContent>
   380→                                  {parties.map(p => (
   381→                                    <SelectItem key={p.code} value={p.code}>
   382→                                      <span className="flex items-center gap-2">
   383→                                        <span className="w-2 h-2 rounded-sm" style={{ backgroundColor: p.color }} />
   384→                                        <span className="font-medium">{p.code}</span>
   385→                                        <span className="text-muted-foreground text-[10px] hidden sm:inline">{p.name}</span>
   386→                                      </span>
   387→                                    </SelectItem>
   388→                                  ))}
   389→                                </SelectContent>
   390→                              </Select>
   391→                            </div>
   392→                            <Input
   393→                              type="number" min="0" placeholder="0"
   394→                              value={pv.votes}
   395→                              onChange={(e) => updatePartyVote(idx, 'votes', e.target.value)}
   396→                              className="w-28 h-8 text-sm tabular-nums text-right"
   397→                            />
   398→                            {partyVotes.length > 2 && (
   399→                              <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-rose" onClick={() => removePartyRow(idx)}>
   400→                                <X className="h-3.5 w-3.5" />
   401→                              </Button>
   402→                            )}
   403→                          </div>
   404→                        );
   405→                      })}
   406→                    </div>
   407→                  </div>
   408→
   409→                  {/* Live totals */}
   410→                  <div className="rounded-lg bg-background border border-border p-3 space-y-2">
   411→                    <div className="flex items-center gap-2 text-xs font-medium">
   412→                      <TrendingUp className="h-3.5 w-3.5 text-emerald" />
   413→                      Live Totals
   414→                    </div>
   415→                    <div className="grid grid-cols-3 gap-2 text-center">
   416→                      <div>
   417→                        <p className="text-lg font-bold tabular-nums text-emerald">{totalVotesCast.toLocaleString()}</p>
   418→                        <p className="text-[10px] text-muted-foreground">Total Cast</p>
   419→                      </div>
   420→                      <div>
   421→                        <p className="text-lg font-bold tabular-nums text-cyan">{rejectedNum}</p>
   422→                        <p className="text-[10px] text-muted-foreground">Rejected</p>
   423→                      </div>
   424→                      <div>
   425→                        <p className="text-lg font-bold tabular-nums text-amber">{accreditedNum > 0 ? Math.round((totalVotesCast / accreditedNum) * 10000) / 100 : 0}%</p>
   426→                        <p className="text-[10px] text-muted-foreground">Turnout</p>
   427→                      </div>
   428→                    </div>
   429→                    {/* Simple bar chart */}
   430→                    {partyVotes.some(p => p.votes) && (
   431→                      <div className="flex h-3 rounded-full overflow-hidden bg-secondary mt-1">
   432→                        {partyVotes.filter(p => p.votes).map((pv) => {
   433→                          const partyInfo = parties.find(p => p.code === pv.party);
   434→                          const pct = totalVotesCast > 0 ? (parseInt(pv.votes) || 0) / totalVotesCast * 100 : 0;
   435→                          return (
   436→                            <div
   437→                              key={pv.party}
   438→                              className="h-full transition-all duration-500"
   439→                              style={{ width: `${pct}%`, backgroundColor: partyInfo?.color || '#555' }}
   440→                              title={`${pv.party}: ${pv.votes}`}
   441→                            />
   442→                          );
   443→                        })}
   444→                      </div>
   445→                    )}
   446→                  </div>
   447→
   448→                  <Button
   449→                    onClick={handleSubmitResults}
   450→                    disabled={submitting || !selectedPU || partyVotes.every(p => !p.votes)}
   451→                    className="w-full h-11 bg-emerald hover:bg-emerald/90 text-emerald-950 font-semibold gap-2"
   452→                  >
   453→                    {submitting && submittedType === 'results' ? (
   454→                      <><Loader2 className="h-4 w-4 animate-spin" /> Submitting...</>
   455→                    ) : submitted && submittedType === 'results' ? (
   456→                      <><CheckCircle2 className="h-4 w-4" /> Results Submitted!</>
   457→                    ) : (
   458→                      <><Send className="h-4 w-4" /> Submit Election Results</>
   459→                    )}
   460→                  </Button>
   461→                </CardContent>
   462→              </Card>
   463→            )}
   464→
   465→            {/* ===== STATISTICS TAB ===== */}
   466→            {activeTab === 'statistics' && (
   467→              <Card>
   468→                <CardContent className="p-4 space-y-4">
   469→                  <div>
   470→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   471→                      <BarChart3 className="h-4 w-4 text-cyan" />
   472→                      Polling Unit Statistics
   473→                    </h3>
   474→                    <p className="text-[11px] text-muted-foreground mt-0.5">
   475→                      Record the operational status and conditions at your polling unit.
   476→                    </p>
   477→                  </div>
   478→
