   149→
   150→function StatCard({
   151→  icon, label, value, color, sub,
   152→}: {
   153→  icon: React.ReactNode; label: string; value: string | number; color: string; sub?: string;
   154→}) {
   155→  const colorMap: Record<string, string> = {
   156→    emerald: 'text-emerald', cyan: 'text-cyan', amber: 'text-amber',
   157→    rose: 'text-rose', purple: 'text-purple', foreground: 'text-foreground',
   158→  };
   159→  return (
   160→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.25 }}>
   161→      <Card className="border-border bg-card/40">
   162→        <CardContent className="p-3 sm:p-4">
   163→          <div className="flex items-center gap-1.5 mb-1.5">
   164→            <span className={colorMap[color] || 'text-muted-foreground'}>{icon}</span>
   165→            <span className="text-[10px] text-muted-foreground uppercase tracking-wider">{label}</span>
   166→          </div>
   167→          <p className={cn('text-lg font-bold tabular-nums', colorMap[color] || 'text-foreground')}>{value}</p>
   168→          {sub && <p className="text-[10px] text-muted-foreground/60 mt-0.5">{sub}</p>}
   169→        </CardContent>
   170→      </Card>
   171→    </motion.div>
   172→  );
   173→}

   179→function ColorBar({ value, max = 100, color }: { value: number; max?: number; color: string }) {
   180→  const pct = Math.min((value / max) * 100, 100);
   181→  return (
   182→    <div className="h-2 bg-secondary rounded-full overflow-hidden w-full min-w-[80px]">
   183→      <motion.div
   184→        initial={{ width: 0 }}
   185→        animate={{ width: `${pct}%` }}
   186→        transition={{ duration: 0.6, ease: 'easeOut' }}
   187→        className="h-full rounded-full"
   188→        style={{ backgroundColor: color }}
   189→      />
   190→    </div>
   191→  );
   192→}

   199→  const queryClient = useQueryClient();
   200→  const { tenantId } = useDashboardStore();
   201→
   202→  // Dialog states

   205→
   206→  // ── Honeypot form state ──
   207→  const [hpName, setHpName] = useState('');
   208→  const [hpState, setHpState] = useState('');
   209→  const [hpLga, setHpLga] = useState('');
   210→  const [hpTrapType, setHpTrapType] = useState('');
   211→  const [hpExpectedResults, setHpExpectedResults] = useState<Array<{ party: string; votes: string }>>([
   212→    { party: '', votes: '' },
   213→  ]);
   214→
   215→  // ── Accessibility form state ──
   216→  const [arPollingUnitId, setArPollingUnitId] = useState('');

   218→    rampAccess: false, brailleBallots: false, signLanguage: false,
   219→    assistiveTech: false, accessibleToilet: false,
   220→  });

   222→  const [arPwdServed, setArPwdServed] = useState('');

   224→  const [arScore, setArScore] = useState([50]);
   225→

   227→  const { data, isLoading, refetch } = useQuery<HoneypotData>({
   228→    queryKey: ['honeypot', tenantId],
   229→    queryFn: () =>
   230→      fetch(`/api/honeypot?tenantId=${tenantId}`).then((r) => r.json()),
   231→    refetchInterval: 15000,
   232→  });
   233→
   234→  // ── Mutation ──
   235→  const mutation = useMutation({
   236→    mutationFn: (body: Record<string, unknown>) =>
   237→      fetch(`/api/honeypot?tenantId=${tenantId}`, {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify(body),
   241→      }).then((r) => r.json()),
   242→    onSuccess: () => {
   243→      queryClient.invalidateQueries({ queryKey: ['honeypot', tenantId] });

   245→    },
   246→    onError: () => {
   247→      toast.error('Failed to complete action');
   248→    },
   249→  });
   250→
   251→  const isMutating = mutation.isPending;
   252→
   253→  // ── Derived data ──
   254→  const honeypots = data?.honeypots || [];
   255→  const accessibility = data?.accessibility || [];
   256→  const stats = data?.stats;
   257→  const biometricSummary = data?.biometricSummary;
   258→  const trapEffectiveness = data?.trapEffectiveness;
   259→  const biometricAgents = useMemo(
   260→    () =>
   261→      [...(biometricSummary?.agents || [])].sort(
   262→        (a, b) => b.biometricRiskScore - a.biometricRiskScore,
   263→      ),
   264→    [biometricSummary],
   265→  );
   266→
   267→  // ── Honeypot dialog handlers ──
   268→  const resetHoneypotForm = () => {
   269→    setHpName('');
   270→    setHpState('');
   271→    setHpLga('');
   272→    setHpTrapType('');
   273→    setHpExpectedResults([{ party: '', votes: '' }]);
   274→  };
   275→
   276→  const handleAddResultRow = () => {
   277→    setHpExpectedResults((prev) => [...prev, { party: '', votes: '' }]);
   278→  };
   279→
   280→  const handleRemoveResultRow = (idx: number) => {
   281→    setHpExpectedResults((prev) => prev.filter((_, i) => i !== idx));
   282→  };
   283→
   284→  const handleResultChange = (idx: number, field: 'party' | 'votes', value: string) => {
   285→    setHpExpectedResults((prev) =>
   286→      prev.map((r, i) => (i === idx ? { ...r, [field]: value } : r)),
   287→    );
   288→  };
   289→
   290→  const handleSubmitHoneypot = () => {
   291→    if (!hpName.trim() || !hpState.trim() || !hpTrapType) {
   292→      toast.error('Please fill in name, state, and trap type');
   293→      return;
   294→    }
   295→    const validResults = hpExpectedResults
   296→      .filter((r) => r.party.trim() && r.votes.trim())
   297→      .map((r) => ({ party: r.party.trim(), votes: Number(r.votes) }));
   298→    mutation.mutate(
   299→      {
   300→        action: 'CREATE_HONEYPOT',
   301→        name: hpName.trim(),
   302→        state: hpState.trim(),
   303→        lga: hpLga.trim() || null,
   304→        trapType: hpTrapType,
   305→        expectedResults: validResults,
   306→      },
   307→      {
   308→        onSuccess: () => {
   309→          setHoneypotDialogOpen(false);
   310→          resetHoneypotForm();
   311→        },
   312→      },
   313→    );
   314→  };
   315→
   316→  // ── Accessibility dialog handlers ──
   317→  const resetAccessibilityForm = () => {
   318→    setArPollingUnitId('');
   319→    setArFeatures({
   320→      rampAccess: false, brailleBallots: false, signLanguage: false,
   321→      assistiveTech: false, accessibleToilet: false,
   322→    });
   323→    setArBarriers([]);
   324→    setArPwdServed('');
   325→    setArPwdTurnedAway('');
   326→    setArScore([50]);
   327→  };
   328→
   329→  const toggleBarrier = (barrier: string) => {
   330→    setArBarriers((prev) =>
   331→      prev.includes(barrier) ? prev.filter((b) => b !== barrier) : [...prev, barrier],
   332→    );
   333→  };
   334→
   335→  const toggleFeature = (key: keyof typeof arFeatures) => {
   336→    setArFeatures((prev) => ({ ...prev, [key]: !prev[key] }));
   337→  };
   338→
   339→  const handleSubmitAccessibility = () => {
   340→    if (!arPollingUnitId.trim()) {
   341→      toast.error('Please enter a polling unit ID');
   342→      return;
   343→    }
   344→    mutation.mutate(
   345→      {
   346→        action: 'CREATE_ACCESSIBILITY_REPORT',
   347→        pollingUnitId: arPollingUnitId.trim(),
   348→        features: arFeatures,
   349→        barrierTypes: arBarriers,
   350→        pwdVotersServed: Number(arPwdServed) || 0,
   351→        pwdVotersTurnedAway: Number(arPwdTurnedAway) || 0,
   352→        overallScore: arScore[0],
   353→      },
   354→      {
   355→        onSuccess: () => {
   356→          setAccessibilityDialogOpen(false);
   357→          resetAccessibilityForm();
   358→        },
   359→      },
   360→    );
   361→  };
   362→
   363→  // ── Render ──
   364→  return (
   365→    <div className="h-full flex flex-col p-4 sm:p-6 gap-4 overflow-hidden">
   366→      {/* Header */}
   367→      <div className="flex items-center justify-between flex-wrap gap-3 shrink-0">
   368→        <div>
   369→          <h2 className="text-lg font-semibold flex items-center gap-2">
   370→            <Radar className="h-5 w-5 text-purple" />
   371→            Honeypot Stations, Biometrics & PWD Tracking
   372→          </h2>
   373→          <p className="text-sm text-muted-foreground mt-0.5">
   374→            Decoy traps, agent biometric verification, and accessibility compliance
   375→          </p>
   376→        </div>
   377→        <Button
   378→          variant="outline"
   379→          size="sm"
   380→          onClick={() => refetch()}
   381→          className="gap-1.5 text-xs"
   382→        >
   383→          <Activity className="h-3.5 w-3.5" />
   384→          Refresh
   385→        </Button>
   386→      </div>
   387→
   388→      {isLoading ? (
   389→        <div className="flex-1 flex items-center justify-center">
   390→          <Loader2 className="h-6 w-6 animate-spin text-purple" />
   391→        </div>
   392→      ) : (
   393→        <Tabs defaultValue="honeypot" className="flex-1 min-h-0 flex flex-col">
   394→          <TabsList className="shrink-0">
   395→            <TabsTrigger value="honeypot" className="gap-1.5 text-xs">
   396→              <ShieldAlert className="h-3.5 w-3.5" />
   397→              Honeypot Stations
   398→            </TabsTrigger>
   399→            <TabsTrigger value="biometrics" className="gap-1.5 text-xs">
   400→              <Fingerprint className="h-3.5 w-3.5" />
   401→              Biometrics
   402→            </TabsTrigger>

   404→              <Accessibility className="h-3.5 w-3.5" />
   405→              PWD / Accessibility
   406→            </TabsTrigger>
   407→          </TabsList>

   409→          {/* ═══════════ TAB 1: Honeypot Stations ═══════════ */}

   420→                icon={<CircleDot className="h-3.5 w-3.5" />}

   422→                value={stats?.activeHoneypots ?? 0}

   459→            {/* Honeypot cards grid */}

   469→                    <motion.div

   475→                      <Card
   476→                        className={cn(
   477→                          'border transition-colors',
   478→                          hp.alertTriggered
   479→                            ? 'border-rose/50 bg-rose/5'
   480→                            : hp.deviationDetected
   481→                              ? 'border-amber/40 bg-amber/5'
   482→                              : 'border-border bg-card/40',
   483→                        )}
   484→                      >
   485→                        <CardContent className="p-4 space-y-3">

   488→                            <div className="flex-1 min-w-0">

   522→                                    <span className="h-2 w-2 rounded-full bg-muted-foreground/40" />
   523→                                    <span className="text-[10px] text-muted-foreground">Inactive</span>
   524→                                  </>
   525→                                )}
   526→                              </div>

   530→                          {/* Deviation banner */}
