    99→    99→function timeAgo(dateStr: string | null): string {
   100→   100→  if (!dateStr) return 'Never';
   101→   101→  const diff = Date.now() - new Date(dateStr).getTime();
   102→   102→  const mins = Math.floor(diff / 60000);
   103→   103→  if (mins < 1) return 'Just now';
   104→   104→  if (mins < 60) return `${mins}m ago`;
   105→   105→  const hrs = Math.floor(mins / 60);
   106→   106→  if (hrs < 24) return `${hrs}h ago`;
   107→   107→  return `${Math.floor(hrs / 24)}d ago`;
   108→   108→}
