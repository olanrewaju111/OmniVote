   100→function formatDate(d: string | null | undefined): string {
   101→  if (!d) return '—';
   102→  return new Date(d).toLocaleDateString('en-NG', {
   103→    day: '2-digit',
   104→    month: 'short',
   105→    hour: '2-digit',
   106→    minute: '2-digit',
   107→  });
   108→}

   200→/* ------------------------------------------------------------------ */
   201→/*  Circular Gauge                                                     */
   202→/* ------------------------------------------------------------------ */
   203→
   204→function SecurityGauge({ score }: { score: number }) {
   205→  const radius = 54;
   206→  const circumference = 2 * Math.PI * radius;
   207→  const offset = circumference - (score / 100) * circumference;
   208→  const color = scoreColor(score);
   209→  const stroke = scoreStroke(score);
   210→
   211→  return (
   212→    <div className="relative flex items-center justify-center">
   213→      <svg width="140" height="140" viewBox="0 0 140 140" className="-rotate-90">
   214→        <circle
   215→          cx="70" cy="70" r={radius}
   216→          fill="none" strokeWidth="8"
   217→          className="stroke-muted/30"
   218→        />
   219→        <motion.circle
   220→          cx="70" cy="70" r={radius}
   221→          fill="none" strokeWidth="8"
   222→          strokeLinecap="round"
   223→          className={stroke}
   224→          initial={{ strokeDashoffset: circumference }}
   225→          animate={{ strokeDashoffset: offset }}
   226→          transition={{ duration: 1.2, ease: 'easeOut' }}
   227→          strokeDasharray={circumference}
   228→          strokeDashoffset={offset}
   229→        />
   230→      </svg>
   231→      <div className="absolute flex flex-col items-center">
   232→        <motion.span
   233→          className={cn('text-3xl font-bold tabular-nums', color)}
   234→          initial={{ opacity: 0, scale: 0.8 }}
   235→          animate={{ opacity: 1, scale: 1 }}
   236→          transition={{ delay: 0.3, duration: 0.5 }}
   237→        >
   238→          {score}
   239→        </motion.span>
   240→        <span className="text-[10px] text-muted-foreground">/ 100</span>
   241→      </div>
   242→    </div>
   243→  );
   244→}

   439→
   440→function EventLogTab({ data, resolveMutation }: {
   441→  data: SecurityData;
   442→  resolveMutation: { mutate: (vars: { eventId: string; resolvedById: string | null }) => void; isPending: boolean };
   443→}) {
   444→  const [severityFilter, setSeverityFilter] = useState<string>('ALL');
   445→  const [typeFilter, setTypeFilter] = useState<string>('ALL');
   446→  const [expandedId, setExpandedId] = useState<string | null>(null);
   447→
   448→  const eventTypes = useMemo(() => {
   449→    const types = new Set(data.events.map(e => e.eventType));
   450→    return Array.from(types).sort();
   451→  }, [data.events]);
   452→
   453→  const filtered = useMemo(() => {
   454→    return data.events.filter(e => {
   455→      if (severityFilter !== 'ALL' && e.severity !== severityFilter) return false;
   456→      if (typeFilter !== 'ALL' && e.eventType !== typeFilter) return false;
   457→      return true;
   458→    });
   459→  }, [data.events, severityFilter, typeFilter]);
   460→
   461→  const currentUser = useDashboardStore(s => s.user);
   462→
   463→  const handleResolve = (eventId: string) => {
   464→    resolveMutation.mutate({
   465→      eventId,
   466→      resolvedById: currentUser?.id || null,
   467→    });
   468→  };
   469→
   470→  return (
   471→    <div className="flex flex-col gap-3 flex-1 overflow-hidden">
   472→      {/* Filter row */}
   473→      <div className="flex items-center gap-3 shrink-0 flex-wrap">
   474→        <Select value={severityFilter} onValueChange={setSeverityFilter}>
   475→          <SelectTrigger className="w-[160px] h-8 text-xs">
   476→            <SelectValue placeholder="Severity" />
   477→          </SelectTrigger>
   478→          <SelectContent>
   479→            <SelectItem value="ALL">All Severities</SelectItem>
   480→            <SelectItem value="INFO">Info</SelectItem>
   481→            <SelectItem value="WARNING">Warning</SelectItem>
   482→            <SelectItem value="CRITICAL">Critical</SelectItem>
   483→          </SelectContent>
   484→        </Select>
   485→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   486→          <SelectTrigger className="w-[180px] h-8 text-xs">
   487→            <SelectValue placeholder="Event Type" />
   488→          </SelectTrigger>
   489→          <SelectContent>
   490→            <SelectItem value="ALL">All Types</SelectItem>
   491→            {eventTypes.map(t => (
   492→              <SelectItem key={t} value={t}>{t.replace(/_/g, ' ')}</SelectItem>
   493→            ))}
   494→          </SelectContent>
   495→        </Select>
   496→        <span className="text-[10px] text-muted-foreground ml-auto">
   497→          {filtered.length} of {data.events.length} events
   498→        </span>
   499→      </div>
   500→
   501→      {/* Table */}
   502→      <div className="flex-1 overflow-y-auto rounded-lg border border-border">
   503→        <Table>
   504→          <TableHeader>
   505→            <TableRow className="hover:bg-transparent">
   506→              <TableHead className="text-[10px] h-8 w-28">Time</TableHead>
   507→              <TableHead className="text-[10px] h-8 w-36">Type</TableHead>
   508→              <TableHead className="text-[10px] h-8 w-24">Severity</TableHead>
   509→              <TableHead className="text-[10px] h-8">Description</TableHead>
   510→              <TableHead className="text-[10px] h-8 w-28">IP Address</TableHead>
   511→              <TableHead className="text-[10px] h-8 w-28">User</TableHead>
   512→              <TableHead className="text-[10px] h-8 w-20">Status</TableHead>
   513→              <TableHead className="text-[10px] h-8 w-16" />
   514→            </TableRow>
   515→          </TableHeader>
   516→          <TableBody>
   517→            {filtered.map(event => (
   518→              <>
   519→                <TableRow
   520→                  key={event.id}
   521→                  className={cn(
   522→                    'cursor-pointer transition-colors',
   523→                    expandedId === event.id && 'bg-card/60',
   524→                  )}
   525→                  onClick={() => setExpandedId(expandedId === event.id ? null : event.id)}
   526→                >
   527→                  <TableCell className="text-[10px] py-2 text-muted-foreground">
   528→                    {formatDate(event.createdAt)}
   529→                  </TableCell>
   530→                  <TableCell className="text-xs py-2">
   531→                    <div className="flex items-center gap-1.5">
   532→                      {eventTypeIcon(event.eventType, 'h-3.5 w-3.5')}
   533→                      <span>{event.eventType.replace(/_/g, ' ')}</span>

   882→}) {
   883→  const policies = data.policies;
   884→  const [encryptionEnabled, setEncryptionEnabled] = useState(policies?.encryptionEnabled ?? false);
   885→  const [twoFactorEnabled, setTwoFactorEnabled] = useState(policies?.twoFactorEnabled ?? false);
   886→  const [sessionTimeout, setSessionTimeout] = useState(policies?.sessionTimeoutMin ?? 30);
   887→  const [dataRetention, setDataRetention] = useState(policies?.dataRetentionDays ?? 365);
   888→  const [auditLogRetention, setAuditLogRetention] = useState(policies?.auditLogRetentionDays ?? 90);
   889→  const [ipList, setIpList] = useState<string[]>(policies?.ipWhitelist ?? []);
   890→  const [newIp, setNewIp] = useState('');
   891→
   892→  const handleAddIp = () => {
   893→    const trimmed = newIp.trim();
   894→    if (!trimmed) return;
   895→    if (ipList.includes(trimmed)) {
   896→      toast.error('IP address already in whitelist');
   897→      return;
   898→    }
   899→    const ipRegex = /^(\d{1,3}\.){3}\d{1,3}(\/\d{1,2})?$/;
   900→    if (!ipRegex.test(trimmed)) {
   901→      toast.error('Invalid IP address format');
   902→      return;
   903→    }
   904→    setIpList(prev => [...prev, trimmed]);
   905→    setNewIp('');
   906→  };
   907→
   908→  const handleRemoveIp = (ip: string) => {
   909→    setIpList(prev => prev.filter(i => i !== ip));
   910→  };
   911→
   912→  const handleSave = () => {
   913→    policyMutation.mutate({
   914→      action: 'UPDATE_POLICY',
   915→      encryptionEnabled,
   916→      twoFactorEnabled,
   917→      sessionTimeoutMin: sessionTimeout,
   918→      ipWhitelist: JSON.stringify(ipList),
   919→      dataRetentionDays: dataRetention,
   920→    });
   921→  };
   922→
   923→  const hasChanges = policies && (
   924→    encryptionEnabled !== policies.encryptionEnabled ||
   925→    twoFactorEnabled !== policies.twoFactorEnabled ||
   926→    sessionTimeout !== policies.sessionTimeoutMin ||
   927→    dataRetention !== policies.dataRetentionDays ||
   928→    auditLogRetention !== policies.auditLogRetentionDays ||
   929→    JSON.stringify(ipList) !== JSON.stringify(policies.ipWhitelist)
   930→  );
   931→

   933→    return (
   934→      <div className="flex-1 flex items-center justify-center text-xs text-muted-foreground">
   935→        No policy data available
   936→      </div>
   937→    );
   938→  }
   939→
   940→  return (
   941→    <div className="space-y-4 overflow-y-auto flex-1 pr-1">
   942→      {/* Encryption & 2FA */}
   943→      <Card className="bg-card/40 border-border">
   944→        <CardContent className="p-4 space-y-4">
   945→          <h3 className="text-xs font-semibold flex items-center gap-2 text-muted-foreground">
   946→            <Shield className="h-3.5 w-3.5" />
   947→            Authentication & Encryption
   948→          </h3>
   949→
   950→          <div className="flex items-center justify-between">
   951→            <div className="flex items-center gap-3">
   952→              <Lock className="h-4 w-4 text-cyan" />
   953→              <div>
   954→                <Label className="text-xs font-medium">Encryption at Rest (AES-256)</Label>
   955→                <p className="text-[10px] text-muted-foreground">Encrypt all sensitive data stored in the database</p>
   956→              </div>
   957→            </div>
   958→            <Switch
   959→              checked={encryptionEnabled}
   960→              onCheckedChange={setEncryptionEnabled}
   961→            />
   962→          </div>
   963→
   964→          <Separator className="bg-border" />
   965→
   966→          <div className="flex items-center justify-between">
   967→            <div className="flex items-center gap-3">
   968→              <ShieldCheck className="h-4 w-4 text-emerald" />
   969→              <div>
   970→                <Label className="text-xs font-medium">Two-Factor Authentication</Label>
   971→                <p className="text-[10px] text-muted-foreground">Require 2FA for all user accounts in this tenant</p>
   972→              </div>

   974→            <Switch
   975→              checked={twoFactorEnabled}

   977→            />
   978→          </div>
   979→
   980→          <Separator className="bg-border" />
   981→
   982→          <div className="flex items-center justify-between">
