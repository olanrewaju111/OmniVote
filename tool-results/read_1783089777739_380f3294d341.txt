   204→   204→function SecurityGauge({ score }: { score: number }) {
   205→   205→  const radius = 54;
   206→   206→  const circumference = 2 * Math.PI * radius;
   207→   207→  const offset = circumference - (score / 100) * circumference;
   208→   208→  const color = scoreColor(score);
   209→   209→  const stroke = scoreStroke(score);
   210→   210→
   211→   211→  return (
   212→   212→    <div className="relative flex items-center justify-center">
   213→   213→      <svg width="140" height="140" viewBox="0 0 140 140" className="-rotate-90">
   214→   214→        <circle
   215→   215→          cx="70" cy="70" r={radius}
   216→   216→          fill="none" strokeWidth="8"
   217→   217→          className="stroke-muted/30"
   218→   218→        />
   219→   219→        <motion.circle
   220→   220→          cx="70" cy="70" r={radius}
   221→   221→          fill="none" strokeWidth="8"
   222→   222→          strokeLinecap="round"
   223→   223→          className={stroke}
   224→   224→          initial={{ strokeDashoffset: circumference }}
   225→   225→          animate={{ strokeDashoffset: offset }}
   226→   226→          transition={{ duration: 1.2, ease: 'easeOut' }}
   227→   227→          strokeDasharray={circumference}
   228→   228→          strokeDashoffset={offset}
   229→   229→        />
   230→   230→      </svg>
   231→   231→      <div className="absolute flex flex-col items-center">
   232→   232→        <motion.span
   233→   233→          className={cn('text-3xl font-bold tabular-nums', color)}
   234→   234→          initial={{ opacity: 0, scale: 0.8 }}
   235→   235→          animate={{ opacity: 1, scale: 1 }}
   236→   236→          transition={{ delay: 0.3, duration: 0.5 }}
   237→   237→        >
   238→   238→          {score}
   239→   239→        </motion.span>
   240→   240→        <span className="text-[10px] text-muted-foreground">/ 100</span>
   241→   241→      </div>
   242→   242→    </div>
   243→   243→  );
   244→   244→}
