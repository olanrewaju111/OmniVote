     1→'use client';
     2→
     3→import { Card, CardContent } from '@/components/ui/card';
     4→import { Badge } from '@/components/ui/badge';
     5→import { Button } from '@/components/ui/button';
     6→import {
     7→  Dialog, DialogContent, DialogDescription, DialogFooter,
     8→  DialogHeader, DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import {
    11→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    12→} from '@/components/ui/select';
    13→import { Input } from '@/components/ui/input';
    14→import {
    15→  Building2, Users, Settings, Shield, Vote, Loader2, MapPin, Save,
    16→  RotateCcw, Plus, Trash2, UserPlus, Mail, ChevronRight, Globe,
    17→  Eye, UserCheck, Radio, AlertTriangle, Pencil, AlertCircle,
    18→} from 'lucide-react';
    19→import { useState, useEffect, useCallback } from 'react';
    20→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
    21→import { cn } from '@/lib/utils';
    22→import { toast } from 'sonner';
    23→import { fetchJson } from '@/lib/api';
    24→import { useDashboardStore, type UserRole } from '@/store/dashboard';
    25→
    26→// ---- Shared types & constants ----
    27→const REGION_PRESETS = [
    28→  { label: 'Nigeria (full)', minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0 },
    29→  { label: 'Lagos', minLat: 6.3, maxLat: 6.7, minLng: 3.2, maxLng: 3.5 },
    30→  { label: 'Abuja FCT', minLat: 8.8, maxLat: 9.2, minLng: 7.2, maxLng: 7.7 },
    31→  { label: 'Kano', minLat: 11.8, maxLat: 12.4, minLng: 8.3, maxLng: 8.9 },
    32→  { label: 'Rivers', minLat: 4.5, maxLat: 5.2, minLng: 6.5, maxLng: 7.1 },
    33→  { label: 'Southeast', minLat: 4.8, maxLat: 7.0, minLng: 6.8, maxLng: 8.2 },
    34→  { label: 'Southwest', minLat: 6.0, maxLat: 9.0, minLng: 2.5, maxLng: 5.0 },
    35→  { label: 'Northwest', minLat: 10.0, maxLat: 14.0, minLng: 3.0, maxLng: 9.0 },
    36→  { label: 'North-Central', minLat: 7.0, maxLat: 10.5, minLng: 3.0, maxLng: 9.5 },
    37→  { label: 'South-South', minLat: 4.3, maxLat: 6.5, minLng: 5.0, maxLng: 8.5 },
    38→  { label: 'Northeast', minLat: 8.0, maxLat: 14.0, minLng: 10.0, maxLng: 15.0 },
    39→];
    40→
    41→interface MapBoundsData {
    42→  minLat: number; maxLat: number; minLng: number; maxLng: number; label: string;
    43→}
    44→
    45→const VALID_ROLES: UserRole[] = ['SUPER_ADMIN', 'TENANT_ADMIN', 'ANALYST', 'TRUST_SAFETY', 'FIELD_AGENT'];
    46→
    47→const ROLE_BADGE: Record<string, string> = {
    48→  SUPER_ADMIN: 'border-emerald/30 text-emerald bg-emerald/10',
    49→  TENANT_ADMIN: 'border-cyan/30 text-cyan bg-cyan/10',
    50→  ANALYST: 'border-amber/30 text-amber bg-amber/10',
    51→  TRUST_SAFETY: 'border-rose/30 text-rose bg-rose/10',
    52→  FIELD_AGENT: 'border-border text-muted-foreground bg-card',
    53→};
    54→
    55→const ROLE_ICONS: Record<string, React.ReactNode> = {
    56→  SUPER_ADMIN: <Shield className="h-3.5 w-3.5" />,
    57→  TENANT_ADMIN: <Users className="h-3.5 w-3.5" />,
    58→  ANALYST: <Eye className="h-3.5 w-3.5" />,
    59→  TRUST_SAFETY: <UserCheck className="h-3.5 w-3.5" />,
    60→  FIELD_AGENT: <Radio className="h-3.5 w-3.5" />,
    61→};
    62→
    63→// ---- Main Component ----
    64→export function TenantManagement() {
    65→  const { user, tenantId } = useDashboardStore();
    66→  const queryClient = useQueryClient();
    67→  const isSuperAdmin = user?.role === 'SUPER_ADMIN';
    68→  const isTenantAdmin = user?.role === 'TENANT_ADMIN';
    69→  const isAdmin = isSuperAdmin || isTenantAdmin;
    70→
    71→  // ===================== SUPER_ADMIN: Platform Tenants =====================
    72→  const { data: allTenants, isLoading: tenantsLoading, isError: tenantsIsError } = useQuery({
    73→    queryKey: ['all-tenants'],
    74→    queryFn: async () => {
    75→      const d = await fetchJson<{ tenants?: unknown[] }>('/api/tenants');
    76→      return d.tenants || [];
    77→    },
    78→    enabled: isSuperAdmin,
    79→  });
    80→
    81→  // ===================== TENANT_ADMIN: Own tenant settings =====================
    82→  const { data: settings, isLoading: settingsLoading, isError: settingsIsError } = useQuery({
    83→    queryKey: ['tenant-settings', tenantId],
    84→    queryFn: async () => {
    85→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`);
    86→    },
    87→    enabled: !!tenantId && isAdmin,
    88→  });
    89→
    90→  // ===================== Common: Tenant Users =====================
    91→  const { data: tenantUsersData, isLoading: usersLoading, isError: usersIsError } = useQuery({
    92→    queryKey: ['tenant-users', tenantId],
    93→    queryFn: async () => {
    94→      const d = await fetchJson<{ users?: unknown[] }>(`/api/tenants/users?tenantId=${tenantId}`);
    95→      return d.users || [];
    96→    },
    97→    enabled: !!tenantId && isAdmin,
    98→  });
    99→
   100→  const currentBounds: MapBoundsData | null = settings?.mapBounds || null;
   101→  const tenantUsers = tenantUsersData || [];
   102→
   103→  // ===================== State =====================
   104→  // Map config dialog
   105→  const [mapConfigOpen, setMapConfigOpen] = useState(false);
   106→  const [mapLabel, setMapLabel] = useState('');
   107→  const [minLat, setMinLat] = useState('');
   108→  const [maxLat, setMaxLat] = useState('');
   109→  const [minLng, setMinLng] = useState('');
   110→  const [maxLng, setMaxLng] = useState('');
   111→
   112→  // Create tenant dialog
   113→  const [createTenantOpen, setCreateTenantOpen] = useState(false);
   114→  const [newTenantName, setNewTenantName] = useState('');
   115→  const [newTenantSlug, setNewTenantSlug] = useState('');
   116→  const [newTenantColor, setNewTenantColor] = useState('#10b981');
   117→  const [newTenantScope, setNewTenantScope] = useState('PRESIDENTIAL');
   118→  const [newAdminName, setNewAdminName] = useState('');
   119→  const [newAdminEmail, setNewAdminEmail] = useState('');
   120→
   121→  // Add user dialog
   122→  const [addUserOpen, setAddUserOpen] = useState(false);
   123→  const [newUserName, setNewUserName] = useState('');
   124→  const [newUserEmail, setNewUserEmail] = useState('');
   125→  const [newUserRole, setNewUserRole] = useState<string>('FIELD_AGENT');
   126→
   127→  // User role change dialog
   128→  const [roleChangeOpen, setRoleChangeOpen] = useState(false);
   129→  const [roleChangeUser, setRoleChangeUser] = useState<{ id: string; name: string; role: string } | null>(null);
   130→  const [newRole, setNewRole] = useState('');
   131→
   132→  // Delete confirmation
   133→  const [deleteConfirm, setDeleteConfirm] = useState<{ type: 'tenant' | 'user'; item: { id: string; name: string } } | null>(null);
   134→
   135→  // Active sub-tab for SUPER_ADMIN
   136→  const [subTab, setSubTab] = useState<'tenants' | 'my-tenant' | 'map'>('tenants');
   137→
   138→  // ===================== Effects =====================
   139→  useEffect(() => {
   140→    if (mapConfigOpen && currentBounds) {
   141→      setMapLabel(currentBounds.label || '');
   142→      setMinLat(String(currentBounds.minLat));
   143→      setMaxLat(String(currentBounds.maxLat));
   144→      setMinLng(String(currentBounds.minLng));
   145→      setMaxLng(String(currentBounds.maxLng));
   146→    } else if (mapConfigOpen) {
   147→      setMapLabel('');
   148→      setMinLat(''); setMaxLat(''); setMinLng(''); setMaxLng('');
   149→    }
   150→  }, [mapConfigOpen, currentBounds]);
   151→
   152→  // Auto-derive slug from tenant name
   153→  useEffect(() => {
   154→    if (createTenantOpen && newTenantName && !newTenantSlug) {
   155→      setNewTenantSlug(newTenantName.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, ''));
   156→    }
   157→  }, [createTenantOpen, newTenantName, newTenantSlug]);
   158→
   159→  // ===================== Mutations =====================
   160→  // Save map bounds
   161→  const saveMapMutation = useMutation({
   162→    mutationFn: async (bounds: MapBoundsData) => {
   163→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`, {
   164→        method: 'PUT',
   165→        headers: { 'Content-Type': 'application/json' },
   166→        body: JSON.stringify({ mapBounds: bounds }),
   167→      });
   168→    },
   169→    onSuccess: () => {
   170→      queryClient.invalidateQueries({ queryKey: ['tenant-settings', tenantId] });
   171→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   172→      setMapConfigOpen(false);
   173→      toast.success('Map area saved. The map will update automatically.');
   174→    },
   175→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to save map configuration'),
   176→  });
   177→
   178→  // Create tenant
   179→  const createTenantMutation = useMutation({
   180→    mutationFn: (data: { name: string; slug: string; primaryColor: string; scope?: string; adminName: string; adminEmail: string }) =>
   181→      fetchJson('/api/tenants', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify(data),
   185→      }),
   186→    onSuccess: (data) => {
   187→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   188→      setCreateTenantOpen(false);
   189→      setNewTenantName(''); setNewTenantSlug(''); setNewTenantColor('#10b981');
   190→      setNewTenantScope('PRESIDENTIAL');
   191→      setNewAdminName(''); setNewAdminEmail('');
   192→      toast.success(`Tenant "${data.tenant.name}" created. Admin: ${data.admin.email}`);
   193→    },
   194→    onError: (err) => toast.error(err?.message || 'Failed to create tenant'),
   195→  });
   196→
   197→  // Delete tenant
   198→  const deleteTenantMutation = useMutation({
   199→    mutationFn: (id: string) =>
   200→      fetchJson(`/api/tenants?id=${id}`, { method: 'DELETE' }),
   201→    onSuccess: () => {
   202→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   203→      setDeleteConfirm(null);
   204→      toast.success('Tenant deleted successfully');
   205→    },
   206→    onError: (err) => toast.error(err?.message || 'Failed to delete tenant'),
   207→  });
   208→
   209→  // Toggle tenant active
   210→  const toggleTenantMutation = useMutation({
   211→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   212→      fetchJson('/api/tenants', {
   213→        method: 'PUT',
   214→        headers: { 'Content-Type': 'application/json' },
   215→        body: JSON.stringify({ id, isActive }),
   216→      }),
   217→    onSuccess: () => {
   218→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   219→      toast.success('Tenant status updated');
   220→    },
   221→    onError: (err) => toast.error(err?.message || 'Failed to update tenant'),
   222→  });
   223→
   224→  // Add user
   225→  const addUserMutation = useMutation({
   226→    mutationFn: (data: { tenantId: string; name: string; email: string; role: string }) =>
   227→      fetchJson('/api/tenants/users', {
   228→        method: 'POST',
   229→        headers: { 'Content-Type': 'application/json' },
   230→        body: JSON.stringify(data),
   231→      }),
   232→    onSuccess: () => {
   233→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   234→      setAddUserOpen(false);
   235→      setNewUserName(''); setNewUserEmail(''); setNewUserRole('FIELD_AGENT');
   236→      toast.success('User added successfully');
   237→    },
   238→    onError: (err) => toast.error(err?.message || 'Failed to add user'),
   239→  });
   240→
   241→  // Change user role
   242→  const changeRoleMutation = useMutation({
   243→    mutationFn: ({ id, role }: { id: string; role: string }) =>
   244→      fetchJson('/api/tenants/users', {
   245→        method: 'PATCH',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({ id, role }),
   248→      }),
   249→    onSuccess: () => {
   250→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   251→      setRoleChangeOpen(false);
   252→      setRoleChangeUser(null);
   253→      toast.success('User role updated');
   254→    },
   255→    onError: (err) => toast.error(err?.message || 'Failed to update role'),
   256→  });
   257→
   258→  // Remove user
   259→  const deleteUserMutation = useMutation({
   260→    mutationFn: (id: string) =>
   261→      fetchJson(`/api/tenants/users?id=${id}`, { method: 'DELETE' }),
   262→    onSuccess: () => {
   263→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   264→      setDeleteConfirm(null);
   265→      toast.success('User removed');
   266→    },
   267→    onError: (err) => toast.error(err?.message || 'Failed to remove user'),
   268→  });
   269→
   270→  // ===================== Handlers =====================
   271→  const handleSaveMap = () => {
   272→    const minLatN = parseFloat(minLat);
   273→    const maxLatN = parseFloat(maxLat);
   274→    const minLngN = parseFloat(minLng);
   275→    const maxLngN = parseFloat(maxLng);
   276→    if ([minLatN, maxLatN, minLngN, maxLngN].some(isNaN)) { toast.error('All coordinates must be valid numbers'); return; }
   277→    if (minLatN >= maxLatN || minLngN >= maxLngN) { toast.error('minLat must be < maxLat, and minLng must be < maxLng'); return; }
   278→    saveMapMutation.mutate({ minLat: minLatN, maxLat: maxLatN, minLng: minLngN, maxLng: maxLngN, label: mapLabel || 'Custom Area' });
   279→  };
   280→
   281→  const handlePreset = (preset: typeof REGION_PRESETS[number]) => {
   282→    setMapLabel(preset.label);
   283→    setMinLat(String(preset.minLat));
   284→    setMaxLat(String(preset.maxLat));
   285→    setMinLng(String(preset.minLng));
   286→    setMaxLng(String(preset.maxLng));
   287→  };
   288→
   289→  const handleResetMap = () => {
   290→    saveMapMutation.mutate({ minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0, label: 'Nigeria' });
   291→  };
   292→
   293→  const handleCreateTenant = () => {
   294→    if (!newTenantName || !newTenantSlug || !newAdminName || !newAdminEmail) {
   295→      toast.error('All fields are required'); return;
   296→    }
   297→    createTenantMutation.mutate({
   298→      name: newTenantName, slug: newTenantSlug, primaryColor: newTenantColor,
   299→      scope: newTenantScope,
   300→      adminName: newAdminName, adminEmail: newAdminEmail,
   301→    });
   302→  };
   303→
   304→  const handleAddUser = () => {
   305→    if (!newUserName || !newUserEmail || !newUserRole) { toast.error('All fields are required'); return; }
   306→    addUserMutation.mutate({ tenantId: tenantId!, name: newUserName, email: newUserEmail, role: newUserRole });
   307→  };
   308→
   309→  const handleRoleChange = () => {
   310→    if (!roleChangeUser || !newRole) return;
   311→    changeRoleMutation.mutate({ id: roleChangeUser.id, role: newRole });
   312→  };
   313→
   314→  // ===================== Restricted access =====================
   315→  if (!isAdmin) {
   316→    return (
   317→      <div className="h-full flex items-center justify-center p-6">
   318→        <div className="text-center space-y-2">
   319→          <Shield className="h-8 w-8 text-muted-foreground/30 mx-auto" />
   320→          <p className="text-sm text-muted-foreground">Access restricted to Super Admin and Tenant Admin</p>
   321→        </div>
   322→      </div>
   323→    );
   324→  }
   325→
   326→  const isLoading = isSuperAdmin ? tenantsLoading : settingsLoading;
   327→  const hasError = isSuperAdmin ? tenantsIsError : settingsIsError;
   328→
   329→  // ===================== Render =====================
   330→  return (
   331→    <div className="h-full flex flex-col overflow-hidden">
   332→      {/* Header */}
   333→      <div className="px-4 sm:px-6 pt-4 pb-3 shrink-0">
   334→        <div className="flex items-center justify-between flex-wrap gap-3">
   335→          <div>
   336→            <h2 className="text-lg font-semibold flex items-center gap-2">
   337→              <Building2 className="h-5 w-5 text-emerald" />
   338→              {isSuperAdmin ? 'Platform Management' : 'Organization Settings'}
   339→            </h2>
   340→            <p className="text-sm text-muted-foreground mt-0.5">
   341→              {isSuperAdmin
   342→                ? 'Create and manage tenant organizations, their users, and configurations'
   343→                : 'Manage your organization settings, users, and map configuration'}
   344→            </p>
   345→          </div>
   346→          {isSuperAdmin && (
   347→            <Button
   348→              size="sm"
   349→              className="gap-1.5 text-xs bg-emerald hover:bg-emerald/90 text-emerald-950 h-8"
   350→              onClick={() => setCreateTenantOpen(true)}
   351→            >
   352→              <Plus className="h-3.5 w-3.5" /> New Tenant
   353→            </Button>
   354→          )}
   355→        </div>
   356→
   357→        {/* Sub-tabs for SUPER_ADMIN */}
   358→        {isSuperAdmin && (
   359→          <div className="flex items-center gap-1 mt-3 bg-muted/30 rounded-lg p-0.5 w-fit">
   360→            {([
   361→              { id: 'tenants' as const, label: 'All Tenants', icon: <Globe className="h-3.5 w-3.5" /> },
   362→              { id: 'my-tenant' as const, label: 'My Organization', icon: <Building2 className="h-3.5 w-3.5" /> },
   363→              { id: 'map' as const, label: 'Map Config', icon: <MapPin className="h-3.5 w-3.5" /> },
   364→            ]).map(tab => (
   365→              <button
   366→                key={tab.id}
   367→                onClick={() => setSubTab(tab.id)}
   368→                className={cn(
   369→                  'flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium transition-colors',
   370→                  subTab === tab.id ? 'bg-card shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   371→                )}
   372→              >
   373→                {tab.icon} {tab.label}
   374→              </button>
   375→            ))}
   376→          </div>
   377→        )}
   378→      </div>
   379→
   380→      {/* Scrollable content */}
   381→      <div className="flex-1 overflow-y-auto px-4 sm:px-6 pb-6 space-y-4 min-h-0">
   382→
   383→        {isLoading ? (
   384→          <div className="flex-1 flex items-center justify-center py-20">
   385→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   386→          </div>
   387→        ) : hasError ? (
   388→          <div className="flex-1 flex flex-col items-center justify-center py-20 text-center p-6">
   389→            <AlertCircle className="h-10 w-10 text-destructive mb-3" />
   390→            <p className="text-sm text-muted-foreground">Failed to load data. Please try again.</p>
   391→            <Button variant="outline" size="sm" className="mt-3" onClick={() => window.location.reload()}>
   392→              Retry
   393→            </Button>
   394→          </div>
   395→        ) : isSuperAdmin ? (
   396→          /* ========== SUPER_ADMIN VIEWS ========== */
   397→          <>
   398→            {subTab === 'tenants' && (
   399→              <SuperAdminTenantsView
   400→                tenants={allTenants || []}
   401→                onToggle={(id, isActive) => toggleTenantMutation.mutate({ id, isActive })}
   402→                onDelete={(item) => setDeleteConfirm({ type: 'tenant', item })}
   403→                onManageUsers={() => setSubTab('my-tenant')}
   404→              />
   405→            )}
   406→            {subTab === 'my-tenant' && (
   407→              <TenantUsersView
   408→                settings={settings}
   409→                tenantUsers={tenantUsers}
   410→                userRole={user?.role || ''}
   411→                onOpenUserDialog={() => setAddUserOpen(true)}
   412→                onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   413→                onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   414→                onOpenMapConfig={() => { setSubTab('map'); setTimeout(() => setMapConfigOpen(true), 100); }}
   415→              />
   416→            )}
   417→            {subTab === 'map' && (
   418→              <MapConfigSection
   419→                currentBounds={currentBounds}
   420→                onConfigure={() => setMapConfigOpen(true)}
   421→              />
   422→            )}
   423→          </>
   424→        ) : (
   425→          /* ========== TENANT_ADMIN VIEW ========== */
   426→          <>
   427→            <TenantUsersView
   428→              settings={settings}
   429→              tenantUsers={tenantUsers}
   430→              userRole={user?.role || ''}
   431→              onOpenUserDialog={() => setAddUserOpen(true)}
   432→              onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   433→              onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   434→              onOpenMapConfig={() => setMapConfigOpen(true)}
   435→            />
   436→            <MapConfigSection
   437→              currentBounds={currentBounds}
   438→              onConfigure={() => setMapConfigOpen(true)}
   439→            />
   440→            <InfoNotices />
   441→          </>
   442→        )}
   443→      </div>
   444→
   445→      {/* ===== MAP CONFIG DIALOG ===== */}
   446→      <Dialog open={mapConfigOpen} onOpenChange={setMapConfigOpen}>
   447→        <DialogContent className="sm:max-w-lg">
   448→          <DialogHeader>
   449→            <DialogTitle className="flex items-center gap-2">
   450→              <MapPin className="h-5 w-5 text-cyan" />
   451→              Configure Map Area
   452→            </DialogTitle>
   453→            <DialogDescription>
   454→              Set the geographic bounding box for the Polling Unit Map. Use a preset or enter custom WGS84 coordinates.
   455→            </DialogDescription>
   456→          </DialogHeader>
   457→          <div className="space-y-4 py-2">
   458→            <div className="space-y-1.5">
   459→              <label className="text-xs font-medium text-muted-foreground">Area Label</label>
   460→              <Input placeholder="e.g. Lagos, Southeast, Kano Central" value={mapLabel} onChange={(e) => setMapLabel(e.target.value)} className="h-9 text-sm" />
   461→            </div>
   462→            <div className="space-y-1.5">
   463→              <label className="text-xs font-medium text-muted-foreground">Quick Preset</label>
   464→              <Select onValueChange={(v) => { const preset = REGION_PRESETS.find(p => p.label === v); if (preset) handlePreset(preset); }}>
   465→                <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select a region preset..." /></SelectTrigger>
   466→                <SelectContent>
