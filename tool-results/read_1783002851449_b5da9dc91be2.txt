   226→function generateMediaUrls(type: string, severity: string): string {
   227→  const urls: string[] = [];
   228→  // Most incidents get 1-3 images
   229→  const numImages = rand(1, 3);
   230→  urls.push(...pickN(SAMPLE_IMAGES, numImages));
   231→
   232→  // Medium+ severity may have video
   233→  if (['MEDIUM', 'HIGH', 'CRITICAL'].includes(severity) && Math.random() > 0.4) {
   234→    urls.push(pick(SAMPLE_VIDEOS));
   235→  }
   236→
   237→  // All incidents get a voice note (field agents often send voice reports)
   238→  if (Math.random() > 0.3) {
   239→    urls.push(pick(SAMPLE_VOICE));
   240→  }
   241→  // Security/tech incidents more likely to have video
   242→  if (['DEEPFAKE_SUSPECT', 'CIB_DETECTED', 'BALLOT_STUFFING', 'VIOLENCE'].includes(type)) {
   243→    urls.push(pick(SAMPLE_VIDEOS));
   244→    urls.push(...pickN(SAMPLE_IMAGES, rand(1, 2)));
   245→  }
   246→
   247→  return JSON.stringify([...new Set(urls)]);
   248→}
   249→
   250→function generatePartyResults(parties: { party: string; name: string; votes: number; color: string }[], validVotes: number) {
   251→  const shuffled = parties.map(p => ({ ...p, votes: rand(5, Math.floor(validVotes * 0.4)) }));
   252→  const total = shuffled.reduce((s, p) => s + p.votes, 0);
   253→  return shuffled.map(p => ({ ...p, votes: Math.round(p.votes / total * validVotes) }))
   254→    .sort((a, b) => b.votes - a.votes);
   255→}

   257→function getDescForType(type: string): string {
   258→  const group = RICH_DESCRIPTIONS.find(g => g.type === type);
   259→  return group ? pick(group.descs) : pick(RICH_DESCRIPTIONS[0].descs);
   260→}

   262→// ─── Seed Functions per Tenant ──────────────────────────────────────
   263→
   264→async function seedTenantData(
   265→  tenantId: string,
   266→  parties: { party: string; name: string; votes: number; color: string }[],
   267→  config: { resultsTarget: number; incidentsTarget: number; messagesTarget: number }
   268→) {
   269→  const agents = await db.user.findMany({
   270→    where: { tenantId, role: 'FIELD_AGENT' },
   271→    select: { id: true, name: true, email: true },
   272→  });
   273→  const admins = await db.user.findMany({
   274→    where: { tenantId, role: { in: ['SUPER_ADMIN', 'TENANT_ADMIN', 'ANALYST'] } },
   275→    select: { id: true, name: true },
   276→  });
   277→  const pus = await db.pollingUnit.findMany({
   278→    where: { election: { tenantId } },
   279→    select: { id: true, name: true, code: true, state: true, lga: true, ward: true, latitude: true, longitude: true, registeredVoters: true },
   280→  });
   281→
   282→  if (agents.length === 0 || pus.length === 0) {
   283→    console.log(`    Skipping ${tenantId} — no agents (${agents.length}) or PUs (${pus.length})`);
   284→    return;
   285→  }
   286→
   287→  // Check existing counts
   288→  const existingResults = await db.electionResult.count({ where: { tenantId } });
   289→  const existingIncidents = await db.incident.count({ where: { tenantId } });
   290→  const existingMessages = await db.agentMessage.count({ where: { tenantId } });
   291→
   292→  console.log(`    Tenant ${tenantId}: ${agents.length} agents, ${pus.length} PUs`);
   293→  console.log(`      Existing: ${existingResults} results, ${existingIncidents} incidents, ${existingMessages} messages`);
   294→
   295→  // ─── 1. ELECTION RESULTS ───────────────────────────────────────
   296→  const resultsToCreate = Math.max(0, config.resultsTarget - existingResults);
   297→  console.log(`      Creating ${resultsToCreate} results...`);
   298→
   299→  for (let i = 0; i < resultsToCreate; i++) {
   300→    const pu = pick(pus);
   301→    const agent = pick(agents);
   302→    const reg = pu.registeredVoters || rand(200, 1200);
   303→    const turnout = Math.random() * 0.45 + 0.2;
   304→    const accredited = Math.floor(reg * turnout);
   305→    const validVotes = rand(Math.floor(accredited * 0.7), accredited);
   306→    const rejected = rand(2, Math.floor(validVotes * 0.08));
   307→    const totalCast = validVotes + rejected;
   308→
   309→    const partyResults = generatePartyResults(parties, validVotes);
   310→    const notes = pick(RESULT_NOTES);
   311→    const hasIncident = Math.random() > 0.75;
   312→
   313→    await db.electionResult.create({
   314→      data: {
   315→        tenantId, pollingUnitId: pu.id, reportedById: agent.id,
   316→        accreditedVoters: accredited,
   317→        totalValidVotes: validVotes,
   318→        rejectedBallots: rejected,
   319→        totalVotesCast: totalCast,
   320→        partyResults: JSON.stringify(partyResults),
   321→        bvasUsed: Math.random() > 0.08,
   322→        materialsArrivedOnTime: Math.random() > 0.15,
   323→        securityPresent: Math.random() > 0.1,
   324→        violenceOccurred: hasIncident,
   325→        notes: `${notes} — Reported by ${agent.name}`,
   326→        verified: Math.random() > 0.6,
   327→        verifiedById: Math.random() > 0.6 ? pick(admins).id : null,
   328→        submittedAt: new Date(Date.now() - rand(120000, 14400000)),
   329→      },
   330→    });
   331→
   332→    // Auto-close the PU
   333→    await db.pollingUnit.update({
   334→      where: { id: pu.id },
   335→      data: { status: pick(['CLOSED', 'CLOSED', 'CLOSED', 'FLAGGED']), totalVotes: totalCast, turnout: Math.round(turnout * 100) / 100 },
   336→    });
   337→  }
   338→
   339→  // ─── 2. INCIDENTS WITH MEDIA ──────────────────────────────────
   340→  const incidentsToCreate = Math.max(0, config.incidentsTarget - existingIncidents);
   341→  console.log(`      Creating ${incidentsToCreate} incidents...`);
   342→
   343→  const incidentTypes = RICH_DESCRIPTIONS.map(r => r.type);
   344→
   345→  for (let i = 0; i < incidentsToCreate; i++) {
   346→    const agent = pick(agents);
   347→    const pu = pick(pus);
   348→    const type = pick(incidentTypes);
   349→    const isSecurity = ['DEEPFAKE_SUSPECT', 'CIB_DETECTED', 'GEO_ANOMALY'].includes(type);
   350→    const isSevere = ['VIOLENCE', 'SNATCHED_BALLOT', 'BALLOT_STUFFING'].includes(type);
   351→
   352→    const severity = isSevere ? pick(['HIGH', 'CRITICAL'])
   353→      : isSecurity ? pick(['HIGH', 'CRITICAL'])
   354→      : pick(['LOW', 'LOW', 'MEDIUM', 'MEDIUM', 'HIGH']);
   355→
   356→    const status = isSecurity ? 'QUARANTINED'
   357→      : severity === 'CRITICAL' ? pick(['ESCALATED', 'PENDING'])
   358→      : pick(['PENDING', 'PENDING', 'REVIEWED', 'REVIEWED', 'ESCALATED', 'DISMISSED']);
   359→
   360→    const desc = getDescForType(type);
   361→    const mediaUrls = generateMediaUrls(type, severity);
   362→    const gpsAnomaly = type === 'GEO_ANOMALY' ? true : Math.random() > 0.92;
   363→
   364→    const inc = await db.incident.create({
   365→      data: {
   366→        tenantId,
   367→        pollingUnitId: pu.id,
   368→        reportedById: agent.id,
   369→        type,
   370→        severity,
   371→        status,
   372→        description: desc,
   373→        mediaUrls,
   374→        gpsLatitude: gpsAnomaly ? pu.latitude + (Math.random() - 0.5) * 0.15 : pu.latitude + (Math.random() - 0.5) * 0.02,
   375→        gpsLongitude: gpsAnomaly ? pu.longitude + (Math.random() - 0.5) * 0.15 : pu.longitude + (Math.random() - 0.5) * 0.02,
   376→        gpsAnomaly,
   377→        aiSummary: isSecurity
   378→          ? `AI ANALYSIS: ${type.replace(/_/g, ' ')} detected. Confidence: ${rand(85, 99)}%. Pattern matches known threat signatures. Automated quarantine applied.`

   380→        aiFlags: JSON.stringify(isSecurity

   392→
   393→    // Create alert for significant incidents
   394→    if (severity === 'HIGH' || severity === 'CRITICAL' || isSevere || isSecurity) {

   457→      : trigger === 'INCIDENT_FOLLOWUP' ? pick(['HIGH', 'URGENT'])
   458→      : trigger === 'INFRACTION_REMINDER' ? pick(['NORMAL', 'HIGH'])

   520→  // Set some agents as truly offline (no data for hours)
   521→  const offlineAgents = pickN(agents.filter(a => !idleAgents.find(ia => ia.id === a.id)), Math.max(1, Math.floor(agents.length * 0.15)));
   522→  for (const agent of offlineAgents) {
   523→    await db.user.update({

   525→      data: {
   526→        isOnline: false,
   527→        lastSeenAt: new Date(Date.now() - rand(7200000, 28800000)), // 2 to 8 hours ago
   528→      },
   529→    });
   530→  }
