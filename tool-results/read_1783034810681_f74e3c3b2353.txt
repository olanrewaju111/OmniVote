   128→// ── Helpers ──────────────────────────────────────────────────────────
   129→function formatRelativeTime(d: string | Date | null): string {
   130→  if (!d) return 'Never';
   131→  const date = new Date(d);
   132→  const now = Date.now();
   133→  const diff = Math.floor((now - date.getTime()) / 60000);
   134→  if (diff < 1) return 'Just now';
   135→  if (diff < 60) return `${diff}m ago`;
   136→  if (diff < 1440) return `${Math.floor(diff / 60)}h ago`;
   137→  return date.toLocaleDateString('en-NG', { day: '2-digit', month: 'short' });
   138→}
   139→
   140→function formatDeadlineCountdown(deadline: string): { text: string; isOverdue: boolean } {
   141→  const diff = new Date(deadline).getTime() - Date.now();
   142→  const mins = Math.floor(Math.abs(diff) / 60000);
   143→  if (diff > 0) {
   144→    return { text: `${mins}m left`, isOverdue: false };
   145→  }
   146→  return { text: `OVERDUE by ${mins}m`, isOverdue: true };
   147→}

   149→function batteryIcon(level: number | null) {
   150→  if (level === null) return <Battery className="h-3.5 w-3.5 text-muted-foreground" />;
   151→  if (level > 60) return <BatteryFull className="h-3.5 w-3.5 text-emerald" />;
   152→  if (level > 20) return <BatteryWarning className="h-3.5 w-3.5 text-amber" />;
   153→  return <BatteryLow className="h-3.5 w-3.5 text-rose" />;
   154→}

   157→  if (level === null) return 'text-muted-foreground';
   158→  if (level > 60) return 'text-emerald';
   159→  if (level > 20) return 'text-amber';
   160→  return 'text-rose';
   161→}

   163→function statusBadge(status: string) {
   164→  switch (status) {
   165→    case 'CHECKED_IN':
   166→      return <Badge className="border-emerald/30 bg-emerald/10 text-emerald text-[10px]">CHECKED IN</Badge>;
   167→    case 'CHECKED_OUT':
   168→      return <Badge className="border-cyan/30 bg-cyan/10 text-cyan text-[10px]">CHECKED OUT</Badge>;
   169→    case 'SOS_TRIGGERED':
   170→      return <Badge className="border-rose/30 bg-rose/10 text-rose text-[10px] glow-rose animate-pulse-dot">SOS</Badge>;
   171→    case 'EXPIRED':
   172→      return <Badge className="border-amber/30 bg-amber/10 text-amber text-[10px]">EXPIRED</Badge>;
   173→    default:
   174→      return <Badge variant="outline" className="text-[10px]">{status}</Badge>;
   175→  }
   176→}

   755→// ── KPI Card Sub-component ───────────────────────────────────────────
   756→function KpiCard({ label, value, icon, accent, delay }: {
   757→  label: string; value: number; icon: React.ReactNode; accent: 'emerald' | 'rose' | 'amber' | 'cyan'; delay: number;
   758→}) {
   759→  const glowClass = accent === 'rose' ? 'glow-rose' : accent === 'amber' ? 'glow-amber' : accent === 'emerald' ? 'glow-emerald' : '';
   760→  return (
   761→    <motion.div variants={item} transition={{ delay: delay * 0.05 }}>
   762→      <Card className={cn('bg-card/40 border-border rounded-xl', glowClass, value > 0 && (accent === 'rose' || accent === 'amber') ? '' : '')}>
   763→        <CardContent className="p-4 flex items-center gap-3">
   764→          <div className={cn(
   765→            'flex items-center justify-center h-9 w-9 rounded-lg shrink-0',
   766→            accent === 'emerald' && 'bg-emerald/10',
   767→            accent === 'rose' && 'bg-rose/10',
   768→            accent === 'amber' && 'bg-amber/10',
   769→            accent === 'cyan' && 'bg-cyan/10',
   770→          )}>
   771→            {icon}
   772→          </div>
   773→          <div>
   774→            <div className="text-xl font-bold leading-none">{value}</div>
   775→            <div className="text-[11px] text-muted-foreground mt-0.5">{label}</div>
   776→          </div>
   777→        </CardContent>
   778→      </Card>
   779→    </motion.div>
   780→  );
