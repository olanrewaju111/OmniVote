   138→
   139→const RISK_CELL_COLORS: Record<string, { bg: string; text: string }> = {
   140→  LOW: { bg: 'rgba(16,185,129,0.20)', text: '#10b981' },
   141→  MEDIUM: { bg: 'rgba(245,158,11,0.30)', text: '#f59e0b' },
   142→  HIGH: { bg: 'rgba(239,68,68,0.30)', text: '#ef4444' },
   143→  CRITICAL: { bg: 'rgba(220,38,38,0.50)', text: '#dc2626' },
   144→};
   145→
   146→function riskCellColor(level: string) {
   147→  return RISK_CELL_COLORS[level] ?? RISK_CELL_COLORS.LOW;
   148→}
   149→
   150→function riskBarColor(score: number): string {
   151→  if (score >= 0.8) return '#ef4444';
   152→  if (score >= 0.6) return '#f97316';
   153→  if (score >= 0.3) return '#f59e0b';
   154→  return '#10b981';
   155→}
   156→
   157→function riskBadgeVariant(level: string): string {
   158→  switch (level) {
   159→    case 'CRITICAL':
   160→      return 'bg-rose-600/20 text-rose-400 border-rose-600/30';
   161→    case 'HIGH':
   162→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   163→    case 'MEDIUM':
   164→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   165→    default:
   166→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   167→  }
   168→}
   169→
   170→function statusBadgeVariant(status: string): string {
   171→  switch (status) {
   172→    case 'RUNNING':
   173→      return 'bg-cyan-500/20 text-cyan-400 border-cyan-500/30';
   174→    case 'COMPLETED':
   175→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   176→    case 'PENDING':
   177→      return 'bg-amber-500/20 text-amber-400 border-amber-500/30';
   178→    case 'FAILED':
   179→      return 'bg-rose-500/20 text-rose-400 border-rose-500/30';
   180→    default:
   181→      return 'bg-muted text-muted-foreground border-border';
   182→  }
   183→}
   184→
   185→function formatDate(dateStr: string): string {
   186→  const d = new Date(dateStr);
   187→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   188→}
   189→
   190→function formatDay(dateStr: string): string {
   191→  const d = new Date(dateStr);
   192→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
   193→}

   212→
   213→export function FlashpointWargame() {
   214→  const { tenantId } = useDashboardStore();
   215→  const queryClient = useQueryClient();
   216→
   217→  // Wargame state
   218→  const [selectedScenarioId, setSelectedScenarioId] = useState<string | null>(null);
   219→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   220→  const [newTitle, setNewTitle] = useState('');
   221→  const [newDescription, setNewDescription] = useState('');
   222→  const [newRole, setNewRole] = useState('RED_TEAM');
   223→  const [newParams, setNewParams] = useState('{\n  \n}');
   224→  const [completeScore, setCompleteScore] = useState('');
   225→
   226→  /* ---- Data fetching ---- */
   227→  const { data, isLoading, isError, refetch } = useQuery<FlashpointData>({
   228→    queryKey: ['flashpoint', tenantId],
   229→    queryFn: () =>
   230→      fetchJson(`/api/flashpoint?tenantId=${tenantId}`),
   231→    refetchInterval: 120_000, // SSE handles real-time; 2-min safety-net fallback
   232→  });
   233→
   234→  const mutation = useMutation({
   235→    mutationFn: (body: Record<string, unknown>) =>
   236→      fetchJson('/api/flashpoint?tenantId=' + tenantId, {
   237→        method: 'POST',
   238→        headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify(body),
   240→      }),
   241→    onSuccess: () => {
   242→      queryClient.invalidateQueries({ queryKey: ['flashpoint', tenantId] });
   243→    },
   244→    onError: () => {
   245→      toast.error('Flashpoint operation failed. Please try again.');
   246→    },
   247→  });
   248→
   249→  /* ---- Derived data ---- */
   250→  const stats = data?.stats;
   251→  const forecasts = data?.forecasts ?? [];
   252→  const scenarios = data?.scenarios ?? [];
   253→  const heatmapData = data?.heatmapData ?? [];
   254→  const topFactors = data?.topContributingFactors ?? [];
   255→
   256→  // Unique states sorted alphabetically
   257→  const uniqueStates = useMemo(() => {
   258→    const set = new Set(heatmapData.map((d) => d.state));
   259→    return Array.from(set).sort();
   260→  }, [heatmapData]);
   261→
   262→  // Unique dates sorted
   263→  const uniqueDates = useMemo(() => {
   264→    const set = new Set(heatmapData.map((d) => d.date));
   265→    return Array.from(set).sort();
   266→  }, [heatmapData]);
   267→
   268→  // Build lookup map for heatmap cells
   269→  const heatmapMap = useMemo(() => {
   270→    const map = new Map<string, (typeof heatmapData)[number]>();
   271→    for (const d of heatmapData) {
   272→      map.set(`${d.state}|${d.date}`, d);
   273→    }
   274→    return map;
   275→  }, [heatmapData]);
   276→
   277→  // State risk averages for bar chart
   278→  const stateRisks = useMemo(() => {
   279→    const map = new Map<string, { overall: number; level: string; count: number }>();
   280→    for (const f of forecasts) {
   281→      const existing = map.get(f.state);
   282→      if (existing) {
   283→        existing.overall = (existing.overall * existing.count + f.riskScores.overall) / (existing.count + 1);
   284→        existing.count += 1;
   285→        // Upgrade risk level if higher
   286→        const order = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
   287→        if (order.indexOf(f.riskLevel) > order.indexOf(existing.level)) {
   288→          existing.level = f.riskLevel;
   289→        }
   290→      } else {
   291→        map.set(f.state, { overall: f.riskScores.overall, level: f.riskLevel, count: 1 });
   292→      }
   293→    }
   294→    return Array.from(map.entries())
   295→      .map(([state, v]) => ({ state, ...v }))
   296→      .sort((a, b) => b.overall - a.overall);
   297→  }, [forecasts]);
   298→
   299→  // Top factor
   300→  const topFactor = topFactors.length > 0 ? topFactors[0].factor : '—';
   301→
   302→  // Selected scenario
   303→  const selectedScenario = scenarios.find((s) => s.id === selectedScenarioId) ?? null;
   304→
   305→  // Max factor count for bar scaling
   306→  const maxFactorCount = Math.max(...topFactors.map((f) => f.count), 1);
   307→
   308→  /* ---- Handlers ---- */
   309→  const handleCreateScenario = () => {
   310→    if (!newTitle.trim()) {
   311→      toast.error('Title is required');
   312→      return;
   313→    }
   314→    let parsedParams: Record<string, unknown> = {};
   315→    try {
   316→      parsedParams = newParams.trim() ? JSON.parse(newParams) : {};
   317→    } catch {
   318→      toast.error('Invalid JSON in parameters');
   319→      return;
   320→    }
   321→    mutation.mutate(
   322→      { action: 'RUN_WARGAME', title: newTitle, description: newDescription, playerRole: newRole, parameters: parsedParams },
   323→      {
   324→        onSuccess: () => {
   325→          toast.success('Scenario created successfully');
   326→          setCreateDialogOpen(false);
   327→          setNewTitle('');
   328→          setNewDescription('');
   329→          setNewRole('RED_TEAM');
   330→          setNewParams('{\n  \n}');
   331→        },
   332→        onError: () => toast.error('Failed to create scenario'),
   333→      },
   334→    );
   335→  };
   336→
   337→  const handleCompleteScenario = () => {
   338→    if (!selectedScenarioId || !completeScore) {
   339→      toast.error('Score is required');
   340→      return;
   341→    }
   342→    const score = parseFloat(completeScore);
   343→    if (isNaN(score) || score < 0 || score > 100) {
   344→      toast.error('Score must be between 0 and 100');
   345→      return;
   346→    }
   347→    mutation.mutate(
   348→      { action: 'COMPLETE_WARGAME', scenarioId: selectedScenarioId, score },
   349→      {
   350→        onSuccess: () => {
   351→          toast.success('Scenario completed');
   352→          setCompleteScore('');
   353→          setSelectedScenarioId(null);
   354→        },
   355→        onError: () => toast.error('Failed to complete scenario'),
   356→      },
   357→    );
   358→  };
   359→
   360→  /* ---- Loading ---- */
   361→  if (isLoading) {
   362→    return (
   363→      <div className="h-full flex flex-col items-center justify-center gap-3">
   364→        <Loader2 className="size-8 text-emerald-500 animate-spin" />
   365→        <p className="text-sm text-muted-foreground">Loading flashpoint data…</p>
   366→      </div>
   367→    );
   368→  }
   369→
   370→  if (isError) {
   371→    return (
   372→      <div className="flex flex-col items-center justify-center h-full min-h-[200px] text-center p-6">
   373→        <AlertCircle className="h-10 w-10 text-destructive mb-3" />
   374→        <p className="text-sm text-muted-foreground">Failed to load data. Please try again.</p>
   375→        <Button variant="outline" size="sm" className="mt-3" onClick={() => window.location.reload()}>
   376→          Retry
   377→        </Button>
   378→      </div>
   379→    );
   380→  }
   381→
   382→  return (
   383→    <div className="h-full flex flex-col gap-4">
   384→      <Tabs defaultValue="forecast" className="h-full flex flex-col gap-3">
   385→        <TabsList>
   386→          <TabsTrigger value="forecast" className="gap-1.5">
   387→            <Flame className="size-4" />
   388→            Risk Forecast
   389→          </TabsTrigger>
   390→          <TabsTrigger value="wargame" className="gap-1.5">
   391→            <Swords className="size-4" />
   392→            Wargame Simulator
   393→          </TabsTrigger>
   394→        </TabsList>
   395→
   396→        {/* ================================================================ */}
   397→        {/* TAB 1 — Risk Forecast                                           */}
   398→        {/* ================================================================ */}
   399→        <TabsContent value="forecast" className="flex-1 min-h-0 flex flex-col">
   400→          <motion.div
   401→            variants={staggerContainer}
   402→            initial="initial"
   403→            animate="animate"
   404→            className="flex flex-col gap-4 flex-1 min-h-0"
   405→          >
   406→            {/* --- Stats Row --- */}
   407→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   408→              <motion.div variants={fadeIn}>
   409→                <Card className="h-full">
   410→                  <CardContent className="p-4 flex flex-col gap-1">
   411→                    <div className="flex items-center gap-2 text-muted-foreground">
   412→                      <Activity className="size-4" />
   413→                      <span className="text-xs font-medium">Total Forecasts</span>
   414→                    </div>
   415→                    <p className="text-2xl font-bold">{stats?.totalForecasts ?? 0}</p>
   416→                  </CardContent>
   417→                </Card>
   418→              </motion.div>
   419→
   420→              <motion.div variants={fadeIn}>
   421→                <Card className="h-full border-red-500/20">
   422→                  <CardContent className="p-4 flex flex-col gap-1">
   423→                    <div className="flex items-center gap-2 text-red-400">
   424→                      <AlertTriangle className="size-4" />
   425→                      <span className="text-xs font-medium">High Risk States</span>
   426→                    </div>
   427→                    <p className="text-2xl font-bold text-red-400">{stats?.highRiskStates?.length ?? 0}</p>
   428→                  </CardContent>
   429→                </Card>
   430→              </motion.div>
   431→
   432→              <motion.div variants={fadeIn}>
   433→                <Card className="h-full border-rose-500/20">
   434→                  <CardContent className="p-4 flex flex-col gap-1">
   435→                    <div className="flex items-center gap-2 text-rose-400">
   436→                      <ShieldAlert className="size-4" />
   437→                      <span className="text-xs font-medium">Critical</span>
   438→                    </div>
   439→                    <p className="text-2xl font-bold text-rose-400">{stats?.byRiskLevel?.CRITICAL ?? 0}</p>
   440→                  </CardContent>
   441→                </Card>
   442→              </motion.div>
   443→
   444→              <motion.div variants={fadeIn}>
   445→                <Card className="h-full">
   446→                  <CardContent className="p-4 flex flex-col gap-1">
   447→                    <div className="flex items-center gap-2 text-cyan-400">
   448→                      <TrendingUp className="size-4" />
   449→                      <span className="text-xs font-medium">Avg Confidence</span>
   450→                    </div>
   451→                    <p className="text-2xl font-bold text-cyan-400">
   452→                      {stats?.avgConfidence != null ? `${(stats.avgConfidence * 100).toFixed(0)}%` : '—'}
   453→                    </p>
   454→                  </CardContent>
   455→                </Card>
   456→              </motion.div>
   457→
   458→              <motion.div variants={fadeIn}>
   459→                <Card className="h-full">
   460→                  <CardContent className="p-4 flex flex-col gap-1">
   461→                    <div className="flex items-center gap-2 text-amber-400">
   462→                      <Zap className="size-4" />
   463→                      <span className="text-xs font-medium">Top Factor</span>
   464→                    </div>
   465→                    <p className="text-sm font-semibold truncate">{topFactor}</p>
   466→                  </CardContent>
   467→                </Card>
   468→              </motion.div>
   469→            </div>
   470→
   471→            {/* --- Main Two Column Layout --- */}
   472→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 min-h-0">
   473→              {/* LEFT: 7-Day Heatmap */}
   474→              <motion.div variants={fadeIn}>
   475→                <Card className="h-full flex flex-col">
   476→                  <CardContent className="p-4 flex flex-col gap-3 flex-1 min-h-0">
   477→                    <div className="flex items-center justify-between">
   478→                      <h3 className="text-sm font-semibold flex items-center gap-2">
   479→                        <CalendarDays className="size-4 text-emerald-400" />
   480→                        7-Day Risk Heatmap
   481→                      </h3>
   482→                      <Button variant="ghost" size="sm" onClick={() => refetch()} className="h-7 px-2 text-xs">
   483→                        <Crosshair className="size-3 mr-1" />
   484→                        Refresh
   485→                      </Button>
   486→                    </div>
   487→
   488→                    {uniqueStates.length === 0 || uniqueDates.length === 0 ? (
   489→                      <div className="flex-1 flex items-center justify-center text-muted-foreground text-sm">
   490→                        No forecast data available
   491→                      </div>
   492→                    ) : (
   493→                      <ScrollArea className="flex-1">
   494→                        <div className="min-w-fit">
   495→                          {/* Header row: empty + dates */}
   496→                          <div
   497→                            className="grid gap-[2px] mb-[2px]"
   498→                            style={{
   499→                              gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   500→                            }}
   501→                          >
   502→                            <div className="text-[10px] text-muted-foreground font-medium px-1" />
   503→                            {uniqueDates.map((date) => (
   504→                              <div
   505→                                key={date}
   506→                                className="text-[10px] text-muted-foreground font-medium text-center truncate px-1"
   507→                              >
   508→                                {formatDay(date)}
   509→                              </div>
   510→                            ))}
   511→                          </div>
   512→
   513→                          {/* Data rows: state + cells */}
   514→                          {uniqueStates.map((state) => (
   515→                            <div
   516→                              key={state}
   517→                              className="grid gap-[2px] mb-[2px]"
   518→                              style={{
   519→                                gridTemplateColumns: `140px repeat(${uniqueDates.length}, minmax(52px, 1fr))`,
   520→                              }}
   521→                            >
   522→                              <div className="flex items-center text-xs font-medium truncate px-1 text-foreground/80">
   523→                                {state}
   524→                              </div>
   525→                              {uniqueDates.map((date) => {
   526→                                const cell = heatmapMap.get(`${state}|${date}`);
   527→                                const colors = cell ? riskCellColor(cell.riskLevel) : { bg: 'rgba(255,255,255,0.05)', text: '#666' };
   528→                                return (
   529→                                  <TooltipProvider key={`${state}|${date}`}>
   530→                                    <Tooltip>
   531→                                      <TooltipTrigger asChild>
   532→                                        <div
   533→                                          className="flex items-center justify-center rounded-sm text-[11px] font-semibold cursor-default transition-transform hover:scale-110 h-8"
   534→                                          style={{
   535→                                            backgroundColor: colors.bg,
   536→                                            color: colors.text,
   537→                                          }}
   538→                                        >
   539→                                          {cell ? cell.overall.toFixed(1) : '—'}
   540→                                        </div>
   541→                                      </TooltipTrigger>
   542→                                      {cell && (
   543→                                        <TooltipContent side="top" className="bg-popover text-popover-foreground border">
   544→                                          <div className="text-xs space-y-1 min-w-[140px]">
   545→                                            <div className="font-semibold">{state} — {formatDate(date)}</div>
   546→                                            <Separator className="my-1" />
   547→                                            <div className="flex justify-between">
   548→                                              <span className="text-muted-foreground">Violence</span>
   549→                                              <span className="font-mono">{cell.violence.toFixed(2)}</span>
   550→                                            </div>
   551→                                            <div className="flex justify-between">
   552→                                              <span className="text-muted-foreground">Intimidation</span>
   553→                                              <span className="font-mono">{cell.intimidation.toFixed(2)}</span>
   554→                                            </div>
   555→                                            <div className="flex justify-between">
   556→                                              <span className="text-muted-foreground">Logistics</span>
   557→                                              <span className="font-mono">{cell.logistics.toFixed(2)}</span>
   558→                                            </div>
   559→                                            <Separator className="my-1" />
   560→                                            <div className="flex justify-between font-semibold">
   561→                                              <span>Overall</span>
