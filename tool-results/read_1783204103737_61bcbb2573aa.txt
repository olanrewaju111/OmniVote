     1→'use client';

     6→import {

     9→} from '@/components/ui/dialog';

    41→interface MapBoundsData {

    46→
    47→const ROLE_BADGE: Record<string, string> = {
    48→  SUPER_ADMIN: 'border-emerald/30 text-emerald bg-emerald/10',

    50→  ANALYST: 'border-amber/30 text-amber bg-amber/10',

    53→};
    54→
    55→const ROLE_ICONS: Record<string, React.ReactNode> = {
    56→  SUPER_ADMIN: <Shield className="h-3.5 w-3.5" />,
    57→  TENANT_ADMIN: <Users className="h-3.5 w-3.5" />,
    58→  ANALYST: <Eye className="h-3.5 w-3.5" />,
    59→  TRUST_SAFETY: <UserCheck className="h-3.5 w-3.5" />,
    60→  FIELD_AGENT: <Radio className="h-3.5 w-3.5" />,
    61→};
    62→
    63→// ---- Main Component ----
    64→export function TenantManagement() {
    65→  const { user, tenantId } = useDashboardStore();
    66→  const queryClient = useQueryClient();
    67→  const isSuperAdmin = user?.role === 'SUPER_ADMIN';
    68→  const isTenantAdmin = user?.role === 'TENANT_ADMIN';
    69→  const isAdmin = isSuperAdmin || isTenantAdmin;
    70→
    71→  // ===================== SUPER_ADMIN: Platform Tenants =====================
    72→  const { data: allTenants, isLoading: tenantsLoading, isError: tenantsIsError } = useQuery({
    73→    queryKey: ['all-tenants'],
    74→    queryFn: async () => {
    75→      const d = await fetchJson<{ tenants?: unknown[] }>('/api/tenants');
    76→      return d.tenants || [];
    77→    },
    78→    enabled: isSuperAdmin,
    79→  });
    80→
    81→  // ===================== TENANT_ADMIN: Own tenant settings =====================
    82→  const { data: settings, isLoading: settingsLoading, isError: settingsIsError } = useQuery({
    83→    queryKey: ['tenant-settings', tenantId],
    84→    queryFn: async () => {
    85→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`);
    86→    },
    87→    enabled: !!tenantId && isAdmin,
    88→  });
    89→
    90→  // ===================== Common: Tenant Users =====================
    91→  const { data: tenantUsersData, isLoading: usersLoading, isError: usersIsError } = useQuery({
    92→    queryKey: ['tenant-users', tenantId],
    93→    queryFn: async () => {
    94→      const d = await fetchJson<{ users?: unknown[] }>(`/api/tenants/users?tenantId=${tenantId}`);
    95→      return d.users || [];
    96→    },
    97→    enabled: !!tenantId && isAdmin,
    98→  });
    99→
   100→  const currentBounds: MapBoundsData | null = settings?.mapBounds || null;
   101→  const tenantUsers = tenantUsersData || [];
   102→
   103→  // ===================== State =====================
   104→  // Map config dialog
   105→  const [mapConfigOpen, setMapConfigOpen] = useState(false);
   106→  const [mapLabel, setMapLabel] = useState('');
   107→  const [minLat, setMinLat] = useState('');
   108→  const [maxLat, setMaxLat] = useState('');
   109→  const [minLng, setMinLng] = useState('');
   110→  const [maxLng, setMaxLng] = useState('');
   111→
   112→  // Create tenant dialog
   113→  const [createTenantOpen, setCreateTenantOpen] = useState(false);
   114→  const [newTenantName, setNewTenantName] = useState('');
   115→  const [newTenantSlug, setNewTenantSlug] = useState('');
   116→  const [newTenantColor, setNewTenantColor] = useState('#10b981');
   117→  const [newTenantScope, setNewTenantScope] = useState('PRESIDENTIAL');
   118→  const [newAdminName, setNewAdminName] = useState('');
   119→  const [newAdminEmail, setNewAdminEmail] = useState('');
   120→
   121→  // Add user dialog
   122→  const [addUserOpen, setAddUserOpen] = useState(false);
   123→  const [newUserName, setNewUserName] = useState('');
   124→  const [newUserEmail, setNewUserEmail] = useState('');
   125→  const [newUserRole, setNewUserRole] = useState<string>('FIELD_AGENT');
   126→
   127→  // User role change dialog
   128→  const [roleChangeOpen, setRoleChangeOpen] = useState(false);
   129→  const [roleChangeUser, setRoleChangeUser] = useState<{ id: string; name: string; role: string } | null>(null);
   130→  const [newRole, setNewRole] = useState('');
   131→
   132→  // Scope change dialog (SUPER_ADMIN only)
   133→  const [scopeChangeOpen, setScopeChangeOpen] = useState(false);
   134→  const [newScope, setNewScope] = useState('');
   135→
   136→  // Delete confirmation
   137→  const [deleteConfirm, setDeleteConfirm] = useState<{ type: 'tenant' | 'user'; item: { id: string; name: string } } | null>(null);
   138→
   139→  // Active sub-tab for SUPER_ADMIN
   140→  const [subTab, setSubTab] = useState<'tenants' | 'my-tenant' | 'map'>('tenants');
   141→
   142→  // ===================== Effects =====================
   143→  useEffect(() => {
   144→    if (mapConfigOpen && currentBounds) {
   145→      setMapLabel(currentBounds.label || '');
   146→      setMinLat(String(currentBounds.minLat));
   147→      setMaxLat(String(currentBounds.maxLat));
   148→      setMinLng(String(currentBounds.minLng));
   149→      setMaxLng(String(currentBounds.maxLng));
   150→    } else if (mapConfigOpen) {
   151→      setMapLabel('');
   152→      setMinLat(''); setMaxLat(''); setMinLng(''); setMaxLng('');
   153→    }
   154→  }, [mapConfigOpen, currentBounds]);
   155→
   156→  // Auto-derive slug from tenant name
   157→  useEffect(() => {
   158→    if (createTenantOpen && newTenantName && !newTenantSlug) {
   159→      setNewTenantSlug(newTenantName.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, ''));
   160→    }
   161→  }, [createTenantOpen, newTenantName, newTenantSlug]);
   162→
   163→  // ===================== Mutations =====================
   164→  // Save map bounds
   165→  const saveMapMutation = useMutation({
   166→    mutationFn: async (bounds: MapBoundsData) => {
   167→      return fetchJson(`/api/tenant-settings?tenantId=${tenantId}`, {
   168→        method: 'PUT',
   169→        headers: { 'Content-Type': 'application/json' },
   170→        body: JSON.stringify({ mapBounds: bounds }),
   171→      });
   172→    },
   173→    onSuccess: () => {
   174→      queryClient.invalidateQueries({ queryKey: ['tenant-settings', tenantId] });
   175→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   176→      setMapConfigOpen(false);
   177→      toast.success('Map area saved. The map will update automatically.');
   178→    },
   179→    onError: (err) => toast.error(err instanceof Error ? err.message : 'Failed to save map configuration'),
   180→  });
   181→
   182→  // Create tenant
   183→  const createTenantMutation = useMutation({
   184→    mutationFn: (data: { name: string; slug: string; primaryColor: string; scope?: string; adminName: string; adminEmail: string }) =>
   185→      fetchJson('/api/tenants', {
   186→        method: 'POST',
   187→        headers: { 'Content-Type': 'application/json' },
   188→        body: JSON.stringify(data),
   189→      }),
   190→    onSuccess: (data) => {
   191→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   192→      setCreateTenantOpen(false);
   193→      setNewTenantName(''); setNewTenantSlug(''); setNewTenantColor('#10b981');
   194→      setNewTenantScope('PRESIDENTIAL');
   195→      setNewAdminName(''); setNewAdminEmail('');
   196→      toast.success(`Tenant "${data.tenant.name}" created. Admin: ${data.admin.email}`);
   197→    },
   198→    onError: (err) => toast.error(err?.message || 'Failed to create tenant'),
   199→  });
   200→
   201→  // Delete tenant
   202→  const deleteTenantMutation = useMutation({
   203→    mutationFn: (id: string) =>
   204→      fetchJson(`/api/tenants?id=${id}`, { method: 'DELETE' }),
   205→    onSuccess: () => {
   206→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   207→      setDeleteConfirm(null);
   208→      toast.success('Tenant deleted successfully');
   209→    },
   210→    onError: (err) => toast.error(err?.message || 'Failed to delete tenant'),
   211→  });
   212→
   213→  // Toggle tenant active
   214→  const toggleTenantMutation = useMutation({
   215→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   216→      fetchJson('/api/tenants', {
   217→        method: 'PUT',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({ id, isActive }),
   220→      }),
   221→    onSuccess: () => {
   222→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   223→      toast.success('Tenant status updated');
   224→    },
   225→    onError: (err) => toast.error(err?.message || 'Failed to update tenant'),
   226→  });
   227→
   228→  // Add user
   229→  const addUserMutation = useMutation({
   230→    mutationFn: (data: { tenantId: string; name: string; email: string; role: string }) =>
   231→      fetchJson('/api/tenants/users', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify(data),
   235→      }),
   236→    onSuccess: () => {
   237→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   238→      setAddUserOpen(false);
   239→      setNewUserName(''); setNewUserEmail(''); setNewUserRole('FIELD_AGENT');
   240→      toast.success('User added successfully');
   241→    },
   242→    onError: (err) => toast.error(err?.message || 'Failed to add user'),
   243→  });
   244→
   245→  // Change user role
   246→  const changeRoleMutation = useMutation({
   247→    mutationFn: ({ id, role }: { id: string; role: string }) =>
   248→      fetchJson('/api/tenants/users', {
   249→        method: 'PATCH',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ id, role }),
   252→      }),
   253→    onSuccess: () => {
   254→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   255→      setRoleChangeOpen(false);
   256→      setRoleChangeUser(null);
   257→      toast.success('User role updated');
   258→    },
   259→    onError: (err) => toast.error(err?.message || 'Failed to update role'),
   260→  });
   261→
   262→  // Change tenant scope (SUPER_ADMIN only)
   263→  const changeScopeMutation = useMutation({
   264→    mutationFn: (scope: string) =>
   265→      fetchJson(`/api/tenant-settings?tenantId=${tenantId}`, {
   266→        method: 'PUT',
   267→        headers: { 'Content-Type': 'application/json' },
   268→        body: JSON.stringify({ scope }),
   269→      }),
   270→    onSuccess: () => {
   271→      queryClient.invalidateQueries({ queryKey: ['tenant-settings', tenantId] });
   272→      queryClient.invalidateQueries({ queryKey: ['all-tenants'] });
   273→      setScopeChangeOpen(false);
   274→      toast.success('Tenant scope updated successfully');
   275→    },
   276→    onError: (err) => toast.error(err?.message || 'Failed to update scope'),
   277→  });
   278→
   279→  // Remove user
   280→  const deleteUserMutation = useMutation({
   281→    mutationFn: (id: string) =>
   282→      fetchJson(`/api/tenants/users?id=${id}`, { method: 'DELETE' }),
   283→    onSuccess: () => {
   284→      queryClient.invalidateQueries({ queryKey: ['tenant-users'] });
   285→      setDeleteConfirm(null);
   286→      toast.success('User removed');
   287→    },
   288→    onError: (err) => toast.error(err?.message || 'Failed to remove user'),
   289→  });
   290→
   291→  // ===================== Handlers =====================
   292→  const handleSaveMap = () => {
   293→    const minLatN = parseFloat(minLat);
   294→    const maxLatN = parseFloat(maxLat);
   295→    const minLngN = parseFloat(minLng);
   296→    const maxLngN = parseFloat(maxLng);
   297→    if ([minLatN, maxLatN, minLngN, maxLngN].some(isNaN)) { toast.error('All coordinates must be valid numbers'); return; }
   298→    if (minLatN >= maxLatN || minLngN >= maxLngN) { toast.error('minLat must be < maxLat, and minLng must be < maxLng'); return; }
   299→    saveMapMutation.mutate({ minLat: minLatN, maxLat: maxLatN, minLng: minLngN, maxLng: maxLngN, label: mapLabel || 'Custom Area' });
   300→  };
   301→
   302→  const handlePreset = (preset: typeof REGION_PRESETS[number]) => {
   303→    setMapLabel(preset.label);
   304→    setMinLat(String(preset.minLat));
   305→    setMaxLat(String(preset.maxLat));
   306→    setMinLng(String(preset.minLng));
   307→    setMaxLng(String(preset.maxLng));
   308→  };
   309→
   310→  const handleResetMap = () => {
   311→    saveMapMutation.mutate({ minLat: 4.0, maxLat: 14.0, minLng: 2.5, maxLng: 15.0, label: 'Nigeria' });
   312→  };
   313→
   314→  const handleCreateTenant = () => {
   315→    if (!newTenantName || !newTenantSlug || !newAdminName || !newAdminEmail) {
   316→      toast.error('All fields are required'); return;
   317→    }
   318→    createTenantMutation.mutate({
   319→      name: newTenantName, slug: newTenantSlug, primaryColor: newTenantColor,
   320→      scope: newTenantScope,
   321→      adminName: newAdminName, adminEmail: newAdminEmail,
   322→    });
   323→  };
   324→
   325→  const handleAddUser = () => {
   326→    if (!newUserName || !newUserEmail || !newUserRole) { toast.error('All fields are required'); return; }
   327→    addUserMutation.mutate({ tenantId: tenantId!, name: newUserName, email: newUserEmail, role: newUserRole });
   328→  };
   329→
   330→  const handleRoleChange = () => {
   331→    if (!roleChangeUser || !newRole) return;
   332→    changeRoleMutation.mutate({ id: roleChangeUser.id, role: newRole });
   333→  };
   334→
   335→  // ===================== Restricted access =====================
   336→  if (!isAdmin) {
   337→    return (
   338→      <div className="h-full flex items-center justify-center p-6">
   339→        <div className="text-center space-y-2">
   340→          <Shield className="h-8 w-8 text-muted-foreground/30 mx-auto" />
   341→          <p className="text-sm text-muted-foreground">Access restricted to Super Admin and Tenant Admin</p>
   342→        </div>
   343→      </div>
   344→    );
   345→  }
   346→
   347→  const isLoading = isSuperAdmin ? tenantsLoading : settingsLoading;
   348→  const hasError = isSuperAdmin ? tenantsIsError : settingsIsError;
   349→
   350→  // ===================== Render =====================
   351→  return (
   352→    <div className="h-full flex flex-col overflow-hidden">
   353→      {/* Header */}
   354→      <div className="px-4 sm:px-6 pt-4 pb-3 shrink-0">
   355→        <div className="flex items-center justify-between flex-wrap gap-3">
   356→          <div>
   357→            <h2 className="text-lg font-semibold flex items-center gap-2">
   358→              <Building2 className="h-5 w-5 text-emerald" />
   359→              {isSuperAdmin ? 'Platform Management' : 'Organization Settings'}
   360→            </h2>
   361→            <p className="text-sm text-muted-foreground mt-0.5">
   362→              {isSuperAdmin
   363→                ? 'Create and manage tenant organizations, their users, and configurations'
   364→                : 'Manage your organization settings, users, and map configuration'}
   365→            </p>
   366→          </div>
   367→          {isSuperAdmin && (
   368→            <Button
   369→              size="sm"
   370→              className="gap-1.5 text-xs bg-emerald hover:bg-emerald/90 text-emerald-950 h-8"
   371→              onClick={() => setCreateTenantOpen(true)}
   372→            >
   373→              <Plus className="h-3.5 w-3.5" /> New Tenant
   374→            </Button>
   375→          )}
   376→        </div>
   377→
   378→        {/* Sub-tabs for SUPER_ADMIN */}
   379→        {isSuperAdmin && (
   380→          <div className="flex items-center gap-1 mt-3 bg-muted/30 rounded-lg p-0.5 w-fit">
   381→            {([
   382→              { id: 'tenants' as const, label: 'All Tenants', icon: <Globe className="h-3.5 w-3.5" /> },
   383→              { id: 'my-tenant' as const, label: 'My Organization', icon: <Building2 className="h-3.5 w-3.5" /> },
   384→              { id: 'map' as const, label: 'Map Config', icon: <MapPin className="h-3.5 w-3.5" /> },
   385→            ]).map(tab => (
   386→              <button
   387→                key={tab.id}
   388→                onClick={() => setSubTab(tab.id)}
   389→                className={cn(
   390→                  'flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium transition-colors',
   391→                  subTab === tab.id ? 'bg-card shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   392→                )}
   393→              >
   394→                {tab.icon} {tab.label}
   395→              </button>
   396→            ))}
   397→          </div>
   398→        )}
   399→      </div>
   400→
   401→      {/* Scrollable content */}
   402→      <div className="flex-1 overflow-y-auto px-4 sm:px-6 pb-6 space-y-4 min-h-0">
   403→
   404→        {isLoading ? (
   405→          <div className="flex-1 flex items-center justify-center py-20">
   406→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   407→          </div>
   408→        ) : hasError ? (
   409→          <div className="flex-1 flex flex-col items-center justify-center py-20 text-center p-6">
   410→            <AlertCircle className="h-10 w-10 text-destructive mb-3" />
   411→            <p className="text-sm text-muted-foreground">Failed to load data. Please try again.</p>
   412→            <Button variant="outline" size="sm" className="mt-3" onClick={() => window.location.reload()}>
   413→              Retry
   414→            </Button>
   415→          </div>
   416→        ) : isSuperAdmin ? (
   417→          /* ========== SUPER_ADMIN VIEWS ========== */
   418→          <>
   419→            {subTab === 'tenants' && (
   420→              <SuperAdminTenantsView
   421→                tenants={allTenants || []}
   422→                onToggle={(id, isActive) => toggleTenantMutation.mutate({ id, isActive })}
   423→                onDelete={(item) => setDeleteConfirm({ type: 'tenant', item })}
   424→                onManageUsers={() => setSubTab('my-tenant')}
   425→              />
   426→            )}
   427→            {subTab === 'my-tenant' && (
   428→              <TenantUsersView
   429→                settings={settings}
   430→                tenantUsers={tenantUsers}
   431→                userRole={user?.role || ''}
   432→                onOpenUserDialog={() => setAddUserOpen(true)}
   433→                onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   434→                onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   435→                onOpenMapConfig={() => { setSubTab('map'); setTimeout(() => setMapConfigOpen(true), 100); }}
   436→                onChangeScope={() => { setNewScope(settings?.scope || 'PRESIDENTIAL'); setScopeChangeOpen(true); }}
   437→                canChangeScope={true}
   438→              />
   439→            )}
   440→            {subTab === 'map' && (
   441→              <MapConfigSection
   442→                currentBounds={currentBounds}
   443→                onConfigure={() => setMapConfigOpen(true)}
   444→              />
   445→            )}
   446→          </>
   447→        ) : (
   448→          /* ========== TENANT_ADMIN VIEW ========== */
   449→          <>
   450→            <TenantUsersView
   451→              settings={settings}
   452→              tenantUsers={tenantUsers}
   453→              userRole={user?.role || ''}
   454→              onOpenUserDialog={() => setAddUserOpen(true)}
   455→              onRoleChange={(u) => { setRoleChangeUser(u); setNewRole(u.role); setRoleChangeOpen(true); }}
   456→              onDeleteUser={(u) => setDeleteConfirm({ type: 'user', item: { id: u.id, name: u.name } })}
   457→              onOpenMapConfig={() => setMapConfigOpen(true)}
   458→              onChangeScope={() => { setNewScope(settings?.scope || 'PRESIDENTIAL'); setScopeChangeOpen(true); }}
   459→              canChangeScope={false}
   460→            />
   461→            <MapConfigSection
   462→              currentBounds={currentBounds}
   463→              onConfigure={() => setMapConfigOpen(true)}
   464→            />
   465→            <InfoNotices />
   466→          </>
   467→        )}
   468→      </div>
   469→
