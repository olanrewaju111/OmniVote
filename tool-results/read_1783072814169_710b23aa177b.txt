    99→
   100→function timeAgo(dateStr: string | null): string {
   101→  if (!dateStr) return 'Never';
   102→  const diff = Date.now() - new Date(dateStr).getTime();
   103→  const mins = Math.floor(diff / 60000);
   104→  if (mins < 1) return 'Just now';
   105→  if (mins < 60) return `${mins}m ago`;
   106→  const hrs = Math.floor(mins / 60);
   107→  if (hrs < 24) return `${hrs}h ago`;
   108→  return `${Math.floor(hrs / 24)}d ago`;
   109→}
